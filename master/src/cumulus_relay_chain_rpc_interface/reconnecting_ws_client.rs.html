72/// Format url and force addition of a port
73fn url_to_string_with_port(url: Url) -> Option<String> {
74	// This is already validated on CLI side, just defensive here
75	if (url.scheme() != "ws" && url.scheme() != "wss") || url.host_str().is_none() {
76		tracing::warn!(target: LOG_TARGET, ?url, "Non-WebSocket URL or missing host.");
77		return None;
78	}
79
80	// Either we have a user-supplied port or use the default for 'ws' or 'wss' here
81	Some(format!(
82		"{}://{}:{}{}{}",
83		url.scheme(),
84		url.host_str()?,
85		url.port_or_known_default()?,
86		url.path(),
87		url.query().map(|query| format!("?{}", query)).unwrap_or_default()
88	))
89}

104/// A prepared preferred server awaiting validation against the active server's latest height.

106	connection: RpcConnection,
107	/// Latest buffered best head, consumed during probing and published when switching.
108	best_header: RelayHeader,
109}

111/// Try to find a new RPC server to connect to. Uses a naive retry
112/// logic that does an exponential backoff in between iterations
113/// through all URLs from the list. It uses a constant to tell how
114/// many iterations of connection attempts to all URLs we allow. We
115/// return early when a connection and its subscriptions are ready.
116async fn connect_next_available_rpc_server(
117	urls: &[String],
118	starting_position: usize,
119) -> Result<RpcConnection, ()> {
120	tracing::debug!(target: LOG_TARGET, starting_position, "Connecting to RPC server.");
121
122	let mut prev_iteration: u32 = 0;
123	for (counter, url) in urls
124		.iter()
125		.cycle()
126		.skip(starting_position)
127		.take(urls.len() * DEFAULT_EXTERNAL_RPC_CONN_RETRIES)
128		.enumerate()
129	{
130		// If we reached the end of the urls list, backoff before retrying
131		// connections to the entire list once more.
132		let Ok(current_iteration) = (counter / urls.len()).try_into() else {
133			tracing::error!(target: LOG_TARGET, "Too many connection attempts to the RPC servers, aborting...");
134			break;
135		};
136		if current_iteration > prev_iteration {
137			// Safe conversion given we convert positive i32s which are lower than u64::MAX.
138			tokio::time::sleep(Duration::from_millis(
139				DEFAULT_SLEEP_TIME_MS_BETWEEN_RETRIES *
140					DEFAULT_SLEEP_EXP_BACKOFF_BETWEEN_RETRIES.pow(prev_iteration) as u64,
141			))
142			.await;
143			prev_iteration = current_iteration;
144		}
145
146		let index = (starting_position + counter) % urls.len();
147		tracing::info!(
148			target: LOG_TARGET,
149			attempt = current_iteration,
150			index,
151			url,
152			"Trying to connect to next external relaychain node.",
153		);
154		match RpcConnection::connect(index, url).await {
155			Ok(connection) => return Ok(connection),
156			Err(err) => tracing::debug!(target: LOG_TARGET, url, ?err, "Unable to connect."),
157		};
158	}
159
160	tracing::error!(target: LOG_TARGET, "Retrying to connect to any external relaychain node failed.");
161	Err(())
162}

164/// Probe earlier servers in preference order without blocking the active connection.
165/// Yield each candidate so the worker can compare it with its latest best block number and
166/// continue the same round if the candidate is already behind.
167fn probe_preferred_rpc_servers(
168	candidates: Vec<(usize, String)>,
169) -> Fuse<BoxStream<'static, PreferredServer>> {
170	stream::iter(candidates)
171		.then(|(index, url)| async move {
172			match tokio::time::timeout(PREFERRED_RPC_SERVER_PROBE_TIMEOUT, async {
173				let mut connection = RpcConnection::connect(index, &url).await?;
174				let best = &mut connection.subscriptions.best_subscription;
175				// Substrate sends its current best head when the subscription opens. Do not
176				// wait for the candidate to catch up; only consume updates already buffered.
177				let Some(mut best_header) = best.next().await.transpose()? else {
178					return Ok(None);
179				};
180				while let Some(notification) = best.next().now_or_never() {
181					let Some(header) = notification.transpose()? else {
182						return Ok(None);
183					};
184					best_header = header;
185				}
186				Ok::<_, JsonRpseeError>(Some(PreferredServer { connection, best_header }))
187			})
188			.await
189			{
190				Ok(Ok(server)) => server,
191				Ok(Err(error)) => {
192					tracing::debug!(target: LOG_TARGET, url, ?error, "Unable to use preferred external relaychain node.");
193					None
194				},
195				Err(_) => {
196					tracing::debug!(target: LOG_TARGET, url, "Timed out checking preferred external relaychain node.");
197					None
198				},
199			}
200		})
201		.filter_map(ready)
202		.boxed()
203		.fuse()
204}

233			RelayNumber,

271	/// Create a request future that performs an RPC request and sends the results to the caller.
272	/// In case of a dead websocket connection, it returns the original request parameters to
273	/// enable retries.
274	fn create_request(
275		&self,
276		method: String,
277		params: ArrayParams,
278		response_sender: OneshotSender<Result<JsonValue, JsonRpseeError>>,
279	) -> BoxFuture<'static, Result<(), RpcDispatcherMessage>> {
280		let future_client = self.client.clone();
281		async move {
282			let resp = future_client.request(&method, params.clone()).await;
283
284			// We should only return the original request in case
285			// the websocket connection is dead and requires a restart.
286			// Other errors should be forwarded to the request caller.
287			if let Err(JsonRpseeError::RestartNeeded(_)) = resp {
288				return Err(RpcDispatcherMessage::Request(method, params, response_sender));
289			}
290
291			if let Err(err) = response_sender.send(resp) {
292				tracing::debug!(
293					target: LOG_TARGET,
294					?err,
295					"Recipient no longer interested in request result"
296				);
297			}
298			Ok(())
299		}
300		.boxed()
301	}

309impl ReconnectingWebsocketWorker {
310	/// Create new worker. Returns the worker and a channel to register new listeners.
311	pub async fn new(
312		urls: Vec<Url>,
313	) -> (ReconnectingWebsocketWorker, TokioSender<RpcDispatcherMessage>) {
314		let urls = urls.into_iter().filter_map(url_to_string_with_port).collect();
315
316		let (tx, rx) = tokio_channel(100);
317		let worker = ReconnectingWebsocketWorker {
318			ws_urls: urls,
319			client_receiver: rx,
320			imported_header_listeners: Vec::new(),
321			finalized_header_listeners: Vec::new(),
322			best_header_listeners: Vec::new(),
323			preferred_server_probe_interval: PREFERRED_RPC_SERVER_PROBE_INTERVAL,
324		};
325		(worker, tx)
326	}
327
328	/// Run this worker to drive notification streams.
329	/// The worker does the following:
330	/// - Listen for [`RpcDispatcherMessage`], perform requests and register new listeners for the
331	///   notification streams
332	/// - Distribute incoming import, best head and finalization notifications to registered
333	///   listeners. If an error occurs during sending, the receiver has been closed and we remove
334	///   the sender from the list.
335	/// - Find a new valid RPC server to connect to in case the websocket connection is terminated.
336	///   If the worker is not able to connect to an RPC server from the list, the worker shuts
337	///   down.
338	/// - While connected to an RPC server other than the first one in the list, periodically check
339	///   the servers listed before it and switch back to the first one that is reachable and has
340	///   caught up with the best block seen so far.
341	pub async fn run(mut self) {
342		let mut pending_requests = FuturesUnordered::new();
343
344		let urls = std::mem::take(&mut self.ws_urls);
345		let Ok(mut active) = connect_next_available_rpc_server(&urls, 0).await else {
346			tracing::error!(target: LOG_TARGET, "No valid RPC url found. Stopping RPC worker.");
347			return;
348		};
349
350		let mut imported_blocks_cache = LruMap::new(ByLength::new(40));
351		let mut should_reconnect = ConnectionStatus::Connected;
352		let mut last_seen_finalized_num: RelayNumber = 0;
353		let mut last_seen_best_num: RelayNumber = 0;
354		let mut preferred_server_probe_timer = tokio::time::interval_at(
355			tokio::time::Instant::now() + self.preferred_server_probe_interval,
356			self.preferred_server_probe_interval,
357		);
358		let mut preferred_server_probe = stream::empty().boxed().fuse();
359		loop {
360			if let ConnectionStatus::ReconnectRequired(failed_request) = should_reconnect {
361				// Requests retain their own clients. Leave them pending while replacing the
362				// connection; a slow request on an older server must not delay failover.
363				let Ok(connection) =
364					connect_next_available_rpc_server(&urls, active.index + 1).await
365				else {
366					tracing::error!(target: LOG_TARGET, "Unable to reconnect, stopping worker.");
367					return;
368				};
369				active = connection;
370				preferred_server_probe = stream::empty().boxed().fuse();
371				preferred_server_probe_timer.reset();
372				if let Some(RpcDispatcherMessage::Request(method, params, sender)) = failed_request
373				{
374					pending_requests.push(active.create_request(method, params, sender));
375				}
376				should_reconnect = ConnectionStatus::Connected;
377			}
378
379			tokio::select! {
380				evt = self.client_receiver.recv() => match evt {
381					Some(RpcDispatcherMessage::RegisterBestHeadListener(tx)) => {
382						self.best_header_listeners.push(tx);
383					},
384					Some(RpcDispatcherMessage::RegisterImportListener(tx)) => {
385						self.imported_header_listeners.push(tx)
386					},
387					Some(RpcDispatcherMessage::RegisterFinalizationListener(tx)) => {
388						self.finalized_header_listeners.push(tx)
389					},
390					Some(RpcDispatcherMessage::Request(method, params, response_sender)) => {
391						pending_requests.push(active.create_request(method, params, response_sender));
392					},
393					None => {
394						tracing::error!(target: LOG_TARGET, "RPC client receiver closed. Stopping RPC Worker.");
395						return;
396					}
397				},
398				should_retry = pending_requests.next(), if !pending_requests.is_empty() => {
399					if let Some(Err(req)) = should_retry {
400						// Connection failures are retried on the current server. A failure
401						// from an old connection must not displace a healthy active one.
402						if !active.client.is_connected() {
403							should_reconnect = ConnectionStatus::ReconnectRequired(Some(req));
404						} else if let RpcDispatcherMessage::Request(method, params, sender) = req {
405							pending_requests.push(active.create_request(method, params, sender));
406						}
407					}
408				},
409				_ = preferred_server_probe_timer.tick(),
410					if active.index > 0 && preferred_server_probe.is_terminated() =>
411				{
412					let candidates = urls[..active.index]
413						.iter()
414						.cloned()
415						.enumerate()
416						.collect();
417					preferred_server_probe = probe_preferred_rpc_servers(candidates);
418				},
419				preferred_server = preferred_server_probe.next(), if !preferred_server_probe.is_terminated() => {
420					let Some(server) = preferred_server else {
421						preferred_server_probe_timer.reset();
422						continue;
423					};
424					// Validate against the latest head, not a snapshot from the start of the
425					// round. Rejection leaves the remaining candidates available for polling.
426					if server.best_header.number < last_seen_best_num {
427						continue;
428					}
429					tracing::info!(
430						target: LOG_TARGET,
431						index = server.connection.index,
432						url = %urls[server.connection.index],
433						previous_index = active.index,
434						"Switched back to preferred external relay chain node.",
435					);
436					active = server.connection;
437					// Publish the head used for validation, not the older startup notifications
438					// consumed by the probe. Subsequent best-head changes may include reorgs.
439					last_seen_best_num = server.best_header.number;
440					distribute_header(server.best_header, &mut self.best_header_listeners);
441					preferred_server_probe = stream::empty().boxed().fuse();
442					preferred_server_probe_timer.reset();
443				},
444				import_event = active.subscriptions.import_subscription.next() => {
445					match import_event {
446						Some(Ok(header)) => {
447							let hash = header.hash();
448							if imported_blocks_cache.peek(&hash).is_some() {
449								tracing::debug!(
450									target: LOG_TARGET,
451									number = header.number,
452									?hash,
453									"Duplicate imported block header. This might happen after switching to a new RPC node. Skipping distribution."
454								);
455								continue;
456							}
457							imported_blocks_cache.insert(hash, ());
458							distribute_header(header, &mut self.imported_header_listeners);
459						},
460						None => {
461							tracing::error!(target: LOG_TARGET, "Subscription closed.");
462							should_reconnect = ConnectionStatus::ReconnectRequired(None);
463						},
464						Some(Err(error)) => {
465							tracing::error!(target: LOG_TARGET, ?error, "Error in RPC subscription.");
466							should_reconnect = ConnectionStatus::ReconnectRequired(None);
467						},
468					}
469				},
470				best_header_event = active.subscriptions.best_subscription.next() => {
471					match best_header_event {
472						Some(Ok(header)) => {
473							last_seen_best_num = header.number;
474							distribute_header(header, &mut self.best_header_listeners)
475						},
476						None => {
477							tracing::error!(target: LOG_TARGET, "Subscription closed.");
478							should_reconnect = ConnectionStatus::ReconnectRequired(None);
479						},
480						Some(Err(error)) => {
481							tracing::error!(target: LOG_TARGET, ?error, "Error in RPC subscription.");
482							should_reconnect = ConnectionStatus::ReconnectRequired(None);
483						},
484					}
485				}
486				finalized_event = active.subscriptions.finalized_subscription.next() => {
487					match finalized_event {
488						Some(Ok(header)) if header.number > last_seen_finalized_num => {
489							last_seen_finalized_num = header.number;
490							distribute_header(header, &mut self.finalized_header_listeners);
491						},
492						Some(Ok(header)) => {
493							tracing::debug!(
494								target: LOG_TARGET,
495								number = header.number,
496								last_seen_finalized_num,
497								"Duplicate finalized block header. This might happen after switching to a new RPC node. Skipping distribution."
498							);
499						},
500						None => {
501							tracing::error!(target: LOG_TARGET, "Subscription closed.");
502							should_reconnect = ConnectionStatus::ReconnectRequired(None);
503						},
504						Some(Err(error)) => {
505							tracing::error!(target: LOG_TARGET, ?error, "Error in RPC subscription.");
506							should_reconnect = ConnectionStatus::ReconnectRequired(None);
507						},
508					}
509				}
510			}
511		}
512	}
513}

549	/// Send the current best header to the subscriber until it goes away.
550	async fn serve_heads(
551		pending: PendingSubscriptionSink,
552		best: Arc<AtomicU32>,
553		subscriptions: Arc<AtomicUsize>,
554		gate: Option<Arc<Semaphore>>,
555		notification_gate: Option<Arc<Semaphore>>,
556		notifications: Arc<AtomicUsize>,
557	) -> SubscriptionResult {
558		subscriptions.fetch_add(1, Ordering::SeqCst);
559		if let Some(gate) = gate {
560			gate.acquire().await?.forget();
561		}
562		let sink = pending.accept().await?;
563		// Returning an error would close the subscription on the client side, so only stop once
564		// the client is gone.
565		loop {
566			if let Some(gate) = &notification_gate {
567				gate.acquire().await?.forget();
568			}
569			let head = SubscriptionMessage::from_json(&header(best.load(Ordering::SeqCst)))?;
570			if sink.send(head).await.is_err() {
571				return Ok(());
572			}
573			notifications.fetch_add(1, Ordering::SeqCst);
574			tokio::time::sleep(Duration::from_millis(50)).await;
575		}
576	}

603		fn start(port: u16, name: &'static str, best: u32, hang: bool) -> Self {
604			Self::start_with_gates(port, name, best, hang, ResponseGates::default())
605		}
606
607		fn start_with_gates(
608			port: u16,
609			name: &'static str,
610			best: u32,
611			hang: bool,
612			gates: ResponseGates,
613		) -> Self {
614			let best = Arc::new(AtomicU32::new(best));
615			let best_notifications = Arc::new(AtomicUsize::new(0));
616			let subscriptions = Arc::new(AtomicUsize::new(0));
617			let hang_requests = Arc::new(AtomicUsize::new(0));
618
619			let mut module = RpcModule::new(());
620			module.register_method("test_whoami", move |_, _, _| name).unwrap();
621			let requests = hang_requests.clone();
622			module
623				.register_async_method("test_hang", move |_, _, _| {
624					let requests = requests.clone();
625					async move {
626						requests.fetch_add(1, Ordering::SeqCst);
627						if hang {
628							futures::future::pending::<()>().await;
629						}
630						name
631					}
632				})
633				.unwrap();
634
635			for (subscribe, notification, unsubscribe) in [
636				("chain_subscribeAllHeads", "chain_allHead", "chain_unsubscribeAllHeads"),
637				("chain_subscribeNewHeads", "chain_newHead", "chain_unsubscribeNewHeads"),
638				(
639					"chain_subscribeFinalizedHeads",
640					"chain_finalizedHead",
641					"chain_unsubscribeFinalizedHeads",
642				),
643			] {
644				let (best, subscriptions) = (best.clone(), subscriptions.clone());
645				let gate = gates.subscriptions.clone();
646				let (notification_gate, notifications) = if subscribe == "chain_subscribeNewHeads" {
647					(gates.best_heads.clone(), best_notifications.clone())
648				} else {
649					(None, Arc::new(AtomicUsize::new(0)))
650				};
651				module
652					.register_subscription(
653						subscribe,
654						notification,
655						unsubscribe,
656						move |_, pending, _, _| {
657							serve_heads(
658								pending,
659								best.clone(),
660								subscriptions.clone(),
661								gate.clone(),
662								notification_gate.clone(),
663								notifications.clone(),
664							)
665						},
666					)
667					.unwrap();
668			}
669
670			// Run on a dedicated runtime, so that dropping it kills the node.
671			let listener = std::net::TcpListener::bind(("127.0.0.1", port)).unwrap();
672			listener.set_nonblocking(true).unwrap();
673			let runtime = tokio::runtime::Builder::new_multi_thread()
674				.worker_threads(1)
675				.enable_all()
676				.build()
677				.unwrap();
678			let server = {
679				let _guard = runtime.enter();
680				Server::builder().build_from_tcp(listener).unwrap().start(module)
681			};
682
683			Self {
684				best,
685				best_notifications,
686				subscriptions,
687				hang_requests,
688				runtime: Some(runtime),
689				_server: server,
690			}
691		}
692	}

695		fn drop(&mut self) {
696			// Dropping a runtime from within the test's runtime would panic.
697			if let Some(runtime) = self.runtime.take() {
698				runtime.shutdown_background();
699			}
700		}

703	/// Pick `N` distinct unused ports.
704	fn unused_ports<const N: usize>() -> [u16; N] {
705		let mut ports = Vec::new();
706		while ports.len() < N {
707			let port = portpicker::pick_unused_port().unwrap();
708			if !ports.contains(&port) {
709				ports.push(port);
710			}
711		}
712		ports.try_into().unwrap()
713	}

715	/// Start a worker for local servers on `ports`, checking preferred servers every 100ms.
716	async fn start_worker(ports: &[u16]) -> TokioSender<RpcDispatcherMessage> {
717		let urls = ports
718			.iter()
719			.map(|port| Url::parse(&format!("ws://127.0.0.1:{port}")).unwrap())
720			.collect();
721		let (mut worker, sender) = ReconnectingWebsocketWorker::new(urls).await;
722		worker.preferred_server_probe_interval = Duration::from_millis(100);
723		tokio::spawn(worker.run());
724		sender
725	}

727	async fn request(worker: &TokioSender<RpcDispatcherMessage>, method: &str) -> JsonValue {
728		let (tx, rx) = futures::channel::oneshot::channel();
729		worker
730			.send(RpcDispatcherMessage::Request(method.into(), ArrayParams::new(), tx))
731			.await
732			.unwrap();
733		tokio::time::timeout(TEST_TIMEOUT, rx).await.unwrap().unwrap().unwrap()
734	}

737		tokio::time::timeout(TEST_TIMEOUT, async {
738			while !condition() {
739				tokio::time::sleep(Duration::from_millis(10)).await;
740			}
741		})
742		.await
743		.unwrap();
744	}

762	}

767			.send(RpcDispatcherMessage::RegisterBestHeadListener(tx))

769			.expect("worker is running");

774				}
775			}
776			panic!("best header stream closed");

779		.expect("worker receives the expected best header");
780	}

983	async fn switch_publishes_latest_buffered_best_head_and_preserves_later_reorgs() {
984		let [port_a, port_b] = unused_ports();
985		let _b = MockRelayNode::start(port_b, "B", 101, false);
986		let worker = start_worker(&[port_a, port_b]).await;

990		worker
991			.send(RpcDispatcherMessage::RegisterBestHeadListener(tx))
992			.await
993			.expect("worker is running");
994
995		// Allow the import and best subscriptions to open, but hold the final subscription
996		// until two best-head notifications have been queued on the candidate connection.
997		let subscriptions = Arc::new(Semaphore::new(2));
998		let notifications = Arc::new(Semaphore::new(0));
999		let a = MockRelayNode::start_with_gates(
1000			port_a,
1001			"A",
1002			100,
1003			false,
1004			ResponseGates {
1005				subscriptions: Some(subscriptions.clone()),
1006				best_heads: Some(notifications.clone()),
1007			},
1008		);
1009		wait_until(|| a.subscriptions.load(Ordering::SeqCst) == 3).await;
1010		notifications.add_permits(1);
1011		wait_until(|| a.best_notifications.load(Ordering::SeqCst) == 1).await;
1012		a.best.store(102, Ordering::SeqCst);
1013		notifications.add_permits(1);
1014		wait_until(|| a.best_notifications.load(Ordering::SeqCst) == 2).await;
1015		subscriptions.add_permits(1);
1016
1017		tokio::time::timeout(Duration::from_secs(5), async {
1018			loop {
1019				let head = heads.next().await.expect("best-head stream stays open");
1020				assert!(head.number >= 101, "handoff must not replay the stale initial head");
1021				if head.number == 102 {
1022					break;
1023				}
1024			}
1025			wait_until_served_by(&worker, "A").await;
1026		})
1027		.await
1028		.expect("handoff publishes the buffered current head without waiting for another block");
1029
1030		// Once installed, a lower best head can represent a genuine reorg. Forward it,
1031		// and do not publish the retained handoff header a second time.
1032		a.best.store(100, Ordering::SeqCst);
1033		notifications.add_permits(1);
1034		let head = tokio::time::timeout(Duration::from_secs(5), heads.next())
1035			.await
1036			.expect("reorg notification arrives")
1037			.expect("best-head stream stays open");
1038		assert_eq!(head.number, 100);
1039	}
