17//! Classify outgoing `AllMessages` into [`Effect`] (asserted) or [`Query`] (mock-answered).
18//!
19//! A single `AllMessages` can correspond to several entries in the observation stream — most
20//! commonly because a single `NetworkBridgeTxMessage::SendRequests` carries multiple `Requests`,
21//! or `SendCollationMessages` carries multiple peer-batches. The classifier returns a flat
22//! `Vec<Classified>` accordingly; callers iterate.
23//!
24//! The outer `AllMessages` enum is large and includes families the collator-protocol does not
25//! interact with at all (`ApprovalDistribution`, `BitfieldDistribution`, ...). The outer match
26//! therefore keeps a wildcard arm that panics with a contract-violation message; that's the
27//! intended failure mode for genuinely undeclared egress.
28//!
29//! Inner matches on small, stable enums (`NetworkBridgeTxMessage`, `Requests`,
30//! `ReportPeerMessage`) are **exhaustive** — when an upstream variant is added, the build
31//! breaks on the next compile and we make a conscious decision rather than discovering it via
32//! a panicked test.
33//!
34//! See `polkadot/node/network/collator-protocol/src/...` for the production emission sites
35//! that ground each variant.
36

54/// Result of classifying a single outgoing `AllMessages`. One message may classify into
55/// multiple entries (see module docs); callers iterate the returned `Vec`.
56#[derive(Debug)]
57pub enum Classified {
58	/// The message is an observable effect — record it.
59	Effect(Effect),
60	/// The message is an information-gathering query — forward to responder.
61	Query(Query),
62}
63
64/// Peek at an outgoing `AllMessages` without consuming it and emit any [`Effect`] descriptions
65/// implied by the message.
66///
67/// Used by the router for **dual-delivery** messages: outbound messages that are *both* an
68/// observable Effect tests assert on *and* an input to a real auxiliary subsystem (e.g.
69/// `CandidateBackingMessage::Second{...}`). The router records these effects manually when an
70/// aux slot accepts the message; otherwise the regular [`classify`] path runs and emits them
71/// in the consume direction.
72///
73/// Returns an empty `Vec` for variants that have no Effect interpretation (queries, plain
74/// aux-only messages).
75pub fn peek_effects(msg: &AllMessages) -> Vec<Effect> {
76	match msg {
77		AllMessages::CandidateBacking(CandidateBackingMessage::Second {
78			scheduling_parent,
79			candidate,
80			..
81		}) => vec![Effect::SecondCandidate {
82			scheduling_parent: *scheduling_parent,
83			candidate_hash: candidate.hash(),
84			para: candidate.descriptor.para_id(),
85		}],
86		_ => Vec::new(),
87	}
88}

90/// Walk an outgoing `AllMessages`, classify it. Panics on undeclared egress (a contract
91/// violation, not a test bug).
92///
93/// `pending` is the side table where outgoing-fetch response senders are parked: any
94/// `NetworkBridgeTxMessage::SendRequests` produces an `Effect::SendRequest` with a fresh
95/// [`RequestId`] and the embedded `oneshot::Sender` is moved into `pending` so tests can
96/// later resolve it via `Sim::respond_fetch`.
97///
98/// [`RequestId`]: crate::contract::RequestId
99pub fn classify(msg: AllMessages, pending: &mut PendingFetches) -> Vec<Classified> {
100	match msg {
101		AllMessages::CandidateBacking(inner) => from_candidate_backing(inner),
102		AllMessages::NetworkBridgeTx(inner) => from_network_bridge_tx(inner, pending),
103		AllMessages::RuntimeApi(inner) => vec![Classified::Query(Query::Runtime(inner))],
104		AllMessages::ChainApi(inner) => vec![Classified::Query(Query::ChainApi(inner))],
105		// A prospective-parachains message is a query family some configs answer (e.g. a
106		// `QueryScript` with a `.prospective(..)` handler, or prospective-parachains itself
107		// under test). In the collator config it is instead consumed by a real
108		// prospective-parachains aux slot, so it should never reach `classify` — the router
109		// only falls through to here when no slot claimed the message. If it does reach here
110		// in a collator scenario, the responder chain has no prospective handler and the tail
111		// `PanicResponder` will surface it by name: the fix is to register the prospective aux
112		// slot, not to script the query.
113		AllMessages::ProspectiveParachains(inner) => {
114			vec![Classified::Query(Query::Prospective(inner))]
115		},
116		other => panic!(
117			"collator-protocol emitted undeclared egress: {:?}\n\
118			 If this is a legitimate effect, add a variant to `Effect` and a classifier arm.",
119			other
120		),
121	}
122}

124fn from_candidate_backing(msg: CandidateBackingMessage) -> Vec<Classified> {
125	match msg {
126		CandidateBackingMessage::Second { scheduling_parent, candidate, pvd: _, pov: _ } => {
127			vec![Classified::Effect(Effect::SecondCandidate {
128				scheduling_parent,
129				candidate_hash: candidate.hash(),
130				para: candidate.descriptor.para_id(),
131			})]
132		},
133		msg @ CandidateBackingMessage::CanSecond(..) => {
134			vec![Classified::Query(Query::CanSecond(msg))]
135		},
136		// Other CandidateBackingMessage variants (`GetBackableCandidates`, `Statement`) are not
137		// emitted by the collator-protocol.
138		other => panic!(
139			"collator-protocol emitted unexpected CandidateBackingMessage variant: {:?}",
140			other
141		),
142	}
143}

145fn from_network_bridge_tx(
146	msg: NetworkBridgeTxMessage,
147	pending: &mut PendingFetches,
148) -> Vec<Classified> {
149	match msg {
150		NetworkBridgeTxMessage::ReportPeer(report) => from_report_peer(report),
151		NetworkBridgeTxMessage::DisconnectPeers(peers, peer_set) => {
152			vec![Classified::Effect(Effect::DisconnectPeers {
153				peers: peers.into_iter().collect::<BTreeSet<_>>(),
154				peer_set,
155			})]
156		},
157		NetworkBridgeTxMessage::ConnectToValidators { validator_ids, peer_set, failed: _ } => {
158			vec![Classified::Effect(Effect::ConnectValidators {
159				validator_ids: validator_ids.into_iter().collect::<BTreeSet<_>>(),
160				peer_set,
161			})]
162		},
163		NetworkBridgeTxMessage::SendCollationMessage(peers, proto) => {
164			let kind = wire_kind_from_collation_protocol(&proto);
165			vec![Classified::Effect(Effect::SendCollation { peers, kind })]
166		},
167		NetworkBridgeTxMessage::SendCollationMessages(batches) => batches
168			.into_iter()
169			.map(|(peers, proto)| {
170				let kind = wire_kind_from_collation_protocol(&proto);
171				Classified::Effect(Effect::SendCollation { peers, kind })
172			})
173			.collect(),
174		NetworkBridgeTxMessage::SendRequests(requests, _) => {
175			requests.into_iter().map(|r| classify_request(r, pending)).collect()
176		},
177
178		// The collator-protocol never sends validation-protocol messages or connect/extend
179		// resolved-validators commands. If that changes, we want to know at compile time, not
180		// at panic time.
181		NetworkBridgeTxMessage::SendValidationMessage(..) |
182		NetworkBridgeTxMessage::SendValidationMessages(..) |
183		NetworkBridgeTxMessage::ConnectToResolvedValidators { .. } |
184		NetworkBridgeTxMessage::AddToResolvedValidators { .. } => {
185			panic!("collator-protocol emitted unexpected NetworkBridgeTxMessage variant: {:?}", msg)
186		},
187	}
188}

190fn from_report_peer(msg: ReportPeerMessage) -> Vec<Classified> {
191	match msg {
192		ReportPeerMessage::Single(peer, change) => vec![Classified::Effect(Effect::Reputation {
193			peer,
194			bucket: RepBucket::from_raw(&change),
195		})],
196		ReportPeerMessage::Batch(map) => map
197			.into_iter()
198			// A net-zero accumulated magnitude is a no-op (offsetting cost + benefit) — drop
199			// it rather than emit a spurious `Reputation` effect. Non-zero magnitudes bucket
200			// via the same logic as the single-report path.
201			.filter_map(|(peer, magnitude)| {
202				RepBucket::from_magnitude(magnitude)
203					.map(|bucket| Classified::Effect(Effect::Reputation { peer, bucket }))
204			})
205			.collect(),
206	}
207}

209fn classify_request(req: Requests, pending: &mut PendingFetches) -> Classified {
210	match req {
211		Requests::CollationFetchingV1(out) => {
212			let to = recipient_to_peer_id(&out.peer);
213			let request_id = pending.register(out.pending_response);
214			Classified::Effect(Effect::SendRequest {
215				request_id,
216				to,
217				kind: ReqKind::CollationFetchingV1,
218				candidate_hash: None,
219			})
220		},
221		Requests::CollationFetchingV2(out) => {
222			let to = recipient_to_peer_id(&out.peer);
223			let candidate_hash = out.payload.candidate_hash;
224			let request_id = pending.register(out.pending_response);
225			Classified::Effect(Effect::SendRequest {
226				request_id,
227				to,
228				kind: ReqKind::CollationFetchingV2,
229				candidate_hash: Some(candidate_hash),
230			})
231		},
232		Requests::CollationFetchingV3(out) => {
233			let to = recipient_to_peer_id(&out.peer);
234			let output_head_data_hash = out.payload.output_head_data_hash;
235			let request_id = pending.register(out.pending_response);
236			Classified::Effect(Effect::SendRequestV3 {
237				request_id,
238				to,
239				kind: ReqKind::CollationFetchingV3,
240				output_head_data_hash,
241			})
242		},
243
244		// Other request kinds are emitted by other subsystems, never by collator-protocol.
245		// Exhaustive arms here mean a new variant on the upstream `Requests` enum breaks the
246		// build until we've consciously routed it.
247		Requests::ChunkFetching(_) |
248		Requests::PoVFetchingV1(_) |
249		Requests::AvailableDataFetchingV1(_) |
250		Requests::DisputeSendingV1(_) |
251		Requests::AttestedCandidateV2(_) => {
252			panic!("collator-protocol emitted unexpected request kind: {:?}", req)
253		},
254	}
255}

257fn wire_kind_from_collation_protocol(
258	proto: &polkadot_node_network_protocol::VersionedCollationProtocol,
259) -> WireMsgKind {
260	use polkadot_node_network_protocol::{
261		v1::CollationProtocol as V1, v2::CollationProtocol as V2,
262		v3_collation::CollationProtocol as V3,
263	};
264	match proto {
265		CollationProtocols::V1(V1::CollatorProtocol(msg)) => match msg {
266			protocol_v1::CollatorProtocolMessage::Declare(_, para, _) => {
267				WireMsgKind::Declare { para: *para }
268			},
269			protocol_v1::CollatorProtocolMessage::AdvertiseCollation(rp) => {
270				WireMsgKind::Advertise {
271					summary: AdvertisementSummary {
272						scheduling_parent: *rp,
273						candidate_hash: None,
274						parent_head_hash: None,
275						output_head_hash: None,
276					},
277				}
278			},
279			protocol_v1::CollatorProtocolMessage::CollationSeconded(rp, _) => {
280				WireMsgKind::CollationSeconded { relay_parent: *rp }
281			},
282		},
283		CollationProtocols::V2(V2::CollatorProtocol(msg)) => match msg {
284			protocol_v2::CollatorProtocolMessage::Declare(_, para, _) => {
285				WireMsgKind::Declare { para: *para }
286			},
287			protocol_v2::CollatorProtocolMessage::AdvertiseCollation {
288				scheduling_parent,
289				candidate_hash,
290				parent_head_data_hash,
291			} => WireMsgKind::Advertise {
292				summary: AdvertisementSummary {
293					scheduling_parent: *scheduling_parent,
294					candidate_hash: Some(*candidate_hash),
295					parent_head_hash: Some(*parent_head_data_hash),
296					output_head_hash: None,
297				},
298			},
299			protocol_v2::CollatorProtocolMessage::CollationSeconded(rp, _) => {
300				WireMsgKind::CollationSeconded { relay_parent: *rp }
301			},
302		},
303		CollationProtocols::V3(V3::CollatorProtocol(msg)) => match msg {
304			protocol_v3::CollatorProtocolMessage::Declare(_, para, _) => {
305				WireMsgKind::Declare { para: *para }
306			},
307			protocol_v3::CollatorProtocolMessage::AdvertiseCollation {
308				scheduling_parent,
309				candidate_hash,
310				parent_head_data_hash,
311				..
312			} => WireMsgKind::Advertise {
313				summary: AdvertisementSummary {
314					scheduling_parent: *scheduling_parent,
315					candidate_hash: Some(*candidate_hash),
316					parent_head_hash: Some(*parent_head_data_hash),
317					output_head_hash: None,
318				},
319			},
320			protocol_v3::CollatorProtocolMessage::CollationSeconded(rp, _) => {
321				WireMsgKind::CollationSeconded { relay_parent: *rp }
322			},
323		},
324		CollationProtocols::V4(protocol_v4::AdvertiseSegment {
325			scheduling_parent,
326			candidates,
327			..
328		}) => {
329			let tip =
330				candidates.last().expect("subsystem never emits an empty segment advertisement");
331			WireMsgKind::Advertise {
332				summary: AdvertisementSummary {
333					scheduling_parent: *scheduling_parent,
334					candidate_hash: None,
335					parent_head_hash: Some(tip.parent_head_data_hash),
336					output_head_hash: Some(tip.output_head_data_hash),
337				},
338			}
339		},
340	}
341}

343fn recipient_to_peer_id(
344	recipient: &polkadot_node_network_protocol::request_response::outgoing::Recipient,
345) -> sc_network_types::PeerId {
346	use polkadot_node_network_protocol::request_response::outgoing::Recipient;
347	match recipient {
348		Recipient::Peer(p) => *p,
349		Recipient::Authority(_) => {
350			panic!("collator-protocol fetches always target a PeerId, not an AuthorityDiscoveryId")
351		},
352	}
353}

362	fn one(c: Vec<Classified>) -> Classified {
363		assert_eq!(c.len(), 1, "expected exactly one classified entry");
364		c.into_iter().next().unwrap()
365	}

367	#[test]
368	fn report_peer_single_classifies_as_reputation_effect() {
369		let peer = sc_network_types::PeerId::random();
370		let change = polkadot_node_network_protocol::ReputationChange::new(i32::MIN, "bad");
371		let msg = AllMessages::NetworkBridgeTx(NetworkBridgeTxMessage::ReportPeer(
372			ReportPeerMessage::Single(peer, change),
373		));
374		match one(classify(msg, &mut PendingFetches::new())) {
375			Classified::Effect(Effect::Reputation { peer: p, bucket }) => {
376				assert_eq!(p, peer);
377				assert_eq!(bucket, RepBucket::Malicious);
378			},
379			other => panic!("unexpected classification: {:?}", other),
380		}
381	}

383	#[test]
384	fn report_peer_batch_emits_one_effect_per_peer() {
385		let p1 = sc_network_types::PeerId::random();
386		let p2 = sc_network_types::PeerId::random();
387		let p3 = sc_network_types::PeerId::random();
388		let mut map = std::collections::HashMap::new();
389		map.insert(p1, -100i32);
390		map.insert(p2, 100i32);
391		map.insert(p3, i32::MIN);
392		let msg = AllMessages::NetworkBridgeTx(NetworkBridgeTxMessage::ReportPeer(
393			ReportPeerMessage::Batch(map),
394		));
395		let out = classify(msg, &mut PendingFetches::new());
396		assert_eq!(out.len(), 3);
397		// Bucket per peer is preserved.
398		let buckets: Vec<RepBucket> = out
399			.into_iter()
400			.map(|c| match c {
401				Classified::Effect(Effect::Reputation { bucket, .. }) => bucket,
402				other => panic!("expected Reputation effect, got {:?}", other),
403			})
404			.collect();
405		assert!(buckets.contains(&RepBucket::Performance));
406		assert!(buckets.contains(&RepBucket::Benefit));
407		assert!(buckets.contains(&RepBucket::Malicious));
408	}
