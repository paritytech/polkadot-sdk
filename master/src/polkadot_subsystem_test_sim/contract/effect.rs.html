17//! `Effect` enum: the protocol-relevant outputs of the collator-protocol subsystem.
18//!
19//! Tests assert on [`Effect`] values exclusively. Each variant carries semantic content (peer
20//! ids, candidate hashes, paraids, message kinds), never raw wire bytes, so refactors to wire
21//! formats do not break tests.
22
23use crate::contract::reputation::RepBucket;
24use polkadot_node_network_protocol::peer_set::PeerSet;
25use polkadot_primitives::{AuthorityDiscoveryId, CandidateHash, Hash, Id as ParaId};
26use sc_network_types::PeerId;
27use std::collections::BTreeSet;
28
29/// Opaque identifier assigned by the harness when the subsystem fires an outgoing request.
30/// Tests pass it to `Sim::respond_fetch` to deliver a response into the oneshot the subsystem
31/// is awaiting.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
33pub struct RequestId(pub u64);
34
35/// A single observable output of the collator-protocol subsystem.
36#[derive(Debug, Clone, PartialEq, Eq)]
37pub enum Effect {
38	/// Subsystem instructed `CandidateBacking` to second a fetched candidate.
39	SecondCandidate {
40		/// Scheduling parent (validator group context).
41		scheduling_parent: Hash,
42		/// Candidate identity.
43		candidate_hash: CandidateHash,
44		/// Para the candidate belongs to.
45		para: ParaId,
46	},
47	/// Subsystem advertised a candidate to a set of peers.
48	SendAdvertisement {
49		/// Recipients.
50		peers: Vec<PeerId>,
51		/// Semantic summary of the advertisement.
52		advertisement: AdvertisementSummary,
53	},
54	/// Subsystem sent a wire-level collation message (Declare, CollationSeconded, ...).
55	SendCollation {
56		/// Recipients.
57		peers: Vec<PeerId>,
58		/// What kind of collation message was sent.
59		kind: WireMsgKind,
60	},
61	/// Subsystem fired an outgoing request.
62	SendRequest {
63		/// Opaque identifier the harness assigned to this request. Tests use this with
64		/// `Sim::respond_fetch(request_id, payload)` to deliver a response into the oneshot
65		/// the subsystem is awaiting.
66		request_id: RequestId,
67		/// Target peer.
68		to: PeerId,
69		/// What kind of request.
70		kind: ReqKind,
71		/// Candidate the request is about, if applicable.
72		candidate_hash: Option<CandidateHash>,
73	},
74	/// Subsystem fired an outgoing request.
75	SendRequestV3 {
76		/// Opaque identifier the harness assigned to this request. Tests use this with
77		/// `Sim::respond_fetch(request_id, payload)` to deliver a response into the oneshot
78		/// the subsystem is awaiting.
79		request_id: RequestId,
80		/// Target peer.
81		to: PeerId,
82		/// What kind of request.
83		kind: ReqKind,
84		/// Output head data hash of the candidate.
85		output_head_data_hash: Hash,
86	},
87	/// Subsystem requested a peer reputation change.
88	Reputation {
89		/// Affected peer.
90		peer: PeerId,
91		/// Bucketed severity (Malicious / Performance / Benefit).
92		bucket: RepBucket,
93	},
94	/// Subsystem instructed the network bridge to maintain connections to a validator set.
95	ConnectValidators {
96		/// Validator authority-discovery ids to connect to.
97		validator_ids: BTreeSet<AuthorityDiscoveryId>,
98		/// Which peer-set the request applies to.
99		peer_set: PeerSet,
100	},
101	/// Subsystem instructed the network bridge to disconnect peers.
102	DisconnectPeers {
103		/// Peers to disconnect.
104		peers: BTreeSet<PeerId>,
105		/// Which peer-set the disconnect applies to.
106		peer_set: PeerSet,
107	},
108	/// Subsystem replied to a previously-received request (e.g. responding to a
109	/// `CollationFetchingRequest`).
110	RequestResponseSent {
111		/// Identifier of the request being responded to.
112		request_id: u64,
113		/// Kind of response sent.
114		kind: RespKind,
115	},
116}
117
118/// Semantic summary of an advertisement carried over the wire. Drops protocol-version
119/// discriminants and raw byte encoding — tests assert against these fields, not wire format.
120#[derive(Debug, Clone, PartialEq, Eq)]
121pub struct AdvertisementSummary {
122	/// Scheduling parent (relay parent for V1, possibly different from the claim's relay parent
123	/// for V2/V3/V4).
124	pub scheduling_parent: Hash,
125	/// Candidate hash, when the advertisement format includes one (V2/V3). `None` for V1 and
126	/// for V4, whose fingerprints deliberately carry no candidate hash.
127	pub candidate_hash: Option<CandidateHash>,
128	/// Hash of the parent head data (V2/V3, and the V4 tip fingerprint).
129	pub parent_head_hash: Option<Hash>,
130	/// Output head hash of the claimed parachain block — the claim identity of a V4
131	/// fingerprint (tip summarized). `None` for V1/V2/V3.
132	pub output_head_hash: Option<Hash>,
133}

135/// Coarse-grained kind of an outbound collation wire message.
136#[derive(Debug, Clone, PartialEq, Eq)]
137pub enum WireMsgKind {
138	/// `Declare(collator_id, para_id, signature)`.
139	Declare {
140		/// Para the collator is declaring on.
141		para: ParaId,
142	},
143	/// `AdvertiseCollation(...)` — semantic summary attached.
144	Advertise {
145		/// Semantic content of the advertisement.
146		summary: AdvertisementSummary,
147	},
148	/// `CollationSeconded(relay_parent, statement)`.
149	CollationSeconded {
150		/// Relay parent the seconded statement is anchored at.
151		relay_parent: Hash,
152	},
153}
154
155/// Coarse-grained kind of an outbound request.
156#[derive(Debug, Clone, PartialEq, Eq)]
157pub enum ReqKind {
158	/// `CollationFetchingRequest` v1 (no candidate hash).
159	CollationFetchingV1,
160	/// `CollationFetchingRequest` v2 (with candidate hash).
161	CollationFetchingV2,
162	/// `CollationFetchingRequest` v3 (no candidate hash, with output head data hash)
163	CollationFetchingV3,
164}
165
166/// Coarse-grained kind of an outbound response.
167#[derive(Debug, Clone, PartialEq, Eq)]
168pub enum RespKind {
169	/// Successful collation fetch response.
170	CollationFetchingOk,
171	/// Error / refusal response.
172	Error,
173}

175/// Helper: convenience for assertions on the seconding effect.
176impl Effect {
177	/// Returns `Some(candidate_hash)` if this is a `SecondCandidate` for the given hash, else
178	/// `None`. Matches the per-Effect convenience matchers described in the design doc.
179	pub fn is_second_of(&self, expected: &CandidateHash) -> bool {
180		matches!(self, Effect::SecondCandidate { candidate_hash, .. } if candidate_hash == expected)
181	}
182
183	/// Returns `Some(candidate_hash)` if this is a `SendRequest` for a `CollationFetching*` of
184	/// the given candidate. Useful when the test does not care which protocol-version is used.
185	pub fn is_fetch_for(&self, expected: &CandidateHash) -> bool {
186		matches!(
187			self,
188			Effect::SendRequest {
189				kind: ReqKind::CollationFetchingV1 | ReqKind::CollationFetchingV2,
190				candidate_hash: Some(c),
191				..
192			} if c == expected
193		)
194	}
195
196	/// Returns `Some(request_id)` if this is a `SendRequest` or `SendRequestV3`. Useful for
197	/// chaining `Sim::expect(...)` with `Sim::respond_fetch(request_id, payload)`.
198	pub fn request_id(&self) -> Option<RequestId> {
199		match self {
200			Effect::SendRequest { request_id, .. } | Effect::SendRequestV3 { request_id, .. } => {
201				Some(*request_id)
202			},
203			_ => None,
204		}
205	}
206}
