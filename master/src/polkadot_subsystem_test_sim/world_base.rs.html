17//! Subsystem-agnostic test world base + the [`HasBase`] trait that gives every
18//! per-tenant `World` the shared scaffolding (`Sim`, `SharedChain`, leaf bookkeeping)
19//! plus a fluent leaf builder.
20//!
21//! Per-tenant consumer crates compose `WorldBase` as a field of their own `World`,
22//! impl [`HasBase`] (two accessor functions), and gain every base method directly:
23//!
24//! ```ignore
25//! pub struct World {
26//!     pub base: WorldBase<MySut>,
27//!     pub tenant_specific_field: ...,
28//! }
29//!
30//! impl HasBase for World {
31//!     type Sut = MySut;
32//!     fn base(&self) -> &WorldBase<Self::Sut> { &self.base }
33//!     fn base_mut(&mut self) -> &mut WorldBase<Self::Sut> { &mut self.base }
34//! }
35//!
36//! // Tenant scenarios call `world.new_block().with_head_data(...).activate()` —
37//! // single fluent builder. Forks via `.from_parent(prev_leaf.hash)`.
38//! ```
39

63	pub hash: Hash,
64	/// Leaf number.

68/// Suite-wide world configuration consumed once at [`WorldBase::start`]. Mid-test
69/// changes flow directly through [`crate::chain::ChainModel`] (`add_session`,
70/// `set_claim_queue_at`, `set_runtime_api_version`, etc.) — single source of truth for
71/// chain/runtime state.
72#[derive(Clone, Debug)]
73pub struct WorldConfig {
74	/// Session index applied to every block produced via `chain.extend(...)`. Mid-test
75	/// session changes go through `chain.add_session(...)` + per-block session overrides.
76	pub session_index: SessionIndex,
77	/// Per-core schedule — write-side scheduling primitive. Each entry is a
78	/// [`CoreSchedule`] that defines how the chain model derives the runtime's
79	/// `ClaimQueue` at every block (the queue rotates per block according to the
80	/// cycle pattern). Per-block claim-queue overrides take precedence; set them
81	/// via [`crate::chain::ChainModel::set_claim_queue_at`] on the chain.
82	pub schedule: Vec<(CoreIndex, CoreSchedule)>,
83	/// Validation-code hash baked into the synthesised backing constraints — must
84	/// match what `make_candidate(.., validation_code_hash)` produces or candidates
85	/// fail constraints' `ValidationCodeMismatch` check.
86	pub validation_code_hash: ValidationCodeHash,
87	/// Optional override for `min_relay_parent_number` in the synthesised backing
88	/// constraints. Defaults to `leaf.number - (scheduling_lookahead - 1)`.
89	pub min_relay_parent_number_override: Option<BlockNumber>,
90	/// Runtime API version reported by the chain model. Lets tests exercise the
91	/// `NotSupported` fallback path for APIs the configured runtime does not yet
92	/// implement (e.g. `AncestorRelayParentInfo`). Defaults to the highest version
93	/// the chain model implements end-to-end.
94	pub runtime_api_version: u32,
95	/// Validators reported in the runtime's session info. Empty by default —
96	/// scenarios that exercise validator-side behaviour (groups, signatures,
97	/// rotation) populate this.
98	pub validators: Vec<ValidatorId>,
99	/// Validator groups reported in the runtime's session info. Empty by default.
100	/// Multiple groups + matching `group_rotation_frequency` enables per-block
101	/// group rotation across cores.
102	pub validator_groups: Vec<Vec<ValidatorIndex>>,
103	/// Group rotation frequency reported in the runtime's session info. Defaults
104	/// to 1 (rotates every block). Set to a large number to keep group 0 stable
105	/// across an ancestry chain.
106	pub group_rotation_frequency: u32,
107	/// Slot of the genesis block. Each subsequent `chain.extend(...)` bumps the
108	/// slot by one. Defaults to 0. Increase this for V3 tests that need
109	/// leaf-parent / leaf at specific slots.
110	pub genesis_slot: Slot,
111	/// Set the `CandidateReceiptV2` node feature flag (`FeatureIndex::CandidateReceiptV2 = 3`).
112	/// V3 advertisements / descriptors are gated on this. Defaults to `false`.
113	pub enable_v3_node_feature: bool,
114}

196/// Subsystem-agnostic shared test-world state: the running `Sim`, the chain model,
197/// the activated leaves, and the suite-wide [`WorldConfig`].
198///
199/// Per-tenant `World` types compose this as a field and impl [`HasBase`] to gain the
200/// shared methods (`new_block`, `deactivate_leaf`, `finalize`, etc.) directly on `world.foo()`.
201pub struct WorldBase<S: SubsystemUnderTest>
202where
203	AllMessages: From<<S::Message as AssociateOutgoing>::OutgoingMessages>,
204	AllMessages: From<S::Message>,
205{
206	/// The driving simulation.
207	pub sim: Sim<S>,
208	/// The chain model — answers Runtime/ChainApi queries. Mutated via
209	/// [`BlockBuilder`] (per-leaf head data + pending availability) and direct
210	/// `chain.lock()` (mid-test session changes, claim-queue overrides, etc.).
211	pub chain: SharedChain,
212	/// All leaves the harness has signalled `ActiveLeaves::start_work` for, in
213	/// activation order.
214	pub leaves: Vec<LeafRef>,
215	/// Suite-wide config. Read-only after `start`; mid-test state changes go through
216	/// `chain` directly.
217	pub config: WorldConfig,
218}
219
220impl<S: SubsystemUnderTest> WorldBase<S>
221where
222	AllMessages: From<<S::Message as AssociateOutgoing>::OutgoingMessages>,
223	AllMessages: From<S::Message>,
224{
225	/// Start a new world from a [`WorldConfig`]. Seeds the chain model with the
226	/// configured genesis slot, session info (validators, groups, rotation), per-core
227	/// claim queue, runtime API version, and node features; installs the chain model +
228	/// a fallback `PanicResponder` as the simulation's responder graph; and spins up
229	/// the `Sim`. No leaves are active until [`HasBase::new_block`] is called.
230	pub fn start(config: WorldConfig) -> Self {
231		let chain = build_chain_model(&config);
232		let mut responder = LayeredResponder::new();
233		responder.push(chain.clone());
234		responder.push(PanicResponder);
235		Self::start_with_responder(responder, chain, config)
236	}
237
238	/// Start a new world with a caller-supplied responder chain. Use when the chain
239	/// model isn't enough — typically because a tenant pushes a tenant-specific
240	/// `AnswerQuery` layer in front of the chain (e.g. a `CanSecond` script).
241	pub fn start_with_responder(
242		responder: LayeredResponder,
243		chain: SharedChain,
244		config: WorldConfig,
245	) -> Self {
246		let sim = Sim::<S>::start(SimConfig::default(), responder);
247		Self { sim, chain, leaves: Vec::new(), config }
248	}
249}

251/// Build a `SharedChain` from a [`WorldConfig`]. Tenants whose responder graph needs
252/// extra layers in front of the chain (e.g. a `CanSecond` query stub) build their
253/// own [`LayeredResponder`] and pass `chain.clone()` into [`WorldBase::start_with_responder`].
254/// Most tenants use [`WorldBase::start`] and don't touch this directly.
255pub fn build_chain_model(config: &WorldConfig) -> SharedChain {
256	let mut chain = ChainModel::new(config.genesis_slot);
257	let session_info = SessionInfo {
258		validators: config.validators.clone(),
259		validator_groups: config.validator_groups.clone(),
260		group_rotation_info: polkadot_primitives::GroupRotationInfo {
261			session_start_block: 0,
262			group_rotation_frequency: config.group_rotation_frequency,
263			now: 0,
264		},
265	};
266	// Register session info at both `0` (default for genesis-disconnected synthetic
267	// ancestors) and `config.session_index` (for blocks the test activates).
268	chain.add_session(0, session_info.clone());
269	if config.session_index != 0 {
270		chain.add_session(config.session_index, session_info);
271	}
272	// Align the genesis block's session with the configured world session so that
273	// `chain.extend(...)` (which inherits the parent's session) produces blocks in
274	// `config.session_index`. Without this, auto-allocated leaves report session 0,
275	// out of sync with `world.session_index()`.
276	let genesis = chain.genesis();
277	chain.set_block_session(genesis, config.session_index);
278	for (core, schedule) in &config.schedule {
279		chain.set_core_schedule(*core, schedule.clone());
280	}
281	chain.set_runtime_api_version(config.runtime_api_version);
282	if config.enable_v3_node_feature {
283		let mut features = polkadot_primitives::NodeFeatures::EMPTY;
284		// FeatureIndex::CandidateReceiptV2 = 3
285		features.resize(4, false);
286		features.set(3, true);
287		chain.set_node_features(features);
288	}
289	SharedChain::new(chain)
290}

292/// Synthesise a permissive backing-constraints record. `valid_watermarks =
293/// vec![leaf.number]` matches `hrmp_watermark = relay_parent_number` in default
294/// `make_candidate(...)` output; `required_parent = head_data` ties acceptance to the
295/// leaf-flavoured head data the test declares.
296pub fn synthesise_constraints(
297	min_relay_parent_number: BlockNumber,
298	valid_watermarks: Vec<BlockNumber>,
299	required_parent: HeadData,
300	validation_code_hash: ValidationCodeHash,
301) -> Constraints {
302	const MAX_POV_SIZE: u32 = 1_000_000;
303	Constraints {
304		min_relay_parent_number,
305		max_pov_size: MAX_POV_SIZE,
306		max_head_data_size: 20480,
307		max_code_size: 1_000_000,
308		ump_remaining: 10,
309		ump_remaining_bytes: 1_000,
310		max_ump_num_per_candidate: 10,
311		dmp_remaining_messages: vec![],
312		hrmp_inbound: InboundHrmpLimitations { valid_watermarks },
313		hrmp_channels_out: vec![],
314		max_hrmp_num_per_candidate: 0,
315		required_parent,
316		validation_code_hash,
317		upgrade_restriction: None,
318		future_validation_code: None,
319	}

322/// Trait every per-tenant `World` impls. Two accessor methods (`base` + `base_mut`)
323/// plus default-impl convenience methods: leaf builder, leaf deactivation, raw
324/// active-leaves signal, suite-wide config accessors.
325pub trait HasBase
326where
327	AllMessages:
328		From<<<Self::Sut as SubsystemUnderTest>::Message as AssociateOutgoing>::OutgoingMessages>,
329	AllMessages: From<<Self::Sut as SubsystemUnderTest>::Message>,
330{
331	/// The subsystem-under-test type the `Sim<S>` inside `WorldBase` is parameterised by.
332	type Sut: SubsystemUnderTest;
333
334	/// Shared base, immutable.
335	fn base(&self) -> &WorldBase<Self::Sut>;
336
337	/// Shared base, mutable.
338	fn base_mut(&mut self) -> &mut WorldBase<Self::Sut>;
339
340	/// Begin building a new block. The block is extended onto the chain's current
341	/// tip via [`ChainModel::extend`]; use [`BlockBuilder::from_parent`] to extend
342	/// from a non-tip parent (forks, or non-linear extensions). Per-(rp, para) head
343	/// data + pending availability + claim queue accumulate on the builder and are
344	/// written to the chain model when the builder finalises.
345	///
346	/// Two terminal verbs:
347	///
348	/// * [`BlockBuilder::register`] — writes the block to the chain model only. The subsystem is
349	///   **not** told. Use this for ancestor blocks that should exist in the chain history but were
350	///   never independently signalled as active leaves (skipped activations, fast-sync gaps,
351	///   blocks that were never tip).
352	///
353	/// * [`BlockBuilder::activate`] — `register` + signal
354	///   `OverseerSignal::ActiveLeaves(start_work(new), deactivated=[parent if it was an active
355	///   leaf])`. Mirrors the production `block_imported` signal: when a child of an active leaf is
356	///   imported, the parent stops being a leaf in the same `ActiveLeavesUpdate`.
357	fn new_block(&mut self) -> BlockBuilder<'_, Self::Sut> {
358		BlockBuilder::new(self.base_mut())
359	}
360
361	/// Deactivate a leaf via `ActiveLeavesUpdate::stop_work`. Mostly useful for
362	/// tests that drive deactivation without an accompanying activation; normal
363	/// linear chain progression deactivates the parent automatically via
364	/// [`BlockBuilder::activate`].
365	fn deactivate_leaf(&mut self, hash: Hash) {
366		let base = self.base_mut();
367		base.sim
368			.signal(OverseerSignal::ActiveLeaves(ActiveLeavesUpdate::stop_work(hash)));
369		base.leaves.retain(|l| l.hash != hash);
370	}
371
372	/// Signal `OverseerSignal::BlockFinalized(hash, number)` followed by an
373	/// `ActiveLeavesUpdate` that prunes orphaned leaves. Mirrors the production
374	/// `block_finalized` path:
375	///
376	/// * any active leaf at `number <= finalized.number` other than the finalized block itself is
377	///   removed (those leaves are on branches that can no longer extend the finalized chain);
378	/// * the finalized block itself stays an active leaf if it currently is one;
379	/// * higher-numbered orphan leaves stay until they fall below a future finalized number —
380	///   production accepts that and so do we.
381	///
382	/// The empty-update case (no leaves to prune) skips the second signal,
383	/// matching production behaviour.
384	fn finalize(&mut self, hash: Hash) {
385		let number = {
386			let chain = self.base().chain.lock();
387			chain.block(&hash).expect("finalized hash must be a registered block").number
388		};
389		let base = self.base_mut();
390		base.sim.signal(OverseerSignal::BlockFinalized(hash, number));
391
392		let mut deactivated = Vec::new();
393		base.leaves.retain(|l| {
394			if l.number <= number && l.hash != hash {
395				deactivated.push(l.hash);
396				false
397			} else {
398				true
399			}
400		});
401		if !deactivated.is_empty() {
402			let update = ActiveLeavesUpdate { activated: None, deactivated: deactivated.into() };
403			base.sim.signal(OverseerSignal::ActiveLeaves(update));
404		}
405	}
406
407	/// Send a raw `OverseerSignal::ActiveLeaves` update. Escape hatch for tests
408	/// that need to construct an update shape outside the [`BlockBuilder`] /
409	/// [`HasBase::deactivate_leaf`] / [`HasBase::finalize`] vocabulary (rare).
410	fn signal_active_leaves(&mut self, update: ActiveLeavesUpdate) {
411		let base = self.base_mut();
412		if let Some(activated) = &update.activated {
413			base.leaves.push(LeafRef { hash: activated.hash, number: activated.number });
414		}
415		for hash in update.deactivated.iter() {
416			base.leaves.retain(|l| l.hash != *hash);
417		}
418		base.sim.signal(OverseerSignal::ActiveLeaves(update));
419	}
420
421	// =====================================================================================
422	// Active-leaf accessors. Subsystem-agnostic: read `base().leaves` + `base().chain`.
423	// =====================================================================================
424
425	/// Hash of the first (and, for most scenarios, only) active leaf. Panics if no
426	/// leaf has been activated.
427	fn leaf(&self) -> Hash {
428		self.base().leaves[0].hash
429	}
430
431	/// Block number of the first active leaf. Panics if no leaf has been activated.
432	fn leaf_number(&self) -> BlockNumber {
433		self.base().leaves[0].number
434	}
435
436	/// Walk back from the first active leaf, returning up to
437	/// `chain.allowed_ancestry_len()` ancestors. `ancestors()[0]` is the leaf's
438	/// parent. Mirrors what real `prospective-parachains` returns for
439	/// `known_allowed_relay_parents_under(leaf)` (excluding the leaf itself).
440	fn ancestors(&self) -> Vec<Hash> {
441		self.ancestors_of(0)
442	}
443
444	/// Walk back from `leaves[idx]`, returning up to
445	/// `chain.allowed_ancestry_len()` ancestors.
446	fn ancestors_of(&self, idx: usize) -> Vec<Hash> {
447		let leaf_hash = self.base().leaves[idx].hash;
448		let chain = self.base().chain.lock();
449		// `allowed_ancestry_len + 1` is the depth budget — `chain.ancestors(_, k)`
450		// returns up to k blocks excluding the queried hash; the implicit view
451		// typically resolves `allowed_ancestry_len` of those.
452		let k = chain.allowed_ancestry_len() as usize + 1;
453		chain.ancestors(leaf_hash, k)
454	}
455
456	// =====================================================================================
457	// Read accessors for suite-wide config — hide field layout of `WorldConfig` so tests
458	// don't break when fields are added or moved.
459	// =====================================================================================
460
461	/// Validation-code hash baked into the synthesised backing constraints.
462	fn validation_code_hash(&self) -> ValidationCodeHash {
463		self.base().config.validation_code_hash
464	}
465
466	/// Session index applied to every block registered by [`BlockBuilder`].
467	fn session_index(&self) -> SessionIndex {
468		self.base().config.session_index
469	}
470
471	/// Optional override for `min_relay_parent_number` in the synthesised backing
472	/// constraints.
473	fn min_relay_parent_number_override(&self) -> Option<BlockNumber> {
474		self.base().config.min_relay_parent_number_override
475	}
476}

478/// Fluent builder for a new block. Accumulates per-para head data + pending
479/// availability, then on `.register()` / `.activate()`:
480///
481/// 1. Allocates the block's hash + number via [`ChainModel::extend`] (or honours an
482///    explicitly-pinned hash via [`Self::with_hash_and_number`]).
483/// 2. Writes per-para head data + pending availability to the chain model.
484/// 3. Synthesises permissive backing constraints (using `world.config`'s `validation_code_hash` +
485///    optional `min_relay_parent_number_override`) and writes them to the chain model.
486/// 4. (`activate` only) Signals `OverseerSignal::ActiveLeaves` with `start_work(new)` plus
487///    `deactivated=[parent]` if the parent was an active leaf — mirroring the production
488///    `block_imported` signal.
489///
490/// Two real-world flows the API mirrors:
491///
492/// * **Tip-following** = repeated `.activate()` calls. Each new block becomes the active leaf; its
493///   parent (the previous tip) is deactivated in the same signal. Models normal block production
494///   where the overseer always emits one `ActiveLeavesUpdate` per imported block.
495///
496/// * **Skipping** = `.register()` for intermediate blocks, `.activate()` only for the next tip the
497///   overseer announces. Models scenarios where the overseer doesn't emit a per-block signal — e.g.
498///   fast-syncing, restart catch-up, the parachains-API gap. Tests use this to verify the subsystem
499///   handles gaps in the activation sequence correctly via implicit view.
500pub struct BlockBuilder<'w, S: SubsystemUnderTest>
501where
502	AllMessages: From<<S::Message as AssociateOutgoing>::OutgoingMessages>,
503	AllMessages: From<S::Message>,
504{
505	base: &'w mut WorldBase<S>,
506	/// Optional explicit (hash, number). When `None`, the builder extends the chain
507	/// at activation time via `chain.extend(parent)`.
508	hash_and_number: Option<(Hash, BlockNumber)>,
509	/// Optional explicit parent for the auto-extend path. `None` means "extend the
510	/// chain's current tip" — the common linear case.
511	parent: Option<Hash>,
512	/// `(para, head_data)` pairs to write via
513	/// `chain.set_backing_constraints_at(block, para, ..)` (head data becomes
514	/// `Constraints::required_parent`).
515	head_data: Vec<(ParaId, HeadData)>,
516	/// `(para, pending)` pairs to write via
517	/// `chain.set_pending_availability_at(block, para, ..)`.
518	pending: Vec<(ParaId, Vec<CandidatePendingAvailability>)>,
519	/// Optional per-block claim-queue override. When set, written via
520	/// `chain.set_claim_queue_at(block, queue)` and takes precedence over the
521	/// suite-wide schedule for this block only — subsequent blocks revert to the
522	/// schedule unless they too set an override.
523	claim_queue: Option<BTreeMap<CoreIndex, VecDeque<ParaId>>>,
524}
525
526impl<'w, S: SubsystemUnderTest> BlockBuilder<'w, S>
527where
528	AllMessages: From<<S::Message as AssociateOutgoing>::OutgoingMessages>,
529	AllMessages: From<S::Message>,
530{
531	fn new(base: &'w mut WorldBase<S>) -> Self {
532		Self {
533			base,
534			hash_and_number: None,
535			parent: None,
536			head_data: Vec::new(),
537			pending: Vec::new(),
538			claim_queue: None,
539		}
540	}

542	/// Pin a literal hash + number for this block instead of having the chain
543	/// auto-allocate one. Use only when the test asserts on a specific hash value
544	/// (rare). When pinning, the caller is responsible for any required ancestor
545	/// registration via direct `chain.lock()` mutation.
546	pub fn with_hash_and_number(mut self, hash: Hash, number: BlockNumber) -> Self {
547		self.hash_and_number = Some((hash, number));
548		self
549	}

576	/// Override the claim queue at this block for a single core. Takes precedence
577	/// over the suite-wide schedule for this block only — subsequent blocks revert
578	/// to the schedule unless they too set an override. Use when modelling
579	/// exceptions to the suite-wide cycle (coretime expiry, on-demand cores,
580	/// scheduler edge cases). Call repeatedly for multi-core overrides.
581	pub fn with_claim_queue_at(
582		mut self,
583		core: CoreIndex,
584		paras: impl IntoIterator<Item = ParaId>,
585	) -> Self {
586		let queue = self.claim_queue.get_or_insert_with(BTreeMap::new);
587		queue.insert(core, paras.into_iter().collect());
588		self
589	}

591	/// Finalise the block on the chain model **without** telling the subsystem.
592	/// Use for ancestor blocks the test wants the chain to know about (so queries
593	/// for them resolve) but that were never independently signalled as active
594	/// leaves — e.g. blocks that were skipped during a fast-sync gap. Returns the
595	/// block's identity.
596	pub fn register(self) -> LeafRef {
597		let (_base, block) = self.flush_to_chain();
598		block
599	}
600
601	/// Finalise the block + signal `OverseerSignal::ActiveLeaves` with
602	/// `start_work(new)` plus `deactivated=[parent]` if the parent is currently an
603	/// active leaf in the harness mirror. Mirrors the production `block_imported`
604	/// signal exactly: a child of the current tip pushes the parent off the active
605	/// set in the same update.
606	///
607	/// If the subsystem under test consumes network-bridge events (i.e.
608	/// [`SubsystemUnderTest::our_view_change`] returns `Some`), this also publishes
609	/// the new view to the subsystem — mirroring the production network bridge's
610	/// view-update fan-out that follows block import.
611	///
612	/// Returns the new active leaf's identity.
613	pub fn activate(self) -> LeafRef {
614		let (base, leaf) = self.flush_to_chain();
615		// Auto-deactivate parent if it's currently an active leaf. Production
616		// `block_imported` does the same: importing a child of an active leaf
617		// emits one `ActiveLeavesUpdate` carrying both `activated: Some(child)`
618		// and `deactivated: [parent]`.
619		let parent_hash = base
620			.chain
621			.lock()
622			.block(&leaf.hash)
623			.expect("just-extended block has a parent_hash")
624			.parent_hash;
625		let mut deactivated: Vec<Hash> = Vec::new();
626		base.leaves.retain(|l| {
627			if l.hash == parent_hash {
628				deactivated.push(l.hash);
629				false
630			} else {
631				true
632			}
633		});
634
635		let update = ActiveLeavesUpdate {
636			activated: Some(new_leaf(leaf.hash, leaf.number)),
637			deactivated: deactivated.into(),
638		};
639		base.sim.signal(OverseerSignal::ActiveLeaves(update));
640		base.leaves.push(leaf);
641
642		// Publish the new view to the subsystem under test, if the adapter consumes
643		// network-bridge events. Mirrors the production sequence: overseer emits
644		// `ActiveLeaves`, network bridge separately broadcasts the new view.
645		let view =
646			polkadot_node_network_protocol::OurView::new(base.leaves.iter().map(|l| l.hash), 0);
647		if let Some(msg) = S::our_view_change(view) {
648			base.sim.send(msg);
649		}
650
651		leaf
652	}
653
654	/// Resolve the block's hash + number (extending the chain if needed) and write
655	/// per-para state + synthesised constraints to the chain model. Returns
656	/// `&mut WorldBase` so `.activate()` can re-borrow the sim.
657	fn flush_to_chain(self) -> (&'w mut WorldBase<S>, LeafRef) {
658		let BlockBuilder { base, hash_and_number, parent, head_data, pending, claim_queue } = self;
659		let validation_code_hash = base.config.validation_code_hash;
660		let min_relay_parent_number_override = base.config.min_relay_parent_number_override;
661
662		let leaf = {
663			let mut chain = base.chain.lock();
664			let (hash, number) = if let Some((hash, number)) = hash_and_number {
665				(hash, number)
666			} else {
667				let parent = parent.unwrap_or_else(|| chain.tip());
668				let hash = chain.extend(parent);
669				let number =
670					chain.block(&hash).expect("just-extended block must be registered").number;
671				(hash, number)
672			};
673			let ancestry_len = (DEFAULT_SCHEDULING_LOOKAHEAD - 1) as u32;
674			let min_relay_parent_number = min_relay_parent_number_override
675				.unwrap_or_else(|| number.saturating_sub(ancestry_len));
676			for (para, head_data) in &head_data {
677				let constraints = synthesise_constraints(
678					min_relay_parent_number,
679					vec![number],
680					head_data.clone(),
681					validation_code_hash,
682				);
683				chain.set_backing_constraints_at(hash, *para, constraints);
684			}
685			for (para, candidates) in pending {
686				let receipts: Vec<CommittedCandidateReceipt> = candidates
687					.iter()
688					.map(|p| CommittedCandidateReceipt {
689						descriptor: p.descriptor.clone(),
690						commitments: p.commitments.clone(),
691					})
692					.collect();
693				chain.set_pending_availability_at(hash, para, receipts);
694			}
695			if let Some(queue) = claim_queue {
696				chain.set_claim_queue_at(hash, queue);
697			}
698			LeafRef { hash, number }
699		};
700		(base, leaf)
701	}
