17//! Controllable stub for `candidate-validation`.
18//!
19//! Real `candidate-validation` runs WASM PVF execution — disqualified for deterministic
20//! tests. The stub answers `CandidateValidationMessage::ValidateFromExhaustive` requests
21//! synchronously based on a [`Verdict`] callback that the test owns.
22//!
23//! # Default behaviour
24//!
25//! [`CandidateValidationStub::always_valid`] returns `ValidationResult::Valid` with
26//! canned-but-consistent commitments derived from the request.
27//!
28//! # Customising
29//!
30//! [`CandidateValidationStub::with_verdict`] takes a closure
31//! `Fn(&CandidateReceipt, &PoV) -> Verdict` for tests that need to reject specific
32//! candidates or simulate execution errors.
33
34use crate::harness::router::{RouteAttempt, SubsystemSlot};
35use futures::{channel::mpsc, future::BoxFuture, FutureExt, SinkExt};
36use polkadot_node_primitives::{InvalidCandidate, PoV, ValidationResult};
37use polkadot_node_subsystem::{
38	messages::{AllMessages, CandidateValidationMessage},
39	FromOrchestra, OverseerSignal,
40};
41use polkadot_primitives::{
42	CandidateCommitments, CandidateHash, CandidateReceiptV2 as CandidateReceipt,
43	PersistedValidationData,
44};
45use std::{
46	collections::HashMap,
47	sync::{Arc, Mutex},
48};

50/// Shared registry mapping `CandidateHash` → `(commitments, pvd)`. The
51/// [`CandidateValidationStub::always_valid`] stub consults this table when validating —
52/// candidates registered here get their bespoke commitments and PVD echoed back, which is
53/// what real backing then seconds.
54///
55/// Without this, every candidate validates with empty `default_commitments`, which makes
56/// the backing pipeline ignore `head_data` threading and breaks fragment-chain scenarios.
57#[derive(Clone, Default)]
58pub struct CandidateOutputs {
59	inner: Arc<Mutex<HashMap<CandidateHash, (CandidateCommitments, PersistedValidationData)>>>,
60}
61
62impl CandidateOutputs {
63	/// Register `(commitments, pvd)` for `hash`. Subsequent validation calls for that
64	/// candidate will reproduce these outputs.
65	pub fn insert(
66		&self,
67		hash: CandidateHash,
68		commitments: CandidateCommitments,
69		pvd: PersistedValidationData,
70	) {
71		self.inner
72			.lock()
73			.expect("CandidateOutputs lock")
74			.insert(hash, (commitments, pvd));
75	}
76
77	fn get(&self, hash: &CandidateHash) -> Option<(CandidateCommitments, PersistedValidationData)> {
78		self.inner.lock().expect("CandidateOutputs lock").get(hash).cloned()
79	}
80}
81
82/// Verdict the stub returns for a single validate request.
83pub enum Verdict {
84	/// Candidate validates with the given commitments and PVD.
85	Valid(CandidateCommitments, PersistedValidationData),
86	/// Candidate is invalid; the wrapped reason surfaces in `ValidationResult::Invalid`.
87	Invalid(InvalidCandidate),
88}
89
90type VerdictFn = dyn FnMut(&CandidateReceipt, &PoV) -> Verdict + Send + 'static;

92/// Stub subsystem for `candidate-validation`. Answers the oneshot in
93/// `CandidateValidationMessage::ValidateFromExhaustive` immediately based on the configured
94/// verdict.
95pub struct CandidateValidationStub {
96	inbound_tx: mpsc::Sender<FromOrchestra<CandidateValidationMessage>>,
97}
98
99impl CandidateValidationStub {
100	/// Stub that approves every candidate. Commitments default to the framework's empty
101	/// shape (no upward messages, no horizontal messages, head data the candidate descriptor
102	/// already implies). For fragment-chain scenarios, register specific outputs via
103	/// [`CandidateOutputs`] — the registry takes precedence over the default.
104	pub fn always_valid<S>(sim: &mut crate::harness::Sim<S>, outputs: CandidateOutputs) -> Self
105	where
106		S: crate::harness::SubsystemUnderTest,
107		AllMessages: From<<S::Message as polkadot_overseer::AssociateOutgoing>::OutgoingMessages>,
108		AllMessages: From<S::Message>,
109	{
110		Self::with_verdict(sim, move |receipt, _| {
111			if let Some((commitments, pvd)) = outputs.get(&receipt.hash()) {
112				return Verdict::Valid(commitments, pvd);
113			}
114			let pvd = crate::builders::fixtures::dummy_pvd();
115			Verdict::Valid(default_commitments(), pvd)
116		})
117	}
118
119	/// Stub with a caller-provided verdict closure. Spawn the stub's worker future on the
120	/// harness executor; the worker pulls inbound messages and replies on each request's
121	/// oneshot synchronously.
122	pub fn with_verdict<S, F>(sim: &mut crate::harness::Sim<S>, verdict: F) -> Self
123	where
124		S: crate::harness::SubsystemUnderTest,
125		F: FnMut(&CandidateReceipt, &PoV) -> Verdict + Send + 'static,
126		AllMessages: From<<S::Message as polkadot_overseer::AssociateOutgoing>::OutgoingMessages>,
127		AllMessages: From<S::Message>,
128	{
129		let (inbound_tx, mut inbound_rx) =
130			mpsc::channel::<FromOrchestra<CandidateValidationMessage>>(0);
131
132		let mut verdict: Box<VerdictFn> = Box::new(verdict);
133		let fut = async move {
134			use futures::StreamExt;
135			while let Some(msg) = inbound_rx.next().await {
136				match msg {
137					FromOrchestra::Signal(OverseerSignal::Conclude) => break,
138					FromOrchestra::Signal(_) => {},
139					FromOrchestra::Communication { msg } => match msg {
140						CandidateValidationMessage::ValidateFromExhaustive {
141							candidate_receipt,
142							pov,
143							response_sender,
144							..
145						} => {
146							let result = match verdict(&candidate_receipt, &pov) {
147								Verdict::Valid(commitments, pvd) => {
148									ValidationResult::Valid(commitments, pvd)
149								},
150								Verdict::Invalid(reason) => ValidationResult::Invalid(reason),
151							};
152							let _ = response_sender.send(Ok(result));
153						},
154						other => panic!(
155							"CandidateValidationStub: unhandled CandidateValidationMessage \
156							 variant: {:?}. Extend the stub when a new message family becomes \
157							 relevant.",
158							other
159						),
160					},
161				}
162			}
163		};
164		sim.executor_mut().spawn(fut.boxed());
165		sim.executor_mut().poll_until_pending();
166		Self { inbound_tx }
167	}
168}
169
170impl SubsystemSlot for CandidateValidationStub {
171	fn name(&self) -> &'static str {
172		"candidate-validation-stub"
173	}
174
175	fn send_signal(&self, signal: OverseerSignal) -> BoxFuture<'static, ()> {
176		let mut tx = self.inbound_tx.clone();
177		async move {
178			let _ = tx.send(FromOrchestra::Signal(signal)).await;
179		}
180		.boxed()
181	}
182
183	fn try_route(&self, msg: AllMessages) -> RouteAttempt {
184		match msg {
185			AllMessages::CandidateValidation(inner) => {
186				let mut tx = self.inbound_tx.clone();
187				let fut = async move {
188					let _ = tx.send(FromOrchestra::Communication { msg: inner }).await;
189				}
190				.boxed();
191				RouteAttempt::Accepted(fut)
192			},
193			other => RouteAttempt::Declined(other),
194		}
195	}
196}

198/// Convenience: empty commitments. The candidate descriptor implies the head data; the rest
199/// is empty.
200fn default_commitments() -> CandidateCommitments {
201	CandidateCommitments {
202		upward_messages: Default::default(),
203		horizontal_messages: Default::default(),
204		new_validation_code: None,
205		head_data: polkadot_primitives::HeadData(vec![1, 2, 3]),
206		processed_downward_messages: 0,
207		hrmp_watermark: 0,
208	}
