43/// Per-block facts the chain model knows.
44#[derive(Clone, Debug)]

58impl BlockInfo {
59	/// Materialise a `Header` for this block. Includes the BABE pre-digest derived from
60	/// `slot` so V3 scheduling-parent validation can extract the slot from the header.
61	pub fn header(&self) -> Header {
62		let pre_digest = PreDigest::SecondaryPlain(SecondaryPlainPreDigest {
63			authority_index: 0,
64			slot: self.slot,
65		});
66		Header {
67			parent_hash: self.parent_hash,
68			number: self.number,
69			state_root: Default::default(),
70			extrinsics_root: Default::default(),
71			digest: Digest { logs: vec![DigestItem::babe_pre_digest(pre_digest)] },
72		}
73	}

76/// In-memory model of the relay chain.
77///
78/// Constructed via [`ChainModel::new`] (genesis only) and grown via [`extend`]. Sessions are
79/// added via [`add_session`]; per-block claim queues via [`set_claim_queue_at`].
80///
81/// The model is single-threaded and owned by the [`Sim`]. Test code mutates it directly.
82///
83/// [`extend`]: ChainModel::extend
84/// [`add_session`]: ChainModel::add_session
85/// [`set_claim_queue_at`]: ChainModel::set_claim_queue_at
86/// [`Sim`]: crate::harness::Sim
87/// Per-core schedule: a finite cycle of paras that repeats. The claim queue at block N
88/// shows positions `[N, N+1, ..., N+lookahead-1]` sampled from `cycle[(start + offset) %
89/// cycle.len()]`. Empty `cycle` means no para is ever scheduled on this core.
90#[derive(Clone, Debug, Default)]
91pub struct CoreSchedule {
92	/// The repeating sequence of paras for this core. `cycle[i]` is the para scheduled at
93	/// block-index `start + i` modulo `cycle.len()`.
94	pub cycle: Vec<ParaId>,
95	/// Block index at which the cycle is anchored. The para at block `n` is
96	/// `cycle[(n - start) % cycle.len()]`, so with the default `start = 0`, `cycle[0]` is the
97	/// para at genesis (block 0), `cycle[1]` at block 1, and so on. Most tests leave this at 0.
98	pub start: u32,
99}
100
101impl CoreSchedule {
102	/// Static schedule: the same para repeats forever.
103	pub fn always(para: ParaId) -> Self {
104		Self { cycle: vec![para], start: 0 }
105	}
106
107	/// Cycling schedule: `cycle[i]` is scheduled at block index `i, len, 2*len, ...`.
108	pub fn cycling(cycle: Vec<ParaId>) -> Self {
109		Self { cycle, start: 0 }
110	}
111
112	/// Para scheduled at block index `block_number`. Returns `None` if cycle is empty.
113	pub fn at(&self, block_number: u32) -> Option<ParaId> {
114		if self.cycle.is_empty() {
115			return None;
116		}
117		let idx = ((block_number.wrapping_sub(self.start)) as usize) % self.cycle.len();
118		Some(self.cycle[idx])
119	}
120}

122/// In-memory model of the relay chain.
123///
124/// Constructed via [`ChainModel::new`] (genesis only) and grown via
125/// [`ChainModel::extend`]. Sessions are added via [`ChainModel::add_session`]; per-core
126/// schedule via [`ChainModel::set_core_schedule`]. Per-block claim-queue overrides are
127/// available via [`ChainModel::set_claim_queue_at`].
128#[derive(Debug)]
129pub struct ChainModel {
130	blocks: BTreeMap<Hash, BlockInfo>,
131	children: BTreeMap<Hash, Vec<Hash>>,
132	sessions: BTreeMap<SessionIndex, SessionInfo>,
133	/// Global schedule: per-core para rotation. The claim queue at any block is computed by
134	/// sampling each core's schedule starting at the block's number.
135	schedule: BTreeMap<CoreIndex, CoreSchedule>,
136	/// Per-block claim-queue overrides. If set, takes precedence over `schedule`. Used by
137	/// tests that need exotic shapes the cycle abstraction can't express.
138	claim_queue_overrides: BTreeMap<Hash, BTreeMap<CoreIndex, VecDeque<ParaId>>>,
139	scheduling_lookahead: u32,
140	async_backing_params: AsyncBackingParams,
141	node_features: NodeFeatures,
142	minimum_backing_votes: u32,
143	pending_availability: BTreeMap<ParaId, Vec<CommittedCandidateReceipt>>,
144	/// Per-relay-parent pending-availability overrides. When set for `(rp, para)`, takes
145	/// precedence over the global `pending_availability` table. Used by tests where
146	/// different leaves report different pending-availability snapshots (e.g. a candidate
147	/// is pending under leaf_b but not yet seeded under sibling leaf_c).
148	pending_availability_at: BTreeMap<(Hash, ParaId), Vec<CommittedCandidateReceipt>>,
149	backing_constraints: BTreeMap<ParaId, Constraints>,
150	/// Per-relay-parent backing-constraint overrides. Takes precedence over the global
151	/// `backing_constraints` table. Used by tests where the same para has different
152	/// `required_parent` head data at different leaves (as in real prospective tests where
153	/// each leaf carries its own per-para `head_data` snapshot).
154	backing_constraints_at: BTreeMap<(Hash, ParaId), Constraints>,
155	/// Per-relay-parent `CandidateEvent` log returned by
156	/// `RuntimeApiRequest::CandidateEvents`. Tests seed this when driving finalization to
157	/// trigger experimental's `+VALID_INCLUDED_CANDIDATE_BUMP` path.
158	candidate_events: BTreeMap<Hash, Vec<CandidateEvent>>,
159	/// Latest finalized block — controls `ChainApiMessage::FinalizedBlockNumber` /
160	/// `FinalizedBlockHash`. Initialised to genesis.
161	finalized: Hash,
162	genesis: Hash,
163	tip: Hash,
164	/// Effective `RuntimeApiRequest` version reported by the model. Controls
165	/// `NotSupported` dispatch for newer APIs. Tests parameterising over runtime
166	/// versions (e.g. exercising both the `AncestorRelayParentInfo` path and the
167	/// chain-header fallback path) override this. Defaults to the highest version the
168	/// chain model implements end-to-end.
169	runtime_api_version: u32,
170}

235	/// Register a block with a caller-chosen hash, parent, and number. Use this when the
236	/// test wants specific hash literals (e.g. `Hash::from_low_u64_be(130)`) instead of the
237	/// synthetic hashes [`Self::extend`] generates.
238	///
239	/// Slot and session are derived from the parent: slot is `parent_slot + 1` and session is
240	/// inherited, **but only if the parent is already registered**. There is no later fixup —
241	/// register parents before children. If the parent is absent at call time (including a
242	/// deliberately detached block with `parent_hash == Hash::zero()`), slot falls back to
243	/// `Slot::from(number)` and session to `0`; pass [`Self::register_block_with_session`] to
244	/// pin the session explicitly in that case.
245	///
246	/// If `hash` is already registered this is a no-op (the test may call `register_block`
247	/// idempotently when walking ancestor chains).
248	///
249	/// Used by per-subsystem test-sim consumer crates (e.g. prospective-parachains) whose
250	/// faithful ports of pre-existing tests use literal block hashes.
251	pub fn register_block(&mut self, hash: Hash, parent_hash: Hash, number: BlockNumber) {
252		self.register_block_with_session(hash, parent_hash, number, None);
253	}
254
255	/// Like [`Self::register_block`] but with an explicit session override. Used when the
256	/// test's whole synthetic chain should live in a specific session (e.g. session 1)
257	/// instead of inheriting from a synthesised genesis-default of 0.
258	pub fn register_block_with_session(
259		&mut self,
260		hash: Hash,
261		parent_hash: Hash,
262		number: BlockNumber,
263		session_override: Option<SessionIndex>,
264	) {
265		if self.blocks.contains_key(&hash) {
266			return;
267		}
268		let (slot, default_session) = match self.blocks.get(&parent_hash) {
269			Some(parent) => (parent.slot + 1, parent.session_index),
270			None => (Slot::from(number as u64), 0),
271		};
272		let session_index = session_override.unwrap_or(default_session);
273		let info = BlockInfo { hash, parent_hash, number, slot, session_index };
274		self.blocks.insert(hash, info);
275		self.children.entry(parent_hash).or_default().push(hash);
276		if number > self.blocks.get(&self.tip).map(|t| t.number).unwrap_or(0) {
277			self.tip = hash;
278		}
279	}

281	/// Append a child block onto `parent`. Slot increments by one, session is inherited from
282	/// the parent. Returns the new block's hash.
283	///
284	/// Calling `extend` multiple times from the same parent produces siblings — each gets
285	/// a distinct hash via a sibling-index mixed into the synthetic hash. The chain model
286	/// supports forks; tests model fork choice by extending different children from a
287	/// shared ancestor and activating each via `extend_and_activate_with`.
288	///
289	/// Panics if `parent` is not known.
290	pub fn extend(&mut self, parent: Hash) -> Hash {
291		let parent_info = self.blocks.get(&parent).cloned().expect("parent block must exist");
292		let number = parent_info.number + 1;
293		let slot = parent_info.slot + 1;
294		let session_index = parent_info.session_index;
295		let sibling_idx = self.children.get(&parent).map(|c| c.len() as u64).unwrap_or(0);
296		let hash = synthetic_child_hash_with_sibling(parent, number, sibling_idx);
297		let info = BlockInfo { hash, parent_hash: parent, number, slot, session_index };
298		self.blocks.insert(hash, info);
299		self.children.entry(parent).or_default().push(hash);
300		self.tip = hash;
301		hash
302	}

346	/// Compute the claim queue at a block: per-core, populate `scheduling_lookahead`
347	/// positions sampled from the schedule starting at the block's number. Returns
348	/// `claim_queue_overrides[block]` if explicitly set.
349	fn derive_claim_queue(&self, block: &Hash) -> BTreeMap<CoreIndex, VecDeque<ParaId>> {
350		if let Some(q) = self.claim_queue_overrides.get(block) {
351			return q.clone();
352		}
353		let info = match self.blocks.get(block) {
354			Some(i) => i,
355			None => return BTreeMap::new(),
356		};
357		let mut out = BTreeMap::new();
358		for (core, sched) in &self.schedule {
359			let mut q = VecDeque::with_capacity(self.scheduling_lookahead as usize);
360			for offset in 0..self.scheduling_lookahead {
361				if let Some(para) = sched.at(info.number + offset) {
362					q.push_back(para);
363				}
364			}
365			if !q.is_empty() {
366				out.insert(*core, q);
367			}
368		}
369		out
370	}

468	/// Walk ancestry of `from`. Yields parent, grandparent, ... up to (but not including) the
469	/// genesis pre-image. Used by `ChainApi::Ancestors`.
470	pub fn ancestors(&self, from: Hash, k: usize) -> Vec<Hash> {
471		let mut out = Vec::with_capacity(k);
472		let mut cursor = from;
473		for _ in 0..k {
474			match self.blocks.get(&cursor) {
475				Some(info) if info.parent_hash != Hash::zero() => {
476					out.push(info.parent_hash);
477					cursor = info.parent_hash;
478				},
479				_ => break,
480			}
481		}
482		out
483	}

485	/// Hash of the finalized block at `number`, found by walking back from the finalized tip
486	/// along parent links. Returns `None` when `number` is above the finalized tip (nothing
487	/// is finalized that high yet) or the chain is malformed. Resolving along the finalized
488	/// chain — rather than scanning all blocks for one at that height — is what keeps forks
489	/// from yielding a non-canonical block.
490	fn finalized_ancestor_at(&self, number: BlockNumber) -> Option<Hash> {
491		let mut cursor = self.finalized;
492		loop {
493			let info = self.blocks.get(&cursor)?;
494			match info.number.cmp(&number) {
495				std::cmp::Ordering::Equal => return Some(info.hash),
496				// Walked below the target without hitting it (gap in the chain) — no match.
497				std::cmp::Ordering::Less => return None,
498				// Still above the target: step to the parent. The genesis pre-image
499				// (`Hash::zero()`) is not a real block, so the `?` above ends the walk.
500				std::cmp::Ordering::Greater => cursor = info.parent_hash,
501			}
502		}
503	}

517	fn answer_runtime_req(&self, parent: Hash, req: RuntimeApiRequest) {
518		let info = self.blocks.get(&parent).unwrap_or_else(|| {
519			panic!("ChainModel: RuntimeApi request for unknown block {:?}", parent)
520		});
521		match req {
522			RuntimeApiRequest::SessionIndexForChild(tx) => {
523				let _ = tx.send(Ok(info.session_index));
524			},
525			RuntimeApiRequest::ClaimQueue(tx) => {
526				let queue = self.derive_claim_queue(&parent);
527				let _ = tx.send(Ok(queue));
528			},
529			RuntimeApiRequest::Validators(tx) => {
530				let _ = tx.send(Ok(self.session_info(info.session_index).validators.clone()));
531			},
532			RuntimeApiRequest::ValidatorGroups(tx) => {
533				let session = self.session_info(info.session_index);
534				let mut rotation = session.group_rotation_info.clone();
535				// Mirror the real runtime: `validator_groups` reports `now = block_number + 1`
536				// (see `runtime_api_impl::v13::validator_groups`). The legacy validator side
537				// trusts this `now` directly; the experimental side recomputes it. Off-by-one
538				// here silently shifts every legacy core-ownership calculation by one block.
539				rotation.now = info.number + 1;
540				let _ = tx.send(Ok((session.validator_groups.clone(), rotation)));
541			},
542			RuntimeApiRequest::SchedulingLookahead(_session, tx) => {
543				let _ = tx.send(Ok(self.scheduling_lookahead));
544			},
545			RuntimeApiRequest::AsyncBackingParams(tx) => {
546				let _ = tx.send(Ok(self.async_backing_params));
547			},
548			RuntimeApiRequest::NodeFeatures(_session, tx) => {
549				let _ = tx.send(Ok(self.node_features.clone()));
550			},
551			RuntimeApiRequest::MinimumBackingVotes(_session, tx) => {
552				let _ = tx.send(Ok(self.minimum_backing_votes));
553			},
554			RuntimeApiRequest::BackingConstraints(para, tx) => {
555				let constraints = self
556					.backing_constraints_at
557					.get(&(parent, para))
558					.cloned()
559					.or_else(|| self.backing_constraints.get(&para).cloned())
560					.unwrap_or_else(default_constraints);
561				let _ = tx.send(Ok(Some(constraints)));
562			},
563			RuntimeApiRequest::CandidatesPendingAvailability(para, tx) => {
564				let candidates = self
565					.pending_availability_at
566					.get(&(parent, para))
567					.cloned()
568					.or_else(|| self.pending_availability.get(&para).cloned())
569					.unwrap_or_default();
570				let _ = tx.send(Ok(candidates));
571			},
572			RuntimeApiRequest::CandidateEvents(tx) => {
573				let events =
574					self.candidate_events.get(&parent).cloned().unwrap_or_default();
575				let _ = tx.send(Ok(events));
576			},
577			RuntimeApiRequest::PersistedValidationData(_para, _assumption, tx) => {
578				// Synthesise a PVD whose shape matches the seconding sanity check: parent
579				// head = empty, relay parent number = block's number, storage root =
580				// Hash::zero(), max_pov_size = 5 MB. Tests that need different shapes should
581				// pre-set the candidate's persisted_validation_data_hash to match.
582				let pvd = PersistedValidationData {
583					parent_head: HeadData(Vec::new()),
584					relay_parent_number: info.number,
585					relay_parent_storage_root: Hash::zero(),
586					max_pov_size: 5 * 1024 * 1024,
587				};
588				let _ = tx.send(Ok(Some(pvd)));
589			},
590			RuntimeApiRequest::DisabledValidators(tx) => {
591				let _ = tx.send(Ok(Vec::new()));
592			},
593			RuntimeApiRequest::AvailabilityCores(tx) => {
594				let _ = tx.send(Ok(Vec::new()));
595			},
596			RuntimeApiRequest::SessionInfo(_session, tx) => {
597				let _ = tx.send(Ok(None));
598			},
599			RuntimeApiRequest::SessionExecutorParams(_session, tx) => {
600				let _ = tx.send(Ok(None));
601			},
602			RuntimeApiRequest::ParaIds(_session, tx) => {
603				// Derive registered paras from the global schedule + per-block claim-queue
604				// overrides. Returning empty here triggers experimental's `prune_paras` to
605				// wipe every score-store entry on every BlockFinalized — see
606				// `peer_manager::PeerManager::prune_registered_paras`.
607				let mut paras: std::collections::BTreeSet<ParaId> = self
608					.schedule
609					.values()
610					.flat_map(|sched| sched.cycle.iter().copied())
611					.collect();
612				for queue in self.claim_queue_overrides.values() {
613					for cores in queue.values() {
614						paras.extend(cores.iter().copied());
615					}
616				}
617				let _ = tx.send(Ok(paras.into_iter().collect()));
618			},
619			RuntimeApiRequest::ValidationCodeBombLimit(_session, tx) => {
620				let _ = tx.send(Ok(60 * 1024 * 1024));
621			},
622			RuntimeApiRequest::MaxRelayParentSessionAge(_session, tx) => {
623				let _ = tx.send(Ok(8));
624			},
625			RuntimeApiRequest::ValidationCodeByHash(_hash, tx) => {
626				// Return a tiny valid-shaped ValidationCode payload. The
627				// always-valid candidate-validation stub doesn't actually run it.
628				let _ = tx.send(Ok(Some(polkadot_primitives::ValidationCode(vec![0x00]))));
629			},
630			RuntimeApiRequest::ValidationCode(_para, _assumption, tx) => {
631				let _ = tx.send(Ok(Some(polkadot_primitives::ValidationCode(vec![0x00]))));
632			},
633			RuntimeApiRequest::ValidationCodeHash(_para, _assumption, tx) => {
634				// Default: return None (no upgrade pending).
635				let _ = tx.send(Ok(None));
636			},
637			RuntimeApiRequest::AncestorRelayParentInfo(_session, queried_relay_parent, tx) => {
638				// If the configured `runtime_api_version` is below the requirement, the
639				// real runtime would return `NotSupported`; mirror that so callers
640				// exercise the chain-header fallback path.
641				if self.runtime_api_version
642					< RuntimeApiRequest::ANCESTOR_RELAY_PARENT_INFO_RUNTIME_REQUIREMENT
643				{
644					let _ = tx.send(Err(
645						polkadot_node_subsystem::errors::RuntimeApiError::NotSupported {
646							runtime_api_name: "AncestorRelayParentInfo",
647						},
648					));
649				} else {
650					// Return None if querying about self (a block isn't in its own
651					// `AllowedRelayParents`); otherwise return the queried block's
652					// RelayParentInfo when we know it. Other-session queries are
653					// disambiguated by the runtime; our chain model only has one
654					// session so we ignore the session arg.
655					let answer = if queried_relay_parent == parent {
656						None
657					} else {
658						self.blocks.get(&queried_relay_parent).map(|info| {
659							polkadot_primitives::vstaging::RelayParentInfo {
660								number: info.number,
661								state_root: Hash::zero(),
662							}
663						})
664					};
665					let _ = tx.send(Ok(answer));
666				}
667			},
668			other => panic!(
669				"ChainModel does not implement RuntimeApiRequest::{:?} yet — extend the model when a subsystem starts asking for it",
670				other
671			),
672		}
673	}
674
675	fn answer_chain_api(&self, msg: ChainApiMessage) {
676		match msg {
677			ChainApiMessage::BlockHeader(hash, tx) => {
678				let header = self.blocks.get(&hash).map(BlockInfo::header);
679				let _ = tx.send(Ok(header));
680			},
681			ChainApiMessage::BlockNumber(hash, tx) => {
682				let number = self.blocks.get(&hash).map(|info| info.number);
683				let _ = tx.send(Ok(number));
684			},
685			ChainApiMessage::Ancestors { hash, k, response_channel } => {
686				let ancestors = self.ancestors(hash, k);
687				let _ = response_channel.send(Ok(ancestors));
688			},
689			ChainApiMessage::FinalizedBlockNumber(tx) => {
690				let n = self.blocks.get(&self.finalized).map(|info| info.number).unwrap_or(0);
691				let _ = tx.send(Ok(n));
692			},
693			ChainApiMessage::FinalizedBlockHash(number, tx) => {
694				// Resolve along the finalized chain, not by a global scan: with forks,
695				// several blocks share a height, and only the one on the path to
696				// `self.finalized` is actually finalized. Walk back from the finalized tip to
697				// the requested height. A height above the finalized tip has no finalized
698				// block yet → `None`, matching the real ChainApi.
699				let hash = self.finalized_ancestor_at(number);
700				let _ = tx.send(Ok(hash));
701			},
702			other => panic!(
703				"ChainModel does not implement ChainApiMessage::{:?} yet — extend the model when a subsystem starts asking for it",
704				other
705			),
706		}
707	}
708}

710impl AnswerQuery for ChainModel {
711	fn try_answer(&mut self, query: Query) -> Option<Query> {
712		match query {
713			Query::Runtime(msg) => {
714				self.answer_runtime(msg);
715				None
716			},
717			Query::ChainApi(msg) => {
718				self.answer_chain_api(msg);
719				None
720			},
721			other => Some(other),
722		}
723	}
724
725	fn answer(&mut self, query: Query) {
726		// Direct-answer surface for unit tests; declines fall through to a panic so test
727		// authors notice when they hand the chain model a query it doesn't own.
728		if let Some(declined) = self.try_answer(query) {
729			panic!(
730				"ChainModel does not handle non-runtime/chain-api queries; declined: {:?}",
731				declined
732			);
733		}
734	}
735}

737/// Shared ownership of a [`ChainModel`].
738///
739/// The harness installs this in its responder chain so the model's mutable state stays
740/// accessible to the test (via [`SharedChain::lock`]) while queries the subsystem fires
741/// against the responder are routed back into the same model.
742#[derive(Clone, Debug)]
743pub struct SharedChain(std::sync::Arc<std::sync::Mutex<ChainModel>>);
744
745impl SharedChain {
746	/// Wrap a chain model for shared use.
747	pub fn new(chain: ChainModel) -> Self {
748		Self(std::sync::Arc::new(std::sync::Mutex::new(chain)))
749	}
750
751	/// Lock the inner model. Panics if poisoned.
752	pub fn lock(&self) -> std::sync::MutexGuard<'_, ChainModel> {
753		self.0.lock().expect("ChainModel mutex poisoned")
754	}
755}
756
757impl AnswerQuery for SharedChain {
758	fn try_answer(&mut self, query: Query) -> Option<Query> {
759		self.lock().try_answer(query)
760	}
761
762	fn answer(&mut self, query: Query) {
763		self.lock().answer(query)
764	}
765}

767/// Permissive default backing constraints. Tests that need stricter shapes pass their own
768/// via [`ChainModel::set_backing_constraints`].
769fn default_constraints() -> Constraints {
770	Constraints {
771		min_relay_parent_number: 0,
772		max_pov_size: 5 * 1024 * 1024,
773		max_code_size: 3 * 1024 * 1024,
774		max_head_data_size: 20 * 1024,
775		ump_remaining: 32,
776		ump_remaining_bytes: 64 * 1024,
777		max_ump_num_per_candidate: 16,
778		dmp_remaining_messages: Vec::new(),
779		// Real constraints reject candidates whose `hrmp_watermark` isn't in
780		// `valid_watermarks` (and isn't strictly greater than the relay parent number).
781		// Test commitments default to `hrmp_watermark = 0`; allow that explicitly so
782		// real prospective accepts plain candidates as fragment-chain members.
783		hrmp_inbound: InboundHrmpLimitations { valid_watermarks: vec![0] },
784		hrmp_channels_out: Vec::new(),
785		max_hrmp_num_per_candidate: 16,
786		required_parent: HeadData(Vec::new()),
787		// Match the validation code hash that `dummy_candidate_receipt_v2_bad_sig`
788		// (used by `Candidate::for_para_at` / `Candidate::builder`) bakes into
789		// every receipt's descriptor. Otherwise real prospective-parachains rejects every
790		// candidate as a `ValidationCodeMismatch` against constraints.
791		validation_code_hash: polkadot_primitives_test_helpers::dummy_validation_code().hash(),
792		upgrade_restriction: None,
793		future_validation_code: None,
794	}
795}

797fn synthetic_child_hash_with_sibling(parent: Hash, number: BlockNumber, sibling: u64) -> Hash {
798	// Deterministic child hash by mixing parent low-u64 with the child number and the
799	// sibling index. Tests do not assert on the exact value; identity is what matters.
800	// Sibling index is shifted into a higher byte so primary children retain the
801	// historical hash for backwards compatibility with existing tests.
802	let parent_low = parent.to_low_u64_be();
803	Hash::from_low_u64_be(parent_low.wrapping_add(0x100_0000 + number as u64 + (sibling << 32)))
804}
