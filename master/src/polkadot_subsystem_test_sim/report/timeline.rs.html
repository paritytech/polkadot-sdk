17//! Build human-readable timeline reports from a [`Recorder`] and a failed assertion.
18
19use crate::{
20	contract::Effect,
21	harness::{
22		observation::{Observation, Stamped},
23		Recorder,
24	},
25};
26use std::{fmt, time::Duration};
27
28/// One slice of the test history from `start` to `start + window`, with a textual description
29/// of what was expected. Calling [`fmt::Display`] yields the canonical failure message.
30pub struct TimelineReport<'a> {
31	/// Description of the assertion that failed (e.g. "Effect::Second{ candidate: 0xab.. }
32	/// within 200ms").
33	pub expected: String,
34	/// Description of the observed outcome (e.g. "timed out at sim_t = 200ms").
35	pub actual: String,
36	/// Sim-time of the start of the window.
37	pub window_start: Duration,
38	/// Length of the assertion window.
39	pub window: Duration,
40	/// Recorder containing the observations.
41	pub recorder: &'a Recorder,
42	/// Optional replay seed for reproducing the failure.
43	pub replay_seed: Option<u64>,
44	/// Optional source location of the assertion (filename:line).
45	pub at: Option<&'a str>,
46	/// Optional hint sentence (e.g. "closest match was Reputation(...) at +2ms").
47	pub hint: Option<String>,
48}
49
50impl fmt::Display for TimelineReport<'_> {
51	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
52		if let Some(at) = self.at {
53			writeln!(f, "  at {}", at)?;
54			writeln!(f)?;
55		}
56		writeln!(f, "  expected:  {}", self.expected)?;
57		writeln!(f, "  actual:    {}", self.actual)?;
58
59		let start = self.window_start;
60		let end = self.window_start + self.window;
61		writeln!(
62			f,
63			"             observed effects since the When step (sim_t = {}ms..{}ms):",
64			start.as_millis(),
65			end.as_millis()
66		)?;
67
68		let entries: Vec<&Stamped<Effect>> = self
69			.recorder
70			.entries()
71			.iter()
72			.filter_map(|o| match o {
73				Observation::Effect(s) => {
74					if s.sim_t >= start && s.sim_t <= end {
75						Some(s)
76					} else {
77						None
78					}
79				},
80			})
81			.collect();
82
83		if entries.is_empty() {
84			writeln!(f, "               <no effects observed in this window>")?;
85		} else {
86			for e in &entries {
87				writeln!(
88					f,
89					"               [{:>5}ms]  {}",
90					e.sim_t.as_millis(),
91					format_effect(&e.value)
92				)?;
93			}
94		}
95
96		if let Some(hint) = &self.hint {
97			writeln!(f)?;
98			writeln!(f, "  hint: {}", hint)?;
99		}
100
101		if let Some(seed) = self.replay_seed {
102			writeln!(f)?;
103			writeln!(f, "  replay:")?;
104			writeln!(
105				f,
106				"    REPLAY_SEED=0x{:x} cargo test -p polkadot-collator-protocol-test-sim",
107				seed
108			)?;
109		}
110
111		Ok(())
112	}
113}

115/// Compact human-readable rendering of an [`Effect`] for the timeline. Exposed so test
116/// code can pretty-print recorder entries when an assertion fires.
117pub fn format_effect(effect: &Effect) -> String {
118	match effect {
119		Effect::SecondCandidate { para, candidate_hash, .. } => {
120			format!("SecondCandidate(para={}, cand={:?})", u32::from(*para), candidate_hash)
121		},
122		Effect::SendAdvertisement { peers, .. } => {
123			format!("SendAdvertisement(peers={})", peers.len())
124		},
125		Effect::SendCollation { peers, kind } => {
126			format!("SendCollation(peers={}, kind={:?})", peers.len(), kind)
127		},
128		Effect::SendRequest { request_id, to, kind, candidate_hash } => format!(
129			"SendRequest(req={:?}, to={}, kind={:?}, cand={:?})",
130			request_id,
131			short_peer(to),
132			kind,
133			candidate_hash
134		),
135		Effect::SendRequestV3 { request_id, to, kind, output_head_data_hash } => format!(
136			"SendRequest(req={:?}, to={}, kind={:?}, cand={:?})",
137			request_id,
138			short_peer(to),
139			kind,
140			output_head_data_hash
141		),
142		Effect::Reputation { peer, bucket } => {
143			format!("Reputation(peer={}, bucket={:?})", short_peer(peer), bucket)
144		},
145		Effect::ConnectValidators { validator_ids, peer_set } => {
146			format!("ConnectValidators(n={}, peer_set={:?})", validator_ids.len(), peer_set)
147		},
148		Effect::DisconnectPeers { peers, peer_set } => {
149			format!("DisconnectPeers(n={}, peer_set={:?})", peers.len(), peer_set)
150		},
151		Effect::RequestResponseSent { request_id, kind } => {
152			format!("RequestResponseSent(req={}, kind={:?})", request_id, kind)
153		},
154	}
155}

157/// Pretty-print every entry in the recorder as one line per effect, prefixed with its
158/// `sim_t` in milliseconds. Tests use this when an assertion fails to dump a readable
159/// timeline rather than `Debug` output.
160pub fn format_timeline(recorder: &Recorder) -> String {
161	let mut out = String::new();
162	out.push_str("Effect timeline (relative to first observation):\n");
163	let entries: Vec<&Stamped<Effect>> = recorder
164		.entries()
165		.iter()
166		.map(|o| match o {
167			Observation::Effect(s) => s,
168		})
169		.collect();
170	if entries.is_empty() {
171		out.push_str("  <no effects observed>\n");
172		return out;
173	}
174	for e in entries {
175		out.push_str(&format!("  [{:>6}ms] {}\n", e.sim_t.as_millis(), format_effect(&e.value),));
176	}
177	out
178}
179
180fn short_peer(p: &sc_network_types::PeerId) -> String {
181	let s = p.to_base58();
182	if s.len() <= 10 {
183		s
184	} else {
185		format!("{}..{}", &s[..6], &s[s.len() - 4..])
186	}
187}

189#[cfg(test)]
190mod tests {
191	use super::*;
192	use crate::contract::{Effect, RepBucket};
193	use sc_network_types::PeerId;
194
195	#[test]
196	fn report_renders_observed_window() {
197		let mut rec = Recorder::new();
198		let p1 = PeerId::random();
199		rec.record_effect(
200			Duration::ZERO,
201			Effect::Reputation { peer: p1, bucket: RepBucket::Performance },
202		);
203		rec.record_effect(
204			Duration::from_millis(2),
205			Effect::Reputation { peer: p1, bucket: RepBucket::Malicious },
206		);
207
208		let report = TimelineReport {
209			expected: "Effect::SecondCandidate within 200ms".into(),
210			actual: "timed out at sim_t = 200ms".into(),
211			window_start: Duration::ZERO,
212			window: Duration::from_millis(200),
213			recorder: &rec,
214			replay_seed: Some(0xdeadbeef),
215			at: Some("tests/foo.rs:42"),
216			hint: Some("closest match was Reputation(.., Malicious) at +2ms".into()),
217		};
218
219		let s = format!("{}", report);
220		assert!(s.contains("at tests/foo.rs:42"));
221		assert!(s.contains("expected:  Effect::SecondCandidate"));
222		assert!(s.contains("actual:    timed out"));
223		assert!(s.contains("[    0ms]  Reputation"));
224		assert!(s.contains("[    2ms]  Reputation"));
225		assert!(s.contains("hint: closest match"));
226		assert!(s.contains("REPLAY_SEED=0xdeadbeef"));
227	}
228
229	#[test]
230	fn report_handles_empty_window() {
231		let rec = Recorder::new();
232		let report = TimelineReport {
233			expected: "Effect::Foo".into(),
234			actual: "timed out".into(),
235			window_start: Duration::ZERO,
236			window: Duration::from_millis(100),
237			recorder: &rec,
238			replay_seed: None,
239			at: None,
240			hint: None,
241		};
242		let s = format!("{}", report);
243		assert!(s.contains("<no effects observed in this window>"));
244	}
