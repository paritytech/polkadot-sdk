17//! Dispatcher: drains outgoing subsystem messages, routes them to either the [`Recorder`]
18//! (effects) or the responder (queries) per the [`crate::contract::classify()`] rule.
19//!
20//! The full responder DSL lands in `crate::responder`. This file defines the dispatch loop and
21//! the [`AnswerQuery`] trait that the responder will implement.
22
23use crate::{
24	contract::{classify, Classified, Query},
25	harness::{pending_fetches::PendingFetches, Recorder},
26};
27use polkadot_node_subsystem::messages::AllMessages;
28use std::time::Duration;
29
30/// Trait implemented by anything that can answer subsystem queries.
31///
32/// Two methods, with one of them having a default implementation:
33///
34/// - [`AnswerQuery::try_answer`] is used by the harness's responder chain. It returns `Some(query)`
35///   if the responder declined the query (so the next layer gets a chance), or `None` if it
36///   consumed it.
37/// - [`AnswerQuery::answer`] is the convenience entry point for unit tests and for responders that
38///   always handle (or always panic on) every query they see.
39///
40/// Most responders implement `answer` only. `try_answer` then defaults to "delegate to
41/// `answer`, never decline". Composable responders (e.g. a chain-model wrapper that only
42/// handles `Runtime`/`ChainApi`) override `try_answer` so the harness can fall through to a
43/// later layer.
44pub trait AnswerQuery {
45	/// Try to answer the query. Return `None` if the query was handled, or `Some(query)` to
46	/// pass it on to the next responder in the chain.
47	///
48	/// Default: delegate to [`answer`] and return `None`.
49	///
50	/// [`answer`]: AnswerQuery::answer
51	fn try_answer(&mut self, query: Query) -> Option<Query> {
52		self.answer(query);
53		None
54	}
55
56	/// Always-consume entry point. Implementations write back to the embedded
57	/// `oneshot::Sender`s. If a query is unexpected, implementations panic with a
58	/// descriptive message — that is a test bug (the test forgot to script a particular
59	/// query path).
60	fn answer(&mut self, _query: Query) {
61		panic!(
62			"AnswerQuery::answer was called on a responder that only implements \
63			 try_answer; declined queries should be handled by a later responder layer"
64		);
65	}
66}
67
68/// A layered responder: tries each layer in order; the first that returns `None` wins. If
69/// all decline, panics.
70///
71/// Use to compose multiple responders. The chain is sometimes hand-built; more often
72/// [`crate::harness::Sim`] populates it for you (chain model + per-test query script + a
73/// default panic-on-unhandled tail).
74pub struct LayeredResponder {
75	layers: Vec<Box<dyn AnswerQuery>>,
76}
77
78impl LayeredResponder {
79	/// New empty chain.
80	pub fn new() -> Self {
81		Self { layers: Vec::new() }
82	}
83
84	/// Append a layer to the end of the chain.
85	pub fn push<R: AnswerQuery + 'static>(&mut self, responder: R) {
86		self.layers.push(Box::new(responder));
87	}
88
89	/// Append a pre-boxed layer.
90	pub fn push_boxed(&mut self, responder: Box<dyn AnswerQuery>) {
91		self.layers.push(responder);
92	}
93}
94
95impl Default for LayeredResponder {
96	fn default() -> Self {
97		Self::new()
98	}
99}
100
101impl AnswerQuery for LayeredResponder {
102	fn try_answer(&mut self, query: Query) -> Option<Query> {
103		let mut current = Some(query);
104		for layer in self.layers.iter_mut() {
105			let q = current.take().expect("loop invariant: current is Some at top of iteration");
106			match layer.try_answer(q) {
107				None => return None,
108				Some(declined) => current = Some(declined),
109			}
110		}
111		// Every layer declined.
112		current
113	}
114
115	fn answer(&mut self, query: Query) {
116		match self.try_answer(query) {
117			None => {},
118			Some(unhandled) => panic!(
119				"LayeredResponder: every layer declined the query; install a fall-through \
120				 layer (e.g. PanicResponder) or extend an existing layer to cover this case. \
121				 Unhandled query: {:?}",
122				unhandled
123			),
124		}
125	}
126}

128/// Drains a single outgoing `AllMessages`, classifies it, and routes:
129/// - Effects → recorded into `recorder`.
130/// - Queries → forwarded to `responder`.
131///
132/// Effects are stamped with `sim_t`, the simulated time elapsed since the start of the
133/// scenario, supplied per `dispatch` call.
134pub struct Dispatcher<'a, R: AnswerQuery + ?Sized> {
135	/// Where effects accumulate.
136	pub recorder: &'a mut Recorder,
137	/// Where queries are routed.
138	pub responder: &'a mut R,
139	/// Side table for pending fetch response senders extracted from `SendRequests`.
140	pub pending: &'a mut PendingFetches,
141}
142
143impl<'a, R: AnswerQuery + ?Sized> Dispatcher<'a, R> {
144	/// Create a new dispatcher.
145	pub fn new(
146		recorder: &'a mut Recorder,
147		responder: &'a mut R,
148		pending: &'a mut PendingFetches,
149	) -> Self {
150		Self { recorder, responder, pending }
151	}
152
153	/// Process a single outbound message. One inbound message can yield multiple classified
154	/// entries (e.g. a batched `SendRequests` or `SendCollationMessages`); the dispatcher
155	/// records / forwards them in order.
156	pub fn dispatch(&mut self, sim_t: Duration, msg: AllMessages) {
157		for c in classify(msg, self.pending) {
158			match c {
159				Classified::Effect(effect) => self.recorder.record_effect(sim_t, effect),
160				Classified::Query(query) => self.responder.answer(query),
161			}
162		}
163	}
164}

173	struct AcceptOnly<F: FnMut(Query)>(F);
174	impl<F: FnMut(Query)> AnswerQuery for AcceptOnly<F> {
175		fn answer(&mut self, query: Query) {
176			(self.0)(query);
177		}
178	}

180	struct DeclineOnly;
181	impl AnswerQuery for DeclineOnly {
182		fn try_answer(&mut self, query: Query) -> Option<Query> {
183			Some(query)
184		}
185	}

187	#[test]
188	fn layered_responder_first_accept_wins() {
189		use std::sync::{Arc, Mutex};
190		let hits_a = Arc::new(Mutex::new(0));
191		let hits_b = Arc::new(Mutex::new(0));
192		let a_clone = hits_a.clone();
193		let b_clone = hits_b.clone();
194		let mut chain = LayeredResponder::new();
195		chain.push(AcceptOnly(move |_| {
196			*a_clone.lock().unwrap() += 1;
197		}));
198		chain.push(AcceptOnly(move |_| {
199			*b_clone.lock().unwrap() += 1;
200		}));
201		let (tx, _rx) = futures::channel::oneshot::channel();
202		chain.answer(Query::ChainApi(ChainApiMessage::FinalizedBlockNumber(tx)));
203		assert_eq!(*hits_a.lock().unwrap(), 1);
204		assert_eq!(*hits_b.lock().unwrap(), 0);
205	}

207	#[test]
208	fn layered_responder_falls_through_on_decline() {
209		use std::sync::{Arc, Mutex};
210		let hits = Arc::new(Mutex::new(0));
211		let hits_clone = hits.clone();
212		let mut chain = LayeredResponder::new();
213		chain.push(DeclineOnly);
214		chain.push(AcceptOnly(move |_| {
215			*hits_clone.lock().unwrap() += 1;
216		}));
217		let (tx, _rx) = futures::channel::oneshot::channel();
218		chain.answer(Query::ChainApi(ChainApiMessage::FinalizedBlockNumber(tx)));
219		assert_eq!(*hits.lock().unwrap(), 1);
220	}

232	struct PanicResponder;
233	impl AnswerQuery for PanicResponder {
234		fn answer(&mut self, query: Query) {
235			panic!("unexpected query: {:?}", query);
236		}

239	#[test]
240	fn effect_message_records_into_recorder() {
241		let mut rec = Recorder::new();
242		let mut resp = PanicResponder;
243		let mut pending = PendingFetches::new();
244		let mut disp = Dispatcher::new(&mut rec, &mut resp, &mut pending);
245		let peer = sc_network_types::PeerId::random();
246		let msg = AllMessages::NetworkBridgeTx(NetworkBridgeTxMessage::DisconnectPeers(
247			vec![peer],
248			PeerSet::Collation,
249		));
250		disp.dispatch(Duration::ZERO, msg);
251		assert_eq!(rec.len(), 1);
252		assert!(matches!(
253			rec.effects().next().unwrap(),
254			Effect::DisconnectPeers { peer_set: PeerSet::Collation, .. }
255		));
256	}

267	#[test]
268	fn query_message_forwards_to_responder() {
269		use polkadot_node_subsystem::messages::ChainApiMessage;
270		let mut rec = Recorder::new();
271		let mut resp = CountingResponder { count: 0 };
272		let mut pending = PendingFetches::new();
273		let mut disp = Dispatcher::new(&mut rec, &mut resp, &mut pending);
274		let (tx, _rx) = futures::channel::oneshot::channel();
275		let msg = AllMessages::ChainApi(ChainApiMessage::FinalizedBlockNumber(tx));
276		disp.dispatch(Duration::ZERO, msg);
277		assert_eq!(rec.len(), 0, "query is not recorded as effect");
278		assert_eq!(resp.count, 1, "query is forwarded to responder");
279	}
