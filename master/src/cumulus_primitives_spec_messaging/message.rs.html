17//! Off-chain wire types and their verifiers.
18//!
19//! The relay chain decodes none of this; it only matches the `StreamsRoot` commitments in the UMP
20//! signals. But the two ends of each protocol are different chains' collators, so the encodings and
21//! the envelope variant indices are part of the wire format.
22//!
23//! - [`leaf_hash`] turns a payload into an MMR leaf. A message is a bounded payload keyed by
24//! `StreamId`; source, stream and position are structural, so there is no message struct.
25//! - Fetch: [`MessagesRequest`] / [`MessagesResponse`], verified by [`verify_messages`] against a
26//! `StreamsRoot` the requester has already verified. The `extension` + `tree_proof` check is the
27//! same one the requires-lift performs.
28//! - Event read: [`EventRequest`] / [`EventResponse`], the single-leaf read of an `Ack` /
29//! `Broadcast` / `Private` stream, verified by [`verify_event`].
30//! - Envelope: [`ExchangeRequest`] / [`ExchangeResponse`] multiplex both over `/spec-msg/exchange`,
31//! verified by [`verify_exchange`], which also binds the response variant to the request's. Variant
32//! indices are frozen.
33

49/// Why verifying an off-chain response failed. [`RootMismatch`](VerifyError::RootMismatch) is
50/// benign (retry under a fresher root); the rest indicate a malformed or forged response.
51#[derive(Clone, Copy, PartialEq, Eq, Debug)]
52pub enum VerifyError {
53	/// An embedded MMR proof (inclusion or extension) did not verify.
54	Proof(ProofError),
55	/// The `StreamsRoot` keyed-tree walk (`tree_proof`) did not resolve.
56	TreeProof,
57	/// Everything verified, but the recomputed `StreamsRoot` is not the requested `under`. The
58	/// peer served under a different root, stale or wrong. Not necessarily malicious.
59	RootMismatch,
60	/// The response's `base` is not the requested `start`. Only surfaced by [`verify_messages`].
61	UnexpectedBase,
62	/// The payloads total more than the request's `max_bytes`, or `max_bytes == 0` and the
63	/// response carries any payload, even an empty one. Only surfaced by [`verify_messages`].
64	ExceedsBudget,
65	/// The response envelope variant does not match the request's. Only surfaced by
66	/// [`verify_exchange`].
67	VariantMismatch,
68	/// A payload exceeded the protocol's single-message bound ([`MAX_SPECULATIVE_MESSAGE_LEN`]).
69	PayloadTooLarge,
70	/// The response is structurally malformed independently of its proofs (e.g. a `start_peaks`
71	/// set inconsistent with `base`, or a `base + len` leaf-counter overflow).
72	MalformedResponse,
73}
74
75impl From<ProofError> for VerifyError {
76	fn from(e: ProofError) -> Self {
77		VerifyError::Proof(e)
78	}
79}

81/// Maximum size in bytes of a single speculative message payload.
82pub const MAX_SPECULATIVE_MESSAGE_LEN: u32 = 102_400;
83
84/// Maximum payloads in one [`MessagesResponse`], enforced at decode. The byte budget does not bound
85/// the count: an empty payload costs one wire byte and no budget, yet is hashed into a leaf, so
86/// without this bound one response could carry millions. A consumer fetches the rest under a later
87/// request.
88pub const MAX_MESSAGES_PER_RESPONSE: u32 = 1024;
89
90/// Hash a payload into a stream MMR leaf: `H(LEAF_TAG ++ leaf_version ++ payload)`. Source, stream
91/// and position are structural and not in the preimage. Versions are hash-disjoint, so only the
92/// correct one reproduces a committed root.
93pub fn leaf_hash(leaf_version: u8, payload: &[u8]) -> Hash {
94	let mut preimage = Vec::with_capacity(2 + payload.len());
95	preimage.push(LEAF_TAG);
96	preimage.push(leaf_version);
97	preimage.extend_from_slice(payload);
98	<SpecHasher as HashT>::hash(&preimage)
99}

101/// Request for a range of a stream's messages, verified under a chosen `StreamsRoot`.
102#[derive(Clone, Encode, Decode, DecodeWithMemTracking, PartialEq, Eq, Debug, TypeInfo)]
103pub struct MessagesRequest {
104	pub stream: StreamId,
105	/// Where to start; typically the receiver's frontier leaf count.
106	pub start: MessagePosition,
107	/// The `StreamsRoot` the response must verify under: one the requester has already
108	/// authenticated. Freshness is the requester's job; re-request under a newer root once
109	/// verified.
110	pub under: StreamsRoot,
111	/// Response size bound; the server may cap harder. `0` requests a payload-free proof (lift
112	/// material).
113	pub max_bytes: u32,
114}
115
116/// Response: payloads from `base` on, plus the proofs binding them and everything before them to
117/// the requested `StreamsRoot`. Nothing is trusted; fabricated peaks, payloads or proofs cannot
118/// reproduce a committed root.
119#[derive(Clone, Encode, Decode, DecodeWithMemTracking, PartialEq, Eq, Debug, TypeInfo)]
120pub struct MessagesResponse {
121	/// Position of the first payload. A hint; a lie fails the proofs.
122	pub base: MessagePosition,
123	/// Leaf-format version for these payloads. A hint; only the correct version reproduces a
124	/// committed root. A response never spans a version change.
125	pub leaf_version: u8,
126	/// The payloads, in MMR order; payload `i` has position `base + i`. At most
127	/// [`MAX_MESSAGES_PER_RESPONSE`], rejected at the length prefix.
128	pub payloads: BoundedVec<Vec<u8>, ConstU32<MAX_MESSAGES_PER_RESPONSE>>,
129	/// The stream's peak set at `base`, for a consumer that holds no frontier. Fabricated peaks
130	/// cannot extend to a committed root.
131	pub start_peaks: BoundedVec<Hash, ConstU32<MAX_MMR_HASHES>>,
132	/// From the frontier recomputed over `payloads` to the stream's current root; the identity
133	/// extension when the payloads already reach it.
134	pub extension: MMRExtensionProof,
135	/// From the stream's root to the `StreamsRoot` the response verifies under.
136	pub tree_proof: StreamProof,
137}
138
139/// Verify a [`MessagesResponse`] against its [`MessagesRequest`]: `resp.base == req.start`, total
140/// payload bytes within `req.max_bytes` (`0` means proofs only), then the payloads under
141/// `req.stream` / `req.under`. On `Ok`, `resp.payloads` are verified. The public entry for the
142/// fetch protocol.
143pub fn verify_messages(req: &MessagesRequest, resp: &MessagesResponse) -> Result<(), VerifyError> {
144	if resp.base != req.start {
145		return Err(VerifyError::UnexpectedBase);
146	}
147	let total = resp.payloads.iter().map(|p| p.len() as u64).fold(0u64, u64::saturating_add);
148	if total > req.max_bytes as u64 || (req.max_bytes == 0 && !resp.payloads.is_empty()) {
149		return Err(VerifyError::ExceedsBudget);
150	}
151	verify_messages_response(req.stream, req.under, resp)
152}
153
154/// Verify a [`MessagesResponse`] for `stream` under `under`: the same check the requires-lift
155/// performs. Crate-internal: it does not bind `resp.base` to a requested start; [`verify_messages`]
156/// does.
157pub(crate) fn verify_messages_response(
158	stream: StreamId,
159	under: StreamsRoot,
160	resp: &MessagesResponse,
161) -> Result<(), VerifyError> {
162	// `base` and `start_peaks` are untrusted; `from_parts` rejects an inconsistent shape and a
163	// `base` past the ceiling.
164	let mut frontier = MmrFrontier::from_parts(resp.start_peaks.to_vec(), resp.base.0)
165		.ok_or(VerifyError::MalformedResponse)?;
166	// The frontier after the payloads must stay within `MAX_MMR_LEAF_COUNT`, or it would not
167	// decode. Checked up front; `try_append` refuses it too.
168	resp.base
169		.0
170		.checked_add(resp.payloads.len() as u64)
171		.filter(|end| *end <= MAX_MMR_LEAF_COUNT)
172		.ok_or(VerifyError::MalformedResponse)?;
173	// Defense-in-depth against a crafted response inflating the hashed input; the transport also
174	// caps total response size.
175	resp.payloads.iter().try_for_each(|p| check_payload_len(p))?;
176
177	for payload in &resp.payloads {
178		frontier
179			.try_append(leaf_hash(resp.leaf_version, payload))
180			.map_err(|_| VerifyError::MalformedResponse)?;
181	}
182
183	let current = resp.extension.calculate_root(&frontier)?;
184	bind_under(stream, under, current, &resp.tree_proof)
185}
186
187fn check_payload_len(payload: &[u8]) -> Result<(), VerifyError> {
188	if payload.len() as u64 > MAX_SPECULATIVE_MESSAGE_LEN as u64 {
189		return Err(VerifyError::PayloadTooLarge);
190	}
191	Ok(())
192}

194/// Walk `tree_proof` from `stream_root` to a `StreamsRoot` and require it to be `under`:
195/// [`TreeProof`](VerifyError::TreeProof) if the walk is malformed,
196/// [`RootMismatch`](VerifyError::RootMismatch) if it ends elsewhere.
197fn bind_under(
198	stream: StreamId,
199	under: StreamsRoot,
200	stream_root: MmrRoot,
201	tree_proof: &StreamProof,
202) -> Result<(), VerifyError> {
203	let root =
204		streams_root_from_proof(stream, stream_root.0, tree_proof).ok_or(VerifyError::TreeProof)?;
205	(root == under).then_some(()).ok_or(VerifyError::RootMismatch)
206}
207
208fn event_leaf(resp: &EventResponse) -> Result<Hash, VerifyError> {
209	check_payload_len(&resp.payload)?;
210	Ok(leaf_hash(resp.leaf_version, &resp.payload))
211}

213/// Request for a single event of a lossy stream (`Ack` / `Broadcast` / `Private`), verified under a
214/// chosen `StreamsRoot`. The server either serves or fails; nothing is resolved server-side.
215#[derive(Clone, Encode, Decode, DecodeWithMemTracking, PartialEq, Eq, Debug, TypeInfo)]
216pub struct EventRequest {
217	pub stream: StreamId,
218	/// The `StreamsRoot` the response must verify under. Always explicit.
219	pub under: StreamsRoot,
220	/// A specific position, or `None` for the head as of `under`.
221	pub at: Option<MessagePosition>,
222}
223
224/// Response carrying one event leaf and the proofs placing it under the requested `StreamsRoot`.
225/// Nothing is trusted. `under` fixes the stream's leaf count, so the head is the leaf at `count -
226/// 1`; an old leaf cannot be served as the head.
227#[derive(Clone, Encode, Decode, DecodeWithMemTracking, PartialEq, Eq, Debug, TypeInfo)]
228pub struct EventResponse {
229	/// The event payload; its position is proven by `inclusion`.
230	pub payload: Vec<u8>,
231	/// Leaf-format version for `payload`. A hint; only the correct version reproduces a committed
232	/// root.
233	pub leaf_version: u8,
234	/// Single-leaf inclusion proof: the sibling path plus the other peaks.
235	pub inclusion: MmrInclusionProof,
236	/// From the stream's root (derived from `inclusion`) to the `StreamsRoot` the response
237	/// verifies under.
238	pub tree_proof: StreamProof,
239}

241/// The `/spec-msg/exchange` request envelope. Variant indices are frozen; foreign implementations
242/// decode by them.
243#[derive(Clone, Encode, Decode, DecodeWithMemTracking, PartialEq, Eq, Debug, TypeInfo)]
244pub enum ExchangeRequest {
245	/// Ordered fetching or lift material. Answered by [`ExchangeResponse::Messages`].
246	#[codec(index = 0)]
247	Messages(MessagesRequest),
248	/// Single-event read. Answered by [`ExchangeResponse::Event`].
249	#[codec(index = 1)]
250	Event(EventRequest),
251}
252
253/// The `/spec-msg/exchange` response envelope. No error variant: a server that cannot serve refuses
254/// at the transport level.
255#[derive(Clone, Encode, Decode, DecodeWithMemTracking, PartialEq, Eq, Debug, TypeInfo)]
256pub enum ExchangeResponse {
257	/// Response to [`ExchangeRequest::Messages`].
258	#[codec(index = 0)]
259	Messages(MessagesResponse),
260	/// Response to [`ExchangeRequest::Event`].
261	#[codec(index = 1)]
262	Event(EventResponse),
263}
264
265/// Outcome of verifying an [`EventResponse`]. A head read yields the stream [`MmrFrontier`], which
266/// lossy consumption records as its interval endpoint; a positional read yields none, since an
267/// inclusion proof reconstructs a root, not a frontier.
268#[derive(Clone, PartialEq, Eq, Debug)]
269pub enum VerifiedEvent {
270	/// Head (register / latest-event) read: the head position and the stream frontier at that
271	/// head.
272	Head { position: MessagePosition, frontier: MmrFrontier },
273	/// Positional read: the requested position.
274	Positional { position: MessagePosition },
275}
276
277/// Verify an [`EventResponse`] against its [`EventRequest`]: `req.at` selects head (`None`) or leaf
278/// `k`, so a head response is never accepted for a positional request or vice versa. On `Ok`,
279/// `resp.payload` is verified. The public entry for event reads.
280pub fn verify_event(
281	req: &EventRequest,
282	resp: &EventResponse,
283) -> Result<VerifiedEvent, VerifyError> {
284	match req.at {
285		None => {
286			let (position, frontier) = verify_event_response(req.stream, req.under, resp)?;
287			Ok(VerifiedEvent::Head { position, frontier })
288		},
289		Some(position) => {
290			verify_positional_event_response(req.stream, req.under, position, resp)?;
291			Ok(VerifiedEvent::Positional { position })
292		},
293	}
294}
295
296/// Verify an [`EventResponse`] as the head of `stream` under `under`; returns the position and the
297/// frontier at that head. Crate-internal; [`verify_event`] is the public entry.
298///
299/// The frontier is returned because lossy consumption needs it as the interval endpoint.
300pub(crate) fn verify_event_response(
301	stream: StreamId,
302	under: StreamsRoot,
303	resp: &EventResponse,
304) -> Result<(MessagePosition, MmrFrontier), VerifyError> {
305	let (position, frontier) = resp.inclusion.verify_head(event_leaf(resp)?)?;
306	bind_under(stream, under, frontier.root(), &resp.tree_proof)?;
307	Ok((position, frontier))
308}
309
310/// Verify an [`EventResponse`] as the leaf at `position` of `stream` under `under`.
311/// Crate-internal; [`verify_event`] is the public entry. Same chain as the head read, with
312/// [`MmrInclusionProof::verify_leaf`].
313pub(crate) fn verify_positional_event_response(
314	stream: StreamId,
315	under: StreamsRoot,
316	position: MessagePosition,
317	resp: &EventResponse,
318) -> Result<(), VerifyError> {
319	let root = resp.inclusion.verify_leaf(position, event_leaf(resp)?)?;
320	bind_under(stream, under, root, &resp.tree_proof)
321}
322
323/// Verified outcome of an [`ExchangeResponse`]: a fetch, or a [`VerifiedEvent`].
324#[derive(Clone, PartialEq, Eq, Debug)]
325pub enum ExchangeVerified {
326	/// Answer to an [`ExchangeRequest::Messages`]; the response's payloads are verified.
327	Messages,
328	/// Answer to an [`ExchangeRequest::Event`]: the verified event.
329	Event(VerifiedEvent),
330}
331
332/// Verify a `/spec-msg/exchange` response against its request. The variants must match
333/// ([`VariantMismatch`](VerifyError::VariantMismatch) otherwise); then dispatches to
334/// [`verify_messages`] or [`verify_event`]. The top-level public entry.
335pub fn verify_exchange(
336	req: &ExchangeRequest,
337	resp: &ExchangeResponse,
338) -> Result<ExchangeVerified, VerifyError> {
339	match (req, resp) {
340		(ExchangeRequest::Messages(mreq), ExchangeResponse::Messages(mresp)) => {
341			verify_messages(mreq, mresp).map(|()| ExchangeVerified::Messages)
342		},
343		(ExchangeRequest::Event(ereq), ExchangeResponse::Event(eresp)) => {
344			verify_event(ereq, eresp).map(ExchangeVerified::Event)
345		},
346		_ => Err(VerifyError::VariantMismatch),
347	}
348}

372	/// The stream root of an `mmr_lib` MMR: its bag bound to the leaf count.
373	fn bound(mmr_size: u64, bag: Hash) -> Hash {
374		MmrRoot::from_bag(get_peak_map(mmr_size), bag).0

377	/// The stream root over the first `n` leaves.
378	fn current_root(leaves: &[Hash], n: usize) -> Hash {
379		let store = MemStore::<Hash>::default();
380		let mut mmr = MemMMR::<Hash, SpecMerge>::new(0, &store);
381		for l in &leaves[..n] {
382			mmr.push(*l).unwrap();
383		}
384		bound(mmr.mmr_size(), mmr.get_root().unwrap())
385	}

389		let store = MemStore::<Hash>::default();

1033			core::array::from_fn(|s| payloads[s].iter().map(|p| leaf_hash(0, p)).collect());
1034		let entries: BTreeMap<_, _> =
1035			(0..2).map(|s| (ids[s], frontier_at(&leaves[s], 6).root().0)).collect();
1036		let under = streams_root(&entries).unwrap();
1037		let streams = core::array::from_fn(|s| {
1038			let (_, proof) = gen_stream_proof(&entries, ids[s]).unwrap();
1039			(ids[s], payloads[s].clone(), leaves[s].clone(), proof)
1040		});
1041		TwoStreams { under, streams }
1042	}

1044	/// The honest fetch of `payloads[base..end]` with a real extension from `end` to 6.
1045	fn fetch(
1046		(_, payloads, leaves, tree_proof): &(StreamId, Vec<Vec<u8>>, Vec<Hash>, StreamProof),
1047		base: usize,
1048		end: usize,
1049	) -> MessagesResponse {
1050		MessagesResponse {
1051			base: MessagePosition(base as u64),
1052			leaf_version: 0,
1053			payloads: payloads[base..end].to_vec().try_into().unwrap(),
1054			start_peaks: BoundedVec::truncate_from(frontier_at(leaves, base).peaks().to_vec()),
1055			extension: ancestry(leaves, end, 6),
1056			tree_proof: tree_proof.clone(),
1057		}
1058	}

1060	/// The honest head read of a 6-leaf stream.
1061	fn head(
1062		(_, payloads, leaves, tree_proof): &(StreamId, Vec<Vec<u8>>, Vec<Hash>, StreamProof),
1063	) -> EventResponse {
1064		let store = MemStore::<Hash>::default();
1065		let mut mmr = MemMMR::<Hash, SpecMerge>::new(0, &store);
1066		let positions: Vec<u64> = leaves.iter().map(|l| mmr.push(*l).unwrap()).collect();
1067		let proof = mmr.gen_proof(vec![positions[5]]).unwrap();
1068		EventResponse {
1069			payload: payloads[5].clone(),
1070			leaf_version: 0,
1071			inclusion: MmrInclusionProof {
1072				mmr_size: mmr_size(6),
1073				items: BoundedVec::truncate_from(proof.proof_items().to_vec()),
1074			},
1075			tree_proof: tree_proof.clone(),
1076		}
1077	}
