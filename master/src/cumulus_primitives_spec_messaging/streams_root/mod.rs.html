17//! `StreamsRoot`: the sender's stream commitment.
18//!
19//! Each block that sends commits one hash, the root of a keyed binary Patricia trie over
20//! `(StreamId, current stream root)` for every active stream. The relay records it as the sender's
21//! `Provides`, and a receiver proves the streams it consumed against it, without a sender-state
22//! proof.
23//!
24//! The key drives the walk, so a proof cannot be moved to another stream
25//! ([`streams_root_from_proof`]).
26//!
27//! Roles: the sender builds the root ([`streams_root`]) and its node serves proofs
28//! ([`gen_stream_proof`]), checking its own header digest ([`read_streams_root`]); a receiver's
29//! lift walks a proof up to the committed root ([`streams_root_from_proof`]).
30

48/// Bit width of a `StreamId` key (the Patricia trie's maximum depth).
49const KEY_BITS: usize = STREAM_ID_LEN * 8;
50
51/// The sender's per-block stream commitment root. Defined relay-side and re-exported here.
52pub use polkadot_primitives::StreamsRoot;
53
54/// One step of a trie walk. The branch direction is not stored.
55#[derive(
56	Clone,
57	Copy,
58	Encode,
59	Decode,
60	DecodeWithMemTracking,
61	MaxEncodedLen,
62	PartialEq,
63	Eq,
64	Debug,
65	scale_info::TypeInfo,
66)]
67pub struct TreeStep {
68	/// Key-bit index this node branches on, `0..KEY_BITS` with `0` the MSB. Part of the inner-node
69	/// hash; the proven key's bit here gives the direction.
70	pub split_bit: u8,
71	pub sibling: Hash,
72}
73
74/// A membership proof that `(stream, stream_root)` is committed under a `StreamsRoot`. Served by
75/// the sender, verified by the receiver.
76#[derive(
77	Clone,
78	Encode,
79	Decode,
80	DecodeWithMemTracking,
81	MaxEncodedLen,
82	PartialEq,
83	Eq,
84	Debug,
85	scale_info::TypeInfo,
86)]
87pub struct StreamProof {
88	/// Branch steps leaf to root, split bits strictly decreasing. Empty for a single-stream tree.
89	/// Bounded at the trie depth, so an over-long proof is rejected at decode.
90	pub steps: BoundedVec<TreeStep, ConstU32<{ KEY_BITS as u32 }>>,
91}
92
93/// Bit `i` of an 8-byte key, `0` the MSB of byte 0.
94fn bit(key: &[u8; STREAM_ID_LEN], i: usize) -> bool {
95	(key[i >> 3] >> (7 - (i & 7))) & 1 == 1
96}
97
98/// First bit index (from the MSB, 0..64) at which two keys differ; returns 64 if identical.
99fn first_diverging_bit(a: &[u8; STREAM_ID_LEN], b: &[u8; STREAM_ID_LEN]) -> usize {
100	(u64::from_be_bytes(*a) ^ u64::from_be_bytes(*b)).leading_zeros() as usize
101}
102
103fn hash_leaf(key: &[u8; STREAM_ID_LEN], stream_root: &Hash) -> Hash {
104	let mut pre = [0u8; 1 + STREAM_ID_LEN + 32];
105	pre[0] = TREE_LEAF_TAG;
106	pre[1..1 + STREAM_ID_LEN].copy_from_slice(key);
107	pre[1 + STREAM_ID_LEN..].copy_from_slice(stream_root.as_bytes());
108	<SpecHasher as HashT>::hash(&pre)
109}
110
111fn hash_inner(split_bit: u8, left: &Hash, right: &Hash) -> Hash {
112	let mut pre = [0u8; 1 + 1 + 32 + 32];
113	pre[0] = TREE_INNER_TAG;
114	pre[1] = split_bit;
115	pre[2..34].copy_from_slice(left.as_bytes());
116	pre[34..].copy_from_slice(right.as_bytes());
117	<SpecHasher as HashT>::hash(&pre)
118}
119
120/// Split a key-sorted slice at its top-level branch: the bit where its first and last keys diverge,
121/// with all `bit == 0` keys before all `bit == 1` keys. Both sides are non-empty. Callers guarantee
122/// `entries.len() >= 2` and distinct keys.
123fn split<'a>(
124	entries: &'a [([u8; STREAM_ID_LEN], Hash)],
125) -> (usize, &'a [([u8; STREAM_ID_LEN], Hash)], &'a [([u8; STREAM_ID_LEN], Hash)]) {
126	let split_bit = first_diverging_bit(&entries[0].0, &entries[entries.len() - 1].0);
127	let p = entries.partition_point(|(k, _)| !bit(k, split_bit));
128	let (left, right) = entries.split_at(p);
129	(split_bit, left, right)
130}
131
132/// Trie root over a key-sorted, non-empty slice.
133fn node_hash(entries: &[([u8; STREAM_ID_LEN], Hash)]) -> Hash {
134	if let [(key, root)] = entries {
135		return hash_leaf(key, root);
136	}
137	let (split_bit, left, right) = split(entries);
138	hash_inner(split_bit as u8, &node_hash(left), &node_hash(right))
139}
140
141/// Encoded keys in trie order.
142///
143/// The trie splits on bytes, so sort by them rather than rely on `Ord`. Unsorted input would leave
144/// one side of `split` empty and make `prove` recurse on its own slice.
145fn keyed(entries: &BTreeMap<StreamId, Hash>) -> Vec<([u8; STREAM_ID_LEN], Hash)> {
146	let mut keyed: Vec<_> =
147		entries.iter().map(|(stream, root)| (stream.to_bytes(), *root)).collect();
148	keyed.sort_unstable_by_key(|(key, _)| *key);
149	keyed
150}
151
152/// The `StreamsRoot` over `(stream, stream_root)` entries; `None` with no active streams. Sender
153/// side. Takes a `BTreeMap` so keys are unique; duplicates would reach `first_diverging_bit` with
154/// identical keys.
155///
156/// For incremental updates in `O(log S)` without rebuilding from scratch in `O(S)`, see
157/// [`update_tree_leaf`] and [`StreamsTree`].
158pub fn streams_root(entries: &BTreeMap<StreamId, Hash>) -> Option<StreamsRoot> {
159	let entries = keyed(entries);
160	if entries.is_empty() {
161		return None;
162	}
163	Some(StreamsRoot(node_hash(&entries)))
164}
165
166/// Walk key-sorted `entries` toward `key`, pushing each branch step root-first, and return the
167/// subtree's root. `key` must be present.
168fn prove(
169	entries: &[([u8; STREAM_ID_LEN], Hash)],
170	key: &[u8; STREAM_ID_LEN],
171	steps: &mut Vec<TreeStep>,
172) -> Hash {
173	if let [(k, root)] = entries {
174		return hash_leaf(k, root);
175	}
176	let (split_bit, left, right) = split(entries);
177	let target_right = bit(key, split_bit);
178	let (sibling, descend) =
179		if target_right { (node_hash(left), right) } else { (node_hash(right), left) };
180	steps.push(TreeStep { split_bit: split_bit as u8, sibling });
181	let child = prove(descend, key, steps);
182	if target_right {
183		hash_inner(split_bit as u8, &sibling, &child)
184	} else {
185		hash_inner(split_bit as u8, &child, &sibling)
186	}
187}
188
189/// Build the trie over `entries` and return `(StreamsRoot, proof)` for `stream`, or `None` if
190/// absent. Sender side.
191pub fn gen_stream_proof(
192	entries: &BTreeMap<StreamId, Hash>,
193	stream: StreamId,
194) -> Option<(StreamsRoot, StreamProof)> {
195	if !entries.contains_key(&stream) {
196		return None;
197	}
198	let entries = keyed(entries);
199	let key = stream.to_bytes();
200	let mut steps = Vec::new();
201	let root = prove(&entries, &key, &mut steps);
202	// `prove` pushes root-first; the wire order is leaf-to-root.
203	steps.reverse();
204	let steps = BoundedVec::try_from(steps).expect("trie depth <= KEY_BITS; qed");
205	Some((StreamsRoot(root), StreamProof { steps }))
206}

208/// Read the sender's `StreamsRoot` from its header digest. `None` if absent or not exactly what the
209/// sender deposits: one `SPMS` item with a 32-byte payload. Foreign readers must agree on
210/// validity, so nothing else is accepted.
211pub fn read_streams_root(digest: &Digest) -> Option<StreamsRoot> {
212	let mut items = digest.logs().iter().filter_map(|item| match item {
213		DigestItem::Consensus(id, payload) if *id == SPMS_ENGINE_ID => Some(payload),
214		_ => None,
215	});
216	let payload = items.next()?;
217	if items.next().is_some() {
218		return None;
219	}
220	StreamsRoot::decode_all(&mut &payload[..]).ok()
221}
222
223/// Fold a membership proof for `(stream, stream_root)` up to the `StreamsRoot` it implies, or
224/// `None` on a malformed step. Callers compare the root themselves (the requires-lift records it,
225/// the fetch checks it against `under`). Each branch direction is `stream`'s bit at `split_bit`, so
226/// a proof built for another stream yields a different root.
227pub fn streams_root_from_proof(
228	stream: StreamId,
229	stream_root: Hash,
230	membership: &StreamProof,
231) -> Option<StreamsRoot> {
232	let key = stream.to_bytes();
233	let mut node = hash_leaf(&key, &stream_root);
234	// Steps are leaf-to-root with strictly decreasing split bits; anything else is rejected.
235	// Uniqueness rests on the key-in-leaf and bit-in-inner commitments, not on this rule. The
236	// length bound is in the type.
237	let mut prev: Option<u8> = None;
238	for step in membership.steps.iter() {
239		let split_bit = step.split_bit as usize;
240		// An out-of-range branch would index the key out of bounds.
241		if split_bit >= KEY_BITS {
242			return None;
243		}
244		if prev.is_some_and(|p| step.split_bit >= p) {
245			return None;
246		}
247		prev = Some(step.split_bit);
248		node = if bit(&key, split_bit) {
249			hash_inner(step.split_bit, &step.sibling, &node)
250		} else {
251			hash_inner(step.split_bit, &node, &step.sibling)
252		};
253	}
254	Some(StreamsRoot(node))
255}

530	fn a_child_off_the_walked_side_is_corrupt() {
531		// The root holds a node on one side and a leaf on the other.
532		let streams = [ch(1000), ch(1001), ch(3000)];
533		let (mut entry, mut store) = (None, BTreeMap::new());
534		for stream in streams {
535			update_tree_leaf(&mut entry, stream, root(0x1), &mut store).unwrap();
536		}
537		let Some(TreeRootEntry::Node { key, .. }) = entry else { panic!("three streams") };
538		let node = store[&key];
539		assert!(
540			matches!(
541				(node.left_child, node.right_child),
542				(TreeChild::Node { .. }, TreeChild::Leaf { .. }) |
543					(TreeChild::Leaf { .. }, TreeChild::Node { .. })
544			),
545			"a node and a leaf under the root"
546		);
547
548		// Swap the root's children: each is now on the side the other's keys walk.
549		let mut swapped = store.clone();
550		swapped.insert(
551			key,
552			InnerNode {
553				left_child: node.right_child,
554				left_hash: node.right_hash,
555				right_child: node.left_child,
556				right_hash: node.left_hash,
557				..node
558			},
559		);
560		for stream in streams {
561			let mut e = entry;
562			assert!(
563				matches!(
564					update_tree_leaf(&mut e, stream, root(0x2), &mut swapped.clone()),
565					Err(TreeError::CorruptNode(_))
566				),
567				"{stream:?}"
568			);
569			assert_eq!(gen_stream_proof_from_nodes(&entry, stream, &swapped), None, "{stream:?}");
570		}
571	}

586	#[test]
587	fn differential_fuzz_random_tree_insertions_and_updates() {
588		let mut seed = 0x85EB_CA6B_472C_1A55u64;
589		let mut rng = || {
590			seed ^= seed << 13;
591			seed ^= seed >> 7;
592			seed ^= seed << 17;
593			seed
594		};
595
596		let mut map = BTreeMap::new();
597		let mut tree = StreamsTree::new();
598
599		// Narrow field ranges give keys that split inside the kind, recipient and domain bytes.
600		let random_stream = |x: u64| match x % 3 {
601			0 => StreamId::Channel {
602				recipient: (((x >> 8) % 64) as u32).into(),
603				domain: (x >> 16) as u8 % 2,
604				num: (x >> 24) as u16 % 4,
605			},
606			1 => StreamId::Ack {
607				recipient: (((x >> 8) % 64) as u32).into(),
608				domain: (x >> 16) as u8 % 2,
609				num: (x >> 24) as u16 % 4,
610			},
611			_ => StreamId::Broadcast {
612				domain: (x >> 8) as u16 % 4,
613				subdomain: (x >> 16) as u8 % 2,
614				num: (x >> 24) as u32 % 64,
615			},
616		};
617
618		// Insert 200 streams with pseudo-random keys and values
619		for _ in 0..200 {
620			let stream = random_stream(rng());
621			let r = Hash::from_low_u64_be(rng());
622			map.insert(stream, r);
623			let tree_root = tree.insert(stream, r).unwrap();
624			assert_eq!(Some(tree_root), streams_root(&map));
625		}
626
627		// Mutate 100 random streams
628		let keys: Vec<_> = map.keys().copied().collect();
629		for _ in 0..100 {
630			let idx = (rng() as usize) % keys.len();
631			let stream = keys[idx];
632			let new_r = Hash::from_low_u64_be(rng());
633			map.insert(stream, new_r);
634			let tree_root = tree.insert(stream, new_r).unwrap();
635			assert_eq!(Some(tree_root), streams_root(&map));
636		}
637
638		// Verify every proof
639		for (stream, sr) in &map {
640			let (r_tree, proof_tree) = tree.gen_stream_proof(*stream).unwrap();
641			let (r_scratch, proof_scratch) = gen_stream_proof(&map, *stream).unwrap();
642			assert_eq!(r_tree, r_scratch);
643			assert_eq!(proof_tree, proof_scratch);
644			assert!(verify_stream_membership(r_tree, *stream, *sr, &proof_tree));
645		}
646
647		// The cached layout does not depend on insertion order.
648		let mut reversed = StreamsTree::new();
649		for (stream, sr) in map.iter().rev() {
650			reversed.insert(*stream, *sr).unwrap();
651		}
652		assert_eq!(reversed, tree);
653	}
