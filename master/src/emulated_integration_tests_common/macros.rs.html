225						<$sender_para as $crate::macros::Chain>::account_data_of(sender.clone()).free;
226					let para_receiver_balance_after =
227						<$receiver_para as $crate::macros::Chain>::account_data_of(receiver.clone()).free;

229					assert_eq!(para_sender_balance_before - $amount - delivery_fees_amount, para_sender_balance_after);
230					assert!(para_receiver_balance_after > para_receiver_balance_before);
231
232					// Update sender balance
233					para_sender_balance_before = <$sender_para as $crate::macros::Chain>::account_data_of(sender.clone()).free;
234				}
235			)+
236		}
237	};
