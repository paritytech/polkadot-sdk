19//! BlockImport wrapper that bitswap-fetches missing TRANSACTION-column entries before
20//! delegating to the inner block import. Tip-sync and body-carrying gap-sync only; warp-sync and
21//! header-only gap-sync pass through.
22//!
23//! [`StorageChainBlockImport::import_block`] dispatches a tip-sync block to one of three paths
24//! based on what the consensus engine has already done:
25//!
26//! 1. The import params already carry executed `StorageChanges`. The wrapper queries the runtime
27//!    API against the post-execution overlay to discover the renew set.
28//! 2. No attached changes. The wrapper executes the block via the runtime API once, then installs
29//!    the resulting `StorageChanges` into the import params so the inner block-import does not
30//!    re-execute.
31//! 3. Body-carrying gap-sync. The renew set is derived from the body itself by tail-hashing against
32//!    runtime-declared metadata, not via runtime execution.

64/// Errors produced by [`StorageChainBlockImport`].
65#[derive(Debug, thiserror::Error)]
66pub enum Error {
67	#[error("call to `TransactionStorageApi::indexed_transactions` runtime API failed: {0}")]
68	IndexedTransactions(sp_api::ApiError),
69	#[error("execution of block {block} via `Core::execute_block` runtime API failed: {error}")]
70	ExecuteBlock { block: String, error: sp_api::ApiError },
71	#[error("failed to access parent state {0} for block execution: {1}")]
72	StateAt(String, sp_api::ApiError),
73	#[error("failed to convert the overlay into storage changes after block execution: {0}")]
74	IntoStorageChanges(String),
75	#[error(
76		"state root after block execution does not match the header: \
77		 header claims {header}, execution produced {executed}"
78	)]
79	StateRootMismatch { header: String, executed: String },
80	#[error("gap-sync block carries no body, cannot restore its indexed transactions")]
81	GapSyncBodyMissing,
82	#[error(
83		"runtime-declared indexed-transaction metadata references extrinsic index {index}, \
84		 but the block body only contains {body_len} extrinsics"
85	)]
86	InvalidExtrinsicIndex { index: u32, body_len: usize },
87	#[error("renew operation from {context} carries a content hash that is not 32 bytes")]
88	MalformedRenewHash { context: &'static str },
89	#[error(
90		"runtime API returned no indexed-transaction metadata for the renewed content \
91		 hash {hash:?} (discovered via {context})"
92	)]
93	MissingRenewMetadata { context: &'static str, hash: ContentHash },
94	#[error("{missing} of {total} indexed transactions could not be fetched via bitswap")]
95	IncompleteFetch { missing: usize, total: usize },
96	#[error("bitswap fetch failed: {0}")]
97	Fetch(#[from] FetchError),
98}

101	fn from(e: Error) -> Self {
102		ConsensusError::Other(Box::new(e))
103	}

106/// A runtime-declared indexed-transaction entry that needs to be bitswap-fetched.
107#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
108pub(crate) struct RenewWant {
109	/// Hash of the indexed data.
110	pub hash: ContentHash,
111	/// Algorithm used to compute `hash`.
112	pub hashing: HashingAlgorithm,
113	/// CID codec.
114	pub cid_codec: u64,
115}
116
117impl From<RenewWant> for Cid {
118	fn from(want: RenewWant) -> Self {
119		let multihash = Multihash::<64>::wrap(want.hashing.multihash_code(), &want.hash)
120			.expect("a 32-byte content hash always fits into a 64-byte multihash");
121		Cid::new_v1(want.cid_codec, multihash)
122	}
123}

151#[async_trait::async_trait]
152impl<Block, Inner, Client> BlockImport<Block> for StorageChainBlockImport<Block, Inner, Client>
153where
154	Block: BlockT<Hash = sc_client_db::DbHash>,
155	Inner: BlockImport<Block, Error = ConsensusError> + Send + Sync,
156	Client: ProvideRuntimeApi<Block>
157		+ BlockBackend<Block>
158		+ HeaderBackend<Block>
159		+ CallApiAt<Block>
160		+ Send
161		+ Sync,
162	Client::Api: TransactionStorageApi<Block> + Core<Block> + ApiExt<Block>,
163{
164	type Error = ConsensusError;
165
166	async fn check_block(
167		&self,
168		block: BlockCheckParams<Block>,
169	) -> Result<ImportResult, Self::Error> {
170		self.inner.check_block(block).await
171	}
172
173	async fn import_block(
174		&self,
175		params: BlockImportParams<Block>,
176	) -> Result<ImportResult, Self::Error> {
177		if !self.should_intercept(&params) {
178			return self.inner.import_block(params).await;
179		}
180
181		if matches!(params.origin, BlockOrigin::GapSync) {
182			return self.import_gap_sync_block(params).await;
183		}
184
185		if params.state_action.as_storage_changes().is_some() {
186			self.import_with_attached_changes(params).await
187		} else {
188			self.import_by_executing_block(params).await
189		}
190	}
191}

193impl<Block, Inner, Client> StorageChainBlockImport<Block, Inner, Client>
194where
195	Block: BlockT<Hash = sc_client_db::DbHash>,
196	Inner: BlockImport<Block, Error = ConsensusError> + Send + Sync,
197	Client: ProvideRuntimeApi<Block>
198		+ BlockBackend<Block>
199		+ HeaderBackend<Block>
200		+ CallApiAt<Block>
201		+ Send
202		+ Sync,
203	Client::Api: TransactionStorageApi<Block> + Core<Block> + ApiExt<Block>,
204{
205	/// True if the block needs bitswap prefetch. Body must be present and the runtime
206	/// must expose `TransactionStorageApi v2+`.
207	fn should_intercept(&self, params: &BlockImportParams<Block>) -> bool {
208		if params.body.is_none() {
209			return false;
210		}
211		match params.origin {
212			BlockOrigin::NetworkInitialSync |
213			BlockOrigin::NetworkBroadcast |
214			BlockOrigin::ConsensusBroadcast |
215			BlockOrigin::Own |
216			BlockOrigin::GapSync => {},
217			BlockOrigin::Genesis | BlockOrigin::File | BlockOrigin::WarpSync => return false,
218		}
219		// Gap-sync parents don't have state, so we call at the finalized hash.
220		let version_state = if matches!(params.origin, BlockOrigin::GapSync) {
221			self.client.info().finalized_hash
222		} else {
223			*params.header.parent_hash()
224		};
225		self.client
226			.runtime_api()
227			.has_api_with::<dyn TransactionStorageApi<Block>, _>(version_state, |v| v >= 2)
228			.unwrap_or(false)
229	}
230
231	/// Import path for blocks whose `BlockImportParams::state_action` already carries
232	/// the executed `StorageChanges` (consensus executed the block before handing it to
233	/// the import queue). Reads the renew set from the attached `transaction_index_changes`
234	/// using the runtime API queried against the post-execution overlay.
235	async fn import_with_attached_changes(
236		&self,
237		params: BlockImportParams<Block>,
238	) -> Result<ImportResult, ConsensusError> {
239		let parent_hash = *params.header.parent_hash();
240		let block_number = *params.header.number();
241
242		let changes = params
243			.state_action
244			.as_storage_changes()
245			.expect("dispatcher gates on as_storage_changes().is_some(); qed");
246		let infos =
247			self.indexed_transactions_with_storage_changes(parent_hash, block_number, changes)?;
248		let renews = verified_renews_from_index_ops(
249			&changes.transaction_index_changes,
250			&infos,
251			"runtime-api overlay",
252		)?;
253		if !renews.is_empty() {
254			log::debug!(
255				target: LOG_TARGET,
256				"block #{block_number:?} ({parent_hash:?}): runtime-API overlay path, \
257				 {} indexed entries, {} renew hashes",
258				infos.len(),
259				renews.len(),
260			);
261		}
262
263		self.apply_prefetch_and_forward(params, renews).await
264	}
265
266	/// Import path for blocks without attached `StorageChanges`. Executes the block via
267	/// the runtime API to discover the renew set, then installs the resulting
268	/// `StorageChanges` into `params.state_action` so the inner block-import does not
269	/// re-execute.
270	async fn import_by_executing_block(
271		&self,
272		mut params: BlockImportParams<Block>,
273	) -> Result<ImportResult, ConsensusError> {
274		let parent_hash = *params.header.parent_hash();
275		let block_number = *params.header.number();
276
277		let (gen_storage_changes, infos) = self.execute_block(&params)?;
278		let renews = verified_renews_from_index_ops(
279			&gen_storage_changes.transaction_index_changes,
280			&infos,
281			"block execution",
282		)?;
283		if !renews.is_empty() {
284			log::debug!(
285				target: LOG_TARGET,
286				"block #{block_number:?} ({parent_hash:?}): block-execution path, \
287				 {} indexed entries, {} renew hashes",
288				infos.len(),
289				renews.len(),
290			);
291		}
292
293		params.state_action =
294			StateAction::ApplyChanges(ConsensusStorageChanges::Changes(gen_storage_changes));
295
296		self.apply_prefetch_and_forward(params, renews).await
297	}

299	/// Shared tail of the tip-block import paths: drop entries already on disk,
300	/// bitswap-fetch the rest, attach them to `params`, and forward to the inner
301	/// block-import.
302	async fn apply_prefetch_and_forward(
303		&self,
304		mut params: BlockImportParams<Block>,
305		renews: HashSet<RenewWant>,
306	) -> Result<ImportResult, ConsensusError> {
307		let missing = self.filter_missing(renews);
308		let payload = self.fetch_all(missing).await?;
309		Self::attach_prefetched(&mut params, payload);
310		self.inner.import_block(params).await
311	}
312
313	/// Gap-sync import path: classifies body extrinsics into synthetic
314	/// `IndexOperation::Insert`/`Renew` ops via tail-hashing against runtime-declared
315	/// metadata, bitswap-fetches missing renew payloads, and attaches both to
316	/// `PrefetchedIndexedTransactions` so the backend can populate the `TRANSACTION`
317	/// column without runtime execution.
318	async fn import_gap_sync_block(
319		&self,
320		mut params: BlockImportParams<Block>,
321	) -> Result<ImportResult, ConsensusError> {
322		let parent_hash = *params.header.parent_hash();
323		let block_number = *params.header.number();
324		let finalized_hash = self.client.info().finalized_hash;
325
326		let infos = self
327			.client
328			.runtime_api()
329			.indexed_transactions(finalized_hash, block_number)
330			.map_err(Error::IndexedTransactions)?;
331		let infos_len = infos.len();
332		let body = params.body.as_ref().ok_or(Error::GapSyncBodyMissing)?;
333		let classified = classify_body::<Block>(&infos, body)?;
334
335		let missing = self.filter_missing(classified.renews);
336		let payload = self.fetch_all(missing).await?;
337
338		if !classified.ops.is_empty() || !payload.is_empty() {
339			log::debug!(
340				target: LOG_TARGET,
341				"gap-sync block #{block_number:?} ({parent_hash:?}, finalized={finalized_hash:?}): \
342				 {infos_len} indexed entries, {} synthetic ops, {} renew payloads",
343				classified.ops.len(),
344				payload.len(),
345			);
346		}
347
348		params.prefetched_indexed_transactions = PrefetchedIndexedTransactions {
349			ops: classified.ops,
350			renew_payloads: payload.into_iter().map(|(h, b)| (sp_core::H256::from(h), b)).collect(),
351		};
352
353		self.inner.import_block(params).await
354	}

356	/// Drops every entry whose data is already in the local TRANSACTION column.
357	fn filter_missing(&self, renews: HashSet<RenewWant>) -> HashSet<RenewWant> {
358		renews
359			.into_iter()
360			.filter(|w| !self.client.has_indexed_transaction(w.hash.into()).unwrap_or(false))
361			.collect()
362	}

364	/// Bitswap-fetch every missing entry. Errors if any entry was not served.
365	async fn fetch_all(
366		&self,
367		missing: HashSet<RenewWant>,
368	) -> Result<Vec<(ContentHash, Vec<u8>)>, Error> {
369		if missing.is_empty() {
370			return Ok(Vec::new());
371		}
372
373		let wants: Vec<RenewWant> = missing.into_iter().collect();
374		let mut acquired = self.fetcher.fetch_many(&wants).await?;
375
376		if acquired.len() != wants.len() {
377			return Err(Error::IncompleteFetch {
378				missing: wants.len() - acquired.len(),
379				total: wants.len(),
380			});
381		}
382
383		let payload: Vec<(ContentHash, Vec<u8>)> = wants
384			.iter()
385			.map(|w| {
386				let data = acquired
387					.remove(&w.hash)
388					.expect("all hashes present; len equality verified above; qed");
389				(w.hash, data)
390			})
391			.collect();
392
393		Ok(payload)
394	}

396	/// Attach prefetched `(content_hash, bytes)` pairs to
397	/// [`BlockImportParams::prefetched_indexed_transactions`] for the backend writer.
398	///
399	/// Only `renew_payloads` is populated: runtime execution produces the actual
400	/// `IndexOperation::Renew` ops, so synthetic ops stay empty here.
401	fn attach_prefetched(
402		params: &mut BlockImportParams<Block>,
403		fetched: Vec<(ContentHash, Vec<u8>)>,
404	) {
405		if fetched.is_empty() {
406			return;
407		}
408		for (hash, _) in &fetched {
409			log::debug!(
410				target: LOG_TARGET,
411				"attaching bitswap-fetched indexed transaction {hash:?} to BlockImportParams",
412			);
413		}
414		params.prefetched_indexed_transactions.renew_payloads =
415			fetched.into_iter().map(|(h, b)| (sp_core::H256::from(h), b)).collect();
416	}

418	/// Query `TransactionStorageApi::indexed_transactions(block_number)` against the
419	/// parent state plus supplied `StorageChanges`, so the runtime reads the
420	/// post-execution state before the block is committed.
421	fn indexed_transactions_with_storage_changes(
422		&self,
423		parent_hash: Block::Hash,
424		block_number: sp_runtime::traits::NumberFor<Block>,
425		changes: &StorageChanges<HashingFor<Block>>,
426	) -> Result<Vec<IndexedTransactionInfo>, Error> {
427		let mut api = self.client.runtime_api();
428		api.set_overlayed_changes(overlay_from_storage_changes::<Block>(changes));
429		api.indexed_transactions(parent_hash, block_number)
430			.map_err(Error::IndexedTransactions)
431	}

433	/// Execute via runtime API once, query indexed metadata on the same `ApiRef`, and obtain
434	/// `StorageChanges`. Caller must reassign `params.state_action` before forwarding so the
435	/// inner block-import does not re-execute.
436	fn execute_block(
437		&self,
438		params: &BlockImportParams<Block>,
439	) -> Result<(StorageChanges<HashingFor<Block>>, Vec<IndexedTransactionInfo>), Error> {
440		let parent_hash = *params.header.parent_hash();
441		let body = params.body.clone().unwrap_or_default();
442		let block = Block::new(params.header.clone(), body);
443
444		let recorder = ProofRecorder::<Block>::default();
445
446		let mut runtime_api = self.client.runtime_api();
447		runtime_api.set_call_context(CallContext::Onchain { import: true });
448		runtime_api.record_proof_with_recorder(recorder.clone());
449		runtime_api.register_extension(ProofSizeExt::new(recorder));
450
451		runtime_api
452			.execute_block(parent_hash, block.into())
453			.map_err(|e| Error::ExecuteBlock {
454				block: format!("{:?}", params.post_hash()),
455				error: e,
456			})?;
457
458		let infos = runtime_api
459			.execute_in_transaction(|api| {
460				TransactionOutcome::Rollback(
461					api.indexed_transactions(parent_hash, *params.header.number()),
462				)
463			})
464			.map_err(Error::IndexedTransactions)?;
465
466		let state = self
467			.client
468			.state_at(parent_hash)
469			.map_err(|e| Error::StateAt(format!("{parent_hash:?}"), e))?;
470
471		let gen_storage_changes = runtime_api
472			.into_storage_changes(&state, parent_hash)
473			.map_err(Error::IntoStorageChanges)?;
474
475		if params.header.state_root() != &gen_storage_changes.transaction_storage_root {
476			return Err(Error::StateRootMismatch {
477				header: format!("{:?}", params.header.state_root()),
478				executed: format!("{:?}", gen_storage_changes.transaction_storage_root),
479			});
480		}
481
482		Ok((gen_storage_changes, infos))
483	}
484}

486/// Returns runtime-verified renew wants for host-call renew operations.
487fn verified_renews_from_index_ops(
488	ops: &[IndexOperation],
489	infos: &[IndexedTransactionInfo],
490	context: &'static str,
491) -> Result<HashSet<RenewWant>, Error> {
492	let mut renews = HashSet::new();
493	for op in ops {
494		let IndexOperation::Renew { hash, .. } = op else { continue };
495		let hash: ContentHash =
496			hash.as_slice().try_into().map_err(|_| Error::MalformedRenewHash { context })?;
497		let info = infos
498			.iter()
499			.find(|info| info.content_hash == hash)
500			.ok_or(Error::MissingRenewMetadata { context, hash })?;
501		renews.insert(RenewWant { hash, hashing: info.hashing, cid_codec: info.cid_codec });
502	}
503	Ok(renews)
504}
505
506fn overlay_from_storage_changes<Block: BlockT>(
507	changes: &StorageChanges<HashingFor<Block>>,
508) -> OverlayedChanges<HashingFor<Block>> {
509	let mut overlay = OverlayedChanges::default();
510	for (key, value) in &changes.main_storage_changes {
511		overlay.set_storage(key.clone(), value.clone());
512	}
513	for (storage_key, changes) in &changes.child_storage_changes {
514		let child_info = ChildInfo::new_default(storage_key);
515		for (key, value) in changes {
516			overlay.set_child_storage(&child_info, key.clone(), value.clone());
517		}
518	}
519	overlay
520}

522/// Result of [`classify_body`]: the synthetic `IndexOperation`s plus the renew wants
523/// whose bytes need to be bitswap-fetched.
524#[derive(Debug)]
525pub(crate) struct ClassifiedBody {
526	pub ops: Vec<IndexOperation>,
527	pub renews: HashSet<RenewWant>,
528}
529
530/// Classifies every `IndexedTransactionInfo` entry against the block body.
531///
532/// Entries whose tail bytes (`body[info.extrinsic_index][len - info.size..]`) hash to
533/// `info.content_hash` under the declared algorithm become `IndexOperation::Insert`; every
534/// other entry becomes an `IndexOperation::Renew` plus a [`RenewWant`] to bitswap-fetch.
535///
536/// Errors if an entry references an extrinsic index outside the body.
537fn classify_body<Block: BlockT>(
538	infos: &[IndexedTransactionInfo],
539	body: &[Block::Extrinsic],
540) -> Result<ClassifiedBody, Error> {
541	let mut ops = Vec::new();
542	let mut renews = HashSet::new();
543
544	for info in infos {
545		let extrinsic_index = info.extrinsic_index;
546		let Some(ext) = body.get(extrinsic_index as usize) else {
547			return Err(Error::InvalidExtrinsicIndex {
548				index: extrinsic_index,
549				body_len: body.len(),
550			});
551		};
552		let encoded = ext.encode();
553		let size = info.size as usize;
554		let matches_tail = encoded.len() >= size && {
555			let tail = &encoded[encoded.len() - size..];
556			info.hashing.hash(tail) == info.content_hash
557		};
558		if matches_tail {
559			ops.push(IndexOperation::Insert {
560				extrinsic: extrinsic_index,
561				hash: info.content_hash.to_vec(),
562				size: info.size,
563			});
564		} else {
565			ops.push(IndexOperation::Renew {
566				extrinsic: extrinsic_index,
567				hash: info.content_hash.to_vec(),
568			});
569			renews.insert(RenewWant {
570				hash: info.content_hash,
571				hashing: info.hashing,
572				cid_codec: info.cid_codec,
573			});
574		}
575	}
576
577	Ok(ClassifiedBody { ops, renews })
578}
