43/// The reasons a received statement is retained, as a bitmask of independent flags.
44///
45/// Each set bit records one reason the local node keeps the statement (DHT affinity, explicit
46/// affinity). A non-empty mask persists the statement under the normal retention rules. An empty
47/// mask marks it transient: held in memory until the next propagation, forwarded once, then dropped
48/// without ever reaching the database.
49#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
50pub struct RetentionReasonMask(u8);
51
52impl RetentionReasonMask {
53	/// No reason to persist: the store keeps the statement only until the next propagation.
54	pub const TRANSIENT: RetentionReasonMask = RetentionReasonMask(0b00);
55	/// The local node is one of the closest DHT replicas for one of the statement's topics.
56	pub const DHT_AFFINITY: RetentionReasonMask = RetentionReasonMask(0b01);
57	/// The local node has explicit affinity for one of the topics.
58	pub const EXPLICIT_AFFINITY: RetentionReasonMask = RetentionReasonMask(0b10);
59
60	/// A mask with every reason set.
61	pub fn persistent() -> Self {
62		RetentionReasonMask(u8::MAX)
63	}
64
65	/// Add `reason` to the mask.
66	pub fn insert(&mut self, reason: RetentionReasonMask) {
67		self.0 |= reason.0;
68	}
69
70	/// Whether `reason` is set.
71	pub fn contains(&self, reason: RetentionReasonMask) -> bool {
72		self.0 & reason.0 == reason.0 && reason.0 != 0
73	}
74
75	/// Whether the statement should be persisted.
76	pub fn is_persistent(&self) -> bool {
77		self.0 != 0
78	}
79}

81/// Shared affinity view to derive a statement's retention mask.
82#[derive(Clone)]
83pub(crate) struct RetentionHandle {
84	dht_affinity: Arc<RwLock<DhtAffinity>>,
85	/// Topics the node has explicit affinity for.
86	topic_affinity: Arc<RwLock<TopicAffinity>>,
87}
88
89impl RetentionHandle {
90	/// A handle seeded with an empty topology, so its resolver persists every statement carrying a
91	/// topic until the orchestrator publishes the learned affinity.
92	pub(crate) fn new(local_peer: PeerId, replication_factor: NonZeroUsize) -> Self {
93		Self {
94			dht_affinity: Arc::new(RwLock::new(DhtAffinity::empty(local_peer, replication_factor))),
95			topic_affinity: Arc::new(RwLock::new(TopicAffinity::default())),
96		}
97	}
98
99	/// The resolver the store calls to derive a statement's retention mask.
100	pub(crate) fn resolver(&self) -> Box<dyn Fn(&Statement) -> RetentionReasonMask + Send + Sync> {
101		let dht_affinity = self.dht_affinity.clone();
102		let topic_affinity = self.topic_affinity.clone();
103		Box::new(move |stmt| {
104			let (Ok(dht), Ok(topics)) = (dht_affinity.read(), topic_affinity.read()) else {
105				log::error!(
106					target: LOG_TARGET,
107					"v2dht: retention affinity lock poisoned; persisting statement defensively",
108				);
109				return RetentionReasonMask::persistent();
110			};
111			let mut mask = RetentionReasonMask::TRANSIENT;
112			if dht.is_affine(stmt) {
113				mask.insert(RetentionReasonMask::DHT_AFFINITY);
114			}
115			if topics.is_affine(stmt) {
116				mask.insert(RetentionReasonMask::EXPLICIT_AFFINITY);
117			}
118			mask
119		})
120	}
121
122	/// Replace the DHT-affinity oracle
123	fn set_dht_affinity(&self, dht_affinity: DhtAffinity) {
124		if let Ok(mut cell) = self.dht_affinity.write() {
125			*cell = dht_affinity;
126		}
127	}
128
129	/// Replace the explicit topic-affinity oracle
130	fn set_topic_affinity(&self, topic_affinity: TopicAffinity) {
131		if let Ok(mut cell) = self.topic_affinity.write() {
132			*cell = topic_affinity;
133		}
134	}
135}

137/// Coordinates the v2 DHT-affinity statement gossip path.
138pub(crate) struct V2DhtOrchestrator {
139	/// Local view of statement-store peers known and connected through network topology events.
140	peers_topology: PeersTopology,
141	/// Tracks the local node's topic affinity and the filters peers advertise.
142	explicit_affinity: ExplicitAffinity,
143	/// Keeps the connected peer set aligned with the peers needed to cover subscriptions.
144	peer_steering: PeerSteering,
145	/// Shared affinity view the store reads to decide statement retention.
146	retention: Option<RetentionHandle>,
147	/// Prometheus metrics.
148	metrics: Option<V2DhtMetrics>,
149}

153		configured_topics: &[Topic],
154		local_peer: PeerId,
155		peers_topology_config: PeersTopologyConfig,
156		protocol: ProtocolName,
157		metrics: Option<V2DhtMetrics>,

162			peer_steering: PeerSteering::new(protocol),
163			retention: None,

168	/// Install the handle the store reads to decide statement retention.
169	pub(crate) fn set_retention_handle(&mut self, handle: RetentionHandle) {
170		self.retention = Some(handle);
171		// Seed both oracles from the current state so configured topics and the learned topology
172		// drive retention before the first peer or subscription event publishes them.
173		self.publish_dht_affinity();
174		self.publish_topic_affinity();
175	}

177	/// The resolver the store calls to derive a statement's retention mask, if retention is wired.
178	pub(crate) fn retention_resolver(
179		&self,
180	) -> Option<Box<dyn Fn(&Statement) -> RetentionReasonMask + Send + Sync>> {
181		self.retention.as_ref().map(RetentionHandle::resolver)
182	}

184	/// Refresh the store's DHT-affinity oracle
185	fn publish_dht_affinity(&self) {
186		if let Some(handle) = &self.retention {
187			handle.set_dht_affinity(self.peers_topology.dht_affinity());
188		}
189	}
190
191	/// Refresh the store's explicit topic-affinity oracle
192	fn publish_topic_affinity(&self) {
193		if let Some(handle) = &self.retention {
194			handle.set_topic_affinity(self.explicit_affinity.topic_affinity());
195		}
196	}

202				self.peers_topology.connected_peers().count(),

205		}

293	/// For each connected peer, the indices of `statements` it should receive.
294	///
295	/// A peer is a target for a statement when it is a DHT routing target for one of the
296	/// statement's topics ([`PeersTopology::routing_targets`]) or its advertised filter matches the
297	/// statement ([`ExplicitAffinity::peer_has_explicit_affinity`]).
298	pub(crate) fn propagation_plan(
299		&self,
300		statements: &[(u64, Hash, Statement)],
301	) -> Vec<(PeerId, Vec<usize>)> {
302		let mut statements_by_peer: HashMap<PeerId, Vec<usize>> = HashMap::new();
303
304		for (index, (_seq, _hash, statement)) in statements.iter().enumerate() {
305			let mut targets: HashSet<PeerId> = HashSet::new();
306
307			for topic in statement.topics() {
308				targets.extend(self.peers_topology.routing_targets(*topic));
309			}
310			for peer in self.peers_topology.connected_peers() {
311				if self.explicit_affinity.peer_has_explicit_affinity(peer, statement) {
312					targets.insert(peer);
313				}
314			}
315
316			for peer_id in targets {
317				statements_by_peer.entry(peer_id).or_default().push(index);
318			}
319		}
320
321		statements_by_peer.into_iter().collect()
322	}

373	fn statement(seed: u8, topics: &[Topic]) -> (u64, Hash, Statement) {
374		let mut statement = Statement::new();
375		statement.set_plain_data(vec![seed]);
376		for (index, topic) in topics.iter().enumerate() {
377			statement.set_topic(index, *topic);
378		}
379		(seed as u64, statement.hash(), statement)
380	}

382	/// Routing targets `propagation_plan` is expected to reach for `topics`, taken from the
383	/// already-tested [`PeersTopology::routing_targets`] rather than recomputed from XOR distances.
384	fn routing_targets(orchestrator: &V2DhtOrchestrator, topics: &[Topic]) -> Vec<PeerId> {
385		let mut targets: Vec<PeerId> = topics
386			.iter()
387			.flat_map(|topic| orchestrator.peers_topology.routing_targets(*topic))
388			.collect();
389		targets.sort();
390		targets.dedup();
391		targets
392	}

651	fn connected_coverage_peer_is_never_disconnected() {
652		let mut orchestrator = V2DhtOrchestrator::new(
653			&[topic(1)],
654			peer(1),
655			topology_config(20, 3),
656			"/statement/test".into(),
657			None,
658		);
659
660		for seed in 2..=10 {
661			orchestrator.on_peers_discovered([peer(seed)]);
662			orchestrator.on_peer_identified(peer(seed), true);
663		}
664
665		// The first tick selects the coverage peers and queues them for connection.
666		orchestrator.on_pending_affinities();
667		let desired = orchestrator.peers_topology.peers_for_topics(&[topic(1)]);
668		assert!(!desired.is_empty());
669
670		// The coverage peers connect, then the next tick recomputes the target.
671		for peer in &desired {
672			orchestrator.on_substream_opened(*peer);
673		}
674		orchestrator.on_pending_affinities();
675
676		// The connected coverage peers stay desired, so peer steering keeps them: nothing left to
677		// connect, nothing to disconnect.
678		assert!(orchestrator.peer_steering.peers_to_connect().is_empty());
679		assert!(orchestrator.peer_steering.peers_to_disconnect().is_empty());
