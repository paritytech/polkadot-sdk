19//! Peer steering for the v2 DHT gossip path.
20//!
21//! Implements [#11935](https://github.com/paritytech/polkadot-sdk/issues/11935): the module that
22//! keeps the node connected to the peers it needs: those covering its subscriptions and its
23//! routing peers.
24//!
25//! The module keeps three sets of peers: the peers with an open substream, the peers the node
26//! needs connected (the desired set), and the peers it holds in the statement protocol's reserved
27//! set. Each refresh moves the reserved set toward the desired one. A desired
28//! peer gets reserved, which dials it and keeps its substream open. A reserved peer no longer
29//! desired is released, which drops a peer this node dialed and leaves a peer that dialed in as
30//! an ordinary inbound peer while an inbound slot is free. A refresh releases at most
31//! [`MAX_DISCONNECT_PERCENT`] of the connected peers, so the set converges over several
32//! refreshes instead of churning at once.
33//! The orchestrator supplies the desired set through
34//! [`PeerSteering::update_peers_needing_connections`].
35//!
36//! Each connected peer carries a score based on its actions. Undesired peers are
37//! dropped lowest-score first.
38
39use crate::LOG_TARGET;
40use sc_network::{multiaddr, types::ProtocolName, NetworkPeers};
41use sc_network_types::PeerId;
42use std::{
43	collections::{HashMap, HashSet},
44	iter,
45};

47/// Per-connection score changes.
48///
49/// The weights are a first cut, not a tuned reputation model.
50pub(crate) mod score {
51	/// A peer delivered a valid statement.
52	pub(crate) const GOOD_ACTION: i32 = 1;
53	/// A peer sent an invalid statement, a duplicate, or an undecodable message.
54	pub(crate) const BAD_ACTION: i32 = -50;
55	/// A peer resent a banned statement.
56	pub(crate) const WASTEFUL_ACTION: i32 = -1;
57}
58
59/// Upper bound, as a percent of the connected peers, on how many connections a single refresh
60/// closes. Capping the churn lets the connected set converge toward the desired one step by step
61/// instead of dropping every undesired connection at once.
62const MAX_DISCONNECT_PERCENT: usize = 20;
63
64/// Keeps the statement protocol's reserved set aligned with the peers the node needs connected.
65///
66/// The desired set comes from the orchestrator and the connected set from the statement substream
67/// events. [`Self::refresh_connections`] edits the reserved set to match the desired one.
68#[derive(Debug)]
69pub(crate) struct PeerSteering {
70	/// Statement protocol whose reserved set the connections are steered through.
71	protocol: ProtocolName,
72	/// Peers with an open statement notification substream.
73	connected: HashMap<PeerId, ConnectedPeer>,
74	/// Monotonic counter handing each new connection its sequence.
75	next_connection_seq: u64,
76	/// Peers the node needs connected: those covering its subscriptions and its routing peers.
77	desired: HashSet<PeerId>,
78	/// Peers held in the statement protocol's reserved set.
79	reserved: HashSet<PeerId>,
80}
81
82/// The state of one connected peer.
83#[derive(Debug)]
84struct ConnectedPeer {
85	/// Sequence at which the peer connected.
86	seq: u64,
87	/// Reputation over this connection.
88	score: i32,
89}
90
91impl PeerSteering {
92	pub(crate) fn new(protocol: ProtocolName) -> Self {
93		Self {
94			protocol,
95			connected: HashMap::new(),
96			next_connection_seq: 0,
97			desired: HashSet::new(),
98			reserved: HashSet::new(),
99		}
100	}
101
102	// === Connection events ===
103
104	/// Record that the statement notification substream to `peer` opened.
105	pub(crate) fn on_substream_opened(&mut self, peer: PeerId) {
106		if !self.connected.contains_key(&peer) {
107			self.connected
108				.insert(peer, ConnectedPeer { seq: self.next_connection_seq, score: 0 });
109			self.next_connection_seq += 1;
110		}
111	}
112
113	/// Record that the statement notification substream to `peer` closed.
114	pub(crate) fn on_substream_closed(&mut self, peer: PeerId) {
115		self.connected.remove(&peer);
116	}
117
118	/// Update peer's connection score.
119	pub(crate) fn update_score(&mut self, peer: PeerId, change: i32) {
120		if let Some(entry) = self.connected.get_mut(&peer) {
121			entry.score = entry.score.saturating_add(change);
122		}
123	}
124
125	/// Peer's current connection score, or `None` if it has no open substream.
126	#[cfg(test)]
127	pub(crate) fn score_of(&self, peer: &PeerId) -> Option<i32> {
128		self.connected.get(peer).map(|entry| entry.score)
129	}
130
131	// === Desired set ===
132
133	/// Replace the set of peers the node needs connected.
134	pub(crate) fn update_peers_needing_connections(
135		&mut self,
136		peers: impl IntoIterator<Item = PeerId>,
137	) {
138		self.desired = peers.into_iter().collect();
139	}
140
141	/// Whether `peer` is held in the statement protocol's reserved set.
142	pub(crate) fn is_reserved(&self, peer: &PeerId) -> bool {
143		self.reserved.contains(peer)
144	}
145
146	#[cfg(test)]
147	pub(crate) fn reserve(&mut self, peers: impl IntoIterator<Item = PeerId>) {
148		self.reserved.extend(peers);
149	}
150
151	// === Reconciliation ===
152
153	/// Desired peers without an open connection.
154	pub(crate) fn peers_to_connect(&self) -> Vec<PeerId> {
155		self.desired
156			.iter()
157			.filter(|peer| !self.connected.contains_key(peer))
158			.copied()
159			.collect()
160	}
161
162	/// Desired peers outside the reserved set.
163	pub(crate) fn peers_to_reserve(&self) -> Vec<PeerId> {
164		self.desired
165			.iter()
166			.filter(|peer| !self.reserved.contains(peer))
167			.copied()
168			.collect()
169	}
170
171	/// Reserved peers no longer desired. A peer without an open substream is released at once. The
172	/// connected ones go at most [`MAX_DISCONNECT_PERCENT`] of the connected peers per refresh, and
173	/// at least one, so the set converges step by step.
174	pub(crate) fn peers_to_release(&self) -> Vec<PeerId> {
175		let limit = (self.connected.len() * MAX_DISCONNECT_PERCENT / 100).max(1);
176		let mut to_release = Vec::new();
177		// Drop the lowest-scored peers first. Among peers of equal score, the longest-connected one
178		// goes first, spreading disconnects across peers.
179		let mut to_disconnect: Vec<(i32, u64, PeerId)> = Vec::new();
180		for peer in self.reserved.iter().filter(|peer| !self.desired.contains(peer)) {
181			match self.connected.get(peer) {
182				Some(entry) => to_disconnect.push((entry.score, entry.seq, *peer)),
183				None => to_release.push(*peer),
184			}
185		}
186		to_disconnect.sort();
187		to_disconnect.truncate(limit);
188		to_release.extend(to_disconnect.into_iter().map(|(_, _, peer)| peer));
189		to_release
190	}
191
192	/// Edit the statement protocol's reserved set on `network` so it matches the desired set.
193	///
194	/// The connected set tracks open substreams, so it updates through the substream events as the
195	/// changes take effect, not here.
196	pub(crate) fn refresh_connections<N: NetworkPeers>(&mut self, network: &N) {
197		let reserve = self.peers_to_reserve();
198		let release = self.peers_to_release();
199
200		log::trace!(
201			target: LOG_TARGET,
202			"peer_steering: refresh_connections reserve {} release {}",
203			reserve.len(),
204			release.len(),
205		);
206
207		// Leaving the reserved set drops a peer this node dialed only while the set has no
208		// outbound slots, otherwise the controller moves it to a free slot and an explicit
209		// disconnect is needed.
210		for peer in release {
211			match network.remove_peers_from_reserved_set(self.protocol.clone(), vec![peer]) {
212				Ok(()) => {
213					self.reserved.remove(&peer);
214				},
215				Err(err) => {
216					log::warn!(target: LOG_TARGET, "peer_steering: release {peer} failed: {err}");
217				},
218			}
219		}
220		for peer in reserve {
221			let addr =
222				iter::once(multiaddr::Protocol::P2p(peer.into())).collect::<multiaddr::Multiaddr>();
223			match network
224				.add_peers_to_reserved_set(self.protocol.clone(), iter::once(addr).collect())
225			{
226				Ok(()) => {
227					self.reserved.insert(peer);
228				},
229				Err(err) => {
230					log::warn!(target: LOG_TARGET, "peer_steering: reserve {peer} failed: {err}");
231				},
232			}
233		}
234	}
235}

242	fn connected_set(steering: &PeerSteering) -> HashSet<PeerId> {
243		steering.connected.keys().copied().collect()
244	}

246	#[test]
247	fn substream_events_maintain_the_connected_set() {
248		let mut steering = PeerSteering::new("/statement/test".into());
249
250		steering.on_substream_opened(peer(1));
251		steering.on_substream_opened(peer(2));
252		assert_eq!(connected_set(&steering), HashSet::from([peer(1), peer(2)]));
253
254		steering.on_substream_closed(peer(1));
255		assert_eq!(connected_set(&steering), HashSet::from([peer(2)]));
256	}

258	/// Opens a substream to each peer and marks it reserved, as a completed refresh leaves it.
259	fn connect_reserved(steering: &mut PeerSteering, seeds: impl IntoIterator<Item = u8>) {
260		for seed in seeds {
261			steering.on_substream_opened(peer(seed));
262			steering.reserve([peer(seed)]);
263		}
264	}

266	#[test]
267	fn reserves_desired_and_releases_undesired() {
268		let mut steering = PeerSteering::new("/statement/test".into());
269		// peer(1) is reserved and desired; peer(2) is reserved but undesired; peer(3) is desired
270		// but not reserved; peer(4) is connected on its own and undesired.
271		connect_reserved(&mut steering, [1, 2]);
272		steering.on_substream_opened(peer(4));
273		steering.update_peers_needing_connections([peer(1), peer(3)]);
274
275		assert_eq!(steering.peers_to_reserve(), vec![peer(3)]);
276		assert_eq!(steering.peers_to_release(), vec![peer(2)]);
277	}

333	#[test]
334	fn score_outranks_connection_age_when_disconnecting() {
335		let mut steering = PeerSteering::new("/statement/test".into());
336		// peer(1) connects first, so connection age alone would drop it before peer(2).
337		connect_reserved(&mut steering, [1, 2]);
338		// The longest-connected peer behaves well, the younger one misbehaves.
339		steering.update_score(peer(1), score::GOOD_ACTION);
340		steering.update_score(peer(2), score::BAD_ACTION);
341
342		// Score outranks age: the misbehaving peer is dropped despite being younger.
343		assert_eq!(steering.peers_to_release(), vec![peer(2)]);
344	}

346	#[test]
347	fn update_score_ignores_unconnected_peers_and_resets_on_reconnect() {
348		let mut steering = PeerSteering::new("/statement/test".into());
349
350		// Reporting a peer without an open substream changes nothing.
351		steering.update_score(peer(1), score::BAD_ACTION);
352		assert!(steering.peers_to_release().is_empty());
353
354		// A fresh connection starts at 0, unaffected by the earlier report. Only the peer scored
355		// down over this connection is dropped first.
356		connect_reserved(&mut steering, [1, 2]);
357		steering.update_score(peer(2), score::BAD_ACTION);
358		assert_eq!(steering.peers_to_release(), vec![peer(2)]);
359	}

361	#[test]
362	fn capped_disconnects_drain_over_successive_refreshes() {
363		let mut steering = PeerSteering::new("/statement/test".into());
364		connect_reserved(&mut steering, 1..=10);
365
366		let mut dropped = HashSet::new();
367		while !steering.connected.is_empty() {
368			let release = steering.peers_to_release();
369			assert!(release.len() <= 2, "each refresh drops at most a fifth");
370			for peer in release {
371				assert!(dropped.insert(peer), "no peer dropped twice");
372				steering.reserved.remove(&peer);
373				steering.on_substream_closed(peer);
374			}
375		}
376
377		assert_eq!(dropped.len(), 10);
378	}
