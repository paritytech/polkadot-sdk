30#[derive(Debug, Clone)]
31pub struct PeersTopologyConfig {
32	/// Number of statement-protocol peers responsible for storing a topic.
33	///
34	/// The DHT-affinity decision uses this to decide whether the local node belongs to the
35	/// K-closest peers for a topic according to the locally learned topology.
36	pub replication_factor: NonZeroUsize,
37	/// Maximum number of connected nodes that we gossip to.
38	///
39	/// This caps `routing_targets`, i.e. the forwarding candidates selected from
40	/// currently connected peers for a topic.
41	pub gossip_target: NonZeroUsize,
42}
43
44#[derive(Debug, Clone)]
45struct PeerInfo {
46	supports_protocol: bool,
47	/// The statement notification substream to the peer is open.
48	connected: bool,
49	/// Cached `peer_key`; the peer id never changes and hashing is costly.
50	key: Key,
51	/// Monotonic timestamp of the most recent event observing this peer; the eviction key for both
52	/// staleness and the capacity backstop.
53	last_seen: Instant,
54}
55
56/// Evict a disconnected peer unseen by any event for this long.
57const PEER_STALENESS_TTL: Duration = Duration::from_secs(2 * 60 * 60);
58
59/// Hard cap on `discovered`; bounds memory when discovery outruns staleness eviction.
60const MAX_KNOWN_PEERS: usize = 8192;
61
62/// Event-fed local view of statement-store peers that ages out peers unseen for
63/// `PEER_STALENESS_TTL`.
64///
65/// The topology is built from peers learned through routing-table updates, identify metadata and
66/// statement notification connections. It computes XOR distances locally over that learned peer
67/// set; it does not issue topic-specific Kademlia lookups.
68///
69/// Topic queries walk the sorted key indexes in increasing XOR distance, without sorting the full
70/// peer set.
71#[derive(Debug, Clone)]
72pub struct PeersTopology {
73	local_peer: PeerId,
74	local_key: Key,
75	config: PeersTopologyConfig,
76	/// Known remote peers, evicted by [`PeersTopology::evict`] once stale or over
77	/// `MAX_KNOWN_PEERS`.
78	discovered: HashMap<PeerId, PeerInfo>,
79	/// XOR-ordered index of the statement-protocol peers: the candidates for DHT storage,
80	/// affinity and forwarding decisions.
81	discovered_index: PeersIndex,
82	/// XOR-ordered index of the connected peers, i.e. those with an open statement notification
83	/// substream.
84	connected: PeersIndex,
85}
86
87impl PeersTopology {
88	pub fn new(local_peer: PeerId, config: PeersTopologyConfig) -> Self {
89		Self {
90			local_peer,
91			local_key: peer_key(&local_peer),
92			config,
93			discovered: HashMap::new(),
94			discovered_index: PeersIndex::default(),
95			connected: PeersIndex::default(),
96		}
97	}
98
99	/// Record that a litep2p routing-table update saw `peers`.
100	///
101	/// `RoutingTableUpdate` is used as the discovery source because litep2p's internal
102	/// Kademlia table is bucket-limited and may discard many discovered peers.
103	pub fn on_peers_discovered(&mut self, peers: impl IntoIterator<Item = PeerId>) {
104		for peer in peers {
105			self.get_or_insert_peer(peer);
106		}
107	}
108
109	/// Record statement-protocol support from identify protocol metadata.
110	///
111	/// Peers that do not support the statement protocol remain known but are excluded from DHT
112	/// storage and forwarding decisions.
113	pub fn on_peer_identified(&mut self, peer: PeerId, supports_statement_protocol: bool) {
114		let info = self.get_or_insert_peer(peer);
115		let key = info.key;
116		info.supports_protocol = supports_statement_protocol;
117		if supports_statement_protocol {
118			self.discovered_index.insert(key, peer);
119		} else {
120			self.discovered_index.remove(key, &peer);
121		}
122	}
123
124	/// Record that the statement notification substream opened.
125	///
126	/// An open substream implies statement-protocol support.
127	pub fn on_substream_opened(&mut self, peer: PeerId) {
128		let info = self.get_or_insert_peer(peer);
129		let key = info.key;
130		info.supports_protocol = true;
131		info.connected = true;
132		self.discovered_index.insert(key, peer);
133		self.connected.insert(key, peer);
134	}
135
136	/// Record that the statement notification substream closed.
137	pub fn on_substream_closed(&mut self, peer: PeerId) {
138		let Some(info) = self.discovered.get_mut(&peer) else { return };
139		let key = info.key;
140		info.connected = false;
141		info.last_seen = Instant::now();
142		self.connected.remove(key, &peer);
143	}
144
145	#[allow(dead_code)]
146	fn is_connected(&self, peer: &PeerId) -> bool {
147		self.discovered.get(peer).is_some_and(|info| info.connected)
148	}
149
150	/// The connected statement-protocol peers, those with an open notification substream.
151	pub fn connected_peers(&self) -> impl Iterator<Item = PeerId> + '_ {
152		self.connected.peers()
153	}
154
155	/// Number of known remote peers, including peers without confirmed statement-protocol support.
156	pub fn known_peers_count(&self) -> usize {
157		self.discovered.len()
158	}
159
160	/// Number of known statement-protocol peers, the DHT storage, affinity and forwarding
161	/// candidates.
162	pub fn dht_eligible_peers_count(&self) -> usize {
163		self.discovered_index.peers().count()
164	}
165
166	/// Closest known statement-protocol peers for `topic`.
167	///
168	/// "Closest" is computed over the locally learned statement-protocol peers, not by querying
169	/// the network for the true global closest peers.
170	#[allow(dead_code)]
171	pub fn closest_known(&self, topic: Topic, limit: usize) -> Vec<PeerId> {
172		self.closest_known_keyed(topic, limit)
173			.into_iter()
174			.map(|(peer, _)| peer)
175			.collect()
176	}

178	/// A standalone DHT-affinity oracle, answering "is the local node a storage replica for a
179	/// topic" off the handler thread.
180	///
181	/// Clones only the data the decision needs — the statement-protocol peer index and the local
182	/// identity — not the full topology.
183	///
184	/// TODO: deep-copies the index on every peer-churn event that republishes the oracle.
185	pub fn dht_affinity(&self) -> DhtAffinity {
186		DhtAffinity {
187			index: self.discovered_index.clone(),
188			local_peer: self.local_peer,
189			local_key: self.local_key,
190			replication_factor: self.config.replication_factor,
191		}
192	}
193
194	/// Connected peers to forward a statement for `topic` to, capped at `gossip_target`: those
195	/// closer to the topic than the local node (routing the statement onward) and those that are
196	/// themselves DHT replicas for it (co-replicas that must store it).
197	pub fn routing_targets(&self, topic: Topic) -> Vec<PeerId> {
198		// TODO: benchmark this per-statement path on large connected sets (see
199		// benches/peers_topology.rs).
200		let local = (self.local_key, self.local_peer);
201		let local_distance = xor_distance(*topic, self.local_key);
202		let k = self.config.replication_factor.get();
203		self.connected
204			.closest(*topic)
205			.take_while(|(peer, key)| {
206				xor_distance(*topic, *key) < local_distance ||
207					is_peer_topic_affine(&self.discovered_index, local, (*key, *peer), k, topic)
208			})
209			.take(self.config.gossip_target.get())
210			.map(|(peer, _)| peer)
211			.collect()
212	}
213
214	/// Local-only explicit-affinity connection candidates for `topics`.
215	///
216	/// This uses only the locally learned topology, avoiding network lookups that would reveal
217	/// explicit-affinity topics, and selects a minimal set of peers covering every topic,
218	/// independent of connection state.
219	pub fn peers_for_topics(&self, topics: &[Topic]) -> Vec<PeerId> {
220		if topics.is_empty() {
221			return Vec::new();
222		}
223
224		let pool_size = self.config.replication_factor.get();
225
226		let closest_pools = topics
227			.iter()
228			.map(|topic| self.closest_known_keyed(*topic, pool_size))
229			.collect::<Vec<_>>();
230
231		let mut uncovered = (0..topics.len()).collect::<HashSet<_>>();
232
233		let mut selected = Vec::new();
234		let limit = topics.len();
235
236		while !uncovered.is_empty() && selected.len() < limit {
237			let Some(best_peer) = best_candidate(topics, &closest_pools, &uncovered, &selected)
238			else {
239				break;
240			};
241
242			selected.push(best_peer);
243			uncovered.retain(|topic_idx| !pool_contains(&closest_pools[*topic_idx], &best_peer));
244		}
245
246		selected
247	}
248
249	/// Insert `peer` if absent, refresh its `last_seen`, and return its record.
250	fn get_or_insert_peer(&mut self, peer: PeerId) -> &mut PeerInfo {
251		let now = Instant::now();
252		let info = self.discovered.entry(peer).or_insert_with(|| PeerInfo {
253			supports_protocol: false,
254			connected: false,
255			key: peer_key(&peer),
256			last_seen: now,
257		});
258		info.last_seen = now;
259		info
260	}
261
262	/// Evict disconnected peers unseen for `PEER_STALENESS_TTL` as of `now`, plus any excess over
263	/// `MAX_KNOWN_PEERS`.
264	pub fn evict(&mut self, now: Instant) {
265		loop {
266			let over_cap = self.discovered.len() > MAX_KNOWN_PEERS;
267			let Some((victim, last_seen)) = self
268				.discovered
269				.iter()
270				.filter(|(_, info)| !info.connected)
271				.min_by_key(|(_, info)| info.last_seen)
272				.map(|(peer, info)| (*peer, info.last_seen))
273			else {
274				return;
275			};
276			if !over_cap && now.saturating_duration_since(last_seen) < PEER_STALENESS_TTL {
277				return;
278			}
279			if let Some(info) = self.discovered.remove(&victim) {
280				self.discovered_index.remove(info.key, &victim);
281			}
282		}
283	}

285	/// `closest_known` paired with each peer's key, so callers that compute further distances
286	/// reuse the key instead of looking it up again.
287	fn closest_known_keyed(&self, topic: Topic, limit: usize) -> Vec<(PeerId, Key)> {
288		self.discovered_index.closest(*topic).take(limit).collect()
289	}
290}

292/// The peer covering the most uncovered topics, breaking ties by the smallest distance to a
293/// covered topic, then by the smallest peer id.
294fn best_candidate(
295	topics: &[Topic],
296	pools: &[Vec<(PeerId, Key)>],
297	uncovered: &HashSet<usize>,
298	selected: &[PeerId],
299) -> Option<PeerId> {
300	uncovered
301		.iter()
302		.flat_map(|topic_idx| pools[*topic_idx].iter().copied())
303		.filter(|(peer, _)| !selected.contains(peer))
304		.collect::<HashSet<_>>()
305		.into_iter()
306		.map(|(peer, key)| {
307			let (covered_count, best_distance) = uncovered
308				.iter()
309				.filter(|topic_idx| pool_contains(&pools[**topic_idx], &peer))
310				.map(|topic_idx| xor_distance(*topics[*topic_idx], key))
311				.fold((0usize, [u8::MAX; 32]), |(count, best), distance| {
312					(count + 1, best.min(distance))
313				});
314			(peer, covered_count, best_distance)
315		})
316		.min_by_key(|&(peer, covered_count, best_distance)| {
317			(Reverse(covered_count), best_distance, peer)
318		})
319		.map(|(peer, ..)| peer)
320}
321
322fn pool_contains(pool: &[(PeerId, Key)], peer: &PeerId) -> bool {
323	pool.iter().any(|(candidate, _)| candidate == peer)
324}
325
326/// Map a peer id into the 32-byte topic key space.
327///
328/// Blake2b-256 spreads peer ids uniformly over the key space. The keys only need to be
329/// consistent across statement-store nodes, which all derive them here; they need not match
330/// litep2p's SHA-256 Kademlia keys because the topology never queries Kademlia by topic.
331fn peer_key(peer: &PeerId) -> Key {
332	sp_crypto_hashing::blake2_256(&peer.to_bytes())
333}
334
335fn xor_distance(a: Key, b: Key) -> Key {
336	let mut distance = [0; 32];
337	for ((distance, a), b) in distance.iter_mut().zip(a).zip(b) {
338		*distance = a ^ b;
339	}
340	distance
341}
342
343/// Whether `candidate` is among the `k` closest to `topic`, ranked against the peers in `index`
344/// plus the local node — i.e. topic-affine, a DHT storage replica for the topic.
345fn is_peer_topic_affine(
346	index: &PeersIndex,
347	local: (Key, PeerId),
348	candidate: (Key, PeerId),
349	k: usize,
350	topic: Topic,
351) -> bool {
352	let candidate_distance = (xor_distance(*topic, candidate.0), candidate.1);
353	let local_node_is_closer = (xor_distance(*topic, local.0), local.1) < candidate_distance;
354	let mut closer = index
355		.closest(*topic)
356		.take_while(|(peer, key)| (xor_distance(*topic, *key), *peer) < candidate_distance)
357		.take(k)
358		.count();
359	// The local node is not in `index`, so count its slot here when it outranks the candidate.
360	// For the oracle's self query (`candidate == local`) this never fires: a node never outranks
361	// itself.
362	if local_node_is_closer {
363		closer += 1;
364	}
365	closer < k
366}

368/// Standalone DHT-affinity oracle: the minimal snapshot needed to answer "is the local node a
369/// storage replica for a topic", shared with the store so it decides retention without the full
370/// [`PeersTopology`].
371#[derive(Clone)]
372pub struct DhtAffinity {
373	index: PeersIndex,
374	local_peer: PeerId,
375	local_key: Key,
376	replication_factor: NonZeroUsize,
377}
378
379impl DhtAffinity {
380	/// An oracle that knows no peers yet, so the local node is the sole replica for every topic.
381	pub fn empty(local_peer: PeerId, replication_factor: NonZeroUsize) -> Self {
382		Self {
383			index: PeersIndex::default(),
384			local_peer,
385			local_key: peer_key(&local_peer),
386			replication_factor,
387		}
388	}
389
390	/// Whether the local node is one of the closest DHT storage replicas for any of the statement's
391	/// topics, over the locally learned statement-store peers.
392	pub fn is_affine(&self, stmt: &Statement) -> bool {
393		stmt.topics().iter().any(|topic| self.is_topic_affine(*topic))
394	}
395
396	/// Whether the local node is one of the closest DHT storage replicas for `topic`.
397	fn is_topic_affine(&self, topic: Topic) -> bool {
398		let local = (self.local_key, self.local_peer);
399		is_peer_topic_affine(&self.index, local, local, self.replication_factor.get(), topic)
400	}
401}

642	#[test]
643	fn queries_match_naive_recomputation() {
644		let local = peer(1);
645		let mut topology = PeersTopology::new(local, topology_config(5, 3));
646		let mut records = Vec::new();
647
648		for seed in 2..=220 {
649			let peer = peer(seed);
650			let supports = seed % 3 != 0;
651			let connected = seed % 4 == 0;
652			topology.on_peers_discovered([peer]);
653			if connected {
654				topology.on_substream_opened(peer);
655			}
656			// Identify after the substream opens: a connected peer may withdraw protocol
657			// support yet must remain a routing candidate.
658			topology.on_peer_identified(peer, supports);
659			records.push((peer, supports, connected));
660		}
661		let dht = topology.dht_affinity();
662
663		for topic_seed in 0..32 {
664			let topic = topic(topic_seed);
665
666			let mut candidates = records
667				.iter()
668				.filter(|(_, supports, _)| *supports)
669				.map(|(peer, ..)| *peer)
670				.collect::<Vec<_>>();
671			candidates.sort_by(|a, b| cmp_distance_then_peer(topic, a, b));
672			assert_eq!(
673				topology.closest_known(topic, 7),
674				candidates.iter().copied().take(7).collect::<Vec<_>>()
675			);
676
677			let mut with_local = candidates.clone();
678			with_local.push(local);
679			with_local.sort_by(|a, b| cmp_distance_then_peer(topic, a, b));
680			let expected_affine = with_local.iter().take(5).any(|peer| *peer == local);
681			assert_eq!(dht.is_affine(&statement_on(topic)), expected_affine);
682
683			let local_distance = distance_to(topic, &local);
684			let mut connected = records
685				.iter()
686				.filter(|(_, _, connected)| *connected)
687				.map(|(peer, ..)| *peer)
688				.filter(|peer| distance_to(topic, peer) < local_distance)
689				.collect::<Vec<_>>();
690			connected.sort_by(|a, b| cmp_distance_then_peer(topic, a, b));
691			connected.truncate(3);
692			assert_eq!(topology.routing_targets(topic), connected);
693		}
694	}
