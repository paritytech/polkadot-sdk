51/// Creates a dedicated tokio runtime for RPC operations.
52///
53/// This runtime isolates RPC blocking operations from the rest of the node
54/// by limiting the number of blocking threads to `max_connections`.
55pub fn create_rpc_runtime(max_connections: u32) -> std::io::Result<tokio::runtime::Runtime> {
56	tokio::runtime::Builder::new_multi_thread()
57		.thread_name("rpc")
58		.enable_all()
59		.max_blocking_threads((max_connections as usize).max(1))
60		.on_thread_start(|| {
61			sc_utils::metrics::TOKIO_THREADS_ALIVE.inc();
62			sc_utils::metrics::TOKIO_THREADS_TOTAL.inc();
63			middleware::RPC_THREADS_ALIVE.inc();
64			middleware::RPC_THREADS_TOTAL.inc();
65		})
66		.on_thread_stop(|| {
67			sc_utils::metrics::TOKIO_THREADS_ALIVE.dec();
68			middleware::RPC_THREADS_ALIVE.dec();
69		})
70		.build()
71}

73/// Spawn handle for RPC tasks that uses the dedicated RPC runtime.
74///
75/// This ensures all RPC-related task spawning (including rpc-spec-v2 APIs like
76/// chainHead and transactionWatch) runs on the isolated RPC runtime rather than
77/// the main node runtime.
78#[derive(Clone)]
79pub struct RpcSpawnHandle {
80	handle: tokio::runtime::Handle,
81}
82
83impl RpcSpawnHandle {
84	/// Create a new RpcSpawnHandle from a tokio runtime handle.
85	pub fn new(handle: tokio::runtime::Handle) -> Self {
86		Self { handle }
87	}
88}
89
90impl sp_core::traits::SpawnNamed for RpcSpawnHandle {
91	fn spawn_blocking(
92		&self,
93		_name: &'static str,
94		_group: Option<&'static str>,
95		future: BoxFuture<'static, ()>,
96	) {
97		let handle = self.handle.clone();
98		self.handle.spawn_blocking(move || {
99			handle.block_on(future);
100		});
101	}
102
103	fn spawn(
104		&self,
105		_name: &'static str,
106		_group: Option<&'static str>,
107		future: BoxFuture<'static, ()>,
108	) {
109		self.handle.spawn(future);
110	}
111}

113/// Type to encapsulate the server handle and listening address.
114pub struct Server {
115	/// Handle to the rpc server
116	handle: ServerHandle,
117	/// Listening address of the server
118	listen_addrs: Vec<SocketAddr>,
119	/// Dedicated RPC runtime (kept alive for the lifetime of the server)
120	rpc_runtime: Option<tokio::runtime::Runtime>,
121}
122
123impl Server {
124	/// Creates a new Server.
125	pub fn new(
126		handle: ServerHandle,
127		listen_addrs: Vec<SocketAddr>,
128		rpc_runtime: tokio::runtime::Runtime,
129	) -> Server {
130		Server { handle, listen_addrs, rpc_runtime: Some(rpc_runtime) }
131	}
132
133	/// Returns the `jsonrpsee::server::ServerHandle` for this Server. Can be used to stop the
134	/// server.
135	pub fn handle(&self) -> &ServerHandle {
136		&self.handle
137	}
138
139	/// The listen address for the running RPC service.
140	pub fn listen_addrs(&self) -> &[SocketAddr] {
141		&self.listen_addrs
142	}
143
144	/// Returns the spawn handle for tasks on the dedicated RPC runtime.
145	pub fn spawn_handle(&self) -> Arc<dyn sp_core::traits::SpawnNamed> {
146		Arc::new(RpcSpawnHandle::new(
147			self.rpc_runtime
148				.as_ref()
149				.expect("rpc_runtime is only taken in Drop; qed")
150				.handle()
151				.clone(),
152		))
153	}
154}
155
156impl Drop for Server {
157	fn drop(&mut self) {
158		// This doesn't not wait for the server to be stopped but fires the signal.
159		let _ = self.handle.stop();
160
161		// Use `shutdown_background()` to avoid blocking, which would panic if
162		// we are being dropped from within an async context.
163		if let Some(runtime) = self.rpc_runtime.take() {
164			runtime.shutdown_background();
165		}
166	}
167}

177/// RPC server configuration.
178pub struct Config<M: Send + Sync + 'static> {
179	/// RPC interfaces to start.
180	pub endpoints: Vec<RpcEndpoint>,
181	/// Metrics.
182	pub metrics: Option<RpcMetrics>,
183	/// RPC API module.
184	pub rpc_api: RpcModule<M>,
185	/// Subscription ID provider.
186	pub id_provider: Option<Box<dyn SubscriptionIdProvider>>,
187	/// RPC logger capacity (default: 1024).
188	pub request_logger_limit: u32,
189	/// Dedicated RPC runtime.
190	pub rpc_runtime: tokio::runtime::Runtime,
191}

193/// Guards the dedicated RPC runtime while [`start_server`] is setting the server up.
194///
195/// `start_server` is awaited from within a runtime context, so dropping the RPC runtime on an
196/// early error return would panic: `Runtime::drop` blocks until the blocking pool has shut down,
197/// which tokio forbids on a runtime thread. Routing every exit path through this guard tears the
198/// runtime down with `shutdown_background()` instead, just like [`Server::drop`] does.
199struct RuntimeGuard(Option<tokio::runtime::Runtime>);
200
201impl RuntimeGuard {
202	fn new(runtime: tokio::runtime::Runtime) -> Self {
203		Self(Some(runtime))
204	}
205
206	fn handle(&self) -> tokio::runtime::Handle {
207		self.0
208			.as_ref()
209			.expect("runtime is only taken in `into_inner` and `Drop`; qed")
210			.handle()
211			.clone()
212	}
213
214	/// Hands the runtime over to the [`Server`], which owns it from then on.
215	fn into_inner(mut self) -> tokio::runtime::Runtime {
216		self.0.take().expect("runtime is only taken here and in `Drop`; qed")
217	}
218}
219
220impl Drop for RuntimeGuard {
221	fn drop(&mut self) {
222		if let Some(runtime) = self.0.take() {
223			// Same trade-off `Server::drop` accepts
224			runtime.shutdown_background();
225		}
226	}
227}

232	stop_handle: StopHandle,
233	metrics: Option<RpcMetrics>,
234	tokio_handle: tokio::runtime::Handle,
235}

237/// Start RPC server listening on given address.
238pub async fn start_server<M>(config: Config<M>) -> Result<Server, Box<dyn StdError + Send + Sync>>
239where
240	M: Send + Sync,
241{
242	let Config { endpoints, metrics, rpc_api, id_provider, request_logger_limit, rpc_runtime } =
243		config;
244
245	// Held as a guard so that every early error return below shuts the runtime down
246	let rpc_runtime = RuntimeGuard::new(rpc_runtime);
247	let rpc_handle = rpc_runtime.handle();
248
249	let (stop_handle, server_handle) = stop_channel();
250	let rpc_api = build_rpc_api(rpc_api);
251	// Bound the metrics `method` label to the registered methods so its
252	// cardinality stays finite (unknown names collapse to `"unknown"`).
253	let metrics = metrics.map(|m| {
254		let known: Vec<&'static str> = rpc_api.method_names().collect();
255		m.with_known_methods(known)
256	});
257	let cfg = PerConnection {
258		methods: rpc_api.into(),
259		metrics,
260		tokio_handle: rpc_handle.clone(),
261		stop_handle,
262	};
263
264	let mut local_addrs = Vec::new();
265
266	for endpoint in endpoints {
267		let allowed_to_fail = endpoint.is_optional;
268		let local_addr = endpoint.listen_addr;
269
270		let mut listener = match endpoint.bind().await {
271			Ok(l) => l,
272			Err(e) if allowed_to_fail => {
273				log::debug!(target: "rpc", "JSON-RPC server failed to bind optional address: {:?}, error: {:?}", local_addr, e);
274				continue;
275			},
276			Err(e) => return Err(e),
277		};
278		let local_addr = listener.local_addr();
279		local_addrs.push(local_addr);
280		let cfg = cfg.clone();
281
282		let RpcSettings {
283			batch_config,
284			max_connections,
285			max_payload_in_mb,
286			max_payload_out_mb,
287			max_buffer_capacity_per_connection,
288			max_subscriptions_per_connection,
289			rpc_methods,
290			rate_limit_trust_proxy_headers,
291			rate_limit_whitelisted_ips,
292			host_filter,
293			cors,
294			rate_limit,
295		} = listener.rpc_settings();
296
297		let http_middleware = tower::ServiceBuilder::new()
298			.option_layer(host_filter)
299			// Proxy `GET /health, /health/readiness` requests to the internal
300			// `system_health` method.
301			.layer(NodeHealthProxyLayer::default())
302			.layer(cors);
303
304		let mut builder = jsonrpsee::server::Server::builder()
305			.max_request_body_size(max_payload_in_mb.saturating_mul(MEGABYTE))
306			.max_response_body_size(max_payload_out_mb.saturating_mul(MEGABYTE))
307			.max_connections(max_connections)
308			.max_subscriptions_per_connection(max_subscriptions_per_connection)
309			.enable_ws_ping(
310				PingConfig::new()
311					.ping_interval(Duration::from_secs(30))
312					.inactive_limit(Duration::from_secs(60))
313					.max_failures(3),
314			)
315			.set_http_middleware(http_middleware)
316			.set_message_buffer_capacity(max_buffer_capacity_per_connection)
317			.set_batch_request_config(batch_config)
318			.custom_tokio_runtime(rpc_handle.clone());
319
320		if let Some(provider) = id_provider.clone() {
321			builder = builder.set_id_provider(provider);
322		} else {
323			builder = builder.set_id_provider(RandomStringIdProvider::new(16));
324		};
325
326		let service_builder = builder.to_service_builder();
327		let deny_unsafe = deny_unsafe(&local_addr, &rpc_methods);
328
329		rpc_handle.spawn(async move {
330			loop {
331				let (sock, remote_addr) = tokio::select! {
332					res = listener.accept() => {
333						match res {
334							Ok(s) => s,
335							Err(e) => {
336								log::debug!(target: "rpc", "Failed to accept connection: {:?}", e);
337								continue;
338							}
339						}
340					}
341					_ = cfg.stop_handle.clone().shutdown() => break,
342				};
343
344				let ip = remote_addr.ip();
345				let cfg2 = cfg.clone();
346				let service_builder2 = service_builder.clone();
347				let rate_limit_whitelisted_ips2 = rate_limit_whitelisted_ips.clone();
348
349				let svc =
350					tower::service_fn(move |mut req: http::Request<hyper::body::Incoming>| {
351						req.extensions_mut().insert(deny_unsafe);
352
353						let PerConnection { methods, metrics, tokio_handle, stop_handle } =
354							cfg2.clone();
355						let service_builder = service_builder2.clone();
356
357						let proxy_ip =
358							if rate_limit_trust_proxy_headers { get_proxy_ip(&req) } else { None };
359
360						let rate_limit_cfg = if rate_limit_whitelisted_ips2
361							.iter()
362							.any(|ips| ips.contains(proxy_ip.unwrap_or(ip)))
363						{
364							log::debug!(target: "rpc", "ip={ip}, proxy_ip={:?} is trusted, disabling rate-limit", proxy_ip);
365							None
366						} else {
367							if !rate_limit_whitelisted_ips2.is_empty() {
368								log::debug!(target: "rpc", "ip={ip}, proxy_ip={:?} is not trusted, rate-limit enabled", proxy_ip);
369							}
370							rate_limit
371						};
372
373						let is_websocket = ws::is_upgrade_request(&req);
374						let transport_label = if is_websocket { "ws" } else { "http" };
375
376						let middleware_layer = match (metrics, rate_limit_cfg) {
377							(None, None) => None,
378							(Some(metrics), None) => Some(
379								MiddlewareLayer::new()
380									.with_metrics(Metrics::new(metrics, transport_label)),
381							),
382							(None, Some(rate_limit)) =>
383								Some(MiddlewareLayer::new().with_rate_limit_per_minute(rate_limit)),
384							(Some(metrics), Some(rate_limit)) => Some(
385								MiddlewareLayer::new()
386									.with_metrics(Metrics::new(metrics, transport_label))
387									.with_rate_limit_per_minute(rate_limit),
388							),
389						};
390
391						let rpc_middleware = RpcServiceBuilder::new()
392							.rpc_logger(request_logger_limit)
393							.option_layer(middleware_layer.clone());
394						let mut svc = service_builder
395							.set_rpc_middleware(rpc_middleware)
396							.build(methods, stop_handle);
397
398						async move {
399							if is_websocket {
400								let on_disconnect = svc.on_session_closed();
401
402								// Spawn a task to handle when the connection is closed.
403								tokio_handle.spawn(async move {
404									let now = std::time::Instant::now();
405									middleware_layer.as_ref().map(|m| m.ws_connect());
406									on_disconnect.await;
407									middleware_layer.as_ref().map(|m| m.ws_disconnect(now));
408								});
409							}
410
411							// https://github.com/rust-lang/rust/issues/102211 the error type can't be inferred
412							// to be `Box<dyn std::error::Error + Send + Sync>` so we need to
413							// convert it to a concrete type as workaround.
414							svc.call(req).await.map_err(|e| BoxError::from(e))
415						}
416					});
417
418				cfg.tokio_handle.spawn(serve_with_graceful_shutdown(
419					sock,
420					svc,
421					cfg.stop_handle.clone().shutdown(),
422				));
423			}
424		});
425	}
426
427	if local_addrs.is_empty() {
428		return Err(Box::new(ListenAddrError));
429	}
430
431	// The previous logging format was before
432	// `Running JSON-RPC server: addr=127.0.0.1:9944, allowed origins=["*"]`
433	//
434	// The new format is `Running JSON-RPC server: addr=<addr1, addr2, .. addr_n>`
435	// with the exception that for a single address it will be `Running JSON-RPC server: addr=addr,`
436	// with a trailing comma.
437	//
438	// This is to make it work with old scripts/utils that parse the logs.
439	log::info!("Running JSON-RPC server: addr={}", format_listen_addrs(&local_addrs));
440
441	Ok(Server::new(server_handle, local_addrs, rpc_runtime.into_inner()))
442}
