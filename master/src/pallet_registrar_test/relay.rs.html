156pub struct MockSwap;
157impl OnSwap for MockSwap {
158	fn on_swap(one: ParaId, other: ParaId) {
159		let mut swap_data = SwapData::get();
160		let one_data = swap_data.remove(&one).unwrap_or_default();
161		let other_data = swap_data.remove(&other).unwrap_or_default();
162		swap_data.insert(one, other_data);
163		swap_data.insert(other, one_data);
164		SwapData::set(swap_data);
165	}

353/// Drive the chain to `n`, rotating sessions every [`BLOCKS_PER_SESSION`] blocks.
354///
355/// Lifted from `paras_registrar/mock.rs`: onboarding only takes effect on a session boundary, so
356/// nothing registers without this.
357pub fn run_to_block(n: BlockNumber) {
358	System::run_to_block_with::<AllPalletsWithSystem>(
359		n,
360		frame_system::RunToBlockHooks::default().before_finalize(|bn| {
361			if (bn + 1) % BLOCKS_PER_SESSION == 0 {
362				let session_index = shared::CurrentSessionIndex::<Runtime>::get() + 1;
363				let keys = crate::VALIDATORS.iter().map(|v| v.public().into()).collect();
364
365				shared::Pallet::<Runtime>::set_session_index(session_index);
366				shared::Pallet::<Runtime>::set_active_validators_ascending(keys);
367
368				Parachains::test_on_new_session();
369			}
370		}),
371	);
372}
373
374/// Advance whole sessions.
375pub fn run_to_session(n: BlockNumber) {
376	run_to_block(n * BLOCKS_PER_SESSION);
377}

379/// Have the validators approve `code`, so onboarding can proceed.
380///
381/// Copied from `polkadot_runtime_common::mock::conclude_pvf_checking`, which is `#[cfg(test)]`
382/// and therefore not importable.
383pub fn conclude_pvf_checking(validation_code: &ValidationCode, session_index: u32) {
384	use polkadot_primitives::PvfCheckStatement;
385
386	let num_required = polkadot_primitives::supermajority_threshold(crate::VALIDATORS.len());
387	crate::VALIDATORS.iter().enumerate().take(num_required).for_each(|(idx, key)| {
388		let statement = PvfCheckStatement {
389			accept: true,
390			subject: validation_code.hash(),
391			session_index,
392			validator_index: (idx as u32).into(),
393		};
394		let signature = key.sign(&statement.signing_payload());
395		let _ = paras::Pallet::<Runtime>::include_pvf_check_statement(
396			frame_system::Origin::<Runtime>::None.into(),
397			statement,
398			signature.into(),
399		);
400	});
