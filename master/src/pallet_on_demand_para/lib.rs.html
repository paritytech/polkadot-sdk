18//! # On-demand Coretime pallet
19//!
20//! Sale of on-demand Coretime from the Coretime chain.
21//!
22//! The Relay chain owns the actual on-demand order queue; this pallet is a front-end for it which
23//! lives alongside [`pallet-broker`](https://docs.rs/pallet-broker) on the Coretime chain:
24//!
25//! - Users call [`Pallet::place_order`] here and pay the spot price in the local currency.
26//! - Because the real queue is one hop away, the spot price cannot be read directly. Instead the
27//!   pallet keeps a local estimate of the queue's depth ([`QueueState`]) which grows with every
28//!   order placed and shrinks by an assumed drain rate per Relay-chain block. The spot price is
29//!   derived from that estimate using [`PriceParameters`].
30//! - Orders accepted within a block are accumulated in [`PendingBatch`] and forwarded to the Relay
31//!   chain in one go on finalization, via [`QueueOnDemandOrders`].
32//! - The spot prices paid are collected in the pallet's pot and booked as [`Revenue`] against the
33//!   timeslice the order was placed in. `pallet-broker` claims the revenue of each timeslice once
34//!   it has ended, along with that of any earlier timeslice which could not be paid out before,
35//!   through [`fp_coretime::revenue::OnDemandRevenue`], and pays it out to the contributors to the
36//!   Instantaneous Coretime Pool.
37//!
38//!   NOTE: It is important to make sure that this pallet is ordered before `ParachainSystem` in the
39//!   runtime - otherwise the messages will not be sent in the block in which they are created,
40//!   introducing additional latency.

321	impl<T: Config> Pallet<T> {
322		/// Configure the pallet.
323		///
324		/// - `origin`: Must be Root or pass `AdminOrigin`.
325		/// - `config`: The configuration for this pallet.
326		#[pallet::call_index(0)]
327		pub fn configure(
328			origin: OriginFor<T>,
329			config: PriceParametersOf<T>,
330		) -> DispatchResultWithPostInfo {
331			T::AdminOrigin::ensure_origin_or_root(origin)?;
332			config.validate::<T>()?;
333			PriceConfig::<T>::put(config);
334			Ok(Pays::No.into())
335		}
336
337		/// Place an on-demand Coretime order for `para_id`.
338		///
339		/// The caller is charged the current estimated spot price, which must not exceed
340		/// `max_amount`. The order is forwarded to the Relay chain at the end of the block.
341		///
342		/// - `origin`: Must be a signed account with enough funds to pay the spot price.
343		/// - `para_id`: The parachain to schedule.
344		/// - `max_amount`: The maximum spot price the caller is willing to pay.
345		#[pallet::call_index(1)]
346		#[pallet::weight(
347			<T as Config>::WeightInfo::place_order()
348			.saturating_add(T::WeightInfo::on_finalize_block_per_order())
349		)]
350		pub fn place_order(
351			origin: OriginFor<T>,
352			para_id: TaskId,
353			max_amount: BalanceOf<T>,
354		) -> DispatchResult {
355			let who = ensure_signed(origin)?;
356			// TODO(ahm-v2): add a check that the para_id is valid
357			Self::do_place_order(who, para_id, max_amount)
358		}
359
360		/// Enable on-demand Coretime sales.
361		///
362		/// Until they are enabled, no orders are accepted. Once they are, they cannot be disabled
363		/// again, and calling this does nothing.
364		///
365		/// This should be called after the AHM v2 migration is completed.
366		///
367		/// - `origin`: Must be Root or pass `AdminOrigin`.
368		#[pallet::call_index(2)]
369		pub fn enable(origin: OriginFor<T>) -> DispatchResultWithPostInfo {
370			T::AdminOrigin::ensure_origin_or_root(origin)?;
371			if !Enabled::<T>::get() {
372				Enabled::<T>::put(true);
373				Self::deposit_event(Event::<T>::Enabled);
374			}
375			Ok(Pays::No.into())
376		}
377	}

379	#[cfg(any(feature = "try-runtime", test))]
380	impl<T: Config> Pallet<T> {
381		/// Check the invariants of the pallet's storage.
382		pub(crate) fn do_try_state() -> Result<(), sp_runtime::TryRuntimeError> {
383			let revenue = Revenue::<T>::get();
384
385			// No revenue is booked for the future.
386			let now = T::RelayBlockNumberProvider::current_block_number();
387			let current: Timeslice = (now / T::TimeslicePeriod::get()).saturated_into();
388			ensure!(
389				revenue.last_key_value().map_or(true, |(&when, _)| when <= current),
390				"Revenue is booked against a timeslice which has not started yet"
391			);
392
393			// The pot holds the revenue booked. If it does not, the revenue is held back until the
394			// pot is topped up.
395			let booked = revenue
396				.values()
397				.fold(BalanceOf::<T>::zero(), |booked, &amount| booked.saturating_add(amount));
398			let reducible = T::Currency::reducible_balance(&Self::account_id(), Preserve, Polite);
399			ensure!(reducible >= booked, "The pot holds less than can be paid out as revenue");
400
401			Ok(())
402		}
403	}

405	impl<T: Config> Pallet<T> {
406		/// The account holding the revenue from on-demand Coretime sales.
407		pub fn account_id() -> T::AccountId {
408			T::PalletId::get().into_account_truncating()
409		}
410
411		pub(crate) fn do_place_order(
412			who: T::AccountId,
413			para_id: TaskId,
414			max_amount: BalanceOf<T>,
415		) -> DispatchResult {
416			// Fail early if the parachain does not yet accept orders (so the migration is not yet
417			// complete).
418			ensure!(Enabled::<T>::get(), Error::<T>::Disabled);
419			// Fail early if the batch is already full.
420			ensure!(
421				PendingBatch::<T>::decode_len().unwrap_or(0) < T::MaxBatchSize::get() as usize,
422				Error::<T>::BatchFull
423			);
424			// Fail early if the account can't cover the declared max_amount.
425			ensure!(
426				T::Currency::reducible_balance(&who, Preserve, Polite) >= max_amount,
427				Error::<T>::InsufficientFunds
428			);
429
430			let pool_cores = T::PoolCapacityProvider::pool_cores();
431			// Fail early if the pool is empty.
432			ensure!(pool_cores > 0, Error::<T>::EmptyPool);
433
434			let now = T::RelayBlockNumberProvider::current_block_number();
435			let mut queue_state = QueueState::<T>::get()
436				.unwrap_or(QueueTracker { outstanding_orders: 0, last_updated: now });
437			let pricing_config = PriceConfig::<T>::get();
438
439			// Assume the Relay chain has drained part of the queue since we last looked at it.
440			let elapsed = now.saturating_sub(queue_state.last_updated).saturated_into();
441
442			let drained_orders = pricing_config
443				.drain_rate_per_block
444				.saturating_mul(elapsed)
445				.saturating_mul(pool_cores);
446			let outstanding_orders = queue_state.outstanding_orders.saturating_sub(drained_orders);
447
448			ensure!(outstanding_orders < pricing_config.order_cap, Error::<T>::QueueFull);
449
450			let spot_price = T::PricingProvider::spot_price(&pricing_config, outstanding_orders)?;
451
452			ensure!(spot_price <= max_amount, Error::<T>::SpotPriceHigherThanMaxAmount);
453
454			// Charge the sending account the spot price. The funds stay in the pot until the
455			// revenue they represent is claimed.
456			T::Currency::transfer(&who, &Self::account_id(), spot_price, Preserve)?;
457			let when: Timeslice = (now / T::TimeslicePeriod::get()).saturated_into();
458			Self::book_revenue(when, spot_price);
459
460			// Add the order to the batch that gets sent to the Relay chain on finalization.
461			PendingBatch::<T>::try_mutate(|batch| {
462				batch
463					.try_push(EnqueuedOrder { para_id, ordered_at: now })
464					// should not happen (we check if it's full at the start), but won't hurt to
465					// handle the error
466					.map_err(|_| Error::<T>::BatchFull)
467			})?;
468
469			queue_state.outstanding_orders = outstanding_orders.saturating_add(1);
470			queue_state.last_updated = now;
471			QueueState::<T>::put(queue_state);
472
473			Self::deposit_event(Event::<T>::OrderPlaced { para_id, spot_price, ordered_by: who });
474
475			Ok(())
476		}
477
478		/// Book `amount` of revenue against timeslice `when`.
479		fn book_revenue(when: Timeslice, amount: BalanceOf<T>) {
480			let booked = Revenue::<T>::get().try_mutate(|revenue| {
481				revenue.entry(when).or_insert_with(Zero::zero).saturating_accrue(amount);
482				if revenue.len() <= T::MaxUnclaimedTimeslices::get() as usize {
483					return;
484				}
485
486				// The revenue of the oldest timeslice has not been claimed for too long; keep it
487				// with the next oldest one.
488				let Some((overdue, overdue_amount)) = revenue.pop_first() else { return };
489				log::warn!(
490					target: LOG_TARGET,
491					"Unclaimed revenue of timeslice {overdue} merged into a later timeslice",
492				);
493				// This will never lose revenue: if we had to remove an entry above, the number of
494				// entries must have been over the limit, which is at least 2. So there are at least
495				// two entries left at this point, and first_entry will never return None.
496				if let Some(mut oldest) = revenue.first_entry() {
497					oldest.get_mut().saturating_accrue(overdue_amount);
498				}
499			});
500			match booked {
501				Some(revenue) => Revenue::<T>::put(revenue),
502				// At most one timeslice too many is booked, and then the oldest one is merged away.
503				None => {
504					frame_support::defensive!("Revenue could not be booked");
505				},
506			}
507		}
508	}

524	impl<T: Config> Pallet<T> {
525		/// Implements [`OnDemandRevenue::claim_revenue`], returning the paid out revenue and the
526		/// number of timeslices whose revenue was claimable.
527		fn do_claim_revenue(
528			up_to: Timeslice,
529			beneficiary: &T::AccountId,
530		) -> (BTreeMap<Timeslice, BalanceOf<T>>, u32) {
531			let mut revenue = Revenue::<T>::get().into_inner();
532			let pot = Self::account_id();
533			let maximum_transfer_amount = T::Currency::reducible_balance(&pot, Preserve, Polite);
534
535			let mut claimable_entries = 0u32;
536			let mut total = BalanceOf::<T>::zero();
537			let mut total_ignoring_max_transfer_amount = BalanceOf::<T>::zero();
538			let mut paid = BTreeMap::new();
539			let mut is_enough_funds_for_transfer = true;
540			for (&when, &amount) in revenue.range(..=up_to) {
541				claimable_entries += 1;
542
543				total_ignoring_max_transfer_amount =
544					total_ignoring_max_transfer_amount.saturating_add(amount);
545
546				if total_ignoring_max_transfer_amount > maximum_transfer_amount {
547					is_enough_funds_for_transfer = false;
548				}
549
550				if is_enough_funds_for_transfer {
551					total = total_ignoring_max_transfer_amount;
552					paid.insert(when, amount);
553				}
554			}
555
556			// Early return when nothing to be claimed: everything up to `up_to` is already paid
557			// out, so it is settled even without revenue.
558			if claimable_entries == 0 {
559				return (BTreeMap::from([(up_to, Zero::zero())]), 0);
560			}
561
562			if is_enough_funds_for_transfer {
563				paid.entry(up_to).or_default();
564			} else {
565				log::error!(
566					target: LOG_TARGET,
567					"The pot holds less than the {:?} of revenue booked up to timeslice {}; paying \
568					 out {:?} instead.",
569					total_ignoring_max_transfer_amount, up_to, total,
570				);
571				Self::deposit_event(Event::<T>::UnexpectedLowFunds {
572					attempted_to_claim: total_ignoring_max_transfer_amount,
573					claimable: maximum_transfer_amount,
574					paid_out: total,
575				});
576			}
577
578			if !total.is_zero() {
579				if let Err(err) = T::Currency::transfer(&pot, beneficiary, total, Preserve) {
580					log::error!(
581						target: LOG_TARGET,
582						"Paying out {total:?} of revenue failed: {err:?}",
583					);
584					return (BTreeMap::new(), claimable_entries);
585				}
586			}
587
588			// Everything up to the last timeslice paid out has been claimed, so drop it.
589			if let Some((&last_paid, _)) = paid.last_key_value() {
590				revenue = revenue.split_off(&last_paid);
591				// In the unlikely case that last_paid is Timeslice::MAX, doing this separately
592				// instead of splitting at last_paid + 1 avoids an overflow.
593				revenue.remove(&last_paid);
594			}
595			Revenue::<T>::put(
596				BoundedBTreeMap::try_from(revenue)
597					.expect("entries were only removed from a bounded map; qed"),
598			);
599
600			for (&when, &amount) in paid.iter().filter(|(_, amount)| !amount.is_zero()) {
601				Self::deposit_event(Event::<T>::RevenueClaimed {
602					when,
603					amount,
604					beneficiary: beneficiary.clone(),
605				});
606			}
607
608			(paid, claimable_entries)
609		}
610	}
611}
