40/// The parameters used for pricing on-demand orders.
41#[derive(
42	Encode, Decode, DecodeWithMemTracking, Clone, PartialEq, Eq, Debug, TypeInfo, MaxEncodedLen,
43)]
44pub struct PriceParameters<Balance> {
45	/// The maximum number of outstanding orders beyond which we reject new orders.
46	pub order_cap: u32,
47	/// The number of outstanding orders assumed to be drained per Relay-chain block, per pool
48	/// core.
49	pub drain_rate_per_block: u32,
50	/// The spot price increase per outstanding order in the queue.
51	pub price_step: Perbill,
52	/// The base spot price when the queue is empty.
53	pub base_fee: Balance,
54}
55
56impl<T> Default for PriceParameters<T>
57where
58	T: From<u32>,
59{
60	fn default() -> Self {
61		Self {
62			order_cap: DEFAULT_ORDER_CAP,
63			drain_rate_per_block: DEFAULT_DRAIN_RATE_PER_BLOCK,
64			price_step: Perbill::from_percent(DEFAULT_PRICE_STEP),
65			base_fee: T::from(DEFAULT_BASE_FEE),
66		}
67	}
68}
69
70impl<Balance> PriceParameters<Balance> {
71	pub(crate) fn validate<T>(&self) -> Result<(), Error<T>>
72	where
73		T: Config,
74		T::Currency: Inspect<<T as SConfig>::AccountId, Balance = Balance>,
75		Balance: frame_support::traits::tokens::Balance,
76	{
77		ensure!(
78			T::PricingProvider::spot_price(self, self.order_cap).is_ok(),
79			Error::<T>::OrderPriceCanOverflow
80		);
81		Ok(())
82	}
83}

117/// Type for determining the prices of on-demand orders
118pub trait PricingProvider<T: Balance> {
119	/// Returns the current spot price of an on-demand order.
120	fn spot_price(price_config: &PriceParameters<T>, queue_depth: u32) -> Result<T, DispatchError>;
121}
122
123/// A type implementing the default algorithm for determining the order price.
124pub struct DefaultPricingProvider;
125
126impl<T: Balance> PricingProvider<T> for DefaultPricingProvider {
127	fn spot_price(price_config: &PriceParameters<T>, queue_depth: u32) -> Result<T, DispatchError> {
128		// Every order already outstanding raises the price by `price_step`.
129		let price_adjustment = ensure_pow(
130			FixedU128::one().saturating_add(FixedU128::from_perbill(price_config.price_step)),
131			queue_depth as usize,
132		)?;
133		Ok(price_adjustment.saturating_mul_int(price_config.base_fee))
134	}
135}
