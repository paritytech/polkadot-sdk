17//! Vote import logic.
18//!
19//! This module encapsulates the actual logic for importing new votes and provides easy access of
20//! the current state for votes for a particular candidate.
21//!
22//! In particular there is `CandidateVoteState` which tells what can be concluded for a particular
23//! set of votes. E.g. whether a dispute is ongoing, whether it is confirmed, concluded, ..
24//!
25//! Then there is `ImportResult` which reveals information about what changed once additional votes
26//! got imported on top of an existing `CandidateVoteState` and reveals "dynamic" information, like
27//! whether due to the import a dispute was raised/got confirmed, ...
28

44/// (Session) environment of a candidate.
45pub struct CandidateEnvironment<'a> {
46	/// The scheduling session the candidate appeared in.
47	scheduling_session: SessionIndex,
48	/// Session for above index.
49	session: &'a SessionInfo,
50	/// Validator indices controlled by this node.
51	controlled_indices: HashSet<ValidatorIndex>,
52	/// Indices of on-chain disabled validators at the `scheduling_parent` combined
53	/// with the off-chain state.
54	disabled_indices: HashSet<ValidatorIndex>,
55}
56
57#[overseer::contextbounds(DisputeCoordinator, prefix = self::overseer)]
58impl<'a> CandidateEnvironment<'a> {
59	/// Create `CandidateEnvironment`.
60	///
61	/// Return: `None` in case session is outside of session window.
62	pub async fn new<Context>(
63		ctx: &mut Context,
64		runtime_info: &'a mut RuntimeInfo,
65		scheduling_session: SessionIndex,
66		scheduling_parent: Hash,
67		disabled_offchain: impl IntoIterator<Item = ValidatorIndex>,
68		controlled_indices: &mut ControlledValidatorIndices,
69	) -> Option<CandidateEnvironment<'a>> {
70		// We use the scheduling parent here to have consensus on disabled state among validators.
71		// If this fetch fails because e.g. we have never seen the fork of the candidate, we are
72		// still fine, because we have spam protection for these cases in place anyways.
73		let disabled_onchain = runtime_info
74			.get_disabled_validators(ctx.sender(), scheduling_parent)
75			.await
76			.unwrap_or_else(|err| {
77				gum::info!(target: LOG_TARGET, ?err, "Failed to get disabled validators");
78				Vec::new()
79			});
80
81		// Using the scheduling parent here is fine, because we warm the cache on active leaves
82		// update, thus this call will succeed even if the scheduling parent's state is not
83		// available.
84		let session = match runtime_info
85			.get_session_info_by_index(ctx.sender(), scheduling_parent, scheduling_session)
86			.await
87		{
88			Ok(extended_session_info) => &extended_session_info.session_info,
89			Err(_) => return None,
90		};
91
92		let n_validators = session.validators.len();
93		let byzantine_threshold = polkadot_primitives::byzantine_threshold(n_validators);
94		// combine on-chain with off-chain disabled validators
95		// process disabled validators in the following order:
96		// - on-chain disabled validators
97		// - prioritized order of off-chain disabled validators
98		// deduplicate the list and take at most `byzantine_threshold` validators
99		let disabled_indices = {
100			let mut d: HashSet<ValidatorIndex> = HashSet::new();
101			for v in disabled_onchain.into_iter().chain(disabled_offchain.into_iter()) {
102				if d.len() == byzantine_threshold {
103					break;
104				}
105				d.insert(v);
106			}
107			d
108		};
109
110		let controlled_indices = controlled_indices
111			.get(scheduling_session, &session.validators)
112			.map_or(HashSet::new(), |index| HashSet::from([index]));
113
114		Some(Self { scheduling_session, session, controlled_indices, disabled_indices })
115	}
116

119		&self.session.validators
120	}

122	/// `SessionInfo` for the candidate's session.

125	}

143/// Whether or not we already issued some statement about a candidate.
144pub enum OwnVoteState {
145	/// Our votes, if any.
146	Voted(Vec<(ValidatorIndex, (DisputeStatement, ValidatorSignature))>),
147
148	/// We are not a parachain validator in the session.
149	///
150	/// Hence we cannot vote.
151	CannotVote,
152}
153
154impl OwnVoteState {
155	fn new(votes: &CandidateVotes, env: &CandidateEnvironment) -> Self {
156		let controlled_indices = env.controlled_indices();
157		if controlled_indices.is_empty() {
158			return Self::CannotVote;
159		}
160
161		let our_valid_votes = controlled_indices
162			.iter()
163			.filter_map(|i| votes.valid.raw().get_key_value(i))
164			.map(|(index, (kind, sig))| {
165				(*index, (DisputeStatement::Valid(kind.clone()), sig.clone()))
166			});
167		let our_invalid_votes = controlled_indices
168			.iter()
169			.filter_map(|i| votes.invalid.get_key_value(i))
170			.map(|(index, (kind, sig))| (*index, (DisputeStatement::Invalid(*kind), sig.clone())));
171
172		Self::Voted(our_valid_votes.chain(our_invalid_votes).collect())
173	}
174
175	/// Is a vote from us missing but we are a validator able to vote?
176	fn vote_missing(&self) -> bool {
177		match self {
178			Self::Voted(votes) if votes.is_empty() => true,
179			Self::Voted(_) | Self::CannotVote => false,
180		}
181	}
182
183	/// Get own approval votes, if any.
184	///
185	/// Empty iterator means, no approval votes. `None` means, there will never be any (we cannot
186	/// vote).
187	fn approval_votes(
188		&self,
189	) -> Option<impl Iterator<Item = (ValidatorIndex, &ValidatorSignature)>> {
190		match self {
191			Self::Voted(votes) => Some(votes.iter().filter_map(|(index, (kind, sig))| {
192				if let DisputeStatement::Valid(ValidDisputeStatementKind::ApprovalChecking) = kind {
193					Some((*index, sig))
194				} else {
195					None
196				}
197			})),
198			Self::CannotVote => None,
199		}
200	}
201
202	/// Get our votes if there are any.
203	///
204	/// Empty iterator means, no votes. `None` means, there will never be any (we cannot
205	/// vote).
206	fn votes(&self) -> Option<&Vec<(ValidatorIndex, (DisputeStatement, ValidatorSignature))>> {
207		match self {
208			Self::Voted(votes) => Some(&votes),
209			Self::CannotVote => None,
210		}
211	}
212}

214/// Complete state of votes for a candidate.
215///
216/// All votes + information whether a dispute is ongoing, confirmed, concluded, whether we already
217/// voted, ...
218pub struct CandidateVoteState<Votes> {
219	/// Votes already existing for the candidate + receipt.
220	votes: Votes,
221
222	/// Information about own votes:
223	own_vote: OwnVoteState,
224
225	/// Current dispute status, if there is any.
226	dispute_status: Option<DisputeStatus>,
227
228	/// Are there `byzantine threshold + 1` invalid votes
229	byzantine_threshold_against: bool,
230}
231
232impl CandidateVoteState<CandidateVotes> {
233	/// Create an empty `CandidateVoteState`
234	///
235	/// in case there have not been any previous votes.
236	pub fn new_from_receipt(candidate_receipt: CandidateReceipt) -> Self {
237		let votes = CandidateVotes {
238			candidate_receipt,
239			valid: ValidCandidateVotes::new(),
240			invalid: BTreeMap::new(),
241		};
242		Self {
243			votes,
244			own_vote: OwnVoteState::CannotVote,
245			dispute_status: None,
246			byzantine_threshold_against: false,
247		}
248	}
249
250	/// Create a new `CandidateVoteState` from already existing votes.
251	pub fn new(votes: CandidateVotes, env: &CandidateEnvironment, now: Timestamp) -> Self {
252		let own_vote = OwnVoteState::new(&votes, env);
253
254		let n_validators = env.validators().len();
255
256		let supermajority_threshold = polkadot_primitives::supermajority_threshold(n_validators);
257
258		// We have a dispute, if we have votes on both sides, with at least one invalid vote
259		// from non-disabled validator or with votes on both sides and confirmed.
260		let has_non_disabled_invalid_votes =
261			votes.invalid.keys().any(|i| !env.disabled_indices().contains(i));
262		let byzantine_threshold = polkadot_primitives::byzantine_threshold(n_validators);
263		let votes_on_both_sides = !votes.valid.raw().is_empty() && !votes.invalid.is_empty();
264		let is_confirmed =
265			votes_on_both_sides && (votes.voted_indices().len() > byzantine_threshold);
266		let is_disputed =
267			is_confirmed || (has_non_disabled_invalid_votes && !votes.valid.raw().is_empty());
268
269		let (dispute_status, byzantine_threshold_against) = if is_disputed {
270			let mut status = DisputeStatus::active();
271			if is_confirmed {
272				status = status.confirm();
273			};
274			let concluded_for = votes.valid.raw().len() >= supermajority_threshold;
275			if concluded_for {
276				status = status.conclude_for(now);
277			};
278
279			let concluded_against = votes.invalid.len() >= supermajority_threshold;
280			if concluded_against {
281				status = status.conclude_against(now);
282			};
283			(Some(status), votes.invalid.len() > byzantine_threshold)
284		} else {
285			(None, false)
286		};
287
288		Self { votes, own_vote, dispute_status, byzantine_threshold_against }
289	}

291	/// Import fresh statements.
292	///
293	/// Result will be a new state plus information about things that changed due to the import.
294	pub fn import_statements(
295		self,
296		env: &CandidateEnvironment,
297		statements: Vec<(SignedDisputeStatement, ValidatorIndex)>,
298		now: Timestamp,
299	) -> ImportResult {
300		let (mut votes, old_state) = self.into_old_state();
301
302		let mut new_invalid_voters = Vec::new();
303		let mut imported_invalid_votes = 0;
304		let mut imported_valid_votes = 0;
305
306		let expected_candidate_hash = votes.candidate_receipt.hash();
307
308		for (statement, val_index) in statements {
309			if env
310				.validators()
311				.get(val_index)
312				.map_or(true, |v| v != statement.validator_public())
313			{
314				gum::error!(
315					target: LOG_TARGET,
316					?val_index,
317					session= ?env.scheduling_session,
318					claimed_key = ?statement.validator_public(),
319					"Validator index doesn't match claimed key",
320				);
321
322				continue;
323			}
324			if statement.candidate_hash() != &expected_candidate_hash {
325				gum::error!(
326					target: LOG_TARGET,
327					?val_index,
328					session= ?env.scheduling_session,
329					given_candidate_hash = ?statement.candidate_hash(),
330					?expected_candidate_hash,
331					"Vote is for unexpected candidate!",
332				);
333				continue;
334			}
335			if statement.session_index() != env.scheduling_session() {
336				gum::error!(
337					target: LOG_TARGET,
338					?val_index,
339					session= ?env.scheduling_session,
340					given_candidate_hash = ?statement.candidate_hash(),
341					?expected_candidate_hash,
342					"Vote is for unexpected session!",
343				);
344				continue;
345			}
346
347			match statement.statement() {
348				DisputeStatement::Valid(valid_kind) => {
349					let fresh = votes.valid.insert_vote(
350						val_index,
351						valid_kind.clone(),
352						statement.into_validator_signature(),
353					);
354					if fresh {
355						imported_valid_votes += 1;
356					}
357				},
358				DisputeStatement::Invalid(invalid_kind) => {
359					let fresh = votes
360						.invalid
361						.insert(val_index, (*invalid_kind, statement.into_validator_signature()))
362						.is_none();
363					if fresh {
364						new_invalid_voters.push(val_index);
365						imported_invalid_votes += 1;
366					}
367				},
368			}
369		}
370
371		let new_state = Self::new(votes, env, now);
372
373		ImportResult {
374			old_state,
375			new_state,
376			imported_invalid_votes,
377			imported_valid_votes,
378			imported_approval_votes: 0,
379			new_invalid_voters,
380		}
381	}

407impl<V> CandidateVoteState<V> {
408	/// Whether or not we have an ongoing dispute.
409	pub fn is_disputed(&self) -> bool {
410		self.dispute_status.is_some()
411	}

414	///
415	/// This checks whether there is a dispute ongoing and we have more than byzantine threshold
416	/// votes.
417	pub fn is_confirmed(&self) -> bool {
418		self.dispute_status.map_or(false, |s| s.is_confirmed_concluded())
419	}
420
421	/// Are we a validator in the session, but have not yet voted?
422	pub fn own_vote_missing(&self) -> bool {

424	}

426	/// Own approval votes if any:
427	pub fn own_approval_votes(
428		&self,
429	) -> Option<impl Iterator<Item = (ValidatorIndex, &ValidatorSignature)>> {
430		self.own_vote.approval_votes()
431	}
432
433	/// Get own votes if there are any.
434	pub fn own_votes(
435		&self,
436	) -> Option<&Vec<(ValidatorIndex, (DisputeStatement, ValidatorSignature))>> {
437		self.own_vote.votes()
438	}
439
440	/// Whether or not there is a dispute and it has already enough valid votes to conclude.
441	pub fn has_concluded_for(&self) -> bool {
442		self.dispute_status.map_or(false, |s| s.has_concluded_for())
443	}
444
445	/// Whether or not there is a dispute and it has already enough invalid votes to conclude.
446	pub fn has_concluded_against(&self) -> bool {
447		self.dispute_status.map_or(false, |s| s.has_concluded_against())
448	}
449
450	/// Get access to the dispute status, in case there is one.
451	pub fn dispute_status(&self) -> &Option<DisputeStatus> {
452		&self.dispute_status
453	}
454
455	/// Access to underlying votes.
456	pub fn votes(&self) -> &V {
457		&self.votes
458	}

461/// An ongoing statement/vote import.
462pub struct ImportResult {
463	/// The state we had before importing new statements.
464	old_state: CandidateVoteState<()>,
465	/// The new state after importing the new statements.
466	new_state: CandidateVoteState<CandidateVotes>,
467	/// New invalid voters as of this import.
468	new_invalid_voters: Vec<ValidatorIndex>,
469	/// Number of successfully imported valid votes.
470	imported_invalid_votes: u32,
471	/// Number of successfully imported invalid votes.
472	imported_valid_votes: u32,
473	/// Number of approval votes imported via `import_approval_votes()`.
474	///
475	/// And only those: If normal import included approval votes, those are not counted here.
476	///
477	/// In other words, without a call `import_approval_votes()` this will always be 0.
478	imported_approval_votes: u32,
479}
480
481impl ImportResult {
482	/// Whether or not anything has changed due to the import.
483	pub fn votes_changed(&self) -> bool {
484		self.imported_valid_votes != 0 || self.imported_invalid_votes != 0
485	}
486
487	/// The dispute state has changed in some way.
488	///
489	/// - freshly disputed
490	/// - freshly confirmed
491	/// - freshly concluded (valid or invalid)
492	pub fn dispute_state_changed(&self) -> bool {
493		self.is_freshly_disputed() || self.is_freshly_confirmed() || self.is_freshly_concluded()
494	}
495
496	/// State as it was before import.
497	pub fn old_state(&self) -> &CandidateVoteState<()> {
498		&self.old_state
499	}
500
501	/// State after import
502	pub fn new_state(&self) -> &CandidateVoteState<CandidateVotes> {
503		&self.new_state
504	}
505
506	/// New "invalid" voters encountered during import.
507	pub fn new_invalid_voters(&self) -> &Vec<ValidatorIndex> {
508		&self.new_invalid_voters
509	}
510
511	/// Number of imported valid votes.
512	pub fn imported_valid_votes(&self) -> u32 {
513		self.imported_valid_votes
514	}
515
516	/// Number of imported invalid votes.
517	pub fn imported_invalid_votes(&self) -> u32 {
518		self.imported_invalid_votes
519	}
520
521	/// Number of imported approval votes.
522	pub fn imported_approval_votes(&self) -> u32 {
523		self.imported_approval_votes
524	}
525
526	/// Whether we now have a dispute and did not prior to the import.
527	pub fn is_freshly_disputed(&self) -> bool {
528		!self.old_state().is_disputed() && self.new_state().is_disputed()
529	}
530
531	/// Whether we just surpassed the byzantine threshold.
532	pub fn is_freshly_confirmed(&self) -> bool {
533		!self.old_state().is_confirmed() && self.new_state().is_confirmed()
534	}
535
536	/// Whether or not any dispute just concluded valid due to the import.
537	pub fn is_freshly_concluded_for(&self) -> bool {
538		!self.old_state().has_concluded_for() && self.new_state().has_concluded_for()
539	}
540
541	/// Whether or not any dispute just concluded invalid due to the import.
542	pub fn is_freshly_concluded_against(&self) -> bool {
543		!self.old_state().has_concluded_against() && self.new_state().has_concluded_against()
544	}
545
546	/// Whether or not any dispute just concluded either invalid or valid due to the import.
547	pub fn is_freshly_concluded(&self) -> bool {
548		self.is_freshly_concluded_against() || self.is_freshly_concluded_for()
549	}
550
551	/// Whether or not the invalid vote count for the dispute went beyond the byzantine threshold
552	/// after the last import
553	pub fn has_fresh_byzantine_threshold_against(&self) -> bool {
554		!self.old_state().byzantine_threshold_against &&
555			self.new_state().byzantine_threshold_against
556	}
557
558	/// Modify this `ImportResult`s, by importing additional approval votes.
559	///
560	/// Both results and `new_state` will be changed as if those approval votes had been in the
561	/// original import.
562	pub fn import_approval_votes(
563		self,
564		env: &CandidateEnvironment,
565		approval_votes: HashMap<ValidatorIndex, (Vec<CandidateHash>, ValidatorSignature)>,
566		now: Timestamp,
567	) -> Self {
568		let Self {
569			old_state,
570			new_state,
571			new_invalid_voters,
572			mut imported_valid_votes,
573			imported_invalid_votes,
574			mut imported_approval_votes,
575		} = self;
576
577		let (mut votes, _) = new_state.into_old_state();
578
579		for (index, (candidate_hashes, sig)) in approval_votes.into_iter() {
580			debug_assert!(
581				{
582					let pub_key = &env
583						.session_info()
584						.validators
585						.get(index)
586						.expect("indices are validated by approval-voting subsystem; qed");
587					let session_index = env.scheduling_session();
588					candidate_hashes.contains(&votes.candidate_receipt.hash()) &&
589						DisputeStatement::Valid(
590							ValidDisputeStatementKind::ApprovalCheckingMultipleCandidates(
591								candidate_hashes.clone(),
592							),
593						)
594						.check_signature(
595							pub_key,
596							*candidate_hashes
597								.first()
598								.expect("Valid votes have at least one candidate; qed"),
599							session_index,
600							&sig,
601						)
602						.is_ok()
603				},
604				"Signature check for imported approval votes failed! This is a serious bug. Session: {:?}, candidate hash: {:?}, validator index: {:?}",
605				env.scheduling_session(),
606				votes.candidate_receipt.hash(),
607				index
608			);
609			if votes.valid.insert_vote(
610				index,
611				// There is a hidden dependency here between approval-voting and this subsystem.
612				// We should be able to start emitting
613				// ValidDisputeStatementKind::ApprovalCheckingMultipleCandidates only after:
614				// 1. Runtime have been upgraded to know about the new format.
615				// 2. All nodes have been upgraded to know about the new format.
616				// Once those two requirements have been met we should be able to increase
617				// max_approval_coalesce_count to values greater than 1.
618				if candidate_hashes.len() > 1 {
619					ValidDisputeStatementKind::ApprovalCheckingMultipleCandidates(candidate_hashes)
620				} else {
621					ValidDisputeStatementKind::ApprovalChecking
622				},
623				sig,
624			) {
625				imported_valid_votes += 1;
626				imported_approval_votes += 1;
627			}
628		}
629
630		let new_state = CandidateVoteState::new(votes, env, now);
631
632		Self {
633			old_state,
634			new_state,
635			new_invalid_voters,
636			imported_valid_votes,
637			imported_invalid_votes,
638			imported_approval_votes,
639		}
640	}
641
642	/// All done, give me those votes.
643	///
