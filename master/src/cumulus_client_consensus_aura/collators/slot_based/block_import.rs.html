40/// The aux storage key used to store the ignored nodes for the given block hash.
41fn ignored_nodes_key<H: Encode>(block_hash: H) -> Vec<u8> {
42	(b"cumulus_slot_based_nodes_to_ignore", block_hash).encode()
43}
44
45/// Prepare a transaction to write the ignored nodes to the aux storage.
46///
47/// Returns the key-value pairs that need to be written to the aux storage.
48fn prepare_ignored_nodes_transaction<Block: BlockT>(
49	block_hash: Block::Hash,
50	ignored_nodes: ProofRecorderIgnoredNodes<Block>,
51) -> impl Iterator<Item = (Vec<u8>, Vec<u8>)> {
52	let key = ignored_nodes_key(block_hash);
53	let encoded_nodes = <ProofRecorderIgnoredNodes<Block> as Encode>::encode(&ignored_nodes);
54
55	[(key, encoded_nodes)].into_iter()
56}
57
58/// Load the ignored nodes associated with a block.
59fn load_ignored_nodes<Block: BlockT, B: AuxStore>(
60	backend: &B,
61	block_hash: Block::Hash,
62) -> ClientResult<Option<ProofRecorderIgnoredNodes<Block>>> {
63	match backend.get_aux(&ignored_nodes_key(block_hash))? {
64		None => Ok(None),
65		Some(t) => ProofRecorderIgnoredNodes::<Block>::decode(&mut &t[..])
66			.map(Some)
67			.map_err(|e| ClientError::Backend(format!("Failed to decode ignored nodes: {}", e))),
68	}
69}

71/// Handle for receiving the block and the storage proof from the [`SlotBasedBlockImport`].
72///
73/// This handle should be passed to [`Params`](super::Params) or can also be dropped if the node is
74/// not running as collator.
75pub struct SlotBasedBlockImportHandle<Block> {
76	receiver: TracingUnboundedReceiver<(Block, StorageProof)>,
77}
78
79impl<Block> SlotBasedBlockImportHandle<Block> {
80	/// Returns the next item.
81	///
82	/// The future will never return when the internal channel is closed.
83	pub async fn next(&mut self) -> (Block, StorageProof) {
84		loop {
85			if self.receiver.is_terminated() {
86				futures::pending!()
87			} else if let Some(res) = self.receiver.next().await {
88				return res;
89			}
90		}
91	}
92}

94/// Register the clean up method for cleaning ignored nodes from blocks on which no further blocks
95/// will be imported.
96fn register_ignored_nodes_cleanup<C, Block>(client: Arc<C>)
97where
98	C: PreCommitActions<Block> + HeaderBackend<Block> + 'static,
99	Block: BlockT,
100{
101	let client_for_closure = client.clone();
102	let on_finality = move |notification: &FinalityNotification<Block>| -> AuxDataOperations {
103		// The old finalized block is the parent of the first block in the tree route,
104		// or the parent of the finalized block if the tree route is empty.
105		let old_finalized_hash = notification
106			.tree_route
107			.first()
108			.and_then(|hash| client_for_closure.header(*hash).ok().flatten())
109			.map(|h| *h.parent_hash())
110			.unwrap_or_else(|| *notification.header.parent_hash());
111
112		notification
113			.stale_blocks
114			.iter()
115			// Delete the ignored nodes for all stale blocks.
116			.map(|b| (ignored_nodes_key(b.hash), None))
117			// We can not delete the ignored nodes for the finalized block, because blocks can still
118			// be imported on top of this block. However, once multiple blocks are finalized at
119			// once, blocks on the route to the finalized parent can no longer become parents
120			// either.
121			.chain(
122				notification
123					.tree_route
124					.iter()
125					.copied()
126					.map(|hash| (ignored_nodes_key(hash), None)),
127			)
128			.chain(std::iter::once((ignored_nodes_key(old_finalized_hash), None)))
129			.collect()
130	};
131
132	client.register_finality_action(Box::new(on_finality));
133}

135/// Special block import for the slot based collator.
136pub struct SlotBasedBlockImport<Block: BlockT, BI, Client> {
137	inner: BI,
138	client: Arc<Client>,
139	sender: TracingUnboundedSender<(Block, StorageProof)>,
140}
141
142impl<Block: BlockT, BI, Client> SlotBasedBlockImport<Block, BI, Client> {
143	/// Create a new instance.
144	///
145	/// The returned [`SlotBasedBlockImportHandle`] needs to be passed to the
146	/// [`Params`](super::Params), so that this block import instance can communicate with the
147	/// collation task. If the node is not running as a collator, just dropping the handle is fine.
148	pub fn new(inner: BI, client: Arc<Client>) -> (Self, SlotBasedBlockImportHandle<Block>)
149	where
150		Client: PreCommitActions<Block> + HeaderBackend<Block> + 'static,
151	{
152		let (sender, receiver) = tracing_unbounded("SlotBasedBlockImportChannel", 1000);
153
154		register_ignored_nodes_cleanup(client.clone());
155
156		(Self { sender, client, inner }, SlotBasedBlockImportHandle { receiver })
157	}
158
159	/// Get the [`ProofRecorderIgnoredNodes`] for `parent`.
160	///
161	/// If `parent` was not part of the same block bundle, the [`ProofRecorderIgnoredNodes`] are not
162	/// required and `None` will be returned.
163	fn get_ignored_nodes(
164		&self,
165		parent: Block::Hash,
166		core_info: &CoreInfo,
167		bundle_info: &BlockBundleInfo,
168		relay_block_identifier: &RelayBlockIdentifier,
169	) -> Option<ProofRecorderIgnoredNodes<Block>>
170	where
171		Client: AuxStore + HeaderBackend<Block> + Send + Sync,
172	{
173		let parent_header = self.client.header(parent).ok().flatten()?;
174		let parent_core_info = CumulusDigestItem::find_core_info(parent_header.digest())?;
175		let parent_bundle_info = CumulusDigestItem::find_block_bundle_info(parent_header.digest())?;
176		let parent_relay_block_identifier =
177			CumulusDigestItem::find_relay_block_identifier(parent_header.digest())?;
178
179		if parent_relay_block_identifier != *relay_block_identifier {
180			tracing::trace!(target: LOG_TARGET, ?parent_relay_block_identifier, ?relay_block_identifier, "Relay block identifier doesn't match");
181			return None;
182		}
183
184		if parent_core_info != *core_info {
185			tracing::trace!(target: LOG_TARGET, ?parent_core_info, ?core_info, "Core info doesn't match");
186			return None;
187		}
188
189		if parent_bundle_info.index.saturating_add(1) != bundle_info.index {
190			tracing::trace!(target: LOG_TARGET, ?parent_bundle_info, ?bundle_info, "Block is not a child, based on the index");
191			return None;
192		}
193
194		match load_ignored_nodes::<Block, _>(&*self.client, parent) {
195			Ok(nodes) => nodes,
196			Err(error) => {
197				tracing::trace!(target: LOG_TARGET, ?parent, ?error, "Failed to load `IgnoredNodes` from aux store");
198				None
199			},
200		}
201	}
202
203	/// Execute the given block and collect the storage proof.
204	///
205	/// We need to execute the block on this level here, because we are collecting the storage
206	/// proofs and combining them for blocks on the same core. So, blocks on the same core do not
207	/// need to include the same trie nodes multiple times and thus, not wasting storage proof size.
208	///
209	/// The proof must be recorded in exactly the same manner as during block building, because the
210	/// proof size is tracked via `ProofSizeExt` and affects runtime state. Without identical proof
211	/// recording, the computed state root would differ and block import would fail.
212	fn execute_block_and_collect_storage_proof(
213		&self,
214		params: &mut sc_consensus::BlockImportParams<Block>,
215	) -> Result<(), sp_consensus::Error>
216	where
217		Client: ProvideRuntimeApi<Block>
218			+ CallApiAt<Block>
219			+ AuxStore
220			+ HeaderBackend<Block>
221			+ Send
222			+ Sync,
223		Client::StateBackend: Send,
224		Client::Api: Core<Block>,
225	{
226		let core_info = CumulusDigestItem::find_core_info(params.header.digest());
227		let bundle_info = CumulusDigestItem::find_block_bundle_info(params.header.digest());
228		let relay_block_identifier =
229			CumulusDigestItem::find_relay_block_identifier(params.header.digest());
230
231		let (Some(core_info), Some(bundle_info), Some(relay_block_identifier)) =
232			(core_info, bundle_info, relay_block_identifier)
233		else {
234			tracing::debug!(
235				target: LOG_TARGET,
236				number = ?params.header.number(),
237				"no bundle digests, skipping execute_block_and_collect_storage_proof",
238			);
239			return Ok(());
240		};
241
242		let parent_hash = *params.header.parent_hash();
243
244		let mut nodes_to_ignore = self
245			.get_ignored_nodes(parent_hash, &core_info, &bundle_info, &relay_block_identifier)
246			.unwrap_or_default();
247
248		let recorder = ProofRecorder::<Block>::with_ignored_nodes(nodes_to_ignore.clone());
249		let proof_size_recorder = RecordingProofSizeProvider::new(recorder.clone());
250
251		let mut runtime_api = self.client.runtime_api();
252
253		// `record_proof_with_recorder` captures trie accesses, while `ProofSizeExt` replays the
254		// proof-size estimations in the same order they were observed during block building.
255		runtime_api.set_call_context(CallContext::Onchain { import: true });
256		runtime_api.record_proof_with_recorder(recorder.clone());
257		runtime_api.register_extension(ProofSizeExt::new(proof_size_recorder.clone()));
258
259		let block = Block::new(params.header.clone(), params.body.clone().unwrap_or_default());
260
261		tracing::debug!(
262			target: LOG_TARGET,
263			?parent_hash,
264			number = ?params.header.number(),
265			?core_info,
266			?bundle_info,
267			"execute_block_and_collect_storage_proof: calling runtime_api.execute_block",
268		);
269
270		runtime_api
271			.execute_block(parent_hash, block.into())
272			.map_err(|e| Box::new(e) as Box<_>)?;
273
274		let storage_proof =
275			runtime_api.extract_proof().expect("Proof recording was enabled above; qed");
276
277		let state = self.client.state_at(parent_hash).map_err(|e| Box::new(e) as Box<_>)?;
278		let gen_storage_changes = runtime_api
279			.into_storage_changes(&state, parent_hash)
280			.map_err(sp_consensus::Error::ChainLookup)?;
281
282		if params.header.state_root() != &gen_storage_changes.transaction_storage_root {
283			return Err(sp_consensus::Error::Other(Box::new(
284				sp_blockchain::Error::InvalidStateRoot,
285			)));
286		}
287
288		// Extend the ignored nodes with the nodes from the storage proof and the generated
289		// storage changes. This ensures that subsequent blocks in the same bundle don't
290		// redundantly include the same trie nodes in their proof.
291		nodes_to_ignore.extend(ProofRecorderIgnoredNodes::<Block>::from_storage_proof::<
292			HashingFor<Block>,
293		>(&storage_proof));
294		nodes_to_ignore.extend(ProofRecorderIgnoredNodes::<Block>::from_memory_db(
295			gen_storage_changes.transaction.clone(),
296		));
297
298		let block_hash = params.post_hash();
299		prepare_ignored_nodes_transaction::<Block>(block_hash, nodes_to_ignore).for_each(
300			|(k, v)| {
301				params.auxiliary.push((k, Some(v)));
302			},
303		);
304
305		// Extract and store proof size recordings
306		let recorded_sizes = proof_size_recorder
307			.recorded_estimations()
308			.into_iter()
309			.map(|size| size as u32)
310			.collect::<Vec<u32>>();
311
312		if !recorded_sizes.is_empty() {
313			prepare_proof_size_recording_aux_data(block_hash, recorded_sizes).for_each(|(k, v)| {
314				params.auxiliary.push((k, Some(v)));
315			});
316		}
317
318		params.state_action =
319			StateAction::ApplyChanges(sc_consensus::StorageChanges::Changes(gen_storage_changes));
320
321		Ok(())
322	}
323}

325impl<Block: BlockT, BI: Clone, Client> Clone for SlotBasedBlockImport<Block, BI, Client> {
326	fn clone(&self) -> Self {
327		Self { inner: self.inner.clone(), client: self.client.clone(), sender: self.sender.clone() }
328	}
329}

331#[async_trait::async_trait]
332impl<Block, BI, Client> BlockImport<Block> for SlotBasedBlockImport<Block, BI, Client>
333where
334	Block: BlockT,
335	BI: BlockImport<Block> + Send + Sync,
336	BI::Error: Into<sp_consensus::Error>,
337	Client:
338		ProvideRuntimeApi<Block> + CallApiAt<Block> + AuxStore + HeaderBackend<Block> + Send + Sync,
339	Client::StateBackend: Send,
340	Client::Api: Core<Block>,
341{
342	type Error = sp_consensus::Error;
343
344	async fn check_block(
345		&self,
346		block: sc_consensus::BlockCheckParams<Block>,
347	) -> Result<sc_consensus::ImportResult, Self::Error> {
348		self.inner.check_block(block).await.map_err(Into::into)
349	}
350
351	async fn import_block(
352		&self,
353		mut params: sc_consensus::BlockImportParams<Block>,
354	) -> Result<sc_consensus::ImportResult, Self::Error> {
355		if !(params.origin == BlockOrigin::Own ||
356			params.with_state() ||
357			params.state_action.skip_execution_checks())
358		{
359			self.execute_block_and_collect_storage_proof(&mut params)?;
360		}
361
362		self.inner.import_block(params).await.map_err(Into::into)
363	}
