18//! Resolving relay-chain data needed to record resubmission entries.
19//!
20//! [`resolve_session`] is shared by the build path (block_builder_task) and the resubmission
21//! backfill task. [`run_resubmission_backfill`] is the task that records resubmission entries for
22//! blocks imported at the tip (built by other collators), doing the relay-chain queries off the
23//! block-import path.
24

44const LOG_TARGET: &str = "aura::resubmission";

46#[derive(Debug, thiserror::Error)]
47pub(crate) enum ResubmissionError {
48	#[error("relay parent header unavailable for {0:?}")]
49	HeaderUnavailable(RelayBlockIdentifier),
50
51	#[error("relay storage root mismatch at number {block_number}")]
52	StorageRootMismatch { block_number: RelayBlockNumber },
53
54	#[error("failed to fetch relay parent header: {0}")]
55	RelayParentHeader(RelayChainError),
56
57	#[error("failed to fetch relay-parent session: {0}")]
58	Session(RelayChainError),
59}
60
61/// Fetch the session index for the given relay parent.
62pub(crate) async fn resolve_session<R: RelayChainInterface + ?Sized>(
63	relay_client: &R,
64	relay_parent: RelayHash,
65) -> Result<SessionIndex, ResubmissionError> {
66	relay_client
67		.session_index_for_child(relay_parent)
68		.await
69		.map_err(ResubmissionError::Session)
70}

72/// Resolve the relay-parent header from a [`RelayBlockIdentifier`] found in a parablock's digest.
73///
74/// Production slot-based blocks carry the relay parent as [`RelayBlockIdentifier::ByStorageRoot`]
75/// (via the relay-parent-storage-root digest), so we look the relay block up by number and confirm
76/// its state root matches — guarding against resolving the wrong fork. [`RelayBlockIdentifier::
77/// ByHash`] is resolved directly.
78async fn resolve_relay_parent<R: RelayChainInterface + ?Sized>(
79	relay_client: &R,
80	identifier: &RelayBlockIdentifier,
81) -> Result<RelayHeader, ResubmissionError> {
82	match identifier {
83		RelayBlockIdentifier::ByHash(relay_parent) => relay_client
84			.header(BlockId::Hash(*relay_parent))
85			.await
86			.map_err(ResubmissionError::RelayParentHeader)?
87			.ok_or_else(|| ResubmissionError::HeaderUnavailable(identifier.clone())),
88		RelayBlockIdentifier::ByStorageRoot { storage_root, block_number } => {
89			let header = relay_client
90				.header(BlockId::Number(*block_number))
91				.await
92				.map_err(ResubmissionError::RelayParentHeader)?
93				.ok_or_else(|| ResubmissionError::HeaderUnavailable(identifier.clone()))?;
94
95			// The canonical relay block at this number must match the storage root recorded in the
96			// digest, otherwise we resolved a different fork.
97			if header.state_root != *storage_root {
98				return Err(ResubmissionError::StorageRootMismatch { block_number: *block_number });
99			}
100
101			Ok(header)
102		},
103	}
104}

106/// Maintain the resubmission store for blocks imported at the tip.
107///
108/// Receives imported `(block, storage proof)` pairs from the [`SlotBasedBlockImportHandle`] and
109/// records a resubmission entry for each via [`backfill_resubmission_entry`], and prunes entries as
110/// blocks are finalized.
111pub(crate) async fn run_resubmission_backfill<Block, RClient, Client>(
112	mut block_import_handle: SlotBasedBlockImportHandle<Block>,
113	relay_client: RClient,
114	para_client: Arc<Client>,
115) where
116	Block: BlockT,
117	RClient: RelayChainInterface,
118	Client: AuxStore + HeaderBackend<Block> + BlockchainEvents<Block>,
119{
120	// Reclaim entries for blocks that were finalized without their prune being observed (e.g. while
121	// the node was down). The notification stream below only covers finalizations from now on.
122	if let Err(err) = prune_missed_finalized_entries::<Block, _>(&*para_client) {
123		tracing::warn!(
124			target: LOG_TARGET,
125			?err,
126			"Failed to prune missed finalized resubmission entries at startup.",
127		);
128	}
129
130	let mut finality_notifications = para_client.finality_notification_stream();
131
132	loop {
133		let import_fut = block_import_handle.next().fuse();
134		let notification_fut = finality_notifications.next().fuse();
135		futures::pin_mut!(import_fut, notification_fut);
136
137		futures::select! {
138			maybe_notification = notification_fut => {
139				let Some(notification) = maybe_notification else {
140					// The finality stream ended; nothing left to prune.
141					break;
142				};
143
144				if let Err(err) = prune_finalized_entries(&*para_client, &notification) {
145					tracing::warn!(
146						target: LOG_TARGET,
147						?err,
148						"Failed to prune finalized resubmission entries.",
149					);
150				}
151			},
152			(block, proof) = import_fut => {
153				backfill_resubmission_entry(
154					&relay_client,
155					&*para_client,
156					block.header(),
157					proof,
158				)
159				.await;
160			},
161		}
162	}
163}

165/// Resolve the relay-chain data an entry needs and write it to the aux store.
166///
167/// Finalized blocks are skipped
168async fn backfill_resubmission_entry<Block, R, Client>(
169	relay_client: &R,
170	para_client: &Client,
171	header: &Block::Header,
172	proof: Arc<StorageProof>,
173) where
174	Block: BlockT,
175	R: RelayChainInterface + ?Sized,
176	Client: AuxStore + HeaderBackend<Block>,
177{
178	let block_hash = header.hash();
179	let number = *header.number();
180
181	if number <= para_client.info().finalized_number {
182		return;
183	}
184
185	let Some(relay_block_identifier) =
186		CumulusDigestItem::find_relay_block_identifier(header.digest())
187	else {
188		tracing::trace!(target: LOG_TARGET, ?block_hash, "Imported block has no relay block identifier; skipping.");
189		return;
190	};
191
192	let relay_parent_header =
193		match resolve_relay_parent(relay_client, &relay_block_identifier).await {
194			Ok(header) => header,
195			Err(err) => {
196				tracing::debug!(
197					target: LOG_TARGET,
198					?block_hash,
199					?err,
200					"Could not resolve relay parent; skipping resubmission entry.",
201				);
202				return;
203			},
204		};
205	let relay_parent = relay_parent_header.hash();
206
207	let relay_parent_session = match resolve_session(relay_client, relay_parent).await {
208		Ok(session) => session,
209		Err(err) => {
210			tracing::debug!(
211				target: LOG_TARGET,
212				?block_hash,
213				?err,
214				"Could not resolve relay-parent session; skipping resubmission entry.",
215			);
216			return;
217		},
218	};
219	if number <= para_client.info().finalized_number {
220		return;
221	}
222
223	let pairs: Vec<_> = prepare_resubmission_aux_data::<Block>(
224		block_hash,
225		proof,
226		relay_parent_header,
227		relay_parent_session,
228	)
229	.collect();
230	let refs: Vec<_> = pairs.iter().map(|(k, v)| (k.as_slice(), v.as_slice())).collect();
231
232	match para_client.insert_aux(&refs, &[]) {
233		Ok(()) => tracing::trace!(
234			target: LOG_TARGET,
235			?block_hash,
236			"Stored resubmission entry for imported block.",
237		),
238		Err(err) => tracing::warn!(
239			target: LOG_TARGET,
240			?block_hash,
241			?err,
242			"Failed to store resubmission entry for imported block.",
243		),
244	}
