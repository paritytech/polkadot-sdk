47fn get_current_relay_slot_at(
48	now: Duration,
49	slot_offset: Duration,
50	relay_chain_slot_duration: Duration,
51) -> Slot {
52	let now = now.saturating_sub(slot_offset);
53	Slot::from_timestamp(
54		Timestamp::from(now),
55		SlotDuration::from_millis(relay_chain_slot_duration.as_millis() as u64),
56	)
57}
58
59fn get_current_relay_slot(slot_offset: Duration, relay_chain_slot_duration: Duration) -> Slot {
60	get_current_relay_slot_at(
61		Timestamp::current().as_duration(),
62		slot_offset,
63		relay_chain_slot_duration,
64	)
65}

67/// Picks a scheduling parent for the next collation under V2 or V3 semantics.
68///
69/// The two policies differ in which relay block they build on, and consequently in
70/// how they tolerate relay block propagation delay. Selected per-call based on whether
71/// V3 is enabled on both the parachain (runtime API) and the relay chain
72/// (`CandidateReceiptV3` node feature):
73///
74/// - **V2**: build on the *current* slot's relay block. Tolerated via a fixed 1s
75///   `slot_offset` — the relay block must arrive within ~1s of the slot starting. If
76///   not, we wait for it before building, so we don't end up using the previous slot's
77///   relay block past our own slot. See
78///   <https://github.com/paritytech/polkadot-sdk/pull/11453>.
79/// - **V3**: build on the *last finished* slot's relay block. No offset hack, no waiting: the relay
80///   block had a full slot to propagate, which is what slots are for. Matches the low-latency v2
81///   design.
82///
83/// Owns the relay chain new-best notification stream so [`Self::wait_for_scheduling_parent`]
84/// can block for a fresh leaf. Initial state is a terminated empty stream. The caller
85/// must call [`Self::ensure_initialized`] before any call to [`Self::wait_for_scheduling_parent`],
86/// in order to make sure that the stream is installed/re-installed if needed.
87pub(crate) struct SchedulingInfo<RelayClient> {
88	best_notifications: Fuse<Pin<Box<dyn Stream<Item = RelayHeader> + Send>>>,
89	relay_slot_duration: Duration,
90	slot_offset: Duration,
91	maybe_best_relay_header: Option<RelayHeader>,
92
93	_phantom: PhantomData<RelayClient>,
94}
95
96impl<RelayClient: RelayChainInterface + 'static> SchedulingInfo<RelayClient> {
97	/// Create a new `SchedulingInfo` with no active notification stream.
98	///
99	/// The caller must call [`Self::ensure_initialized`] before the first
100	/// `wait_for_scheduling_parent` invocation.
101	pub fn new(relay_chain_slot_duration: Duration, slot_offset: Duration) -> Self {
102		let stream: Pin<Box<dyn Stream<Item = RelayHeader> + Send>> =
103			Box::pin(futures::stream::empty());
104		let mut stream = stream.fuse();
105		// Force the fused stream into the terminated state so the first
106		// `should_reinit` call returns `true`.
107		stream.next().now_or_never();
108
109		Self {
110			best_notifications: stream,
111			relay_slot_duration: relay_chain_slot_duration,
112			slot_offset,
113			maybe_best_relay_header: None,
114			_phantom: Default::default(),
115		}
116	}
117
118	async fn get_best_relay_block_data<'a>(
119		relay_client: &RelayClient,
120		relay_chain_data_cache: &'a mut RelayChainDataCache<RelayClient>,
121	) -> Result<&'a RelayChainData, ()> {
122		let best_relay_hash = relay_client.best_block_hash().await.map_err(|_| ())?;
123		relay_chain_data_cache.get_by_hash(best_relay_hash).await.map_err(|_| ())
124	}
125
126	/// `true` if the best-block notification stream is terminated and must be replaced
127	/// before the next `wait_for_scheduling_parent` call.
128	///
129	/// Returns `true` both at startup (the initial stream is a terminated empty stream)
130	/// and after the underlying subscription has ended.
131	fn should_reinit(&self) -> bool {
132		self.best_notifications.is_terminated()
133	}
134
135	pub async fn ensure_initialized<'a>(
136		&'a mut self,
137		relay_client: &RelayClient,
138		relay_chain_data_cache: &'a mut RelayChainDataCache<RelayClient>,
139	) -> Option<&'a RelayChainData> {
140		if !self.should_reinit() {
141			return None;
142		}
143
144		match relay_client.new_best_notification_stream().await {
145			Ok(best_notifications) => {
146				self.best_notifications = best_notifications.fuse();
147			},
148			Err(err) => {
149				tracing::error!(
150					target: LOG_TARGET,
151					?err,
152					"Failed to reset the relay chain best block notification stream. \
153					The next call to `wait_for_scheduling_parent` might fail."
154				);
155			},
156		};
157
158		let best_relay_block_data =
159			match Self::get_best_relay_block_data(relay_client, relay_chain_data_cache).await {
160				Ok(best_relay_block_data) => best_relay_block_data,
161				Err(()) => {
162					tracing::error!(
163						target: LOG_TARGET,
164						"Failed to get the `RelayChainData` for the best relay chain block. \
165						The next call to `wait_for_scheduling_parent` might fail."
166					);
167					return None;
168				},
169			};
170		self.maybe_best_relay_header = Some(best_relay_block_data.relay_header.clone());
171
172		Some(best_relay_block_data)
173	}

175	pub fn is_v3_enabled(
176		v3_enabled_on_para: bool,
177		relay_chain_data: Option<&RelayChainData>,
178	) -> bool {
179		v3_enabled_on_para && relay_chain_data.map_or(false, |data| data.is_v3_enabled())
180	}

182	/// Pick a scheduling parent under the policy described on [`SchedulingInfo`],
183	/// blocking on the notification stream until one is available.
184	///
185	/// V3 is used iff `v3_enabled_on_para` is true *and* the relay chain has the
186	/// `CandidateReceiptV3` node feature set at the candidate block; otherwise V2.
187	/// Under V3, if the best leaf's slot is still in progress, walks back to its
188	/// parent — and aborts when that crosses a BABE epoch boundary, since the
189	/// scheduling parent must share a session with the active leaf.
190	///
191	/// Returns `Some((header, v3_used))`, or `None` on relay client error, a session
192	/// boundary, or a terminated notification stream.
193	pub async fn wait_for_scheduling_parent(
194		&mut self,
195		relay_chain_data_cache: &mut RelayChainDataCache<RelayClient>,
196		v3_enabled_on_para: bool,
197		production_slot: Slot,
198	) -> Option<(RelayHeader, bool)> {
199		let mut maybe_best_relay_header = self.maybe_best_relay_header.take();
200		let (best_relay_slot, best_relay_header_data) = loop {
201			// Drain buffered notifications.
202			while let Some(Some(header)) = self.best_notifications.next().now_or_never() {
203				maybe_best_relay_header = Some(header);
204			}
205
206			let best_relay_header = match maybe_best_relay_header.take() {
207				Some(header) => header,
208				None => self.best_notifications.next().await?,
209			};
210			self.maybe_best_relay_header = Some(best_relay_header.clone());
211			let best_relay_header_data =
212				relay_chain_data_cache.get_by_header(best_relay_header).await.ok()?;
213			let best_relay_slot = get_relay_slot(&best_relay_header_data.relay_header)?;
214
215			let v3_enabled = Self::is_v3_enabled(v3_enabled_on_para, Some(&best_relay_header_data));
216			if v3_enabled {
217				// V3 needs a scheduling parent from a finished slot, which the walk below
218				// derives from this best.
219				break (best_relay_slot, best_relay_header_data);
220			}
221
222			// For v2, we need to loop until we find a scheduling parent associated with a
223			// current slot.
224			if best_relay_slot >= get_current_relay_slot(self.slot_offset, self.relay_slot_duration)
225			{
226				return Some((best_relay_header_data.relay_header.clone(), false));
227			}
228		};
229
230		// v3: walk back to the first finished slot
231		let mut scheduling_parent_data = best_relay_header_data;
232		let mut scheduling_parent_slot = best_relay_slot;
233		while scheduling_parent_slot >= production_slot {
234			// The scheduling parent should be part of the same session as the best
235			// relay block.
236			if sc_consensus_babe::contains_epoch_change::<RelayBlock>(
237				&scheduling_parent_data.relay_header,
238			) {
239				return None;
240			}
241
242			let ancestor_hash = *scheduling_parent_data.relay_header.parent_hash();
243			scheduling_parent_data =
244				relay_chain_data_cache.get_by_hash(ancestor_hash).await.ok()?;
245			scheduling_parent_slot = get_relay_slot(&scheduling_parent_data.relay_header)?
246		}
247
248		Some((scheduling_parent_data.relay_header.clone(), true))
249	}
250}

252/// Why [`SchedulingProofBuilder::build_with_signed_payload`] could not sign.
253// The signed terminal has no production caller until the resubmission path lands. Outside tests
254// only `build` is reachable.
255#[cfg_attr(not(test), allow(dead_code))]
256#[derive(Debug, thiserror::Error)]
257pub(crate) enum SchedulingSignError {
258	#[error("scheduling proof builder is missing the {0}")]
259	MissingInput(&'static str),
260
261	#[error("collator peer id does not fit an ApprovedPeerId")]
262	PeerId,
263
264	#[error("keystore failed to sign the scheduling info payload")]
265	Keystore,
266
267	#[error("keystore holds no key for the slot author")]
268	NoKey,
269
270	#[error("keystore returned a {0}-byte signature, expected 64")]
271	SignatureSize(usize),
272}

274/// Assembles a V3 [`SchedulingProof`], converting collator-side values into the shapes the proof
275/// needs: relay parent descendants become the backwards header chain, and the core assignment plus
276/// collator peer id become a signed [`SchedulingInfoPayload`].
277#[cfg_attr(not(test), allow(dead_code))]
278pub(crate) struct SchedulingProofBuilder<'a, P: Pair> {
279	internal_scheduling_parent_header: RelayHeader,
280	header_chain: Vec<RelayHeader>,
281	core_info: Option<&'a CoreInfo>,
282	peer_id: Option<PeerId>,
283	keystore: Option<&'a KeystorePtr>,
284	_phantom: PhantomData<P>,
285}
286
287#[cfg_attr(not(test), allow(dead_code))]
288impl<'a, P> SchedulingProofBuilder<'a, P>
289where
290	P: Pair,
291	P::Public: AppPublic,
292{
293	/// Start a proof whose internal scheduling parent is `internal_scheduling_parent_header`.
294	///
295	/// Without [`descendants`](Self::descendants) the header chain stays empty, which is the
296	/// `relay_parent_offset = 0` case where the scheduling parent is the internal one.
297	pub(crate) fn new(internal_scheduling_parent_header: RelayHeader) -> Self {
298		Self {
299			internal_scheduling_parent_header,
300			header_chain: Vec::new(),
301			core_info: None,
302			peer_id: None,
303			keystore: None,
304			_phantom: PhantomData,
305		}
306	}
307
308	/// Relay parent descendants, ordered oldest to newest. The proof's header chain runs the other
309	/// way, from the scheduling parent backwards.
310	pub(crate) fn descendants(mut self, descendants: Vec<RelayHeader>) -> Self {
311		self.header_chain = descendants.into_iter().rev().collect();
312		self
313	}
314
315	/// The core the submission targets. Its selector and claim queue offset are what the signature
316	/// commits to.
317	pub(crate) fn for_core(mut self, core_info: &'a CoreInfo) -> Self {
318		self.core_info = Some(core_info);
319		self
320	}
321
322	/// The peer credited for delivering the collation.
323	pub(crate) fn crediting_peer(mut self, peer_id: PeerId) -> Self {
324		self.peer_id = Some(peer_id);
325		self
326	}
327
328	/// The keystore holding the signing key.
329	pub(crate) fn keystore(mut self, keystore: &'a KeystorePtr) -> Self {
330		self.keystore = Some(keystore);
331		self
332	}
333
334	/// Assemble the proof unsigned, for an initial submission
335	/// (`relay_parent == internal scheduling parent`). An initial submission must stay valid
336	/// without any additional signed data. Core selection comes from the block's UMP signals.
337	pub(crate) fn build(self) -> SchedulingProof {
338		SchedulingProof::new(self.header_chain, self.internal_scheduling_parent_header, None)
339	}
340
341	/// Assemble the proof around a signed scheduling info, for a resubmission (`relay_parent` an
342	/// ancestor of the internal scheduling parent). Validators require the signature there.
343	/// Everything it commits to has to be supplied first. `author_pub` is the key eligible for
344	/// the slot at the internal scheduling parent.
345	pub(crate) fn build_with_signed_payload(
346		self,
347		author_pub: &P::Public,
348	) -> Result<SchedulingProof, SchedulingSignError> {
349		let signed_scheduling_info = self.sign(author_pub)?;
350
351		Ok(SchedulingProof::new(
352			self.header_chain,
353			self.internal_scheduling_parent_header,
354			Some(signed_scheduling_info),
355		))
356	}
357
358	fn sign(&self, author_pub: &P::Public) -> Result<SignedSchedulingInfo, SchedulingSignError> {
359		use SchedulingSignError::*;
360
361		let core_info = self.core_info.ok_or(MissingInput("core info"))?;
362		let peer_id = self.peer_id.ok_or(MissingInput("collator peer id"))?;
363		let keystore = self.keystore.ok_or(MissingInput("keystore"))?;
364
365		let payload = SchedulingInfoPayload::new(
366			core_info.selector,
367			core_info.claim_queue_offset.0,
368			ApprovedPeerId::try_from(peer_id.to_bytes()).map_err(|_| PeerId)?,
369			self.internal_scheduling_parent_header.hash(),
370		);
371
372		let signature = keystore
373			.sign_with(
374				<P::Public as AppCrypto>::ID,
375				<P::Public as AppCrypto>::CRYPTO_ID,
376				author_pub.as_slice(),
377				&payload.encode(),
378			)
379			.map_err(|_| Keystore)?
380			.ok_or(NoKey)?;
381
382		let signature = <[u8; 64]>::try_from(signature.as_slice())
383			.map_err(|_| SignatureSize(signature.len()))?;
384
385		Ok(SignedSchedulingInfo { payload, signature })
386	}
387}

554	/// Five headers around `current_slot`: one very old, two from finished slots
555	/// (`current_slot - 2`, `current_slot - 1`) and two from future slots (`+ 10`, `+ 11`).

561		current_slot: u64,

563		let mut node_features = NodeFeatures::from_vec(vec![0; 5]);
564		if v3_enabled {
565			node_features.set(FeatureIndex::CandidateReceiptV3 as usize, true);
566		}

848	/// Consecutive V3 headers (numbers 50, 51, ...) at exactly the given slots, parent-linked, the
849	/// last one being the best block.
850	fn build_v3_chain_with_slots(
851		slots: &[u64],
852	) -> (TestRelayClient, RelayChainDataCache<TestRelayClient>, Vec<RelayHeader>) {
853		let mut node_features = NodeFeatures::from_vec(vec![0; 5]);
854		node_features.set(FeatureIndex::CandidateReceiptV3 as usize, true);
855
856		let mut headers: Vec<RelayHeader> = vec![];
857		for (i, slot) in slots.iter().enumerate() {
858			let parent_hash = headers.last().map(|h| h.hash()).unwrap_or_default();
859			headers.push(tests::relay_header_with_slot(50 + i as u32, parent_hash, *slot));
860		}
861
862		let headers_map = headers.iter().map(|h| (h.hash(), h.clone())).collect();
863		let client = TestRelayClient::new_with_best(headers_map, headers.last().unwrap().hash());
864		let mut cache = RelayChainDataCache::new(client.clone(), 1.into());
865		for header in &headers {
866			cache.set_test_data(header.clone(), vec![], node_features.clone());
867		}
868
869		(client, cache, headers)
870	}
