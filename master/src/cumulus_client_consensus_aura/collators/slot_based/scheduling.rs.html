36fn get_current_relay_slot_at(
37	now: Duration,
38	slot_offset: Duration,
39	relay_chain_slot_duration: Duration,
40) -> Slot {
41	let now = now.saturating_sub(slot_offset);
42	Slot::from_timestamp(
43		Timestamp::from(now),
44		SlotDuration::from_millis(relay_chain_slot_duration.as_millis() as u64),
45	)
46}
47
48fn get_current_relay_slot(slot_offset: Duration, relay_chain_slot_duration: Duration) -> Slot {
49	get_current_relay_slot_at(
50		Timestamp::current().as_duration(),
51		slot_offset,
52		relay_chain_slot_duration,
53	)
54}

56/// Picks a scheduling parent for the next collation under V2 or V3 semantics.
57///
58/// The two policies differ in which relay block they build on, and consequently in
59/// how they tolerate relay block propagation delay. Selected per-call based on whether
60/// V3 is enabled on both the parachain (runtime API) and the relay chain
61/// (`CandidateReceiptV3` node feature):
62///
63/// - **V2**: build on the *current* slot's relay block. Tolerated via a fixed 1s
64///   `slot_offset` — the relay block must arrive within ~1s of the slot starting. If
65///   not, we wait for it before building, so we don't end up using the previous slot's
66///   relay block past our own slot. See
67///   <https://github.com/paritytech/polkadot-sdk/pull/11453>.
68/// - **V3**: build on the *last finished* slot's relay block. No offset hack, no waiting: the relay
69///   block had a full slot to propagate, which is what slots are for. Matches the low-latency v2
70///   design.
71///
72/// Owns the relay chain new-best notification stream so [`Self::wait_for_scheduling_parent`]
73/// can block for a fresh leaf. Initial state is a terminated empty stream. The caller
74/// must call [`Self::ensure_initialized`] before any call to [`Self::wait_for_scheduling_parent`],
75/// in order to make sure that the stream is installed/re-installed if needed.
76pub(crate) struct SchedulingInfo<RelayClient> {
77	best_notifications: Fuse<Pin<Box<dyn Stream<Item = RelayHeader> + Send>>>,
78	relay_slot_duration: Duration,
79	slot_offset: Duration,
80	maybe_best_relay_header: Option<RelayHeader>,
81
82	_phantom: PhantomData<RelayClient>,
83}
84
85impl<RelayClient: RelayChainInterface + 'static> SchedulingInfo<RelayClient> {
86	/// Create a new `SchedulingInfo` with no active notification stream.
87	///
88	/// The caller must call [`Self::ensure_initialized`] before the first
89	/// `wait_for_scheduling_parent` invocation.
90	pub fn new(relay_chain_slot_duration: Duration, slot_offset: Duration) -> Self {
91		let stream: Pin<Box<dyn Stream<Item = RelayHeader> + Send>> =
92			Box::pin(futures::stream::empty());
93		let mut stream = stream.fuse();
94		// Force the fused stream into the terminated state so the first
95		// `should_reinit` call returns `true`.
96		stream.next().now_or_never();
97
98		Self {
99			best_notifications: stream,
100			relay_slot_duration: relay_chain_slot_duration,
101			slot_offset,
102			maybe_best_relay_header: None,
103			_phantom: Default::default(),
104		}
105	}

107	async fn get_best_relay_block_data<'a>(
108		relay_client: &RelayClient,
109		relay_chain_data_cache: &'a mut RelayChainDataCache<RelayClient>,
110	) -> Result<&'a RelayChainData, ()> {
111		let best_relay_hash = relay_client.best_block_hash().await.map_err(|_| ())?;
112		relay_chain_data_cache.get_by_hash(best_relay_hash).await.map_err(|_| ())
113	}
114
115	/// `true` if the best-block notification stream is terminated and must be replaced
116	/// before the next `wait_for_scheduling_parent` call.
117	///
118	/// Returns `true` both at startup (the initial stream is a terminated empty stream)
119	/// and after the underlying subscription has ended.
120	fn should_reinit(&self) -> bool {
121		self.best_notifications.is_terminated()
122	}
123
124	pub async fn ensure_initialized<'a>(
125		&'a mut self,
126		relay_client: &RelayClient,
127		relay_chain_data_cache: &'a mut RelayChainDataCache<RelayClient>,
128	) -> Option<&'a RelayChainData> {
129		if !self.should_reinit() {
130			return None;
131		}
132
133		match relay_client.new_best_notification_stream().await {
134			Ok(best_notifications) => {
135				self.best_notifications = best_notifications.fuse();
136			},
137			Err(err) => {
138				tracing::error!(
139					target: crate::LOG_TARGET,
140					?err,
141					"Failed to reset the relay chain best block notification stream. \
142					The next call to `wait_for_scheduling_parent` might fail."
143				);
144			},
145		};
146
147		let best_relay_block_data =
148			match Self::get_best_relay_block_data(relay_client, relay_chain_data_cache).await {
149				Ok(best_relay_block_data) => best_relay_block_data,
150				Err(()) => {
151					tracing::error!(
152						target: crate::LOG_TARGET,
153						"Failed to get the `RelayChainData` for the best relay chain block. \
154						The next call to `wait_for_scheduling_parent` might fail."
155					);
156					return None;
157				},
158			};
159		self.maybe_best_relay_header = Some(best_relay_block_data.relay_header.clone());
160
161		Some(best_relay_block_data)
162	}

164	pub fn is_v3_enabled(
165		v3_enabled_on_para: bool,
166		relay_chain_data: Option<&RelayChainData>,
167	) -> bool {
168		v3_enabled_on_para && relay_chain_data.map_or(false, |data| data.is_v3_enabled())
169	}
170
171	/// Pick a scheduling parent under the policy described on [`SchedulingInfo`],
172	/// blocking on the notification stream until one is available.
173	///
174	/// V3 is used iff `v3_enabled_on_para` is true *and* the relay chain has the
175	/// `CandidateReceiptV3` node feature set at the candidate block; otherwise V2.
176	/// Under V3, if the best leaf's slot is still in progress, walks back to its
177	/// parent — and aborts when that crosses a BABE epoch boundary, since the
178	/// scheduling parent must share a session with the active leaf.
179	///
180	/// Returns `Some((header, v3_used))`, or `None` on relay client error, a session
181	/// boundary, or a terminated notification stream.
182	pub async fn wait_for_scheduling_parent(
183		&mut self,
184		relay_chain_data_cache: &mut RelayChainDataCache<RelayClient>,
185		v3_enabled_on_para: bool,
186	) -> Option<(RelayHeader, bool)> {
187		let mut maybe_best_relay_header = self.maybe_best_relay_header.take();
188		let (best_relay_slot, best_relay_header_data) = loop {
189			// Drain buffered notifications.
190			while let Some(Some(header)) = self.best_notifications.next().now_or_never() {
191				maybe_best_relay_header = Some(header);
192			}
193
194			let best_relay_header = match maybe_best_relay_header.take() {
195				Some(header) => header,
196				None => self.best_notifications.next().await?,
197			};
198			self.maybe_best_relay_header = Some(best_relay_header.clone());
199			let best_relay_header_data =
200				relay_chain_data_cache.get_by_header(best_relay_header).await.ok()?;
201			let best_relay_slot = get_relay_slot(&best_relay_header_data.relay_header)?;
202
203			let v3_enabled = Self::is_v3_enabled(v3_enabled_on_para, Some(&best_relay_header_data));
204			if v3_enabled {
205				// For scheduling v3 we don't need to loop since we need to return a
206				// scheduling parent associated with a finished slot.
207				break (best_relay_slot, best_relay_header_data);
208			}
209
210			// For v2, we need to loop until we find a scheduling parent associated with a
211			// current slot.
212			if best_relay_slot >= get_current_relay_slot(self.slot_offset, self.relay_slot_duration)
213			{
214				return Some((best_relay_header_data.relay_header.clone(), false));
215			}
216		};
217
218		// v3: walk back to the first finished slot
219		let mut scheduling_parent_data = best_relay_header_data;
220		let mut scheduling_parent_slot = best_relay_slot;
221		while scheduling_parent_slot >=
222			get_current_relay_slot(Duration::ZERO, self.relay_slot_duration)
223		{
224			// The scheduling parent should be part of the same session as the best
225			// relay block.
226			if sc_consensus_babe::contains_epoch_change::<RelayBlock>(
227				&scheduling_parent_data.relay_header,
228			) {
229				return None;
230			}
231
232			let ancestor_hash = *scheduling_parent_data.relay_header.parent_hash();
233			scheduling_parent_data =
234				relay_chain_data_cache.get_by_hash(ancestor_hash).await.ok()?;
235			scheduling_parent_slot = get_relay_slot(&scheduling_parent_data.relay_header)?
236		}
237
238		Some((scheduling_parent_data.relay_header.clone(), true))
239	}
240}
