44/// Parameters for the collation task.
45pub struct Params<Block: BlockT, RClient, CS, Backend, CHP> {
46	/// A handle to the relay-chain client.
47	pub relay_client: RClient,
48	/// The collator key used to sign collations before submitting to validators.
49	pub collator_key: CollatorPair,
50	/// The para's ID.
51	pub para_id: ParaId,
52	/// Whether we should reinitialize the collator config (i.e. we are transitioning to aura).
53	pub reinitialize: bool,
54	/// Collator service interface
55	pub collator_service: CS,
56	/// Receiver channel for collation/segment messages from the block builder task.
57	pub collator_receiver: TracingUnboundedReceiver<CollatorMessage<Block>>,
58	/// When set, the collator will export every produced `POV` to this folder.
59	pub export_pov: Option<PathBuf>,
60	/// The para client's backend, used to hydrate resubmitted segment entries from the
61	/// resubmission store — done here, off the block-production hot path.
62	pub para_backend: Arc<Backend>,
63	/// Validation code hash provider, used while hydrating resubmitted segment entries.
64	pub code_hash_provider: CHP,
65}

67/// Asynchronously executes the collation task for a parachain.
68///
69/// This function initializes the collator subsystems necessary for producing and submitting
70/// collations to the relay chain. It listens for new best relay chain block notifications and
71/// handles collator messages. If our parachain is scheduled on a core and we have a candidate,
72/// the task will build a collation and send it to the relay chain.
73pub async fn run_collation_task<Block, RClient, CS, Backend, CHP>(
74	Params {
75		relay_client,
76		collator_key,
77		para_id,
78		reinitialize,
79		collator_service,
80		mut collator_receiver,
81		export_pov,
82		para_backend,
83		code_hash_provider,
84	}: Params<Block, RClient, CS, Backend, CHP>,
85) where
86	Block: BlockT,
87	CS: CollatorServiceInterface<Block> + Send + Sync + 'static,
88	RClient: RelayChainInterface + Clone + 'static,
89	Backend: sc_client_api::Backend<Block> + 'static,
90	CHP: ValidationCodeHashProvider<Block::Hash> + Send + Sync + 'static,
91{
92	let Ok(mut overseer_handle) = relay_client.overseer_handle() else {
93		tracing::error!(target: LOG_TARGET, "Failed to get overseer handle.");
94		return;
95	};
96
97	cumulus_client_collator::initialize_collator_subsystems(
98		&mut overseer_handle,
99		collator_key,
100		para_id,
101		reinitialize,
102	)
103	.await;
104
105	// Read-side handle over the resubmission aux store, used to hydrate the V3 resubmittable
106	// headers off the block-production hot path. Cheap to hold (wraps the backend `Arc`).
107	let resubmission_store = ResubmissionStore::new(para_backend.clone());
108
109	let mut submitter = CollationSubmitter {
110		collator_service,
111		overseer_handle,
112		export_pov,
113		para_backend,
114		code_hash_provider,
115		resubmission_store,
116	};
117
118	while let Some(message) = collator_receiver.next().await {
119		submitter.submit(message).await;
120	}
121}

123/// Turns [`CollatorMessage`]s into collations and forwards them to the collation-generation
124/// subsystem.
125struct CollationSubmitter<Block: BlockT, CS, Backend, CHP> {
126	collator_service: CS,
127	overseer_handle: OverseerHandle,
128	export_pov: Option<PathBuf>,
129	/// Backend, store, and code-hash provider hydrate the resubmitted headers into
130	/// [`CollationParts`] off the block-production hot path.
131	para_backend: Arc<Backend>,
132	code_hash_provider: CHP,
133	resubmission_store: ResubmissionStore<Block, Backend>,
134}
135
136impl<Block, CS, Backend, CHP> CollationSubmitter<Block, CS, Backend, CHP>
137where
138	Block: BlockT,
139	CS: CollatorServiceInterface<Block>,
140	Backend: sc_client_api::Backend<Block>,
141	CHP: ValidationCodeHashProvider<Block::Hash>,
142{
143	/// Build the collation(s) in `message` and submit them via
144	/// [`CollationGenerationMessage::SubmitSegment`]: a `Collation` as a one-element V2 segment, a
145	/// `Segment` as V3.
146	async fn submit(&mut self, message: CollatorMessage<Block>) {
147		match message {
148			CollatorMessage::Collation { core_index, parts } => {
149				// V2: no scheduling proof, scheduling parent is the relay parent.
150				let Some(segment_collation) = self.build_collation(parts, None).await else {
151					return;
152				};
153				let scheduling_parent = segment_collation.relay_parent;
154
155				tracing::debug!(target: LOG_TARGET, ?core_index, ?scheduling_parent, "Submitting collation for core.");
156
157				self.overseer_handle
158					.send_msg(
159						CollationGenerationMessage::SubmitSegment(SubmitSegmentParams {
160							scheduling_parent,
161							core_index,
162							candidates_descriptor_version: CandidateDescriptorVersion::V2,
163							collations: BoundedVec::truncate_from(vec![segment_collation]),
164						}),
165						"SubmitSegment",
166					)
167					.await;
168			},
169			CollatorMessage::Segment {
170				scheduling_proof,
171				core_index,
172				resubmittable_headers,
173				bundle,
174			} => {
175				// V3: scheduling parent comes from the proof.
176				let scheduling_parent = scheduling_proof.scheduling_parent();
177
178				// Hydrate the resubmitted headers here (proof/body reads), off the block-production
179				// hot path, then prepend them (oldest first) to the freshly-built bundle.
180				let mut all_parts =
181					super::resubmittable_segment::ResubmittableSegment::new(resubmittable_headers)
182						.hydrate(
183							&*self.para_backend,
184							&self.code_hash_provider,
185							&self.resubmission_store,
186						);
187				let resubmitted = all_parts.len();
188				// `bundle` holds at most one freshly-built block.
189				let fresh = usize::from(bundle.is_some());
190				all_parts.extend(bundle);
191				let total = resubmitted + fresh;
192
193				// Parts that fail to build or whose session lookup fails are skipped — they do not
194				// abort the whole segment.
195				let mut collations = Vec::with_capacity(all_parts.len());
196				for parts in all_parts {
197					if let Some(collation) =
198						self.build_collation(parts, Some(scheduling_proof.clone())).await
199					{
200						collations.push(collation);
201					}
202				}
203
204				if collations.is_empty() {
205					tracing::debug!(
206						target: LOG_TARGET,
207						?core_index,
208						resubmitted,
209						fresh,
210						"No collations built for segment; nothing submitted for core.",
211					);
212					return;
213				}
214
215				if collations.len() > MAX_SEGMENT_LEN as usize {
216					tracing::warn!(
217						target: LOG_TARGET,
218						?core_index,
219						segment_len = collations.len(),
220						max = MAX_SEGMENT_LEN,
221						"Segment exceeds MAX_SEGMENT_LEN; truncating.",
222					);
223				}
224
225				tracing::debug!(
226					target: LOG_TARGET,
227					?core_index,
228					segment_len = collations.len(),
229					resubmitted,
230					fresh,
231					dropped = total.saturating_sub(collations.len()),
232					"Submitting segment for core.",
233				);
234
235				self.overseer_handle
236					.send_msg(
237						CollationGenerationMessage::SubmitSegment(SubmitSegmentParams {
238							scheduling_parent,
239							core_index,
240							candidates_descriptor_version: CandidateDescriptorVersion::V3,
241							collations: BoundedVec::truncate_from(collations),
242						}),
243						"SubmitSegment",
244					)
245					.await;
246			},
247		}
248	}
249}

251impl<Block, CS, Backend, CHP> CollationSubmitter<Block, CS, Backend, CHP>
252where
253	Block: BlockT,
254	CS: CollatorServiceInterface<Block>,
255	Backend: sc_client_api::Backend<Block>,
256	CHP: ValidationCodeHashProvider<Block::Hash>,
257{
258	/// Build one `SegmentCollation` from `parts`: build the PoV and export it if configured. The
259	/// session and relay-parent header data are carried in `parts`, resolved at build/hydration
260	/// time. Returns `None` if the collation could not be built.
261	async fn build_collation(
262		&self,
263		parts: CollationParts<Block>,
264		scheduling_proof: Option<SchedulingProof>,
265	) -> Option<SegmentCollation> {
266		let CollationParts {
267			relay_parent,
268			parent_header,
269			blocks,
270			proof,
271			validation_code_hash,
272			validation_data,
273			relay_parent_session,
274		} = parts;
275
276		// The scheduling proof, including its UMP scheduling-tail override, is applied inside
277		// `build_multi_block_collation`.
278		let (collation, block_data) = match self.collator_service.build_multi_block_collation(
279			&parent_header,
280			blocks,
281			proof,
282			scheduling_proof,
283		) {
284			Some(collation) => collation,
285			None => {
286				tracing::warn!(target: LOG_TARGET, ?relay_parent, "Unable to build collation.");
287				return None;
288			},
289		};
290
291		block_data.log_size_info();
292
293		if let MaybeCompressedPoV::Compressed(ref pov) = collation.proof_of_validity {
294			if let Some(pov_path) = self.export_pov.as_ref() {
295				if let Some(header) = block_data.blocks().first().map(|b| b.header()) {
296					export_pov_to_path::<Block>(
297						pov_path,
298						pov.clone(),
299						header.hash(),
300						*header.number(),
301						parent_header.clone(),
302						validation_data.relay_parent_storage_root,
303						validation_data.relay_parent_number,
304						validation_data.max_pov_size,
305					);
306				}
307			}
308
309			tracing::info!(
310				target: LOG_TARGET,
311				block_numbers = ?block_data.blocks().iter().map(|b| *b.header().number()).collect::<Vec<_>>(),
312				"Compressed PoV size: {}kb",
313				pov.block_data.0.len() as f64 / 1024f64,
314			);
315		}
316
317		Some(SegmentCollation {
318			relay_parent,
319			collation,
320			validation_code_hash,
321			session_index: relay_parent_session,
322			validation_data,
323		})
324	}
