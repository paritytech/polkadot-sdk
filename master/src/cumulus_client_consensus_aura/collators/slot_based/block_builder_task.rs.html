130/// Call context for the runtime queries that shape the candidate.
131///
132/// Resolves `:pending_code`, the blob the proposer executes and `ValidationCodeHashProvider` names.
133/// The default offchain context sees `:code` only, shaping the candidate from the outgoing runtime.
134const BLOCK_PRODUCTION_CONTEXT: CallContext = CallContext::Onchain { import: false };
135
136/// A runtime API handle set to [`BLOCK_PRODUCTION_CONTEXT`].
137fn onchain_runtime_api<Block: BlockT, Client>(para_client: &Client) -> ApiRef<'_, Client::Api>
138where
139	Client: ProvideRuntimeApi<Block>,
140	Client::Api: ApiExt<Block>,
141{
142	let mut api = para_client.runtime_api();
143	api.set_call_context(BLOCK_PRODUCTION_CONTEXT);
144	api
145}

147/// The scheduling values that shape the candidate, read from the runtime executing the block.
148#[derive(Clone, Copy, Debug, PartialEq, Eq)]
149struct SchedulingParams {
150	v3_enabled: bool,
151	relay_parent_offset: u32,
152}
153
154/// Read the [`SchedulingParams`] from the runtime that will execute the block.
155///
156/// Must be read at the parent we build on, not the best head: the two straddle a runtime upgrade
157/// whenever `find_parent` walks back into an unbackable segment.
158fn scheduling_params_at<Block: BlockT, Client>(
159	para_client: &Client,
160	at: Block::Hash,
161) -> SchedulingParams
162where
163	Client: ProvideRuntimeApi<Block>,
164	Client::Api: SchedulingV3EnabledApi<Block> + RelayParentOffsetApi<Block> + ApiExt<Block>,
165{
166	let api = onchain_runtime_api(para_client);
167
168	SchedulingParams {
169		v3_enabled: api.scheduling_v3_enabled(at).unwrap_or(false),
170		relay_parent_offset: api.relay_parent_offset(at).unwrap_or_default(),
171	}
172}

174/// Everything the builder needs for one slot, all derived from a single [`SchedulingParams`].
175struct BuildingPrerequisites<Block: BlockT> {
176	/// The relay block the candidate is scheduled at.
177	scheduling_parent_header: RelayHeader,
178	/// Whether scheduling V3 applies to this candidate.
179	v3_enabled: bool,
180	/// Distance from the scheduling parent down to the relay parent.
181	relay_parent_offset: u32,
182	/// The relay parent and the descendants linking it to the scheduling parent.
183	relay_parent_data: RelayParentData,
184	/// The parent to build on, plus the included header at the scheduling parent.
185	parent_search_result: consensus_common::ParentSearchResult<Block>,
186}
187
188/// The relay chain context `params` imply: the scheduling parent, whether V3 applies to it, and the
189/// relay parent with its descendants.
190async fn derive_relay_context<RelayClient>(
191	relay_client: &RelayClient,
192	relay_chain_data_cache: &mut RelayChainDataCache<RelayClient>,
193	scheduling_info: &mut SchedulingInfo<RelayClient>,
194	params: SchedulingParams,
195) -> Option<(RelayHeader, bool, RelayParentData)>
196where
197	RelayClient: RelayChainInterface + 'static,
198{
199	let Some((scheduling_parent_header, v3_enabled)) = scheduling_info
200		.wait_for_scheduling_parent(relay_chain_data_cache, params.v3_enabled)
201		.await
202	else {
203		tracing::warn!(target: LOG_TARGET, "Unable to fetch the scheduling parent hash.");
204		return None;
205	};
206
207	let max_relay_parent_session_age = if v3_enabled {
208		relay_client
209			.max_relay_parent_session_age(scheduling_parent_header.hash())
210			.await
211			.unwrap_or(0)
212	} else {
213		0
214	};
215
216	let Ok(Some(relay_parent_data)) = offset_relay_parent_find_descendants(
217		relay_chain_data_cache,
218		scheduling_parent_header.clone(),
219		params.relay_parent_offset,
220		max_relay_parent_session_age,
221	)
222	.await
223	else {
224		return None;
225	};
226
227	Some((scheduling_parent_header, v3_enabled, relay_parent_data))
228}

230/// Derive the [`BuildingPrerequisites`] for the current slot. `None` means the slot is skipped.
231///
232/// The relay chain context follows the [`SchedulingParams`] of the runtime that executes the block,
233/// which is only known once the parent is settled. So the first phase derives the context from the
234/// para best head in order to run the parent search, and the second re-derives it from the chosen
235/// parent whenever the two disagree, keeping the parent the first phase settled on.
236async fn derive_building_prerequisites<Block, Client, Backend, RelayClient>(
237	para_client: &Client,
238	para_backend: &Backend,
239	relay_client: &RelayClient,
240	relay_chain_data_cache: &mut RelayChainDataCache<RelayClient>,
241	scheduling_info: &mut SchedulingInfo<RelayClient>,
242	para_id: ParaId,
243) -> Option<BuildingPrerequisites<Block>>
244where
245	Block: BlockT,
246	Client: ProvideRuntimeApi<Block> + HeaderBackend<Block>,
247	Client::Api: SchedulingV3EnabledApi<Block> + RelayParentOffsetApi<Block> + ApiExt<Block>,
248	Backend: sc_client_api::Backend<Block>,
249	RelayClient: RelayChainInterface + 'static,
250{
251	let best_hash = para_client.info().best_hash;
252	let best_params = scheduling_params_at(para_client, best_hash);
253
254	let (scheduling_parent_header, v3_enabled, relay_parent_data) =
255		derive_relay_context(relay_client, relay_chain_data_cache, scheduling_info, best_params)
256			.await?;
257
258	let parent_search_params = if v3_enabled {
259		ParentSearchParams::V3 { scheduling_parent: scheduling_parent_header.hash() }
260	} else {
261		ParentSearchParams::V2 { scheduling_parent: relay_parent_data.relay_parent().hash() }
262	};
263	let parent_search_result = crate::collators::find_parent(
264		relay_client,
265		para_backend,
266		para_id,
267		parent_search_params,
268		|parent| {
269			// We never want to build on any "middle block" that isn't the last block in
270			// a core.
271			// When the digest item doesn't exist, we are running in compatibility
272			// mode and all parents are valid.
273			CumulusDigestItem::is_last_block_in_core(parent.digest()).unwrap_or(true)
274		},
275	)
276	.await?;
277
278	let build_parent_hash = parent_search_result.best_parent_header.hash();
279	let build_params = scheduling_params_at(para_client, build_parent_hash);
280	if build_parent_hash == best_hash || build_params == best_params {
281		return Some(BuildingPrerequisites {
282			scheduling_parent_header,
283			v3_enabled,
284			relay_parent_offset: best_params.relay_parent_offset,
285			relay_parent_data,
286			parent_search_result,
287		});
288	}
289
290	tracing::info!(
291		target: LOG_TARGET,
292		best = ?best_hash,
293		?build_parent_hash,
294		"Build parent's runtime disagrees with the para best's, re-deriving the relay chain \
295		context from the build parent.",
296	);
297
298	let (scheduling_parent_header, v3_enabled, relay_parent_data) =
299		derive_relay_context(relay_client, relay_chain_data_cache, scheduling_info, build_params)
300			.await?;
301
302	Some(BuildingPrerequisites {
303		scheduling_parent_header,
304		v3_enabled,
305		relay_parent_offset: build_params.relay_parent_offset,
306		relay_parent_data,
307		parent_search_result,
308	})
309}

311/// Run block-builder.
312pub fn run_block_builder<Block, P, BI, CIDP, Client, Backend, RelayClient, CHP, Proposer, CS>(
313	params: BuilderTaskParams<Block, BI, CIDP, Client, Backend, RelayClient, CHP, Proposer, CS>,
314) -> impl Future<Output = ()> + Send + 'static
315where
316	Block: BlockT,
317	Client: ProvideRuntimeApi<Block>
318		+ UsageProvider<Block>
319		+ BlockOf
320		+ AuxStore
321		+ HeaderBackend<Block>
322		+ BlockBackend<Block>
323		+ Send
324		+ Sync
325		+ 'static,
326	Client::Api: AuraApi<Block, P::Public>
327		+ RelayParentOffsetApi<Block>
328		+ AuraUnincludedSegmentApi<Block>
329		+ TargetBlockRate<Block>
330		+ BlockBuilder<Block>
331		+ cumulus_primitives_core::KeyToIncludeInRelayProof<Block>
332		+ SchedulingV3EnabledApi<Block>,
333	Backend: sc_client_api::Backend<Block> + 'static,
334	RelayClient: RelayChainInterface + Clone + 'static,
335	CIDP: CreateInherentDataProviders<Block, ()> + 'static,
336	CIDP::InherentDataProviders: Send,
337	BI: BlockImport<Block> + ParachainBlockImportMarker + Send + Sync + 'static,
338	Proposer: Environment<Block> + Send + Sync + 'static,
339	CS: CollatorServiceInterface<Block> + Send + Sync + 'static,
340	CHP: consensus_common::ValidationCodeHashProvider<Block::Hash> + Send + Sync + 'static,
341	P: Pair + Send + Sync + 'static,
342	P::Public: AppPublic + Member + Codec,
343	P::Signature: TryFrom<Vec<u8>> + Member + Codec,
344{
345	async move {
346		tracing::info!(target: LOG_TARGET, "Starting slot-based block-builder task.");
347		let BuilderTaskParams {
348			relay_client,
349			create_inherent_data_providers,
350			para_client,
351			keystore,
352			block_import,
353			collator_peer_id,
354			para_id,
355			proposer,
356			collator_service,
357			collator_sender,
358			code_hash_provider,
359			relay_chain_slot_duration,
360			para_backend,
361			slot_offset,
362			max_pov_percentage,
363		} = params;
364
365		let mut slot_timer = SlotTimer::new_with_offset(slot_offset, relay_chain_slot_duration);
366
367		let mut collator = {
368			let params = collator_util::Params {
369				create_inherent_data_providers,
370				block_import,
371				relay_client: relay_client.clone(),
372				keystore: keystore.clone(),
373				collator_peer_id,
374				para_id,
375				proposer,
376				collator_service,
377			};
378
379			Collator::<Block, P, _, _, _, _, _>::new(params)
380		};
381
382		let mut relay_chain_data_cache = RelayChainDataCache::new(relay_client.clone(), para_id);
383		let mut connection_helper = BackingGroupConnectionHelper::new(
384			keystore.clone(),
385			relay_client
386				.overseer_handle()
387				// Should never fail. If it fails, then providing collations to relay chain
388				// doesn't work either. So it is fine to panic here.
389				.expect("Relay chain interface must provide overseer handle."),
390		);
391
392		let mut scheduling_info = SchedulingInfo::new(relay_chain_slot_duration, slot_offset);
393		let maybe_best_relay_block_data = scheduling_info
394			.ensure_initialized(&relay_client, &mut relay_chain_data_cache)
395			.await;
396
397		let v3_enabled = SchedulingInfo::<RelayClient>::is_v3_enabled(
398			scheduling_params_at(&*para_client, para_client.info().best_hash).v3_enabled,
399			maybe_best_relay_block_data,
400		);
401		slot_timer.set_offset_by_scheduling_version(v3_enabled, slot_offset);
402
403		loop {
404			let _ = scheduling_info
405				.ensure_initialized(&relay_client, &mut relay_chain_data_cache)
406				.await;
407
408			// We wait here until the next slot arrives.
409			let Ok(slot_time) = slot_timer.wait_until_next_slot().await else {
410				tracing::error!(target: LOG_TARGET, "Unable to wait for next slot.");
411				return;
412			};
413
414			let Some(BuildingPrerequisites {
415				scheduling_parent_header,
416				v3_enabled,
417				relay_parent_offset,
418				relay_parent_data,
419				parent_search_result,
420			}) = derive_building_prerequisites(
421				&*para_client,
422				&*para_backend,
423				&relay_client,
424				&mut relay_chain_data_cache,
425				&mut scheduling_info,
426				para_id,
427			)
428			.await
429			else {
430				continue;
431			};
432
433			slot_timer.set_offset_by_scheduling_version(v3_enabled, slot_offset);
434
435			let relay_parent_header = relay_parent_data.relay_parent().clone();
436			let relay_parent_hash = relay_parent_header.hash();
437
438			// For the logic that follows we need the included header at the relay parent since
439			// it will be used for checking the unincluded segment len.
440			// Corresponding checks related to the unincluded segment len are also done by the
441			// runtime in the `set_validation_data` inherent, using the relay parent context.
442			let included_header_at_execution = if v3_enabled {
443				match fetch_included_from_relay_chain(
444					&relay_client,
445					&*para_backend,
446					relay_parent_hash,
447					para_id,
448				)
449				.await
450				{
451					Ok(Some((header, _))) => header,
452					Ok(None) => {
453						tracing::error!(
454							target: LOG_TARGET,
455							"Failed to fetch the included header at execution \
456							from the relay chain."
457						);
458						continue;
459					},
460					Err(error) => {
461						tracing::error!(
462							target: LOG_TARGET,
463							?error,
464							"Failed to fetch the included header at execution \
465							from the relay chain."
466						);
467						continue;
468					},
469				}
470			} else {
471				parent_search_result.included_at_scheduling
472			};
473			let initial_parent_hash = parent_search_result.best_parent_header.hash();
474			let initial_parent_header = parent_search_result.best_parent_header;
475			let unincluded_segment_len_at_execution = initial_parent_header
476				.number()
477				.saturating_sub(*included_header_at_execution.number());
478
479			let Ok(para_slot_duration) =
480				crate::slot_duration_at(&*para_client, initial_parent_hash)
481			else {
482				tracing::error!(target: LOG_TARGET, "Failed to fetch slot duration from runtime.");
483				continue;
484			};
485
486			// Use the slot calculated from relay parent
487			let Some(para_slot) = adjust_para_to_relay_parent_slot(
488				relay_parent_data.relay_parent(),
489				relay_chain_slot_duration,
490				para_slot_duration,
491			) else {
492				continue;
493			};
494
495			let Ok(max_pov_size) = relay_chain_data_cache
496				.get_by_hash(relay_parent_hash)
497				.await
498				.map(|d| d.max_pov_size)
499			else {
500				continue;
501			};
502
503			let allowed_pov_size = if let Some(max_pov_percentage) = max_pov_percentage {
504				max_pov_size * max_pov_percentage / 100
505			} else {
506				// Set the block limit to 85% of the maximum PoV size.
507				//
508				// Once https://github.com/paritytech/polkadot-sdk/issues/6020 issue is
509				// fixed, this should be removed.
510				max_pov_size * 85 / 100
511			} as usize;
512
513			// We mainly call this to inform users at genesis if there is a mismatch with the
514			// on-chain data.
515			collator
516				.collator_service()
517				.check_block_status(initial_parent_hash, &initial_parent_header);
518
519			let Some(relay_slot) = get_relay_slot(&relay_parent_header) else { continue };
520
521			let included_hash_at_execution = included_header_at_execution.hash();
522
523			if let Ok(authorities) =
524				onchain_runtime_api(&*para_client).authorities(initial_parent_hash)
525			{
526				connection_helper.update::<P>(para_slot.slot, &authorities).await;
527			}
528
529			let Some(slot_claim) = crate::collators::claim_slot::<_, _, P>(
530				para_slot.slot,
531				para_slot.timestamp,
532				initial_parent_hash,
533				&*para_client,
534				&keystore,
535			)
536			.await
537			else {
538				tracing::debug!(
539					target: LOG_TARGET,
540					?unincluded_segment_len_at_execution,
541					relay_parent = ?relay_parent_hash,
542					relay_parent_num = %relay_parent_header.number(),
543					?included_hash_at_execution,
544					included_num_at_execution = %included_header_at_execution.number(),
545					initial_parent = ?initial_parent_hash,
546					slot = ?para_slot.slot,
547					"Not eligible to claim slot."
548				);
549				continue;
550			};
551
552			tracing::debug!(
553				target: LOG_TARGET,
554				?unincluded_segment_len_at_execution,
555				relay_parent = ?relay_parent_hash,
556				relay_parent_num = %relay_parent_header.number(),
557				relay_parent_offset,
558				?included_hash_at_execution,
559				included_num_at_execution = %included_header_at_execution.number(),
560				initial_parent = ?initial_parent_hash,
561				slot = ?para_slot.slot,
562				"Claiming slot."
563			);
564
565			// Determine claim queue lookup parameters.
566			//
567			// V3: look up at scheduling_parent (fresh RC tip), offset is just
568			// max_claim_queue_offset since the claim queue is already at the tip.
569			//
570			// V1/V2: look up at relay_parent which is relay_parent_offset blocks
571			// behind the tip, so the offset includes relay_parent_offset to
572			// compensate.
573			let maybe_max_claim_queue_offset = onchain_runtime_api(&*para_client)
574				.max_claim_queue_offset(initial_parent_hash)
575				.map(|offset| offset as u32);
576			let (claim_queue_relay_block, claim_queue_offset) = if v3_enabled {
577				// V3: look up at scheduling_parent (fresh tip)
578				(&scheduling_parent_header, maybe_max_claim_queue_offset.unwrap_or(2))
579			} else {
580				// V1/V2: look up at relay_parent, add relay_parent_offset
581				// For the `max_claim_queue_offset` we use a default of `0` for backwards
582				// compatibility when the runtime API is not implemented.
583				let total_offset = relay_parent_offset + maybe_max_claim_queue_offset.unwrap_or(0);
584				(&relay_parent_header, total_offset)
585			};
586			let mut cores = match determine_cores(
587				&mut relay_chain_data_cache,
588				claim_queue_relay_block,
589				para_id,
590				claim_queue_offset,
591			)
592			.await
593			{
594				Ok(Some(core)) => core,
595				Ok(None) => {
596					tracing::debug!(
597						target: LOG_TARGET,
598						relay_parent = ?relay_parent_hash,
599						"No cores scheduled."
600					);
601					continue;
602				},
603				Err(()) => {
604					tracing::error!(
605						target: LOG_TARGET,
606						relay_parent = ?relay_parent_hash,
607						"Failed to determine cores."
608					);
609
610					break;
611				},
612			};
613
614			let number_of_blocks =
615				match para_client.runtime_api().target_block_rate(initial_parent_hash) {
616					Ok(interval) => interval,
617					Err(error) => {
618						tracing::debug!(
619							target: LOG_TARGET,
620							block = ?initial_parent_hash,
621							?error,
622							"Failed to fetch `slot_schedule`, assuming one block per core"
623						);
624
625						// Backwards compatible we use the number of cores as number of blocks.
626						cores.total_cores()
627					},
628				};
629
630			// In total we want to have at max `number_of_blocks` cores to use.
631			cores.truncate_cores(number_of_blocks);
632			let raw_blocks_per_core = (number_of_blocks / cores.total_cores()).max(1);
633			let left_over_blocks = number_of_blocks % cores.total_cores();
634			let blocks_per_cores = (0..cores.total_cores())
635				.map(|i| {
636					// We distribute the left over blocks across the cores.
637					raw_blocks_per_core + u32::from(i < left_over_blocks)
638				})
639				.collect::<Vec<_>>();
640
641			tracing::debug!(
642				target: LOG_TARGET,
643				?blocks_per_cores,
644				core_indices = ?cores.core_indices(),
645				"Core configuration",
646			);
647
648			let mut pov_parent_header = initial_parent_header;
649			let mut pov_parent_hash = initial_parent_hash;
650			let block_time = relay_chain_slot_duration / number_of_blocks;
651
652			for blocks_per_core in blocks_per_cores {
653				let time_for_core = slot_time.time_left() / cores.cores_left();
654
655				match build_collation_for_core(BuildCollationParams {
656					pov_parent_header,
657					pov_parent_hash,
658					relay_parent_header: &relay_parent_header,
659					relay_parent_hash,
660					max_pov_size,
661					para_id,
662					relay_client: &relay_client,
663					code_hash_provider: &code_hash_provider,
664					slot_claim: &slot_claim,
665					collator_sender: &collator_sender,
666					collator: &mut collator,
667					allowed_pov_size,
668					core_info: cores.core_info(),
669					core_index: cores.core_index(),
670					block_time,
671					blocks_per_core,
672					time_for_core,
673					is_last_core_in_parachain_slot: cores.is_last_core() &&
674						slot_time.is_parachain_slot_ending(para_slot_duration.as_duration()),
675					collator_peer_id,
676					relay_parent_data: relay_parent_data.clone(),
677					total_number_of_blocks: number_of_blocks,
678					included_hash_at_execution,
679					relay_slot,
680					para_slot: para_slot.slot,
681					para_client: &*para_client,
682					v3_enabled,
683				})
684				.await
685				{
686					Ok(Some(header)) => {
687						pov_parent_header = header;
688						pov_parent_hash = pov_parent_header.hash();
689					},
690					// Let's wait for the next slot
691					Ok(None) => break,
692					Err(()) => return,
693				}
694
695				if !cores.advance() {
696					break;
697				}
698			}
699		}
700	}
701}

745/// Build a collation for one core.
746///
747/// One collation can be composed of multiple blocks.
748async fn build_collation_for_core<
749	Block: BlockT,
750	P,
751	RelayClient,
752	BI,
753	CIDP,
754	Proposer,
755	CS,
756	CHP,
757	Client,
758>(
759	BuildCollationParams {
760		pov_parent_header,
761		pov_parent_hash,
762		relay_parent_header,
763		relay_parent_hash,
764		max_pov_size,
765		para_id,
766		relay_client,
767		code_hash_provider,
768		slot_claim,
769		collator_sender,
770		collator,
771		allowed_pov_size,
772		core_info,
773		core_index,
774		block_time,
775		blocks_per_core,
776		time_for_core: slot_time_for_core,
777		is_last_core_in_parachain_slot,
778		collator_peer_id,
779		mut relay_parent_data,
780		total_number_of_blocks,
781		included_hash_at_execution,
782		relay_slot,
783		para_slot,
784		para_client,
785		v3_enabled,
786	}: BuildCollationParams<'_, Block, P, RelayClient, BI, CIDP, Proposer, CS, CHP, Client>,
787) -> Result<Option<Block::Header>, ()>
788where
789	RelayClient: RelayChainInterface + 'static,
790	P: Pair,
791	P::Public: AppPublic + Member + Codec,
792	P::Signature: TryFrom<Vec<u8>> + Member + Codec,
793	CIDP: CreateInherentDataProviders<Block, ()> + 'static,
794	CIDP::InherentDataProviders: Send,
795	BI: BlockImport<Block> + ParachainBlockImportMarker + Send + Sync + 'static,
796	Proposer: Environment<Block> + Send + Sync + 'static,
797	CS: CollatorServiceInterface<Block> + Send + Sync + 'static,
798	CHP: consensus_common::ValidationCodeHashProvider<Block::Hash> + Send + Sync + 'static,
799	Client: ProvideRuntimeApi<Block>,
800	Client::Api: AuraUnincludedSegmentApi<Block>
801		+ ApiExt<Block>
802		+ cumulus_primitives_core::KeyToIncludeInRelayProof<Block>,
803{
804	let core_start = Instant::now();
805
806	let validation_data = PersistedValidationData {
807		parent_head: pov_parent_header.encode().into(),
808		relay_parent_number: *relay_parent_header.number(),
809		relay_parent_storage_root: *relay_parent_header.state_root(),
810		max_pov_size,
811	};
812
813	// Check if V3 scheduling is enabled and build scheduling proof if so.
814	let mut scheduling_proof = None;
815	if v3_enabled {
816		// The relay parent descendants are only needed for v2.
817		let descendants = relay_parent_data.take_descendants();
818		// The descendants are ordered from oldest to newest, so we need to reverse them.
819		let header_chain: Vec<_> = descendants.into_iter().rev().collect();
820		let scheduling_parent =
821			header_chain.first().map(|header| header.hash()).unwrap_or(relay_parent_hash);
822
823		tracing::debug!(
824			target: LOG_TARGET,
825			relay_parent = ?relay_parent_hash,
826			?scheduling_parent,
827			header_chain_len = header_chain.len(),
828			"Building V3 collation with scheduling proof",
829		);
830
831		scheduling_proof = Some(SchedulingProof {
832			header_chain,
833			// Initial submission: internal_scheduling_parent == relay_parent, so the
834			// internal scheduling parent header is the relay parent's header itself.
835			internal_scheduling_parent_header: relay_parent_header.clone(),
836			// Initial submission: no signature needed, core selection from UMP signals
837			signed_scheduling_info: None,
838		});
839	}
840
841	let Some(validation_code_hash) = code_hash_provider.code_hash_at(pov_parent_hash) else {
842		tracing::error!(
843			target: LOG_TARGET,
844			?pov_parent_hash,
845			"Could not fetch validation code hash",
846		);
847
848		return Err(());
849	};
850
851	check_validation_code_or_log(&validation_code_hash, para_id, relay_client, relay_parent_hash)
852		.await;
853
854	let session = resolve_session(relay_client, relay_parent_hash)
855		.await
856		.inspect_err(|err| {
857			tracing::warn!(
858				target: LOG_TARGET,
859				?relay_parent_hash,
860				?err,
861				"Could not resolve relay-parent session; resubmission entry will be skipped.",
862			);
863		})
864		.ok();
865
866	let mut blocks = Vec::new();
867	let mut proofs = Vec::new();
868	let mut ignored_nodes = IgnoredNodes::default();
869
870	let mut parent_hash = pov_parent_hash;
871	let mut parent_header = pov_parent_header.clone();
872
873	for block_index in 0..blocks_per_core {
874		// Check if we can build the next block
875		if !crate::collators::can_build_upon::<Block, Client>(
876			parent_hash,
877			included_hash_at_execution,
878			relay_slot,
879			para_slot,
880			para_client,
881		)
882		.await
883		{
884			tracing::debug!(
885				target: LOG_TARGET,
886				?parent_hash,
887				?included_hash_at_execution,
888				"Cannot build next block due to unincluded segment constraints, skipping entire bundle. Will continue at the next slot."
889			);
890
891			return Ok(None);
892		}
893
894		// Create schedule for this block to determine timing decisions
895		let schedule = BlockProductionSchedule::new(
896			block_index,
897			blocks_per_core,
898			total_number_of_blocks,
899			is_last_core_in_parachain_slot,
900		);
901
902		if schedule.should_skip_production() {
903			tracing::debug!(
904				target: LOG_TARGET,
905				"Skipping block production so that the next node is able to import all blocks before its slot."
906			);
907			break;
908		}
909
910		tracing::trace!(
911			target: LOG_TARGET,
912			%block_index,
913			core_index = %core_index.0,
914			"Preparing to build block"
915		);
916
917		let (parachain_inherent_data, other_inherent_data) = match collator
918			.create_inherent_data_with_rp_offset(
919				relay_parent_hash,
920				&validation_data,
921				parent_hash,
922				slot_claim.timestamp(),
923				Some(relay_parent_data.clone()),
924				crate::collators::get_relay_proof_request(para_client, parent_hash),
925				collator_peer_id,
926			)
927			.await
928		{
929			Err(err) => {
930				tracing::error!(target: LOG_TARGET, ?err, "Failed to create inherent data.");
931				return Ok(None);
932			},
933			Ok(x) => x,
934		};
935
936		let storage_proof_recorder =
937			ProofRecorder::<Block>::with_ignored_nodes(ignored_nodes.clone());
938
939		let proof_size_recorder = RecordingProofSizeProvider::new(storage_proof_recorder.clone());
940
941		let mut extra_extensions = Extensions::default();
942		extra_extensions.register(ProofSizeExt::new(proof_size_recorder.clone()));
943
944		let block_production_start = Instant::now();
945		// The time we have left to spent for the block.
946		let time_left_for_block = slot_time_for_core.saturating_sub(core_start.elapsed()) /
947			(blocks_per_core - block_index) as u32;
948
949		// The first block on a multi-block core gets the full remaining core time so that the
950		// runtime's `FullCore` weight mode can actually be utilized. Subsequent blocks are
951		// capped at `block_time` because they only carry fractional weight.
952		//
953		// Single-block cores (blocks_per_core == 1) go through schedule.authoring_duration()
954		// so that slot handover adjustments (e.g., Shorten) are applied on the last core.
955		let authoring_duration = if block_index == 0 && blocks_per_core > 1 {
956			slot_time_for_core.saturating_sub(core_start.elapsed())
957		} else {
958			schedule.authoring_duration(time_left_for_block, block_time)
959		};
960
961		tracing::trace!(
962			target: LOG_TARGET,
963			?authoring_duration,
964			"Building block"
965		);
966
967		let Ok(Some((built_block, mut import_block))) = collator
968			.build_block(BuildBlockAndImportParams {
969				parent_header: &parent_header,
970				slot_claim,
971				additional_pre_digest: vec![
972					CumulusDigestItem::CoreInfo(core_info.clone()).to_digest_item(),
973					CumulusDigestItem::BlockBundleInfo(BlockBundleInfo {
974						index: block_index as u8,
975						is_last: schedule.block_ends_bundle(),
976					})
977					.to_digest_item(),
978				],
979				parachain_inherent_data,
980				extra_inherent_data: other_inherent_data,
981				proposal_duration: authoring_duration,
982				max_pov_size: allowed_pov_size,
983				storage_proof_recorder: storage_proof_recorder.into(),
984				extra_extensions,
985			})
986			.await
987		else {
988			tracing::error!(target: LOG_TARGET, "Unable to build block at slot.");
989			return Ok(None);
990		};
991
992		parent_hash = built_block.block.header().hash();
993		parent_header = built_block.block.header().clone();
994
995		// Extract and add proof size recordings to the import block
996		let recorded_sizes = proof_size_recorder
997			.recorded_estimations()
998			.into_iter()
999			.map(|size| size as u32)
1000			.collect::<Vec<u32>>();
1001
1002		if !recorded_sizes.is_empty() {
1003			prepare_proof_size_recording_aux_data(parent_hash, recorded_sizes).for_each(
1004				|(k, v)| {
1005					import_block.auxiliary.push((k, Some(v)));
1006				},
1007			);
1008		}
1009
1010		let proof = Arc::new(built_block.proof);
1011		if let Some(relay_parent_session) = session {
1012			prepare_resubmission_aux_data::<Block>(
1013				built_block.block.header().hash(),
1014				proof.clone(),
1015				relay_parent_header.clone(),
1016				relay_parent_session,
1017			)
1018			.for_each(|(k, v)| {
1019				import_block.auxiliary.push((k, Some(v)));
1020			});
1021		}
1022
1023		if let Err(error) = collator.import_block(import_block).await {
1024			tracing::error!(target: LOG_TARGET, ?error, "Failed to import built block.");
1025			return Ok(None);
1026		}
1027
1028		// Announce the newly built block to our peers.
1029		collator.collator_service().announce_block(parent_hash, None);
1030
1031		blocks.push(built_block.block);
1032		proofs.push(Arc::unwrap_or_clone(proof));
1033
1034		let full_core_digest = CumulusDigestItem::contains_use_full_core(parent_header.digest());
1035		let runtime_upgrade_digest = parent_header
1036			.digest()
1037			.logs
1038			.iter()
1039			.any(|it| matches!(it, sp_runtime::DigestItem::RuntimeEnvironmentUpdated));
1040
1041		if full_core_digest || runtime_upgrade_digest {
1042			tracing::trace!(
1043				target: LOG_TARGET,
1044				block_hash = ?parent_hash,
1045				time_used_by_block_in_secs = %block_production_start.elapsed().as_secs_f32(),
1046				%full_core_digest,
1047				%runtime_upgrade_digest,
1048				"Stopping block production for core",
1049			);
1050			break;
1051		}
1052
1053		ignored_nodes.extend(IgnoredNodes::from_storage_proof::<HashingFor<Block>>(
1054			proofs.last().expect("We just pushed the proof into the vector; qed"),
1055		));
1056		ignored_nodes.extend(IgnoredNodes::from_memory_db(built_block.backend_transaction));
1057
1058		// If there is still time left for the block in the slot, we sleep the rest of the time.
1059		// This ensures that we have some steady block rate.
1060		if let Some(sleep) = time_left_for_block
1061			.checked_sub(block_production_start.elapsed())
1062			// Let's not sleep for the last block here, to send out the collation as early as
1063			// possible.
1064			.filter(|_| !schedule.is_effective_last_block())
1065		{
1066			tokio::time::sleep(sleep).await;
1067		}
1068	}
1069
1070	if blocks.is_empty() {
1071		tracing::debug!(
1072			target: LOG_TARGET,
1073			?core_index,
1074			relay_parent = ?relay_parent_hash,
1075			"Did not build any blocks, returning"
1076		);
1077
1078		return Ok(None);
1079	}
1080
1081	let proof = StorageProof::merge(proofs);
1082
1083	tracing::trace!(
1084		target: LOG_TARGET,
1085		?core_index,
1086		relay_parent = ?relay_parent_hash,
1087		blocks = ?blocks.iter().map(|b| b.hash()).collect::<Vec<_>>(),
1088		"Sending out PoV"
1089	);
1090
1091	if let Err(err) = collator_sender.unbounded_send(CollatorMessage {
1092		relay_parent: relay_parent_hash,
1093		scheduling_proof,
1094		parent_header: pov_parent_header.clone(),
1095		blocks,
1096		proof,
1097		validation_code_hash,
1098		core_index,
1099		validation_data,
1100	}) {
1101		tracing::error!(target: LOG_TARGET, ?err, "Unable to send block to collation task.");
1102		Err(())
1103	} else {
1104		// Now let's sleep for the rest of the core.
1105		if let Some(sleep) = slot_time_for_core.checked_sub(core_start.elapsed()) {
1106			tokio::time::sleep(sleep).await;
1107		}
1108
1109		Ok(Some(parent_header))
1110	}
1111}

1113/// Translate the slot of the relay parent to the slot of the parachain.
1114fn adjust_para_to_relay_parent_slot(
1115	relay_header: &RelayHeader,
1116	relay_chain_slot_duration: Duration,
1117	para_slot_duration: SlotDuration,
1118) -> Option<SlotInfo> {
1119	let relay_slot = sc_consensus_babe::find_pre_digest::<RelayBlock>(&relay_header)
1120		.map(|babe_pre_digest| babe_pre_digest.slot())
1121		.ok()?;
1122	let new_slot = Slot::from_timestamp(
1123		relay_slot
1124			.timestamp(SlotDuration::from_millis(relay_chain_slot_duration.as_millis() as u64))?,
1125		para_slot_duration,
1126	);
1127	let para_slot = SlotInfo { slot: new_slot, timestamp: new_slot.timestamp(para_slot_duration)? };
1128	tracing::debug!(
1129		target: LOG_TARGET,
1130		timestamp = ?para_slot.timestamp,
1131		slot = ?para_slot.slot,
1132		"Parachain slot adjusted to relay chain.",
1133	);
1134	Some(para_slot)
1135}

1137/// Finds a relay chain parent block at a specified offset from the best block, collecting its
1138/// descendants.
1139///
1140/// # Returns
1141/// * `Ok(RelayParentData)` - Contains the target relay parent and its ordered list of descendants
1142/// * `Err(())` - If any relay chain block header cannot be retrieved
1143///
1144/// The function traverses backwards from the best block until it finds the block at the specified
1145/// offset, collecting all blocks in between to maintain the chain of ancestry.
1146pub async fn offset_relay_parent_find_descendants<RelayClient>(
1147	relay_chain_data_cache: &mut RelayChainDataCache<RelayClient>,
1148	scheduling_parent: RelayHeader,
1149	relay_parent_offset: u32,
1150	max_relay_parent_session_age: u32,
1151) -> Result<Option<RelayParentData>, ()>
1152where
1153	RelayClient: RelayChainInterface + 'static,
1154{
1155	let scheduling_parent_hash = scheduling_parent.hash();
1156	let mut current_relay_header = scheduling_parent;
1157
1158	let mut relay_parent_descendants: VecDeque<RelayHeader> = Default::default();
1159	let mut relay_parent_session_age = 0;
1160	loop {
1161		if current_relay_header.number == 0 {
1162			return Ok(None);
1163		}
1164
1165		if relay_parent_session_age > max_relay_parent_session_age {
1166			tracing::debug!(target: LOG_TARGET,
1167				?scheduling_parent_hash,
1168				ancestor = %current_relay_header.hash(),
1169				ancestor_block_number = current_relay_header.number(),
1170				"max_relay_parent_session_age exceeded."
1171			);
1172			return Ok(None);
1173		}
1174		// If the header contains an epoch change log, it means that it's the first block
1175		// of a new session. So, at the next iteration, we will be at the previous session.
1176		if sc_consensus_babe::contains_epoch_change::<RelayBlock>(&current_relay_header) {
1177			relay_parent_session_age += 1;
1178		}
1179
1180		if relay_parent_descendants.len() == relay_parent_offset as usize {
1181			break;
1182		}
1183		relay_parent_descendants.push_front(current_relay_header.clone());
1184
1185		let next_relay_block =
1186			relay_chain_data_cache.get_by_hash(*current_relay_header.parent_hash()).await?;
1187		let next_relay_header = next_relay_block.relay_header.clone();
1188
1189		current_relay_header = next_relay_header;
1190	}
1191
1192	tracing::debug!(
1193		target: LOG_TARGET,
1194		relay_parent_hash = %current_relay_header.hash(),
1195		relay_parent_num = current_relay_header.number(),
1196		num_descendant = relay_parent_descendants.len(),
1197		"Relay parent descendants."
1198	);
1199
1200	Ok(Some(RelayParentData::new_with_descendants(
1201		current_relay_header,
1202		relay_parent_descendants.into(),
1203	)))
1204}

1207pub struct Cores {
1208	selector: CoreSelector,
1209	claim_queue_offset: ClaimQueueOffset,
1210	core_indices: Vec<CoreIndex>,
1211}
1212
1213impl Cores {
1214	/// Returns the current [`CoreInfo`].
1215	pub fn core_info(&self) -> CoreInfo {
1216		CoreInfo {
1217			selector: self.selector,
1218			claim_queue_offset: self.claim_queue_offset,
1219			number_of_cores: (self.core_indices.len() as u16).into(),
1220		}
1221	}
1222
1223	/// Returns the core indices.
1224	fn core_indices(&self) -> &[CoreIndex] {
1225		&self.core_indices
1226	}
1227
1228	/// Returns the current [`CoreIndex`].
1229	pub fn core_index(&self) -> CoreIndex {
1230		self.core_indices[self.selector.0 as usize]
1231	}
1232
1233	/// Advance to the next available core.
1234	///
1235	/// Returns `false` if there is no core left.
1236	fn advance(&mut self) -> bool {
1237		if self.selector.0 as usize + 1 < self.core_indices.len() {
1238			self.selector.0 += 1;
1239			true
1240		} else {
1241			false
1242		}
1243	}
1244
1245	/// Returns the total number of cores.
1246	pub fn total_cores(&self) -> u32 {
1247		self.core_indices.len() as u32
1248	}
1249
1250	/// Truncate `cores` to `max_cores`.
1251	pub fn truncate_cores(&mut self, max_cores: u32) {
1252		self.core_indices.truncate(max_cores as usize);
1253	}
1254
1255	/// Returns the number of cores left.
1256	fn cores_left(&self) -> u32 {
1257		self.total_cores() - self.selector.0 as u32
1258	}
1259
1260	/// Returns if the current core is the last core.
1261	fn is_last_core(&self) -> bool {
1262		self.cores_left() == 1
1263	}
1264}

1266/// Slot handover adjustment strategy based on total block rate.
1267///
1268/// These adjustments exist because without transaction streaming, the next author
1269/// must sequentially import all blocks before building their own. Each variant
1270/// uses a different strategy to provide import buffer time.
1271// TODO: Once transaction streaming is implemented, this can be removed.
1272#[derive(Debug, Clone, Copy)]
1273enum SlotHandoverAdjustment {
1274	/// 0-1 blocks per slot - no adjustment needed.
1275	/// The next author has plenty of time to import.
1276	None,
1277
1278	/// 2-3 blocks per slot (~2-3s block time) - shorten authoring time.
1279	Shorten {
1280		/// Time adjustment factor of last block authoring time.
1281		time_factor: f32,
1282	},
1283
1284	/// >3 blocks per slot (<2s block time) - skip last block.
1285	///
1286	/// Block time is too fast for time reduction alone, so we skip
1287	/// producing the last block in each parachain slot entirely.
1288	Skip,
1289}
1290
1291impl SlotHandoverAdjustment {
1292	/// Determine the appropriate adjustment based on total blocks per relay slot and blocks per
1293	/// core.
1294	fn from_total_blocks(total_blocks: u32, blocks_per_core: u32) -> Self {
1295		match total_blocks {
1296			0..=1 => Self::None,
1297			2..=3 if blocks_per_core == 1 || blocks_per_core == total_blocks => {
1298				Self::Shorten { time_factor: 0.5 }
1299			},
1300			_ => Self::Skip,
1301		}
1302	}
1303
1304	/// Whether this adjustment skips the last block (vs adjusting time).
1305	fn skips_last_block(&self) -> bool {
1306		matches!(self, Self::Skip)
1307	}
1308}
1309
1310/// Policy object that determines block production timing decisions.
1311///
1312/// Encapsulates the complex timing logic for block production, making decisions
1313/// about when to skip blocks, how long to spend authoring, and when to sleep.
1314#[derive(Debug, Clone, Copy)]
1315struct BlockProductionSchedule {
1316	mode: SlotHandoverAdjustment,
1317	block_index: u32,
1318	blocks_per_core: u32,
1319	is_last_core_in_parachain_slot: bool,
1320}
1321
1322impl BlockProductionSchedule {
1323	fn new(
1324		block_index: u32,
1325		blocks_per_core: u32,
1326		total_blocks: u32,
1327		is_last_core_in_parachain_slot: bool,
1328	) -> Self {
1329		Self {
1330			mode: SlotHandoverAdjustment::from_total_blocks(total_blocks, blocks_per_core),
1331			block_index,
1332			blocks_per_core,
1333			is_last_core_in_parachain_slot,
1334		}
1335	}
1336
1337	/// Whether this is the actual last block index in the core.
1338	fn is_last_block_in_core(&self) -> bool {
1339		self.block_index + 1 == self.blocks_per_core
1340	}
1341
1342	/// Whether this is the second-to-last block index.
1343	fn is_second_to_last(&self) -> bool {
1344		self.block_index + 2 == self.blocks_per_core
1345	}
1346
1347	/// Whether to skip producing this block entirely.
1348	///
1349	/// In Bundling mode, we skip the last block in the parachain slot
1350	/// to give the next author time to import all previous blocks.
1351	fn should_skip_production(&self) -> bool {
1352		self.mode.skips_last_block() &&
1353			self.is_last_block_in_core() &&
1354			self.is_last_core_in_parachain_slot
1355	}
1356
1357	/// Whether this is effectively the last block we'll produce for this core.
1358	///
1359	/// Used for `BundleInfo { is_last }` - validators need to know which
1360	/// block might be final. Also used for sleep decisions - we don't sleep
1361	/// after the last or second-to-last block to speed up the final stretch.
1362	///
1363	/// The second-to-last block is always included because:
1364	/// 1. In Bundling mode on the last core, we skip the actual last block
1365	/// 2. Even when not skipping, avoiding sleep on the last two blocks speeds things up
1366	fn is_effective_last_block(&self) -> bool {
1367		self.is_last_block_in_core() || self.is_second_to_last()
1368	}
1369
1370	/// Whether the node stops block production after this block for this bundle.
1371	///
1372	/// Returns `true` when:
1373	/// - This is the last block in the core, OR
1374	/// - This is the second-to-last and the actual last will be skipped (Skip mode on the last core
1375	///   of the parachain slot).
1376	fn block_ends_bundle(&self) -> bool {
1377		self.is_last_block_in_core() ||
1378			(self.is_second_to_last() &&
1379				self.mode.skips_last_block() &&
1380				self.is_last_core_in_parachain_slot)
1381	}
1382
1383	/// Compute the authoring duration given available time.
1384	fn authoring_duration(&self, time_left: Duration, block_time: Duration) -> Duration {
1385		let adjusted = match &self.mode {
1386			SlotHandoverAdjustment::Shorten { time_factor }
1387				if self.is_last_core_in_parachain_slot =>
1388			{
1389				time_left.mul_f32(*time_factor)
1390			},
1391			_ => time_left,
1392		};
1393
1394		block_time.min(adjusted)
1395	}
1396}

1398/// Determine the cores for the given `para_id`.
1399///
1400/// Takes into account the `parent` core to find the next available cores.
1401pub async fn determine_cores<RI: RelayChainInterface + 'static>(
1402	relay_chain_data_cache: &mut RelayChainDataCache<RI>,
1403	scheduling_parent: &RelayHeader,
1404	para_id: ParaId,
1405	relay_parent_offset: u32,
1406) -> Result<Option<Cores>, ()> {
1407	let claim_queue =
1408		&relay_chain_data_cache.get_by_hash(scheduling_parent.hash()).await?.claim_queue;
1409
1410	let core_indices = claim_queue
1411		.iter_claims_at_depth_for_para(relay_parent_offset as _, para_id)
1412		.collect::<Vec<_>>();
1413
1414	Ok(if core_indices.is_empty() {
1415		None
1416	} else {
1417		Some(Cores {
1418			selector: CoreSelector(0),
1419			claim_queue_offset: ClaimQueueOffset(relay_parent_offset as u8),
1420			core_indices,
1421		})
1422	})
1423}
