34/// A potential parent block returned from [`find_parent_for_building`]
35#[derive(PartialEq, Clone)]
36pub struct ParentSearchResult<B: BlockT> {
37	/// The header of the included block (confirmed on relay chain).
38	pub included_header: B::Header,
39	/// The header of the best parent block to build on.
40	pub best_parent_header: B::Header,
41}
42
43impl<B: BlockT> std::fmt::Debug for ParentSearchResult<B> {
44	fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
45		f.debug_struct("ParentSearchResult")
46			.field("included_number", &self.included_header.number())
47			.field("best_parent_hash", &self.best_parent_header.hash())
48			.field("best_parent_number", &self.best_parent_header.number())
49			.finish()
50	}
51}

53fn get_para_header<Block: BlockT>(
54	backend: &impl Backend<Block>,
55	hash: Block::Hash,
56) -> Option<Block::Header> {
57	let Ok(Some(header)) = backend.blockchain().header(hash) else {
58		tracing::warn!(
59			target: LOG_TARGET,
60			%hash,
61			"Failed to get header for para block.",
62		);
63		return None;
64	};
65
66	Some(header)
67}

69async fn fetch_pvd_header<Block: BlockT>(
70	relay_client: &impl RelayChainInterface,
71	at: RelayHash,
72	para_id: ParaId,
73	occupied_core_assumption: OccupiedCoreAssumption,
74) -> RelayChainResult<Option<Block::Header>> {
75	let maybe_header = relay_client
76		.persisted_validation_data(at, para_id, occupied_core_assumption)
77		.await?
78		.and_then(|pvd| Block::Header::decode(&mut &pvd.parent_head.0[..]).ok());
79
80	Ok(maybe_header)
81}

83/// Fetch the included block from the relay chain.
84pub async fn fetch_included_from_relay_chain<B: BlockT>(
85	relay_client: &impl RelayChainInterface,
86	backend: &impl Backend<B>,
87	at: RelayHash,
88	para_id: ParaId,
89) -> Result<Option<(B::Header, B::Hash)>, RelayChainError> {
90	// Fetch the pending header from the relay chain. We use `OccupiedCoreAssumption::TimedOut`
91	// so that even if there is a pending candidate, we assume it is timed out, and we get the
92	// included head.
93	let Some(included_header) =
94		fetch_pvd_header::<B>(relay_client, at, para_id, OccupiedCoreAssumption::TimedOut).await?
95	else {
96		return Ok(None);
97	};
98
99	let included_hash = included_header.hash();
100	// If the included block is not locally known, we can't do anything.
101	let Some(included_header) = get_para_header(backend, included_hash) else {
102		return Ok(None);
103	};
104	Ok(Some((included_header, included_hash)))
105}

107/// Build an ancestry of relay parents that are acceptable.
108///
109/// An acceptable relay parent is one that is no more than `ancestry_lookback` + 1 blocks below the
110/// relay parent we want to build on. Parachain blocks anchored on relay parents older than that can
111/// not be considered potential parents for block building. They have no chance of still getting
112/// included, so our newly build parachain block would also not get included.
113///
114/// On success, returns a vector of `(header_hash, state_root)` of the relevant relay chain
115/// ancestry blocks.
116async fn build_relay_parent_ancestry(
117	relay_client: &impl RelayChainInterface,
118	relay_parent: RelayHash,
119	ancestry_lookback: usize,
120) -> Result<Vec<(RelayHash, RelayHash)>, RelayChainError> {
121	let mut ancestry = Vec::with_capacity(ancestry_lookback + 1);
122	let mut current_rp = relay_parent;
123	while ancestry.len() <= ancestry_lookback {
124		let Some(header) = relay_client.header(RBlockId::hash(current_rp)).await? else { break };
125
126		ancestry.push((current_rp, *header.state_root()));
127		current_rp = *header.parent_hash();
128
129		// Respect the relay-chain rule not to cross session boundaries.
130		if contains_epoch_change::<RelayBlock>(&header) {
131			break;
132		}
133
134		// don't iterate back into the genesis block.
135		if header.number == 1 {
136			break;
137		}
138	}
139	Ok(ancestry)
140}

142/// Check if a block's relay parent is within the allowed ancestry.
143fn is_relay_parent_in_ancestry<Block: BlockT>(
144	header: &Block::Header,
145	rp_ancestry: &[(RelayHash, RelayHash)],
146) -> bool {
147	let digest = header.digest();
148	let relay_parent = cumulus_primitives_core::extract_relay_parent(digest);
149	let storage_root =
150		cumulus_primitives_core::rpsr_digest::extract_relay_parent_storage_root(digest)
151			.map(|(storage_root, _)| storage_root);
152	if relay_parent.is_none() && storage_root.is_none() {
153		return false;
154	}
155
156	rp_ancestry.iter().any(|(rp_hash, rp_storage_root)| {
157		Some(*rp_hash) == relay_parent || Some(*rp_storage_root) == storage_root
158	})
159}

161/// Find the deepest valid parent block starting from `start`.
162///
163/// The `start` block (pending or included) is always valid by construction.
164/// This function explores its descendants via DFS, returning the deepest block
165/// whose relay-parent is within the allowed ancestry.
166fn find_deepest_valid_parent<Block: BlockT>(
167	backend: &impl Backend<Block>,
168	start_header: Block::Header,
169	start_hash: Block::Hash,
170	rp_ancestry: &[(RelayHash, RelayHash)],
171) -> Block::Header {
172	let mut best = start_header;
173
174	let mut frontier: Vec<Block::Hash> =
175		backend.blockchain().children(start_hash).ok().into_iter().flatten().collect();
176
177	tracing::trace!(
178		target: LOG_TARGET,
179		?start_hash,
180		num_children = frontier.len(),
181		"Searching for deepest valid parent."
182	);
183
184	while let Some(hash) = frontier.pop() {
185		let Ok(Some(header)) = backend.blockchain().header(hash) else { continue };
186
187		if !is_relay_parent_in_ancestry::<Block>(&header, rp_ancestry) {
188			continue;
189		}
190
191		// This block is valid - update best if it's deeper.
192		if header.number() > best.number() {
193			best = header.clone();
194		}
195
196		frontier.extend(backend.blockchain().children(hash).ok().into_iter().flatten());
197	}
198
199	best
200}

202/// Find the best parent block to build on.
203///
204/// This accepts a relay-chain block to be used as an anchor and searches for the best
205/// parachain block to use as a parent for a new block.
206///
207/// The search starts from either the pending block (if one exists) or the included block,
208/// and finds the deepest descendant whose relay-parent is within the allowed ancestry.
209///
210/// Returns `None` if no suitable parent can be found (e.g., included block unknown locally).
211pub async fn find_parent_for_building<Block: BlockT>(
212	relay_client: &impl RelayChainInterface,
213	backend: &impl Backend<Block>,
214	para_id: ParaId,
215	relay_parent: RelayHash,
216) -> RelayChainResult<Option<ParentSearchResult<Block>>> {
217	tracing::trace!(
218		target: LOG_TARGET,
219		?para_id,
220		?relay_parent,
221		"Parent search"
222	);
223
224	// Get the included block.
225	let Some((included_header, included_hash)) =
226		fetch_included_from_relay_chain(relay_client, backend, relay_parent, para_id).await?
227	else {
228		return Ok(None);
229	};
230
231	// Fetch the pending block if one exists.
232	let maybe_pending = {
233		// Fetch the most recent pending header from the relay chain. We use
234		// `OccupiedCoreAssumption::Included` so the candidate pending availability gets enacted
235		// before being returned to us.
236		let maybe_header = fetch_pvd_header::<Block>(
237			relay_client,
238			relay_parent,
239			para_id,
240			OccupiedCoreAssumption::Included,
241		)
242		.await?
243		.filter(|pvd_header| pvd_header.hash() != included_hash);
244
245		// If the pending block is not locally known, we can't proceed.
246		if let Some(header) = maybe_header {
247			let hash = header.hash();
248			let Some(header) = get_para_header(backend, hash) else {
249				return Ok(None);
250			};
251			Some((header, hash))
252		} else {
253			None
254		}
255	};
256	// Determine the starting point for the search.
257	let (start_header, start_hash) =
258		maybe_pending.unwrap_or((included_header.clone(), included_hash));
259
260	let ancestry_lookback = relay_client
261		.scheduling_lookahead(relay_parent)
262		.await
263		.unwrap_or(DEFAULT_SCHEDULING_LOOKAHEAD)
264		.saturating_sub(1) as usize;
265	// Build up the ancestry record of the relay chain to compare against.
266	let rp_ancestry =
267		build_relay_parent_ancestry(relay_client, relay_parent, ancestry_lookback).await?;
268
269	// Search for the deepest valid parent starting from the pending/included block.
270	let best_parent_header =
271		find_deepest_valid_parent(backend, start_header, start_hash, &rp_ancestry);
272
273	Ok(Some(ParentSearchResult { included_header, best_parent_header }))
274}
