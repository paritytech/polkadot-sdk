18//! Per-transaction cold/hot access list.
19//!
20//! The per-frame rollback machinery here (flat journal + checkpoint stack, with
21//! `enter_frame` / `commit_frame` / `rollback_frame` wired into `Stack::run`)
22//! mirrors [`crate::transient_storage::TransientStorage`].
23
24use alloc::vec::Vec;
25use frame_support::{BoundedBTreeSet, BoundedVec};
26use sp_core::{ConstU32, H160};
27
28use crate::{exec::Key, limits};
29
30/// Inline-storage cap for `Slot::VarInline`. Covers word-sized keys (`H160`,
31/// `H256`, `AccountId32`). `Slot` stays 40 bytes for any cap up to ~38, at no
32/// memory cost.
33pub const MAX_INLINE_KEY_LEN: usize = 36;
34
35/// Maximum number of distinct `(address, slot)` entries tracked in the
36/// access list within a single transaction.
37///
38/// Bounds the working memory `AccessList` can allocate per transaction.
39/// EIP-2929 does not specify a structural cap; Ethereum relies on gas to
40/// implicitly bound growth.
41///
42/// Past this cap, new touches bill cold without being added to the set;
43/// slots already tracked continue to bill hot.
44///
45/// Memory grows discontinuously due to the runtime allocator (sc-allocator)
46/// rounding allocations up to power-of-2 size classes.
47///
48/// All figures below are approximate order-of-magnitude estimates. The
49/// Ethereum-gas column shows the EIP-2929 cost of filling the set to that
50/// size via cold SLOADs (2 100 gas each).
51///
52/// | Entries | Fix/Inline (Best) | VarLong (Worst) |     Gas (Ethereum) |
53/// |---------|-------------------|-----------------|--------------------|
54/// |       1 |      ~1.3 KB      |     ~1.4 KB     |          2.1 k gas |
55/// |       2 |      ~1.3 KB      |     ~1.6 KB     |          4.2 k gas |
56/// |       8 |      ~1.5 KB      |     ~2.6 KB     |         16.8 k gas |
57/// |      32 |       ~7 KB       |      ~11 KB     |         67.2 k gas |
58/// |     128 |      ~45 KB       |      ~65 KB     |          269 k gas |
59/// |   2 048 |      ~730 KB      |       ~1 MB     |          4.3 M gas |
60///
61/// Set ~2× above the current PoV-reachable ceiling as a backstop: each
62/// cold access charges ~10 KB `proof_size`, capping a transaction
63/// (~7.5 MiB PoV) at ~770 cold touches.
64pub const MAX_ACCESS_LIST_ENTRIES: usize = 2_048;
65
66/// Worst-case per-entry memory in the `BoundedBTreeSet` + journal, measured against
67/// sc-allocator (8-byte headers, power-of-2 buckets). `Slot::Fix` and
68/// `Slot::VarInline` measure ~366 B; `Slot::VarLong` ~502 B. Rounded up to 512
69/// for headroom.
70pub const MAX_ACCESS_LIST_ENTRY_BYTES: usize = 512;
71
72/// Worst-case total memory the access list can hold per transaction.
73pub const MAX_ACCESS_LIST_BYTES: u32 =
74	MAX_ACCESS_LIST_ENTRIES.saturating_mul(MAX_ACCESS_LIST_ENTRY_BYTES) as u32;
75
76/// Storage slot identifier for an access-list entry.
77#[derive(Ord, PartialOrd, Eq, PartialEq, Debug, Clone)]
78pub enum Slot {
79	/// Fixed 32-byte storage key.
80	Fix([u8; 32]),
81	/// Variable-length key up to [`MAX_INLINE_KEY_LEN`], stored inline to
82	/// avoid the per-entry heap allocation `VarLong` requires, while keeping
83	/// `Slot` size bounded.
84	VarInline { bytes: [u8; MAX_INLINE_KEY_LEN], len: u8 },
85	/// Variable-length key longer than [`MAX_INLINE_KEY_LEN`], up to
86	/// `limits::STORAGE_KEY_BYTES`.
87	VarLong(BoundedVec<u8, ConstU32<{ limits::STORAGE_KEY_BYTES }>>),
88}
89
90impl From<&Key> for Slot {
91	fn from(key: &Key) -> Self {
92		match key {
93			Key::Fix(v) => Slot::Fix(*v),
94			Key::Var(v) => {
95				let raw: &[u8] = v.as_ref();
96				if raw.len() <= MAX_INLINE_KEY_LEN {
97					let mut bytes = [0u8; MAX_INLINE_KEY_LEN];
98					bytes[..raw.len()].copy_from_slice(raw);
99					Slot::VarInline { bytes, len: raw.len() as u8 }
100				} else {
101					Slot::VarLong(v.clone())
102				}
103			},
104		}
105	}
106}

108/// Classification of a storage access for pricing.
109#[cfg_attr(test, derive(PartialEq, Eq))]
110#[derive(Clone, Copy, Debug)]
111pub enum StorageAccessKind {
112	/// Persistent storage, tracked by the access list.
113	Persistent(Warmth),
114	/// Transient storage, not tracked by the access list.
115	Transient,
116}
117
118/// Warmth of a persistent storage access. Describes the slot's state
119/// **before** the access.
120#[derive(Clone, Copy, Debug, Eq, PartialEq)]
121pub enum Warmth {
122	/// Slot was already in the access list before this access.
123	Hot,
124	/// Slot was not in the access list before this access; the touch adds
125	/// it (so the next access to the same slot returns `Hot`). `revertible`
126	/// is true when the entry was journaled under an open frame, so a
127	/// `rollback_frame` can drop it and the slot becomes cold again.
128	Cold { revertible: bool },
129}
130
131impl Warmth {
132	/// Whether this was the first access to the slot this transaction.
133	#[cfg(any(test, feature = "runtime-benchmarks"))]
134	pub(crate) fn is_cold(&self) -> bool {
135		matches!(self, Self::Cold { .. })
136	}
137}

139/// Snapshot of per-transaction access-list counters.
140#[derive(Clone, Copy, Debug, Eq, PartialEq)]
141pub struct AccessListMetrics {
142	/// Currently-hot entries (across all open frames).
143	pub size: usize,
144	/// Total cold touches across the transaction, including ones later rolled back.
145	pub cold: u32,
146	/// Total hot touches across the transaction, including ones later rolled back.
147	pub hot: u32,
148}
149
150/// One entry per `(storage slot, contract address)` accessed in the current tx.
151///
152/// Field order is `slot, address` so the derived `Ord` decides on `slot`
153/// first, the most-discriminating field in the typical access pattern (one
154/// contract touching many slots within a transaction).
155#[derive(Ord, PartialOrd, Eq, PartialEq, Debug, Clone)]
156pub struct AccessEntry {
157	/// Slot identifier.
158	pub slot: Slot,
159	/// Contract whose child trie is being touched.
160	pub address: H160,
161}
162
163/// Per-transaction access list with per-frame rollback support. Layout
164/// follows [`crate::transient_storage::TransientStorage`]: a current-state
165/// set, a flat journal of insertions, and journal-index checkpoints.
166///
167/// # Safety invariant
168///
169/// Callers touch the `AccessList` before charging gas, so reverts must roll back the touches
170/// they made. Without that, an out-of-gas at the cold charge after the touch would leave the slot
171/// warm without the cold charge being paid, and a later access would then be billed hot.
172
173#[derive(Default)]
174pub struct AccessList {
175	/// All currently-hot entries.
176	accessed: BoundedBTreeSet<AccessEntry, ConstU32<{ MAX_ACCESS_LIST_ENTRIES as u32 }>>,
177	/// Flat journal of insertions (in order). Each entry was added by exactly
178	/// one frame; `checkpoints` marks the frame boundaries inside this journal.
179	journal: BoundedVec<AccessEntry, ConstU32<{ MAX_ACCESS_LIST_ENTRIES as u32 }>>,
180	/// Stack of journal indices. `checkpoints.last()` is the index at which
181	/// the current frame started inserting; rolling back means draining
182	/// `journal` from that index and removing those entries from `accessed`.
183	checkpoints: Vec<usize>,
184	/// Total cold touches across the transaction. Includes touches in
185	/// frames that later rolled back.
186	cold_count: u32,
187	/// Total hot touches across the transaction. Includes touches in
188	/// frames that later rolled back.
189	hot_count: u32,
190}
191
192impl AccessList {
193	/// Create an empty access list for a new transaction.
194	pub fn new() -> Self {
195		Self::default()
196	}
197
198	/// Open a new nested frame.
199	///
200	/// This allows to either commit or roll back all touches that are made
201	/// after this call. For every `enter_frame` there must be a matching call
202	/// to either `commit_frame` or `rollback_frame`.
203	pub fn enter_frame(&mut self) {
204		self.checkpoints.push(self.journal.len());
205	}
206
207	/// Commit the top frame.
208	///
209	/// Touches made during that frame stay, but may still be rolled back if a
210	/// parent frame later reverts.
211	///
212	/// # Panics
213	///
214	/// Will panic if there is no open frame.
215	pub fn commit_frame(&mut self) {
216		self.checkpoints.pop().expect("frame open; qed");
217	}
218
219	/// Rollback the top frame.
220	///
221	/// Touches made during that frame are removed from the access list.
222	///
223	/// # Panics
224	///
225	/// Will panic if there is no open frame.
226	pub fn rollback_frame(&mut self) {
227		let checkpoint = self.checkpoints.pop().expect("frame open; qed");
228		for entry in self.journal.drain(checkpoint..) {
229			self.accessed.remove(&entry);
230		}
231	}
232
233	/// Non-mutating sibling of [`touch`](Self::touch). A peek never journals, so
234	/// a cold result is always non-revertible.
235	pub fn peek(&self, entry: &AccessEntry) -> Warmth {
236		if self.accessed.contains(entry) { Warmth::Hot } else { Warmth::Cold { revertible: false } }
237	}
238
239	/// Whether the set is at the entry cap.
240	fn is_full(&self) -> bool {
241		self.accessed.len() >= MAX_ACCESS_LIST_ENTRIES
242	}
243
244	/// Whether a nested-frame checkpoint is open.
245	fn in_nested_frame(&self) -> bool {
246		!self.checkpoints.is_empty()
247	}
248
249	/// Register the entry, returning whether it was cold or hot.
250	///
251	/// Past [`MAX_ACCESS_LIST_ENTRIES`], new entries are billed cold without
252	/// being journaled; previously-hot slots continue to bill hot.
253	pub fn touch(&mut self, entry: AccessEntry) -> Warmth {
254		let kind = if self.is_full() {
255			// Past the cap: bill by membership, but never journal.
256			self.peek(&entry)
257		} else if self
258			.accessed
259			.try_insert(entry.clone())
260			.expect("under cap; checked is_full above; qed")
261		{
262			// Newly inserted: journal it so the owning frame's rollback can drop it.
263			self.journal
264				.try_push(entry)
265				.expect("journal grows in lockstep with accessed and shares its bound; qed");
266			Warmth::Cold { revertible: self.in_nested_frame() }
267		} else {
268			Warmth::Hot
269		};
270
271		match kind {
272			Warmth::Cold { .. } => self.cold_count = self.cold_count.saturating_add(1),
273			Warmth::Hot => self.hot_count = self.hot_count.saturating_add(1),
274		}
275		kind
276	}
277
278	/// Per-transaction metrics snapshot.
279	pub fn metrics(&self) -> AccessListMetrics {
280		AccessListMetrics { size: self.accessed.len(), cold: self.cold_count, hot: self.hot_count }
281	}
282
283	/// Returns the number of open checkpoints.
284	#[cfg(test)]
285	pub fn frame_depth(&self) -> usize {
286		self.checkpoints.len()
287	}
288}

294	#[test]
295	fn nested_commit_then_parent_rollback_drops_all() {
296		let mut al = AccessList::new();
297		let (a, b, c, d) = (
298			AccessEntry { address: H160::zero(), slot: Slot::Fix([0xA; 32]) },
299			AccessEntry { address: H160::zero(), slot: Slot::Fix([0xB; 32]) },
300			AccessEntry { address: H160::zero(), slot: Slot::Fix([0xC; 32]) },
301			AccessEntry { address: H160::zero(), slot: Slot::Fix([0xD; 32]) },
302		);
303
304		// Root frame: cold, but no checkpoint covers it, so it is not revertible.
305		assert_eq!(al.touch(a.clone()), Warmth::Cold { revertible: false }, "A: first touch cold");
306		assert!(!al.touch(a.clone()).is_cold(), "A: second touch hot");
307
308		al.enter_frame();
309		assert_eq!(al.frame_depth(), 1);
310
311		// Inside F1: journaled under the open checkpoint, so it is revertible.
312		assert_eq!(al.touch(b.clone()), Warmth::Cold { revertible: true }, "B in F1: cold");
313		assert!(!al.touch(a.clone()).is_cold(), "A in F1: hot via parent");
314
315		al.enter_frame();
316		assert!(al.touch(c.clone()).is_cold(), "C in F2: cold");
317
318		al.commit_frame();
319		assert_eq!(al.frame_depth(), 1);
320		assert!(!al.peek(&c).is_cold(), "C: survives F2 commit");
321
322		assert!(al.touch(d.clone()).is_cold(), "D in F1: cold");
323		assert_eq!(al.metrics().size, 4);
324
325		al.rollback_frame();
326		assert_eq!(al.frame_depth(), 0);
327		assert!(!al.peek(&a).is_cold(), "A: first frame, survives F1 revert");
328		assert!(al.peek(&b).is_cold(), "B: inserted by F1, rolled back");
329		assert!(al.peek(&c).is_cold(), "C: F2-committed-into-F1, gone when F1 reverts");
330		assert!(al.peek(&d).is_cold(), "D: inserted by F1, rolled back");
331
332		// Counters never decrement, even for entries that later roll back:
333		// A (cold) + B,C,D (cold) -> 4 cold; A,A (hot) -> 2 hot. Only A still hot,
334		// so `size` is 1.
335		assert_eq!(
336			al.metrics(),
337			AccessListMetrics { size: 1, cold: 4, hot: 2 },
338			"counters must include rolled-back touches",
339		);
340	}

342	#[test]
343	fn touch_caps_at_max_entries() {
344		let mut al = AccessList::new();
345		// Fill to the cap with distinct addresses.
346		for i in 0..MAX_ACCESS_LIST_ENTRIES {
347			let address = H160::from_low_u64_be(i as u64);
348			assert!(al.touch(AccessEntry { address, slot: Slot::Fix([0; 32]) }).is_cold());
349		}
350		assert_eq!(al.metrics().size, MAX_ACCESS_LIST_ENTRIES);
351
352		let new_entry = AccessEntry {
353			address: H160::from_low_u64_be(MAX_ACCESS_LIST_ENTRIES as u64),
354			slot: Slot::Fix([0; 32]),
355		};
356		// Past the cap a new entry is billed cold but never journaled, so even
357		// inside an open frame it can't be rolled back.
358		al.enter_frame();
359		assert_eq!(
360			al.touch(new_entry.clone()),
361			Warmth::Cold { revertible: false },
362			"past cap: bills cold, not revertible",
363		);
364		al.commit_frame();
365		assert_eq!(al.metrics().size, MAX_ACCESS_LIST_ENTRIES, "set size stays at cap");
366		assert!(al.peek(&new_entry).is_cold(), "past-cap entry is not tracked");
367
368		assert!(al.touch(new_entry).is_cold(), "past cap re-touch: still cold (not tracked)");
369
370		let existing = AccessEntry { address: H160::zero(), slot: Slot::Fix([0; 32]) };
371		assert!(!al.touch(existing).is_cold(), "existing entry still hot at cap");
372	}

374	#[test]
375	fn peek_does_not_mutate() {
376		let mut al = AccessList::new();
377		let entry = AccessEntry { address: H160::zero(), slot: Slot::Fix([1; 32]) };
378
379		assert!(al.peek(&entry).is_cold(), "untouched entry: cold");
380		assert!(al.peek(&entry).is_cold(), "repeated query: still cold");
381		assert_eq!(
382			al.metrics(),
383			AccessListMetrics { size: 0, cold: 0, hot: 0 },
384			"peek must not bump counters",
385		);
386
387		al.touch(entry.clone());
388
389		assert!(!al.peek(&entry).is_cold(), "after touch: hot");
390		assert!(!al.peek(&entry).is_cold(), "repeated query: still hot");
391		assert_eq!(
392			al.metrics(),
393			AccessListMetrics { size: 1, cold: 1, hot: 0 },
394			"peek must not bump the hot counter",
395		);
396	}
