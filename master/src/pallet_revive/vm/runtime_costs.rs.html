31/// Current approximation of the gas/s consumption considering
32/// EVM execution over compiled WASM (on 4.4Ghz CPU).
33/// Given the 2000ms Weight, from which 75% only are used for transactions,
34/// the total EVM execution gas limit is: GAS_PER_SECOND * 2 * 0.75 ~= 60_000_000.
35const GAS_PER_SECOND: u64 = 40_000_000;
36
37/// Approximate ratio of the amount of Weight per Gas.
38/// u64 works for approximations because Weight is a very small unit compared to
39/// gas.
40const WEIGHT_PER_GAS: u64 = WEIGHT_REF_TIME_PER_SECOND / GAS_PER_SECOND;
41
42/// Smallest init code length the `evm_instantiate*` benchmarks run at.
43pub const BENCH_MIN_EVM_INIT_CODE_LEN: u32 = 1;

199/// How a storage access is priced.
200#[derive(Clone, Copy, Debug, Eq, PartialEq)]
201pub enum StorageAccessKind {
202	/// Persistent storage, priced by its access-list warmth.
203	Persistent(Warmth),
204	/// Transient storage, every access costs the same.
205	Transient,
206}
207
208impl StorageAccessKind {
209	/// Builds the storage access kind. `warmth` is called only for persistent storage.
210	pub fn new(transient: bool, warmth: impl FnOnce() -> Warmth) -> Self {
211		if transient { Self::Transient } else { Self::Persistent(warmth()) }
212	}
213}

215/// For functions that modify storage, benchmarks are performed with one item in the
216/// storage. To account for the worst-case scenario, the weight of the overhead of
217/// writing to or reading from full storage is included. For transient storage writes,
218/// the rollback weight is added to reflect the worst-case scenario for this operation.
219macro_rules! cost_storage {
220    (write_transient, $name:ident $(, $arg:expr )*) => {
221        T::WeightInfo::$name($( $arg ),*)
222            .saturating_add(T::WeightInfo::rollback_transient_storage())
223            .saturating_add(T::WeightInfo::set_transient_storage_full()
224            .saturating_sub(T::WeightInfo::set_transient_storage_empty()))
225    };
226
227    (read_transient, $name:ident $(, $arg:expr )*) => {
228        T::WeightInfo::$name($( $arg ),*)
229            .saturating_add(T::WeightInfo::get_transient_storage_full()
230            .saturating_sub(T::WeightInfo::get_transient_storage_empty()))
231    };
232
233    (write_cold, $name:ident $(, $arg:expr )*) => {
234        T::WeightInfo::$name($( $arg ),*)
235            .saturating_add(T::WeightInfo::set_storage_full()
236            .saturating_sub(T::WeightInfo::set_storage_empty()))
237    };
238
239    (read_cold, $name:ident $(, $arg:expr )*) => {
240        T::WeightInfo::$name($( $arg ),*)
241            .saturating_add(T::WeightInfo::get_storage_full()
242            .saturating_sub(T::WeightInfo::get_storage_empty()))
243    };
244}
245
246macro_rules! cost_args {
247	// cost_args!(name, a, b, c) -> T::WeightInfo::name(a, b, c).saturating_sub(T::WeightInfo::name(0, 0, 0))
248	($name:ident, $( $arg: expr ),+) => {
249		(T::WeightInfo::$name($( $arg ),+).saturating_sub(cost_args!(@call_zero $name, $( $arg ),+)))
250	};
251	// Transform T::WeightInfo::name(a, b, c) into T::WeightInfo::name(0, 0, 0)
252	(@call_zero $name:ident, $( $arg:expr ),*) => {
253		T::WeightInfo::$name($( cost_args!(@replace_token $arg) ),*)
254	};
255	// Replace the token with 0.
256	(@replace_token $_in:tt) => { 0 };
257}

259impl RuntimeCosts {
260	/// Extra ref_time a hot storage access pays to look up the block's overlay.
261	fn hot_storage_overlay_overhead<T: Config>() -> Weight {
262		let per_read = |weight_fn: fn(u32) -> Weight| weight_fn(1).saturating_sub(weight_fn(0));
263		per_read(T::WeightInfo::overlay_probe_full)
264			.saturating_sub(per_read(T::WeightInfo::overlay_probe_empty))
265	}
266
267	/// What a hot write pays on top of the cold read that warmed the slot:
268	/// re-hashing the slot's trie path when the block's storage root is
269	/// computed.
270	fn hot_write_surcharge<T: Config>() -> Weight {
271		let db = T::DbWeight::get();
272		db.writes(1).saturating_sub(db.reads(1))
273	}
274
275	/// Pick the matching storage bench for the access `kind`.
276	fn weight_for_storage_access<T: Config>(
277		op: StorageOp,
278		kind: StorageAccessKind,
279		cold: impl FnOnce() -> Weight,
280		hot: impl FnOnce() -> Weight,
281		transient: impl FnOnce() -> Weight,
282	) -> Weight {
283		match kind {
284			StorageAccessKind::Persistent(Warmth::Cold { revertible }) => {
285				let cost = cold()
286					.saturating_add(T::WeightInfo::access_list_touch_cold_full())
287					.saturating_sub(T::WeightInfo::access_list_touch_cold_empty());
288				if revertible {
289					cost.saturating_add(T::WeightInfo::access_list_rollback_amortization())
290				} else {
291					cost
292				}
293			},
294			StorageAccessKind::Persistent(Warmth::Hot { charged }) => hot()
295				.saturating_add(if charged.covers(op) {
296					Weight::zero()
297				} else {
298					Self::hot_write_surcharge::<T>()
299				})
300				.saturating_add(Self::hot_storage_overlay_overhead::<T>())
301				.saturating_add(T::WeightInfo::access_list_touch_hot_full())
302				.saturating_sub(T::WeightInfo::access_list_touch_hot_single_element()),
303			StorageAccessKind::Transient => transient(),
304		}
305	}
306}

308impl<T: Config> Token<T> for RuntimeCosts {
309	fn influence_lowest_weight_limit(&self) -> bool {
310		true
311	}
312
313	fn weight(&self) -> Weight {
314		use self::RuntimeCosts::*;
315		match *self {
316			HostFn => cost_args!(noop_host_fn, 1),
317			EvmMemoryGrow { words, allocation_words } => {
318				let grow = cost_args!(evm_memory_grow, words);
319				let allocate = if allocation_words == 0 {
320					Weight::zero()
321				} else {
322					T::WeightInfo::evm_memory_reallocate(allocation_words)
323				};
324				grow.saturating_add(allocate)
325			},
326			EvmMemoryCopyOut(len) => {
327				if len == 0 {
328					Weight::zero()
329				} else {
330					T::WeightInfo::evm_memory_copy_out(len)
331				}
332			},
333			// `extcodecopy` charges `CodeSize` separately; subtract it so its read isn't counted
334			// twice.
335			ExtCodeCopy(len) => {
336				T::WeightInfo::extcodecopy(len).saturating_sub(T::WeightInfo::seal_code_size())
337			},
338			CopyToContract(len) => T::WeightInfo::seal_copy_to_contract(len),
339			EvmCopyToContract(len) => cost_args!(seal_copy_to_contract, len),
340			CopyFromContract(len) => T::WeightInfo::seal_return(len),
341			CallDataSize => T::WeightInfo::seal_call_data_size(),
342			ReturnDataSize => T::WeightInfo::seal_return_data_size(),
343			CallDataLoad => T::WeightInfo::seal_call_data_load(),
344			CallDataCopy(len) => T::WeightInfo::seal_call_data_copy(len),
345			Caller => T::WeightInfo::seal_caller(),
346			Origin => T::WeightInfo::seal_origin(),
347			ToAccountId => T::WeightInfo::to_account_id(),
348			CodeHash => T::WeightInfo::seal_code_hash(),
349			CodeSize => T::WeightInfo::seal_code_size(),
350			OwnCodeHash => T::WeightInfo::own_code_hash(),
351			CallerIsOrigin => T::WeightInfo::caller_is_origin(),
352			CallerIsRoot => T::WeightInfo::caller_is_root(),
353			OriginIsRoot => T::WeightInfo::origin_is_root(),
354			Address => T::WeightInfo::seal_address(),
355			RefTimeLeft => T::WeightInfo::seal_ref_time_left(),
356			WeightLeft => T::WeightInfo::weight_left(),
357			Balance => T::WeightInfo::seal_balance(),
358			BalanceOf => T::WeightInfo::seal_balance_of(),
359			ValueTransferred => T::WeightInfo::seal_value_transferred(),
360			MinimumBalance => T::WeightInfo::minimum_balance(),
361			BlockNumber => T::WeightInfo::seal_block_number(),
362			BlockHash => T::WeightInfo::seal_block_hash(),
363			BlockAuthor => T::WeightInfo::seal_block_author(),
364			GasPrice => T::WeightInfo::seal_gas_price(),
365			BaseFee => T::WeightInfo::seal_base_fee(),
366			Now => T::WeightInfo::seal_now(),
367			GasLimit => T::WeightInfo::seal_gas_limit(),
368			Terminate { code_removed } => {
369				// logic only runs if code is removed
370				if code_removed {
371					T::WeightInfo::seal_terminate(code_removed.into())
372						.saturating_add(T::WeightInfo::seal_terminate_logic())
373				} else {
374					T::WeightInfo::seal_terminate(code_removed.into())
375				}
376			},
377			DepositEvent { num_topic, len } => T::WeightInfo::seal_deposit_event(num_topic, len)
378				.saturating_add(T::WeightInfo::on_finalize_block_per_event(len))
379				.saturating_add(Weight::from_parts(
380					limits::EXTRA_EVENT_CHARGE_PER_BYTE.saturating_mul(len.into()).into(),
381					0,
382				)),
383			SetStorage { new_bytes, old_bytes, kind } => Self::weight_for_storage_access::<T>(
384				StorageOp::Write,
385				kind,
386				|| cost_storage!(write_cold, seal_set_storage, new_bytes, old_bytes),
387				|| T::WeightInfo::seal_set_storage_hot(new_bytes, old_bytes),
388				|| cost_storage!(write_transient, seal_set_transient_storage, new_bytes, old_bytes),
389			),
390			ClearStorage { len, kind } => Self::weight_for_storage_access::<T>(
391				StorageOp::Write,
392				kind,
393				|| cost_storage!(write_cold, clear_storage, len),
394				|| T::WeightInfo::clear_storage_hot(len),
395				|| cost_storage!(write_transient, seal_clear_transient_storage, len),
396			),
397			ContainsStorage { len, kind } => Self::weight_for_storage_access::<T>(
398				StorageOp::Read,
399				kind,
400				|| cost_storage!(read_cold, contains_storage, len),
401				|| T::WeightInfo::contains_storage_hot(len),
402				|| cost_storage!(read_transient, seal_contains_transient_storage, len),
403			),
404			GetStorage { len, kind } => Self::weight_for_storage_access::<T>(
405				StorageOp::Read,
406				kind,
407				|| cost_storage!(read_cold, seal_get_storage, len),
408				|| T::WeightInfo::seal_get_storage_hot(len),
409				|| cost_storage!(read_transient, seal_get_transient_storage, len),
410			),
411			TakeStorage { len, kind } => Self::weight_for_storage_access::<T>(
412				StorageOp::Write,
413				kind,
414				|| cost_storage!(write_cold, take_storage, len),
415				|| T::WeightInfo::take_storage_hot(len),
416				|| cost_storage!(write_transient, seal_take_transient_storage, len),
417			),
418			CallBase => T::WeightInfo::seal_call(0),
419			DelegateCallBase => T::WeightInfo::seal_delegate_call(),
420			PrecompileBase => T::WeightInfo::seal_call_precompile(0, 0),
421			PrecompileWithInfoBase => T::WeightInfo::seal_call_precompile(1, 0),
422			PrecompileDecode(len) => cost_args!(seal_call_precompile, 0, len),
423			CallTransferSurcharge { dust_transfer } => {
424				// `Weight::max` is per dimension, so ref time and proof size can each be taken from
425				// either of the two benches.
426				T::WeightInfo::seal_call_transfer(dust_transfer.into())
427					.max(T::WeightInfo::seal_call_transfer_to_existing(dust_transfer.into()))
428					.saturating_sub(T::WeightInfo::seal_call(0))
429			},
430			CallInputCloned(len) => cost_args!(seal_call, len),
431			Instantiate { input_data_len, balance_transfer, dust_transfer } => {
432				let transfer = if balance_transfer || dust_transfer {
433					T::WeightInfo::seal_instantiate_transfer(dust_transfer.into())
434						.saturating_sub(T::WeightInfo::seal_instantiate(0))
435				} else {
436					Weight::zero()
437				};
438				T::WeightInfo::seal_instantiate(input_data_len).saturating_add(transfer)
439			},
440			Create { init_code_len, balance_transfer, dust_transfer } => {
441				let transfer = if balance_transfer || dust_transfer {
442					T::WeightInfo::evm_instantiate_transfer(dust_transfer.into())
443						.saturating_sub(T::WeightInfo::evm_instantiate(BENCH_MIN_EVM_INIT_CODE_LEN))
444				} else {
445					Weight::zero()
446				};
447				T::WeightInfo::evm_instantiate(init_code_len).saturating_add(transfer)
448			},
449			HashSha256(len) => T::WeightInfo::sha2_256(len),
450			Ripemd160(len) => T::WeightInfo::ripemd_160(len),
451			HashKeccak256(len) => T::WeightInfo::seal_hash_keccak_256(len),
452			HashBlake256(len) => T::WeightInfo::hash_blake2_256(len),
453			HashBlake128(len) => T::WeightInfo::hash_blake2_128(len),
454			EcdsaRecovery => T::WeightInfo::ecdsa_recover(),
455			P256Verify => T::WeightInfo::p256_verify(),
456			Sr25519Verify(len) => T::WeightInfo::seal_sr25519_verify(len),
457			Precompile(weight) => weight,
458			EcdsaToEthAddress => T::WeightInfo::seal_ecdsa_to_eth_address(),
459			GetImmutableData(len) => T::WeightInfo::seal_get_immutable_data(len),
460			SetImmutableData(len) => T::WeightInfo::seal_set_immutable_data(len),
461			Bn128Add => T::WeightInfo::bn128_add(),
462			Bn128Mul => T::WeightInfo::bn128_mul(),
463			Bn128Pairing(len) => T::WeightInfo::bn128_pairing(len),
464			Identity(len) => T::WeightInfo::identity(len),
465			Blake2F(rounds) => T::WeightInfo::blake2f(rounds),
466			Modexp(gas) => Weight::from_parts(gas.saturating_mul(WEIGHT_PER_GAS), 0),
467			Delegations { new_accounts, existing_accounts, invalid_accounts } => {
468				T::WeightInfo::process_new_account_authorization(new_accounts)
469					.saturating_add(T::WeightInfo::process_existing_account_authorization(
470						existing_accounts,
471					))
472					.saturating_add(T::WeightInfo::process_invalid_authorization(invalid_accounts))
473			},
474		}
475	}
476}

626
627		let call = |dust_transfer| weight(RuntimeCosts::CallTransferSurcharge { dust_transfer });

650			);
651			assert_eq!(

653				surcharge(true, true),
