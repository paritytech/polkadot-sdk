48/// Hard-coded value returned by the EVM `DIFFICULTY` opcode.
49///
50/// After Ethereum's Merge (Sept 2022), the `DIFFICULTY` opcode was redefined to return
51/// `prevrandao`, a randomness value from the beacon chain. In Substrate pallet-revive
52/// a fixed constant is returned instead for compatibility with contracts that still read this
53/// opcode. The value is aligned with the difficulty hardcoded for PVM contracts.
54pub(crate) const DIFFICULTY: u64 = 2500000000000000_u64;
55
56/// Cost  for a single unit of EVM gas.
57#[derive(Eq, PartialEq, Debug, Clone, Copy)]
58pub struct EVMGas(pub u64);
59
60impl<T: Config> Token<T> for EVMGas {
61	fn weight(&self) -> Weight {
62		let base_cost = T::WeightInfo::evm_jumpdest_opcode(1)
63			.saturating_sub(T::WeightInfo::evm_jumpdest_opcode(0));
64		base_cost.saturating_mul(self.0)
65	}
66}

134/// An op-code in one block of the `evm_dispatch_mix` benchmark.
135pub struct DispatchMixOpcode {
136	/// The op-code's byte.
137	pub opcode: u8,
138	/// What the op-code is charged.
139	cost: EvmOpcodeCosts,
140	/// How many times the op-code occurs in a block.
141	pub repetitions: u64,
142}

144impl DispatchMixOpcode {
145	/// Creates an entry for the dispatch mix.
146	const fn new(opcode: u8, cost: EvmOpcodeCosts, repetitions: u64) -> Self {
147		Self { opcode, cost, repetitions }
148	}

151impl EvmOpcodeCosts {
152	/// The cost measured by the op-code's benchmark, where the interpreter always dispatches to
153	/// the same handler.
154	fn base_weight<T: Config>(&self) -> Weight {
155		let per_opcode = |weight_fn: fn(u32) -> Weight| weight_fn(1).saturating_sub(weight_fn(0));
156		match self {
157			Self::Jump => per_opcode(T::WeightInfo::evm_jump_opcode)
158				.saturating_sub(Self::JumpDest.base_weight::<T>()),
159			Self::Jumpi => per_opcode(T::WeightInfo::evm_jumpi_opcode)
160				.saturating_sub(Self::JumpDest.base_weight::<T>()),
161			Self::JumpDest => per_opcode(T::WeightInfo::evm_jumpdest_opcode),
162			Self::Pc => per_opcode(T::WeightInfo::evm_pc_opcode),
163			Self::Push => per_opcode(T::WeightInfo::evm_push_opcode),
164			Self::Pop => per_opcode(T::WeightInfo::evm_pop_opcode),
165			Self::Dup => per_opcode(T::WeightInfo::evm_dup_opcode),
166			Self::Swap => per_opcode(T::WeightInfo::evm_swap_opcode),
167			Self::ChainId => per_opcode(T::WeightInfo::evm_chainid_opcode),
168			Self::Difficulty => per_opcode(T::WeightInfo::evm_difficulty_opcode),
169			Self::CodeSize => per_opcode(T::WeightInfo::evm_codesize_opcode),
170			Self::CallDataSize => per_opcode(T::WeightInfo::evm_calldatasize_opcode),
171			Self::ReturnDataSize => per_opcode(T::WeightInfo::evm_returndatasize_opcode),
172			Self::CallDataLoad => per_opcode(T::WeightInfo::evm_calldataload_opcode),
173			Self::CallDataCopy => per_opcode(T::WeightInfo::evm_calldatacopy_opcode),
174			Self::CodeCopy => per_opcode(T::WeightInfo::evm_codecopy_opcode),
175			Self::ReturnDataCopy => per_opcode(T::WeightInfo::evm_returndatacopy_opcode),
176			Self::MLoad => per_opcode(T::WeightInfo::evm_mload_opcode),
177			Self::MSize => per_opcode(T::WeightInfo::evm_msize_opcode),
178			Self::MStore => per_opcode(T::WeightInfo::evm_mstore_opcode),
179			Self::MStore8 => per_opcode(T::WeightInfo::evm_mstore8_opcode),
180			Self::MCopy { len } => {
181				// The fixed cost covers the first 64 bytes, so shorter copies pay nothing more.
182				let extra_bytes = len.saturating_sub(64);
183				per_opcode(T::WeightInfo::evm_mcopy_opcode).saturating_add(
184					T::WeightInfo::evm_mcopy_per_byte(extra_bytes)
185						.saturating_sub(T::WeightInfo::evm_mcopy_per_byte(0)),
186				)
187			},
188			Self::Lt => per_opcode(T::WeightInfo::evm_lt_opcode),
189			Self::Gt => per_opcode(T::WeightInfo::evm_gt_opcode),
190			Self::Eq => per_opcode(T::WeightInfo::evm_eq_opcode),
191			Self::IsZero => per_opcode(T::WeightInfo::evm_iszero_opcode)
192				.saturating_sub(Self::Pop.base_weight::<T>()),
193			Self::And => per_opcode(T::WeightInfo::evm_and_opcode),
194			Self::Or => per_opcode(T::WeightInfo::evm_or_opcode),
195			Self::Xor => per_opcode(T::WeightInfo::evm_xor_opcode),
196			Self::Not => per_opcode(T::WeightInfo::evm_not_opcode),
197			Self::Byte => per_opcode(T::WeightInfo::evm_byte_opcode)
198				.saturating_sub(Self::Pop.base_weight::<T>()),
199			Self::Clz => per_opcode(T::WeightInfo::evm_clz_opcode)
200				.saturating_sub(Self::Pop.base_weight::<T>()),
201			Self::Slt => per_opcode(T::WeightInfo::evm_slt_opcode)
202				.saturating_sub(Self::Pop.base_weight::<T>()),
203			Self::Sgt => per_opcode(T::WeightInfo::evm_sgt_opcode)
204				.saturating_sub(Self::Pop.base_weight::<T>()),
205			Self::Shl => per_opcode(T::WeightInfo::evm_shl_opcode)
206				.saturating_sub(Self::Pop.base_weight::<T>()),
207			Self::Shr => per_opcode(T::WeightInfo::evm_shr_opcode)
208				.saturating_sub(Self::Pop.base_weight::<T>()),
209			Self::Sar => per_opcode(T::WeightInfo::evm_sar_opcode)
210				.saturating_sub(Self::Pop.base_weight::<T>()),
211			Self::AddMod => per_opcode(T::WeightInfo::evm_addmod_opcode)
212				.max(per_opcode(T::WeightInfo::evm_addmod_opcode_one_limb_variant))
213				.max(per_opcode(T::WeightInfo::evm_addmod_opcode_three_limb_variant))
214				.max(per_opcode(T::WeightInfo::evm_addmod_opcode_mixed_variant))
215				.max(per_opcode(T::WeightInfo::evm_addmod_opcode_fixed_knuth_variant))
216				.saturating_sub(Self::Pop.base_weight::<T>()),
217			Self::Div => per_opcode(T::WeightInfo::evm_div_opcode)
218				.max(per_opcode(T::WeightInfo::evm_div_opcode_one_limb_variant))
219				.max(per_opcode(T::WeightInfo::evm_div_opcode_three_limb_variant))
220				.max(per_opcode(T::WeightInfo::evm_div_opcode_mixed_variant))
221				.max(per_opcode(T::WeightInfo::evm_div_opcode_fixed_knuth_variant))
222				.saturating_sub(Self::Pop.base_weight::<T>()),
223			Self::Add => per_opcode(T::WeightInfo::evm_add_opcode),
224			Self::Sub => per_opcode(T::WeightInfo::evm_sub_opcode),
225			Self::Mul => per_opcode(T::WeightInfo::evm_mul_opcode),
226			Self::SDiv => per_opcode(T::WeightInfo::evm_sdiv_opcode)
227				.max(per_opcode(T::WeightInfo::evm_sdiv_opcode_one_limb_variant))
228				.max(per_opcode(T::WeightInfo::evm_sdiv_opcode_three_limb_variant))
229				.max(per_opcode(T::WeightInfo::evm_sdiv_opcode_mixed_variant))
230				.max(per_opcode(T::WeightInfo::evm_sdiv_opcode_fixed_knuth_variant))
231				.saturating_sub(Self::Pop.base_weight::<T>()),
232			Self::Mod => per_opcode(T::WeightInfo::evm_mod_opcode)
233				.max(per_opcode(T::WeightInfo::evm_mod_opcode_one_limb_variant))
234				.max(per_opcode(T::WeightInfo::evm_mod_opcode_three_limb_variant))
235				.max(per_opcode(T::WeightInfo::evm_mod_opcode_mixed_variant))
236				.max(per_opcode(T::WeightInfo::evm_mod_opcode_fixed_knuth_variant))
237				.saturating_sub(Self::Pop.base_weight::<T>()),
238			Self::SMod => per_opcode(T::WeightInfo::evm_smod_opcode)
239				.max(per_opcode(T::WeightInfo::evm_smod_opcode_one_limb_variant))
240				.max(per_opcode(T::WeightInfo::evm_smod_opcode_three_limb_variant))
241				.max(per_opcode(T::WeightInfo::evm_smod_opcode_mixed_variant))
242				.max(per_opcode(T::WeightInfo::evm_smod_opcode_fixed_knuth_variant))
243				.saturating_sub(Self::Pop.base_weight::<T>()),
244			Self::MulMod => per_opcode(T::WeightInfo::evm_mulmod_opcode)
245				.saturating_sub(Self::Pop.base_weight::<T>()),
246			Self::Exp { exponent_bits: 0 } => per_opcode(T::WeightInfo::evm_exp_zero_opcode),
247			Self::Exp { exponent_bits } => per_opcode(T::WeightInfo::evm_exp_opcode)
248				.saturating_add(
249					T::WeightInfo::evm_exp_per_bit(exponent_bits.saturating_sub(1))
250						.saturating_sub(T::WeightInfo::evm_exp_per_bit(0)),
251				),
252			Self::SignExtend => per_opcode(T::WeightInfo::evm_signextend_opcode)
253				.saturating_sub(Self::Pop.base_weight::<T>()),
254		}
255	}

257	/// The op-codes in one block of the `evm_dispatch_mix` benchmark, which builds its code from
258	/// this table.
259	///
260	/// These are cheap op-codes whose work doesn't depend on their operands, so in the mix each
261	/// one does exactly the work it's charged for. The repetitions make every block leave the
262	/// stack as high as it found it.
263	pub const DISPATCH_MIX: [DispatchMixOpcode; 14] = [
264		DispatchMixOpcode::new(opcode::PC, Self::Pc, 5),
265		DispatchMixOpcode::new(opcode::CHAINID, Self::ChainId, 5),
266		DispatchMixOpcode::new(opcode::DIFFICULTY, Self::Difficulty, 5),
267		DispatchMixOpcode::new(opcode::CODESIZE, Self::CodeSize, 5),
268		DispatchMixOpcode::new(opcode::CALLDATASIZE, Self::CallDataSize, 5),
269		DispatchMixOpcode::new(opcode::RETURNDATASIZE, Self::ReturnDataSize, 5),
270		DispatchMixOpcode::new(opcode::MSIZE, Self::MSize, 5),
271		DispatchMixOpcode::new(opcode::JUMPDEST, Self::JumpDest, 5),
272		DispatchMixOpcode::new(opcode::NOT, Self::Not, 5),
273		DispatchMixOpcode::new(opcode::POP, Self::Pop, 7),
274		DispatchMixOpcode::new(opcode::AND, Self::And, 7),
275		DispatchMixOpcode::new(opcode::OR, Self::Or, 7),
276		DispatchMixOpcode::new(opcode::XOR, Self::Xor, 7),
277		DispatchMixOpcode::new(opcode::MUL, Self::Mul, 7),
278	];

287/// What a mispredicted dispatch to an op-code's handler costs.
288///
289/// The benchmark of an op-code runs only that op-code, so the interpreter always dispatches to
290/// the same handler. Contracts mix op-codes, which makes many of these dispatches mispredict.
291/// `evm_dispatch_mix` runs [`EvmOpcodeCosts::DISPATCH_MIX`] in a pseudo-random order, and this
292/// charges the part of it that the charges of its op-codes don't cover, per op-code.
293#[derive(Eq, PartialEq, Debug, Clone, Copy)]
294pub struct EvmDispatch;
295
296impl EvmDispatch {
297	/// Checks that every op-code in the mix pays dispatch and the weights leave a charge above
298	/// zero. Otherwise `evm_dispatch_mix` measured no more than the charges of its op-codes, and
299	/// the dispatch charge would silently be zero.
300	pub fn integrity_test<T: Config>() {
301		for entry in EvmOpcodeCosts::DISPATCH_MIX {
302			assert!(
303				instructions::pays_dispatch(entry.opcode),
304				"op-code {:#04x} in `evm_dispatch_mix` must pay the dispatch charge",
305				entry.opcode,
306			);
307		}
308		assert!(
309			Token::<T>::weight(&Self).ref_time() > 0,
310			"`evm_dispatch_mix` must measure more than the charges of its op-codes",
311		);
312	}
313}
314
315impl<T: Config> Token<T> for EvmDispatch {
316	fn weight(&self) -> Weight {
317		let (mix_opcodes, mix_charges) = EvmOpcodeCosts::DISPATCH_MIX.iter().fold(
318			(0, Weight::zero()),
319			|(opcodes, charges), DispatchMixOpcode { cost, repetitions, .. }| {
320				(
321					opcodes + repetitions,
322					charges.saturating_add(cost.base_weight::<T>().saturating_mul(*repetitions)),
323				)
324			},
325		);
326		T::WeightInfo::evm_dispatch_mix(1)
327			.saturating_sub(T::WeightInfo::evm_dispatch_mix(0))
328			.saturating_sub(mix_charges)
329			.saturating_div(mix_opcodes)
330	}
331}

333impl<T: Config> ContractBlob<T> {
334	/// Create a new contract from EVM init code.
335	pub fn from_evm_init_code(code: Vec<u8>, owner: AccountIdOf<T>) -> Result<Self, DispatchError> {
336		if code.len() > revm::primitives::eip3860::MAX_INITCODE_SIZE &&
337			!DebugSettings::is_unlimited_contract_size_allowed::<T>()
338		{
339			return Err(<Error<T>>::BlobTooLarge.into());
340		}
341
342		let code_len = code.len() as u32;
343		let code_info = CodeInfo {
344			owner,
345			deposit: Default::default(),
346			refcount: 0,
347			code_len,
348			code_type: BytecodeType::Evm,
349			behaviour_version: Default::default(),
350		};
351
352		Bytecode::new_raw_checked(Bytes::from(code.to_vec())).map_err(|err| {
353			log::debug!(target: LOG_TARGET, "failed to create evm bytecode from init code: {err:?}" );
354			<Error<T>>::CodeRejected
355		})?;
356
357		// Code hash is not relevant for init code, since it is not stored on-chain.
358		let code_hash = H256::default();
359		Ok(ContractBlob { code, code_info, code_hash })
360	}
361
362	/// Create a new contract from EVM runtime code.
363	pub fn from_evm_runtime_code(
364		code: Vec<u8>,
365		owner: AccountIdOf<T>,
366	) -> Result<Self, DispatchError> {
367		let code_len = code.len() as u32;
368		let deposit = super::calculate_code_deposit::<T>(code_len);
369		Self::from_evm_runtime_code_with_deposit(code, owner, deposit)
370	}
371
372	/// Create a new contract from EVM runtime code with an explicit owner and
373	/// deposit amount.
374	///
375	/// Used for `Origin::Root` uploads: there is no origin account to attribute
376	/// the deposit to, so the caller passes the pallet's own account as a
377	/// sentinel owner (no user can sign as it, so the code can't be removed via
378	/// the owner-gated path) and a zero deposit (both `charge_deposit` and
379	/// `refund_deposit` short-circuit at amount 0).
380	pub fn from_evm_runtime_code_with_deposit(
381		code: Vec<u8>,
382		owner: AccountIdOf<T>,
383		deposit: BalanceOf<T>,
384	) -> Result<Self, DispatchError> {
385		if code.len() > revm::primitives::eip170::MAX_CODE_SIZE &&
386			!DebugSettings::is_unlimited_contract_size_allowed::<T>()
387		{
388			return Err(<Error<T>>::BlobTooLarge.into());
389		}
390
391		// EIP-3541: reject new contract code (runtime code) starting with the 0xEF byte.
392		// Reserved for EIP-7702 delegation indicators; clashing here would let a
393		// constructor return bytes that subsequent calls would misinterpret as a
394		// delegation pointer.
395		if code.first() == Some(&0xEF) {
396			return Err(<Error<T>>::CodeRejected.into());
397		}
398
399		let code_len = code.len() as u32;
400
401		let code_info = CodeInfo {
402			owner,
403			deposit,
404			refcount: 0,
405			code_len,
406			code_type: BytecodeType::Evm,
407			behaviour_version: Default::default(),
408		};
409
410		Bytecode::new_raw_checked(Bytes::from(code.to_vec())).map_err(|err| {
411			log::debug!(target: LOG_TARGET, "failed to create evm bytecode from code: {err:?}" );
412			<Error<T>>::CodeRejected
413		})?;
414
415		let code_hash = H256(sp_io::hashing::keccak_256(&code));
416		Ok(ContractBlob { code, code_info, code_hash })
417	}
418}

420/// Calls the EVM interpreter with the provided bytecode and inputs.
421pub fn call<E: Ext>(bytecode: Bytecode, ext: &mut E, input: Vec<u8>) -> ExecResult {
422	let mut interpreter = Interpreter::new(ExtBytecode::new(bytecode), input, ext);
423	let tracing_enabled = tracing::if_tracing(|t| t.is_execution_tracer()).unwrap_or(false);
424
425	let ControlFlow::Break(halt) = if tracing_enabled {
426		run_plain_with_tracing(&mut interpreter)
427	} else {
428		run_plain(&mut interpreter)
429	};
430	halt.into()
431}
432
433pub fn run_plain<E: Ext>(interpreter: &mut Interpreter<E>) -> ControlFlow<Halt, Infallible> {
434	loop {
435		let opcode = interpreter.bytecode.opcode();
436		interpreter.bytecode.relative_jump(1);
437		exec_instruction(interpreter, opcode)?;
438	}
439}
440
441fn run_plain_with_tracing<E: Ext>(
442	interpreter: &mut Interpreter<E>,
443) -> ControlFlow<Halt, Infallible> {
444	loop {
445		let opcode = interpreter.bytecode.opcode();
446		tracing::if_tracing(|tracer| {
447			let pc = interpreter.bytecode.pc() as u64;
448			tracer.enter_opcode(pc, opcode, interpreter)
449		});
450
451		interpreter.bytecode.relative_jump(1);
452		let res = exec_instruction(interpreter, opcode);
453
454		tracing::if_tracing(|tracer| tracer.exit_step(interpreter, None));
455
456		res?;
457	}
