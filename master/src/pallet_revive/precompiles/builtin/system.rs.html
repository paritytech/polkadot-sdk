31pub struct System<T>(PhantomData<T>);
32
33impl<T: Config> BuiltinPrecompile for System<T> {
34	type T = T;
35	type Interface = ISystem::ISystemCalls;
36	const MATCHER: BuiltinAddressMatcher =
37		BuiltinAddressMatcher::Fixed(NonZero::new(0x900).unwrap());
38	const HAS_CONTRACT_INFO: bool = false;
39
40	fn call(
41		_address: &[u8; 20],
42		input: &Self::Interface,
43		env: &mut impl Ext<T = Self::T>,
44	) -> Result<Vec<u8>, Error> {
45		use ISystem::ISystemCalls;
46		match input {
47			ISystemCalls::terminate(_) if env.is_read_only() => {
48				Err(crate::Error::<T>::StateChangeDenied.into())
49			},
50			ISystemCalls::hashBlake256(ISystem::hashBlake256Call { input }) => {
51				env.frame_meter_mut()
52					.charge_weight_token(RuntimeCosts::HashBlake256(input.len() as u32))?;
53				let output = sp_io::hashing::blake2_256(input.as_bytes_ref());
54				Ok(output.abi_encode())
55			},
56			ISystemCalls::hashBlake128(ISystem::hashBlake128Call { input }) => {
57				env.frame_meter_mut()
58					.charge_weight_token(RuntimeCosts::HashBlake128(input.len() as u32))?;
59				let output = sp_io::hashing::blake2_128(input.as_bytes_ref());
60				Ok(output.abi_encode())
61			},
62			ISystemCalls::toAccountId(ISystem::toAccountIdCall { input }) => {
63				env.frame_meter_mut().charge_weight_token(RuntimeCosts::ToAccountId)?;
64				let account_id = env.to_account_id(&H160::from_slice(input.as_slice()));
65				Ok(account_id.encode().abi_encode())
66			},
67			ISystemCalls::callerIsOrigin(ISystem::callerIsOriginCall {}) => {
68				env.frame_meter_mut().charge_weight_token(RuntimeCosts::CallerIsOrigin)?;
69				let is_origin = env.caller_is_origin(true);
70				Ok(is_origin.abi_encode())
71			},
72			ISystemCalls::callerIsRoot(ISystem::callerIsRootCall {}) => {
73				env.frame_meter_mut().charge_weight_token(RuntimeCosts::CallerIsRoot)?;
74				let is_root = env.caller_is_root(true);
75				Ok(is_root.abi_encode())
76			},
77			ISystemCalls::originIsRoot(ISystem::originIsRootCall {}) => {
78				env.frame_meter_mut().charge_weight_token(RuntimeCosts::OriginIsRoot)?;
79				let is_root = env.origin_is_root();
80				Ok(is_root.abi_encode())
81			},
82			ISystemCalls::ownCodeHash(ISystem::ownCodeHashCall {}) => {
83				env.frame_meter_mut().charge_weight_token(RuntimeCosts::OwnCodeHash)?;
84				let caller = env.caller();
85				let addr = T::AddressMapper::to_address(caller.account_id()?);
86				let output = env.code_hash(&addr.into()).0.abi_encode();
87				Ok(output)
88			},
89			ISystemCalls::minimumBalance(ISystem::minimumBalanceCall {}) => {
90				env.frame_meter_mut().charge_weight_token(RuntimeCosts::MinimumBalance)?;
91				let minimum_balance = env.minimum_balance();
92				Ok(minimum_balance.to_big_endian().abi_encode())
93			},
94			ISystemCalls::weightLeft(ISystem::weightLeftCall {}) => {
95				env.frame_meter_mut().charge_weight_token(RuntimeCosts::WeightLeft)?;
96				let ref_time = env.frame_meter().weight_left().unwrap_or_default().ref_time();
97				let proof_size = env.frame_meter().weight_left().unwrap_or_default().proof_size();
98				let res = (ref_time, proof_size);
99				Ok(res.abi_encode())
100			},
101			ISystemCalls::terminate(ISystem::terminateCall { beneficiary }) => {
102				// no need to adjust gas because this always deletes code
103				env.frame_meter_mut()
104					.charge_weight_token(RuntimeCosts::Terminate { code_removed: true })?;
105				let h160 = H160::from_slice(beneficiary.as_slice());
106				env.terminate_caller(&h160).map_err(Error::try_to_revert::<T>)?;
107				Ok(Vec::new())
108			},
109			ISystemCalls::sr25519Verify(ISystem::sr25519VerifyCall {
110				signature,
111				message,
112				publicKey,
113			}) => {
114				env.frame_meter_mut()
115					.charge_weight_token(RuntimeCosts::Sr25519Verify(message.len() as _))?;
116				let ok = env.sr25519_verify(signature, message, publicKey);
117				Ok(ok.abi_encode())
118			},
119			ISystemCalls::ecdsaToEthAddress(ISystem::ecdsaToEthAddressCall { publicKey }) => {
120				env.frame_meter_mut().charge_weight_token(RuntimeCosts::EcdsaToEthAddress)?;
121				let address =
122					env.ecdsa_to_eth_address(publicKey).map_err(Error::try_to_revert::<T>)?;
123				Ok(address.abi_encode())
124			},
125		}
126	}
127}

160		ExtBuilder::default().build().execute_with(|| {
161			// given
162			let mut call_setup = CallSetup::<Test>::default();
163			let (mut ext, _) = call_setup.ext();
164			let unmapped_address = H160::zero();
165
166			// when
167			let input = ISystem::ISystemCalls::toAccountId(ISystem::toAccountIdCall {
168				input: unmapped_address.0.into(),
169			});
170			let raw_data =
171				<System<Test>>::call(&<System<Test>>::MATCHER.base_address(), &input, &mut ext)

221	#[test]
222	fn sr25519_verify() {
223		use crate::precompiles::alloy::sol_types::sol_data::Bool;
224		ExtBuilder::default().build().execute_with(|| {
225			let _ = <Test as Config>::Currency::set_balance(&ALICE, 100_000_000_000);
226
227			let mut call_setup = CallSetup::<Test>::default();
228			let (mut ext, _) = call_setup.ext();
229
230			let mut call_with = |message: &[u8; 11]| {
231				// Alice's signature for "hello world"
232				#[rustfmt::skip]
233				let signature: [u8; 64] = [
234					184, 49, 74, 238, 78, 165, 102, 252, 22, 92, 156, 176, 124, 118, 168, 116, 247,
235					99, 0, 94, 2, 45, 9, 170, 73, 222, 182, 74, 60, 32, 75, 64, 98, 174, 69, 55, 83,
236					85, 180, 98, 208, 75, 231, 57, 205, 62, 4, 105, 26, 136, 172, 17, 123, 99, 90, 255,
237					228, 54, 115, 63, 30, 207, 205, 131,
238				];
239
240				// Alice's public key
241				#[rustfmt::skip]
242				let public_key: [u8; 32] = [
243					212, 53, 147, 199, 21, 253, 211, 28, 97, 20, 26, 189, 4, 169, 159, 214, 130, 44,
244					133, 88, 133, 76, 205, 227, 154, 86, 132, 231, 165, 109, 162, 125,
245				];
246
247				let weight_before = ext.frame_meter().weight_consumed();
248
249				let input = ISystem::ISystemCalls::sr25519Verify(ISystem::sr25519VerifyCall {
250					signature,
251					message: (*message).into(),
252					publicKey: public_key.into(),
253				});
254				let result =
255					<System<Test>>::call(&<System<Test>>::MATCHER.base_address(), &input, &mut ext)
256						.unwrap();
257
258				let weight_used = ext.frame_meter().weight_consumed() - weight_before;
259				assert!(weight_used.ref_time() > 0, "sr25519_verify should charge weight");
260				assert_eq!(
261					weight_used,
262					Token::<Test>::weight(&RuntimeCosts::Sr25519Verify(message.len() as u32)),
263					"sr25519_verify should charge the expected weight"
264				);
265				result
266			};
267			let result = Bool::abi_decode(&call_with(&b"hello world")).expect("decoding failed");
268			assert!(result);
269			let result = Bool::abi_decode(&call_with(&b"hello worlD")).expect("decoding failed");
270			assert!(!result);
271		});
272	}
