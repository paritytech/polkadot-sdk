18//! Handle encoded RLP ethereum receipts for block hash computation.
19
20use alloc::vec::Vec;
21use alloy_core::rlp;
22use codec::{Decode, Encode, MaxEncodedLen};
23use scale_info::TypeInfo;
24use sp_core::{H160, H256};
25use sp_crypto_hashing::keccak_256;
26
27/// Number of bytes that a bloom stores.
28pub const BLOOM_SIZE_BYTES: usize = 256;
29
30/// Accumulate events (logs) into a stream of RLP encoded bytes.
31/// This is a straight forward implementation that RLP encodes logs as they are added.
32///
33/// The main goal is to generate the RLP-encoded representation of receipts
34/// which is required to compute the receipt root hash, without storing the full receipt
35/// data in memory.
36///
37/// One approach is to store the full receipt in memory, together with the RLP encoding
38/// of the receipt.
39///
40/// However, since we only care about the RLP encoding of the receipt, we can optimize the memory
41/// usage by only storing the RLP encoded value and the logs directly. This effectively saves
42/// the need to store the full receipt (which can grow unboundedly due to the number of logs), and
43/// builds the RLP encoding incrementally as logs are added.
44///
45/// The implementation leverages the RLP encoding details of the receipt:
46///
47/// ```ignore
48/// // Memory representation of the RLP encoded receipt:
49/// [
50/// 	ReceiptHeader ++ rlp(status) ++ rlp(gas) ++ rlp(bloom)
51/// 			++ LogsHeader ++ rlp(log1) ++ rlp(log2) ++ ... ++ rlp(logN)
52/// ]
53/// ```
54///
55/// The optimization comes from the fact that `rlp(log1) ++ rlp(log2) ++ ... ++ rlp(logN)`
56/// can be built incrementally.
57///
58/// On average, from the real ethereum block, this implementation reduces the memory usage by 30%.
59///  `EncodedReceipt Space optimization (on average): 0.6995642434146292`
60pub struct AccumulateReceipt {
61	/// The RLP bytes where the logs are accumulated.
62	pub encoding: Vec<u8>,
63	/// The bloom filter collected from accumulating logs.
64	pub bloom: LogsBloom,
65}
66
67/// What an ethereum transaction has committed to its receipt so far: how much of the
68/// [`AccumulateReceipt`] encoding, the bloom over exactly those logs, and whether a log did not
69/// fit.
70///
71/// Kept in storage while the transaction runs, so that it rolls back with whatever storage layer
72/// rolls back: a contract frame, or a `batch_all` around a plain call whose balance change was
73/// mirrored as a log. The accumulated RLP is one log after another and the bloom is monotone, so
74/// the receipt at any layer is the encoding cut back to this length and this bloom.
75#[derive(Encode, Decode, MaxEncodedLen, TypeInfo, Clone, Copy, Debug, PartialEq, Eq)]
76pub struct CommittedReceiptLogs {
77	/// The length of the encoding the committed logs take up.
78	pub encoded_len: u32,
79	/// The bloom over the committed logs.
80	pub bloom: [u8; BLOOM_SIZE_BYTES],
81	/// Whether a log was left out because it did not fit the receipt.
82	///
83	/// Set when a log would have taken the receipt past
84	/// [`crate::limits::MAX_RECEIPT_LOGS_BYTES`]. Every later log is left out as well, and the
85	/// frame or the transaction the log was emitted in fails with
86	/// [`crate::Error::ReceiptTooLarge`].
87	pub exceeded: bool,
88}
89
90impl Default for CommittedReceiptLogs {
91	fn default() -> Self {
92		Self { encoded_len: 0, bloom: [0u8; BLOOM_SIZE_BYTES], exceeded: false }
93	}
94}
95
96impl AccumulateReceipt {
97	/// Constructs a new [`AccumulateReceipt`].
98	pub const fn new() -> Self {
99		Self { encoding: Vec::new(), bloom: LogsBloom::new() }
100	}
101
102	/// Cut the accumulator back to what `committed` covers, dropping every log a rolled-back
103	/// layer left behind.
104	pub fn reset_to(&mut self, committed: &CommittedReceiptLogs) {
105		self.encoding.truncate(committed.encoded_len as usize);
106		self.bloom = LogsBloom { bloom: committed.bloom };
107	}
108
109	/// What the accumulator holds, as the value to commit.
110	///
111	/// Always returns [`CommittedReceiptLogs::exceeded`] as `false`: the accumulator does not know
112	/// about a log that did not fit. That is correct only while no log has overflowed, which
113	/// `block_storage::capture_into_receipt` ensures by returning before it calls this once the
114	/// flag is set. Do not use it where the result could overwrite a set flag.
115	pub fn committed(&self) -> CommittedReceiptLogs {
116		CommittedReceiptLogs {
117			encoded_len: self.encoding.len() as u32,
118			bloom: self.bloom.bloom,
119			exceeded: false,
120		}
121	}
122
123	/// The number of bytes [`Self::add_log`] appends to the encoding for this log.
124	pub fn encoded_log_len(contract: &H160, data: &[u8], topics: &[H256]) -> usize {
125		let (_, payload_length) = Self::log_lengths(contract, data, topics);
126		payload_length.saturating_add(rlp::length_of_length(payload_length))
127	}
128
129	/// The payload lengths of the log's topic list and of the log itself, without their headers.
130	fn log_lengths(contract: &H160, data: &[u8], topics: &[H256]) -> (usize, usize) {
131		let mut topics_len: usize = 0;
132		for topic in topics {
133			// Topics are represented by 32 bytes. However, their encoding
134			// can produce different lengths depending on their value.
135			topics_len = topics_len.saturating_add(rlp::Encodable::length(&topic.0));
136		}
137		// Account for the size of the list header.
138		let topics_list_header_length =
139			topics_len.saturating_add(rlp::length_of_length(topics_len));
140		// Compute the total payload length of the log.
141		let payload_length = rlp::Encodable::length(&contract.0)
142			.saturating_add(rlp::Encodable::length(&data))
143			.saturating_add(topics_list_header_length);
144		(topics_len, payload_length)
145	}
146
147	/// Add the log into the accumulated receipt.
148	///
149	/// This accrues the log bloom and keeps track of the RLP encoding of the log.
150	pub fn add_log(&mut self, contract: &H160, data: &[u8], topics: &[H256]) {
151		// Accrue the log bloom.
152		self.bloom.accrue_log(contract, topics);
153
154		let (topics_len, payload_length) = Self::log_lengths(contract, data, topics);
155
156		let header = rlp::Header { list: true, payload_length };
157		header.encode(&mut self.encoding);
158		rlp::Encodable::encode(&contract.0, &mut self.encoding);
159		// Encode the topics as a list
160		rlp::Header { list: true, payload_length: topics_len }.encode(&mut self.encoding);
161		for topic in topics {
162			rlp::Encodable::encode(&topic.0, &mut self.encoding);
163		}
164		rlp::Encodable::encode(&data, &mut self.encoding);
165	}
166
167	/// Finalize the accumulated receipt and return the RLP encoded bytes.
168	pub fn encoded_receipt(
169		encoded_logs: Vec<u8>,
170		bloom: LogsBloom,
171		status: bool,
172		gas: u64,
173		transaction_type: Vec<u8>,
174	) -> Vec<u8> {
175		let logs_length = encoded_logs.len();
176		let list_header_length = logs_length + rlp::length_of_length(logs_length);
177
178		let header = rlp::Header {
179			list: true,
180			payload_length: rlp::Encodable::length(&status) +
181				rlp::Encodable::length(&gas) +
182				rlp::Encodable::length(&bloom.bloom) +
183				list_header_length,
184		};
185
186		let mut encoded = transaction_type;
187		header.encode(&mut encoded);
188		rlp::Encodable::encode(&status, &mut encoded);
189		rlp::Encodable::encode(&gas, &mut encoded);
190		rlp::Encodable::encode(&bloom.bloom, &mut encoded);
191
192		let logs_header = rlp::Header { list: true, payload_length: logs_length };
193		logs_header.encode(&mut encoded);
194
195		encoded.extend(encoded_logs);
196
197		encoded
198	}
199}

201/// Bloom log filter compatible with Ethereum implementation.
202///
203/// This structure avoids conversions between substrate to alloy types
204/// to optimally compute the bloom.
205#[derive(Clone, Copy)]
206pub struct LogsBloom {
207	/// The bloom bytes used to store logs.
208	pub bloom: [u8; BLOOM_SIZE_BYTES],
209}
210
211impl Default for LogsBloom {
212	fn default() -> Self {
213		Self::new()
214	}
215}
216
217impl LogsBloom {
218	/// Constructs a new [`LogsBloom`].
219	pub const fn new() -> Self {
220		Self { bloom: [0u8; BLOOM_SIZE_BYTES] }
221	}
222
223	/// Ingests a raw log (event) into the bloom filter.
224	pub fn accrue_log(&mut self, contract: &H160, topics: &[H256]) {
225		Self::m3_2048(&mut self.bloom, contract.as_ref());
226
227		for topic in topics {
228			Self::m3_2048(&mut self.bloom, topic.as_ref());
229		}
230	}
231
232	/// Accrues the input into the bloom filter.
233	pub fn accrue_bloom(&mut self, other: &Self) {
234		for i in 0..BLOOM_SIZE_BYTES {
235			self.bloom[i] |= other.bloom[i];
236		}
237	}
238
239	/// Specialized Bloom filter that sets three bits out of 2048, given an
240	/// arbitrary byte sequence.
241	///
242	/// See Section 4.3.1 "Transaction Receipt" of the
243	/// [Ethereum Yellow Paper][ref] (page 6).
244	///
245	/// [ref]: https://ethereum.github.io/yellowpaper/paper.pdf
246	fn m3_2048(bloom: &mut [u8; 256], bytes: &[u8]) {
247		let hash = keccak_256(bytes);
248		for i in [0, 2, 4] {
249			let bit = (hash[i + 1] as usize + ((hash[i] as usize) << 8)) & 0x7FF;
250			bloom[256 - 1 - bit / 8] |= 1 << (bit % 8);
251		}
252	}
253}

260	#[test]
261	fn test_bloom_accrue_log() {
262		let mut bloom = LogsBloom::new();
263		let data = vec![
264			(H160::repeat_byte(0x01), vec![H256::repeat_byte(0x02), H256::repeat_byte(0x03)]),
265			(H160::repeat_byte(0x04), vec![H256::repeat_byte(0x05), H256::repeat_byte(0x06)]),
266			(H160::repeat_byte(0x07), vec![H256::repeat_byte(0x08), H256::repeat_byte(0x09)]),
267		];
268
269		for (contract, topics) in data.clone() {
270			bloom.accrue_log(&contract, &topics);
271		}
272
273		let mut alloy_bloom = alloy_core::primitives::Bloom::default();
274
275		for (contract, topics) in data {
276			alloy_bloom.accrue_raw_log(
277				contract.0.into(),
278				&topics.iter().map(|t| t.0.into()).collect::<Vec<_>>(),
279			);
280		}
281
282		assert_eq!(bloom.bloom, alloy_bloom.0);
283	}

287		let mut bloom = LogsBloom::new();
288		let mut bloom2 = LogsBloom::new();
289
290		bloom.accrue_log(&H160::repeat_byte(0x01), &[H256::repeat_byte(0x02)]);

293

297			.accrue_raw_log(H160::repeat_byte(0x01).0.into(), &[H256::repeat_byte(0x02).0.into()]);

300		alloy_bloom.accrue_bloom(&alloy_bloom2);
301
302		assert_eq!(bloom.bloom, alloy_bloom.0);

305	#[test]
306	fn encoded_log_len_matches_add_log() {
307		let contract = H160::repeat_byte(0x01);
308		for num_topics in 0..=4u8 {
309			let topics: Vec<H256> = (0..num_topics).map(H256::repeat_byte).collect();
310			for data_len in [0, 1, 55, 56, 255, 256, 65_536] {
311				let data = vec![0x42u8; data_len];
312				let mut receipt = AccumulateReceipt::new();
313				receipt.add_log(&contract, &data, &[]);
314				let before = receipt.encoding.len();
315				receipt.add_log(&contract, &data, &topics);
316				assert_eq!(
317					receipt.encoding.len() - before,
318					AccumulateReceipt::encoded_log_len(&contract, &data, &topics),
319					"{num_topics} topics, {data_len} bytes of data",
320				);
321			}
322		}
323	}

325	#[test]
326	fn test_accumulate_receipt() {
327		let mut receipt = AccumulateReceipt::new();
328
329		receipt.add_log(&H160::repeat_byte(0x01), &[0x01, 0x02], &[H256::repeat_byte(0x02)]);
330		receipt.add_log(&H160::repeat_byte(0x03), &[0x03, 0x04], &[H256::repeat_byte(0x04)]);
331
332		let encoded = AccumulateReceipt::encoded_receipt(
333			receipt.encoding,
334			receipt.bloom,
335			true,
336			21000,
337			vec![],
338		);
339
340		let alloy_receipt = alloy_consensus::Receipt {
341			status: true.into(),
342			cumulative_gas_used: 21000,
343			logs: vec![
344				alloy_core::primitives::Log::new_unchecked(
345					H160::repeat_byte(0x01).0.into(),
346					vec![H256::repeat_byte(0x02).0.into()],
347					vec![0x01, 0x02].into(),
348				),
349				alloy_core::primitives::Log::new_unchecked(
350					H160::repeat_byte(0x03).0.into(),
351					vec![H256::repeat_byte(0x04).0.into()],
352					vec![0x03, 0x04].into(),
353				),
354			],
355		};
356
357		// Check bloom filters.
358		let alloy_bloom = alloy_receipt.bloom_slow();
359		assert_eq!(receipt.bloom.bloom, alloy_bloom.0);
360
361		// Check RLP encoding.
362		let mut alloy_encoded = vec![];
363		alloy_receipt.rlp_encode_with_bloom(&alloy_bloom, &mut alloy_encoded);
364
365		assert_eq!(alloy_encoded, encoded);
366	}
