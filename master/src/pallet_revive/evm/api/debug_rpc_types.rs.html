30/// The type of tracer to use.
31#[derive(TypeInfo, Debug, Clone, Encode, Decode, Serialize, Deserialize, PartialEq)]
32#[serde(tag = "tracer", content = "tracerConfig", rename_all = "camelCase")]
33pub enum TracerType {
34	/// A tracer that traces calls.
35	CallTracer(Option<CallTracerConfig>),
36
37	/// A tracer that traces the prestate.
38	PrestateTracer(Option<PrestateTracerConfig>),
39
40	/// A tracer that traces opcodes and syscalls.
41	ExecutionTracer(Option<ExecutionTracerConfig>),
42}
43
44impl From<CallTracerConfig> for TracerType {
45	fn from(config: CallTracerConfig) -> Self {
46		TracerType::CallTracer(Some(config))
47	}
48}
49
50impl From<PrestateTracerConfig> for TracerType {
51	fn from(config: PrestateTracerConfig) -> Self {
52		TracerType::PrestateTracer(Some(config))
53	}
54}
55
56impl From<ExecutionTracerConfig> for TracerType {
57	fn from(config: ExecutionTracerConfig) -> Self {
58		TracerType::ExecutionTracer(Some(config))
59	}
60}
61
62impl Default for TracerType {
63	fn default() -> Self {
64		TracerType::ExecutionTracer(Some(ExecutionTracerConfig::default()))
65	}
66}

68/// Tracer configuration used to trace calls.
69#[derive(TypeInfo, Debug, Clone, Default, PartialEq)]
70#[cfg_attr(feature = "std", derive(Serialize), serde(rename_all = "camelCase"))]
71pub struct TracerConfig {
72	/// The tracer type.
73	#[cfg_attr(feature = "std", serde(flatten, default))]
74	pub config: TracerType,
75
76	/// Timeout for the tracer.
77	#[cfg_attr(feature = "std", serde(with = "humantime_serde", default))]
78	pub timeout: Option<core::time::Duration>,
79}

81#[cfg(feature = "std")]
82impl<'de> Deserialize<'de> for TracerConfig {
83	fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
84	where
85		D: Deserializer<'de>,
86	{
87		#[derive(Deserialize)]
88		#[serde(rename_all = "camelCase")]
89		struct TracerConfigWithType {
90			#[serde(flatten)]
91			config: TracerType,
92			#[serde(with = "humantime_serde", default)]
93			timeout: Option<core::time::Duration>,
94		}
95
96		#[derive(Deserialize)]
97		#[serde(rename_all = "camelCase")]
98		struct TracerConfigInline {
99			#[serde(flatten, default)]
100			execution_tracer_config: ExecutionTracerConfig,
101			#[serde(with = "humantime_serde", default)]
102			timeout: Option<core::time::Duration>,
103		}
104
105		#[derive(Deserialize)]
106		#[serde(untagged)]
107		enum TracerConfigHelper {
108			WithType(TracerConfigWithType),
109			Inline(TracerConfigInline),
110		}
111
112		match TracerConfigHelper::deserialize(deserializer)? {
113			TracerConfigHelper::WithType(cfg) => {
114				Ok(TracerConfig { config: cfg.config, timeout: cfg.timeout })
115			},
116			TracerConfigHelper::Inline(cfg) => Ok(TracerConfig {
117				config: TracerType::ExecutionTracer(Some(cfg.execution_tracer_config)),
118				timeout: cfg.timeout,
119			}),
120		}
121	}
122}

358/// A prestate Trace
359#[derive(TypeInfo, Encode, Serialize, Decode, Clone, Debug, Eq, PartialEq)]
360#[serde(untagged)]
361pub enum PrestateTrace {
362	/// The Prestate mode returns the accounts necessary to execute a given transaction
363	Prestate(BTreeMap<H160, PrestateTraceInfo>),
364
365	/// The diff mode returns the differences between the transaction's pre and post-state
366	/// The result only contains the accounts that were modified by the transaction
367	DiffMode {
368		/// The state before the call.
369		///  The accounts in the `pre` field will contain all of their basic fields, even if those
370		/// fields have not been modified. For `storage` however, only non-empty slots that have
371		/// been modified will be included
372		pre: BTreeMap<H160, PrestateTraceInfo>,
373		/// The state after the call.
374		/// It only contains the specific fields that were actually modified during the transaction
375		post: BTreeMap<H160, PrestateTraceInfo>,
376	},
377}
378
379impl<'de> Deserialize<'de> for PrestateTrace {
380	fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
381	where
382		D: serde::Deserializer<'de>,
383	{
384		struct PrestateTraceVisitor;
385
386		impl<'de> Visitor<'de> for PrestateTraceVisitor {
387			type Value = PrestateTrace;
388
389			fn expecting(&self, formatter: &mut core::fmt::Formatter) -> core::fmt::Result {
390				formatter.write_str("a map representing either Prestate or DiffMode")
391			}
392
393			fn visit_map<A>(self, mut map: A) -> Result<Self::Value, A::Error>
394			where
395				A: MapAccess<'de>,
396			{
397				let mut pre_map = None;
398				let mut post_map = None;
399				let mut account_map = BTreeMap::new();
400
401				while let Some(key) = map.next_key::<String>()? {
402					match key.as_str() {
403						"pre" => {
404							if pre_map.is_some() {
405								return Err(Error::duplicate_field("pre"));
406							}
407							pre_map = Some(map.next_value::<BTreeMap<H160, PrestateTraceInfo>>()?);
408						},
409						"post" => {
410							if post_map.is_some() {
411								return Err(Error::duplicate_field("post"));
412							}
413							post_map = Some(map.next_value::<BTreeMap<H160, PrestateTraceInfo>>()?);
414						},
415						_ => {
416							let addr: H160 =
417								key.parse().map_err(|_| Error::custom("Invalid address"))?;
418							let info = map.next_value::<PrestateTraceInfo>()?;
419							account_map.insert(addr, info);
420						},
421					}
422				}
423
424				match (pre_map, post_map) {
425					(Some(pre), Some(post)) => {
426						if !account_map.is_empty() {
427							return Err(Error::custom("Mixed diff and prestate mode"));
428						}
429						Ok(PrestateTrace::DiffMode { pre, post })
430					},
431					(None, None) => Ok(PrestateTrace::Prestate(account_map)),
432					_ => Err(Error::custom("diff mode: must have both 'pre' and 'post'")),
433				}
434			}
435		}
436
437		deserializer.deserialize_map(PrestateTraceVisitor)
438	}
439}

472/// Returns true if the map has no `Some` element
473pub fn is_empty<K, V>(map: &BTreeMap<K, Option<V>>) -> bool {
474	!map.values().any(|v| v.is_some())
475}

477/// Serializes a map, skipping `None` values.
478pub fn serialize_map_skip_none<S, K, V>(
479	map: &BTreeMap<K, Option<V>>,
480	serializer: S,
481) -> Result<S::Ok, S::Error>
482where
483	S: Serializer,
484	K: serde::Serialize,
485	V: serde::Serialize,
486{
487	let len = map.values().filter(|v| v.is_some()).count();
488	let mut ser_map = serializer.serialize_map(Some(len))?;
489
490	for (key, opt_val) in map {
491		if let Some(val) = opt_val {
492			ser_map.serialize_entry(key, val)?;
493		}
494	}
495
496	ser_map.end()
497}

547/// The kind of execution step.

558		op: u8,
559		/// EVM stack contents.
560		#[serde(

563			deserialize_with = "deserialize_stack_minimal"

565		stack: Vec<Bytes>,
566		/// EVM memory contents.
567		#[serde(
568			default,
569			skip_serializing_if = "Vec::is_empty",

572		memory: Vec<Bytes>,
573		/// Contract storage changes.
574		#[serde(
575			default,
576			skip_serializing_if = "Option::is_none",
577			serialize_with = "serialize_storage_no_prefix"
578		)]
579		storage: Option<alloc::collections::BTreeMap<Bytes, Bytes>>,
580	},
581	/// A PVM syscall execution.
582	PVMSyscall {
583		/// The executed syscall.
584		#[serde(
585			serialize_with = "serialize_syscall_op",
586			deserialize_with = "deserialize_syscall_op"
587		)]
588		op: u8,

595		#[serde(
596			default,
597			skip_serializing_if = "Option::is_none",
598			with = "super::hex_serde::option"
599		)]
600		returned: Option<u64>,
601	},
602}

610macro_rules! define_opcode_functions {
611	($($op:ident),* $(,)?) => {
612		/// Get opcode name from byte value using opcode names
613		fn get_opcode_name(opcode: u8) -> &'static str {
614			use revm::bytecode::opcode::*;
615			match opcode {
616				$(
617					$op => stringify!($op),
618				)*
619				_ => "INVALID",
620			}
621		}
622
623		/// Get opcode byte from name string
624		fn get_opcode_byte(name: &str) -> Option<u8> {
625			use revm::bytecode::opcode::*;
626			match name {
627				$(
628					stringify!($op) => Some($op),
629				)*
630				_ => None,
631			}
632		}
633	};
634}

788/// Serialize opcode as string using REVM opcode names
789fn serialize_opcode<S>(opcode: &u8, serializer: S) -> Result<S::Ok, S::Error>
790where
791	S: serde::Serializer,
792{
793	let name = get_opcode_name(*opcode);
794	serializer.serialize_str(name)
795}

797/// Serialize a syscall index to its name
798fn serialize_syscall_op<S>(idx: &u8, serializer: S) -> Result<S::Ok, S::Error>
799where
800	S: serde::Serializer,
801{
802	use crate::vm::pvm::env::list_trace_ops;
803	let Some(name_bytes) = list_trace_ops().get(*idx as usize) else {
804		return Err(serde::ser::Error::custom(alloc::format!("Unknown trace op: {idx}")));
805	};
806	let name = core::str::from_utf8(name_bytes).unwrap_or_default();
807	serializer.serialize_str(name)
808}

810/// Deserialize opcode from string using reverse lookup table
811fn deserialize_opcode<'de, D>(deserializer: D) -> Result<u8, D::Error>
812where
813	D: serde::Deserializer<'de>,
814{
815	let s = String::deserialize(deserializer)?;
816	get_opcode_byte(&s)
817		.ok_or_else(|| serde::de::Error::custom(alloc::format!("Unknown opcode: {}", s)))
818}
819
820/// Deserialize syscall from string name to index
821fn deserialize_syscall_op<'de, D>(deserializer: D) -> Result<u8, D::Error>
822where
823	D: serde::Deserializer<'de>,
824{
825	use crate::vm::pvm::env::list_trace_ops;
826	let s = String::deserialize(deserializer)?;
827	let ops = list_trace_ops();
828	ops.iter()
829		.position(|name| core::str::from_utf8(name).unwrap_or_default() == s)
830		.map(|i| i as u8)
831		.ok_or_else(|| serde::de::Error::custom(alloc::format!("Unknown trace op: {}", s)))
832}

834/// A smart contract execution call trace.
835#[derive(
836	TypeInfo, Default, Encode, Decode, Serialize, Deserialize, Clone, Debug, Eq, PartialEq,
837)]
838#[serde(rename_all = "camelCase")]
839pub struct CallTrace {
840	/// Address of the sender.
841	pub from: H160,
842	/// Amount of gas provided for the call.
843	#[serde(with = "super::hex_serde")]
844	pub gas: u64,
845	/// Amount of gas used.
846	#[serde(with = "super::hex_serde")]
847	pub gas_used: u64,
848	/// Address of the receiver.
849	pub to: H160,
850	/// Call input data.
851	pub input: Bytes,
852	/// Return data.
853	#[serde(skip_serializing_if = "Bytes::is_empty")]
854	pub output: Bytes,

856	#[serde(skip_serializing_if = "Option::is_none")]
857	pub error: Option<String>,
858	/// The revert reason, if the call reverted.
859	#[serde(skip_serializing_if = "Option::is_none")]
860	pub revert_reason: Option<String>,
861	/// List of sub-calls.
862	#[serde(skip_serializing_if = "Vec::is_empty")]

864	/// List of logs emitted during the call.
865	#[serde(skip_serializing_if = "Vec::is_empty")]
866	pub logs: Vec<CallLog>,

868	#[serde(skip_serializing_if = "Option::is_none")]
869	pub value: Option<U256>,
870	/// Type of call.
871	#[serde(rename = "type")]
872	pub call_type: CallType,
873	/// Number of child calls entered (for log position calculation)
874	#[serde(skip)]
875	pub child_call_count: u32,
876}

907/// Serialize stack values using minimal hex format (like Geth)
908fn serialize_stack_minimal<S>(stack: &Vec<Bytes>, serializer: S) -> Result<S::Ok, S::Error>
909where
910	S: serde::Serializer,
911{
912	let minimal_values: Vec<String> = stack.iter().map(|bytes| bytes.to_short_hex()).collect();
913	minimal_values.serialize(serializer)
914}
915
916/// Deserialize stack values from minimal hex format
917fn deserialize_stack_minimal<'de, D>(deserializer: D) -> Result<Vec<Bytes>, D::Error>
918where
919	D: serde::Deserializer<'de>,
920{
921	let strings = Vec::<String>::deserialize(deserializer)?;
922	strings
923		.into_iter()
924		.map(|s| {
925			// Parse as U256 to handle minimal hex like "0x0" or "0x4"
926			let s = s.trim_start_matches("0x");
927			let value = sp_core::U256::from_str_radix(s, 16)
928				.map_err(|e| serde::de::Error::custom(alloc::format!("{:?}", e)))?;
929			// Convert to bytes, trimming leading zeros to match serialization
930			let bytes = value.to_big_endian();
931			let trimmed = bytes
932				.iter()
933				.position(|&b| b != 0)
934				.map(|pos| bytes[pos..].to_vec())
935				.unwrap_or_else(|| alloc::vec![0u8]);
936			Ok(Bytes::from(trimmed))
937		})
938		.collect()
939}

941/// Serialize memory values without "0x" prefix (like Geth)
942fn serialize_memory_no_prefix<S>(memory: &Vec<Bytes>, serializer: S) -> Result<S::Ok, S::Error>
943where
944	S: serde::Serializer,

950/// Serialize storage map without "0x" prefix (like Geth)
951fn serialize_storage_no_prefix<S>(
952	storage: &Option<alloc::collections::BTreeMap<Bytes, Bytes>>,
953	serializer: S,
954) -> Result<S::Ok, S::Error>
955where
956	S: serde::Serializer,
957{
958	match storage {
959		None => serializer.serialize_none(),
960		Some(map) => {
961			let mut ser_map = serializer.serialize_map(Some(map.len()))?;
962			for (key, value) in map {
963				ser_map.serialize_entry(&key.to_hex_no_prefix(), &value.to_hex_no_prefix())?;
964			}
965			ser_map.end()
966		},
967	}
