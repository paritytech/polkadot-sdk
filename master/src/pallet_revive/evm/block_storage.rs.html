49/// Result of an Ethereum context call execution.
50pub(crate) struct EthereumCallResult {
51	/// Receipt gas information.
52	pub receipt_gas_info: ReceiptGasInfo,
53	/// The dispatch result with post-dispatch information.
54	pub result: DispatchResultWithPostInfo,
55}
56
57impl EthereumCallResult {
58	/// Create a new `EthereumCallResult` from contract execution details.
59	///
60	/// # Parameters
61	///
62	/// - `signer`: The signer of the transaction
63	/// - `output`: The execution result
64	/// - `gas_consumed`: The weight consumed during execution
65	/// - `base_call_weight`: The base call weight
66	/// - `encoded_len`: The length of the encoded transaction in bytes
67	/// - `info`: Dispatch information used for fee computation
68	/// - `effective_gas_price`: The EVM gas price
69	pub(crate) fn new<T: Config>(
70		signer: AccountIdOf<T>,
71		mut output: ContractResult<ExecReturnValue, BalanceOf<T>>,
72		mut base_call_weight: Weight,
73		encoded_len: u32,
74		info: &DispatchInfo,
75		effective_gas_price: U256,
76	) -> Self {
77		let effective_gas_price = effective_gas_price.max(Pallet::<T>::evm_base_fee());
78
79		if let Ok(retval) = &output.result &&
80			retval.did_revert()
81		{
82			output.result = Err(<Error<T>>::ContractReverted.into());
83		}
84
85		// Refund pre-charged revert event weight if the call succeeds.
86		if output.result.is_ok() {
87			base_call_weight.saturating_reduce(T::WeightInfo::deposit_eth_extrinsic_revert_event())
88		}
89
90		crate::if_tracing(|tracer| {
91			tracer.dispatch_result(base_call_weight, output.weight_consumed);
92		});
93
94		let result = dispatch_result(output.result, output.weight_consumed, base_call_weight);
95		let native_fee = T::FeeInfo::compute_actual_fee(encoded_len, &info, &result);
96		let result = T::FeeInfo::ensure_not_overdrawn(native_fee, result);
97
98		let fee = Pallet::<T>::convert_native_to_evm(match output.storage_deposit {
99			StorageDeposit::Refund(refund) => native_fee.saturating_sub(refund),
100			StorageDeposit::Charge(amount) => native_fee.saturating_add(amount),
101		});
102
103		let (mut gas_used, rest) = fee.div_mod(effective_gas_price);
104		if !rest.is_zero() {
105			gas_used = gas_used.saturating_add(1_u32.into());
106		}
107
108		let tx_cost = gas_used.saturating_mul(effective_gas_price);
109		if tx_cost > fee {
110			let round_up_fee = BalanceWithDust::<BalanceOf<T>>::from_value::<T>(tx_cost - fee)
111				.expect("value fits into BalanceOf<T>; qed");
112			log::debug!(target: LOG_TARGET, "Collecting round_up fee from {signer:?}: {round_up_fee:?}");
113			let _ = burn_with_dust::<T>(&signer, round_up_fee)
114					.inspect_err(|e| log::debug!(target: LOG_TARGET, "Failed to collect round up fee {round_up_fee:?} from {signer:?}: {e:?}"));
115		}
116
117		Self { receipt_gas_info: ReceiptGasInfo { gas_used, effective_gas_price }, result }
118	}
119}

121/// Capture the Ethereum log for the current transaction.
122///
123/// This method does nothing if called from outside of the ethereum context.
124pub fn capture_ethereum_log(contract: &H160, data: &[u8], topics: &[H256]) {
125	receipt::with(|receipt| {
126		receipt.add_log(contract, data, topics);
127	});
128}

130/// Get the receipt details of the current transaction.
131///
132/// This method returns `None` if and only if the function is called
133/// from outside of the ethereum context.
134pub fn get_receipt_details() -> Option<(Vec<u8>, LogsBloom)> {
135	receipt::with(|receipt| {
136		let encoding = core::mem::take(&mut receipt.encoding);
137		let bloom = core::mem::take(&mut receipt.bloom);
138		(encoding, bloom)
139	})
140}

148/// Execute the Ethereum call, and write the block storage transaction details.
149///
150/// # Parameters
151/// - transaction_encoded: The RLP encoded transaction bytes.
152/// - call: A closure that executes the transaction logic and returns an `EthereumCallResult`.
153pub fn with_ethereum_context<T: Config>(
154	transaction_encoded: Vec<u8>,
155	call: impl FnOnce() -> EthereumCallResult,
156) -> DispatchResultWithPostInfo {
157	receipt::using(&mut AccumulateReceipt::new(), || {
158		let (err, receipt_gas_info, post_info) =
159			with_transaction(|| -> TransactionOutcome<Result<_, DispatchError>> {
160				let EthereumCallResult { receipt_gas_info, result } = call();
161				match result {
162					Ok(post_info) => {
163						TransactionOutcome::Commit(Ok((None, receipt_gas_info, post_info)))
164					},
165					Err(err) => TransactionOutcome::Rollback(Ok((
166						Some(err.error),
167						receipt_gas_info,
168						err.post_info,
169					))),
170				}
171			})?;
172
173		if let Some(dispatch_error) = err {
174			deposit_eth_extrinsic_revert_event::<T>(dispatch_error);
175			crate::block_storage::process_transaction::<T>(
176				transaction_encoded,
177				false,
178				receipt_gas_info,
179			);
180			Ok(post_info)
181		} else {
182			// deposit a dummy event in benchmark mode
183			#[cfg(feature = "runtime-benchmarks")]
184			deposit_eth_extrinsic_revert_event::<T>(crate::Error::<T>::BenchmarkingError.into());
185
186			crate::block_storage::process_transaction::<T>(
187				transaction_encoded,
188				true,
189				receipt_gas_info,
190			);
191			Ok(post_info)
192		}
193	})
194}

196fn deposit_eth_extrinsic_revert_event<T: Config>(dispatch_error: DispatchError) {
197	Pallet::<T>::deposit_event(Event::<T>::EthExtrinsicRevert { dispatch_error });
198}

203	EthereumBlock::<T>::kill();
204}

206/// Build the ethereum block and store it into the pallet storage.
207pub fn on_finalize_build_eth_block<T: Config>(block_number: BlockNumberFor<T>) {
208	let block_builder_ir = EthBlockBuilderIR::<T>::get();
209	EthBlockBuilderIR::<T>::kill();
210
211	let (block, receipt_data) =
212		EthereumBlockBuilder::<T>::from_ir(block_builder_ir).build_block(block_number);
213
214	// Put the block hash into storage.
215	BlockHash::<T>::insert(block_number, block.hash);
216
217	// Prune older block hashes.
218	let block_hash_count = BLOCK_HASH_COUNT;
219	let to_remove = block_number.saturating_sub(block_hash_count.into()).saturating_sub(One::one());
220	if !Zero::is_zero(&to_remove) {
221		<BlockHash<T>>::remove(to_remove);
222	}
223	// Store the ETH block into the last block.
224	EthereumBlock::<T>::put(block);
225	// Store the receipt info data for offchain reconstruction.
226	ReceiptInfoData::<T>::put(receipt_data);
227}

229/// Process a transaction payload with extra details.
230/// This stores the RLP encoded transaction and receipt details into storage.
231///
232/// The data is used during the `on_finalize` hook to reconstruct the ETH block.
233pub fn process_transaction<T: Config>(
234	transaction_encoded: Vec<u8>,
235	success: bool,
236	receipt_gas_info: ReceiptGasInfo,
237) {
238	// Method returns `None` only when called from outside of the ethereum context.
239	// This is not the case here, since this is called from within the
240	// ethereum context.
241	let (encoded_logs, bloom) = get_receipt_details().unwrap_or_default();
242
243	let block_builder_ir = EthBlockBuilderIR::<T>::get();
244	let mut block_builder = EthereumBlockBuilder::<T>::from_ir(block_builder_ir);
245
246	block_builder.process_transaction(
247		transaction_encoded,
248		success,
249		receipt_gas_info,
250		encoded_logs,
251		bloom,
252	);
253
254	EthBlockBuilderIR::<T>::put(block_builder.to_ir());
255}

257// The `EthereumBlockBuilder` builds the Ethereum-compatible block by maintaining
258// two incremental hash builders. Each builder accumulates entries until the trie
259// is finalized:
260//  1. `transactions_root` - builds the Merkle root of transaction payloads
261//  2. `receipts_root` - builds the Merkle root of transaction receipts (event logs)
262//
263// The `EthereumBlockBuilder` is serialized and deserialized to and from storage
264// on every transaction via the `EthereumBlockBuilderIR` object. This is needed until
265// the runtime exposes a better API to preserve the state between transactions (ie,
266// the global `environment!` is wiped because each transaction will instantiate a new
267// WASM instance).
268//
269// For this reason, we need to account for the memory used by the `EthereumBlockBuilder`
270// and for the pallet storage consumed by the `EthereumBlockBuilderIR`.
271//
272// ## Memory Usage Analysis
273//
274// The incremental hash builder accumulates entries until the trie is finalized.
275// The last added entry value is kept in memory until it can be hashed.
276// The keys are always ordered and the hashing happens when the next entry is added to
277// the trie. The common prefix of the current and previous keys forms the path into the
278// trie, and together with the value of the previous entry, a hash of 32 bytes is
279// computed.
280//
281// For this reason, the memory usage of the incremental hash builder is no greater
282// than two entries of maximum size, plus some marginal book-keeping overhead
283// (ignored to simplify calculations).
284//
285// `IncrementalHashBuilder = 2 * maximum size of the entry`
286//
287// Additionally, the block builder caches the first entry for each incremental hash.
288// The entry is loaded from storage into RAM when either:
289// - The block is finalized, OR
290// - After 127 transactions.
291// Therefore, an additional entry of maximum size is needed in memory.
292//
293// That gives us 3 items of maximum size per each hash builder.
294//
295// `EthereumBlockBuilder = 3 * (max size of transactions + max size of receipts)`
296// The maximum size of a transaction is limited by
297// `limits::MAX_TRANSACTION_PAYLOAD_SIZE`, while the maximum size of a receipt is
298// limited by `limits::EVENT_BYTES`.
299//
300// Similarly, this is the amount of pallet storage consumed by the
301// `EthereumBlockBuilderIR` object, plus a marginal book-keeping overhead.
302pub fn block_builder_bytes_usage(max_events_size: u32) -> u32 {
303	// A block builder requires 3 times the maximum size of the entry.
304	const MEMORY_COEFFICIENT: u32 = 3;
305
306	// Because events are not capped, and the builder cannot exceed the
307	// number of bytes received, the actual memory usage for receipts is:
308	// `receipts_hash_builder = min(events_per_tx * 3, max_events_size)`
309	// where `max_events_size` can be consumed by a single transaction.
310	// Since we don't know in advance the `events_per_tx`, we'll assume the
311	// worst case scenario.
312	let receipts_hash_builder = max_events_size;
313
314	// `transactions_root` hash builder
315	let transactions_hash_builder =
316		limits::MAX_TRANSACTION_PAYLOAD_SIZE.saturating_mul(MEMORY_COEFFICIENT);
317
318	receipts_hash_builder.saturating_add(transactions_hash_builder)
