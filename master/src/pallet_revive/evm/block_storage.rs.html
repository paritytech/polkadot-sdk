60	pub result: DispatchResultWithPostInfo,

63impl EthereumCallResult {
64	/// Create a new `EthereumCallResult` from contract execution details.
65	///
66	/// # Parameters
67	///
68	/// - `signer`: The signer of the transaction
69	/// - `output`: The execution result
70	/// - `gas_consumed`: The weight consumed during execution
71	/// - `base_call_weight`: The base call weight
72	/// - `encoded_len`: The length of the encoded transaction in bytes
73	/// - `info`: Dispatch information used for fee computation
74	/// - `effective_gas_price`: The EVM gas price
75	pub(crate) fn new<T: Config>(
76		signer: AccountIdOf<T>,
77		mut output: ContractResult<ExecReturnValue, BalanceOf<T>>,
78		mut base_call_weight: Weight,
79		encoded_len: u32,
80		info: &DispatchInfo,
81		effective_gas_price: U256,
82	) -> Self {
83		let effective_gas_price = effective_gas_price.max(Pallet::<T>::evm_base_fee());
84
85		if let Ok(retval) = &output.result &&
86			retval.did_revert()
87		{
88			output.result = Err(<Error<T>>::ContractReverted.into());
89		}
90
91		// Refund pre-charged revert event weight if the call succeeds.
92		if output.result.is_ok() {
93			base_call_weight.saturating_reduce(T::WeightInfo::deposit_eth_extrinsic_revert_event())
94		}
95
96		crate::if_tracing(|tracer| {
97			tracer.dispatch_result(base_call_weight, output.weight_consumed);
98		});
99
100		let result = dispatch_result(output.result, output.weight_consumed, base_call_weight);
101		let native_fee = T::FeeInfo::compute_actual_fee(encoded_len, &info, &result);
102		let result = T::FeeInfo::ensure_not_overdrawn(native_fee, result);
103
104		let fee = Pallet::<T>::convert_native_to_evm(match output.storage_deposit {
105			StorageDeposit::Refund(refund) => native_fee.saturating_sub(refund),
106			StorageDeposit::Charge(amount) => native_fee.saturating_add(amount),
107		});
108
109		let (mut gas_used, rest) = fee.div_mod(effective_gas_price);
110		if !rest.is_zero() {
111			gas_used = gas_used.saturating_add(1_u32.into());
112		}
113
114		let tx_cost = gas_used.saturating_mul(effective_gas_price);
115		if tx_cost > fee {
116			let round_up_fee = BalanceWithDust::<BalanceOf<T>>::from_value::<T>(tx_cost - fee)
117				.expect("value fits into BalanceOf<T>; qed");
118			log::debug!(target: LOG_TARGET, "Collecting round_up fee from {signer:?}: {round_up_fee:?}");
119			let _ = burn_with_dust::<T>(&signer, round_up_fee)
120					.inspect_err(|e| log::debug!(target: LOG_TARGET, "Failed to collect round up fee {round_up_fee:?} from {signer:?}: {e:?}"));
121		}
122
123		Self { receipt_gas_info: ReceiptGasInfo { gas_used, effective_gas_price }, result }
124	}
125}

127/// Add the log to the open ethereum transaction's receipt. Returns `None` outside an ethereum
128/// transaction, where the caller decides what becomes of the log.
129///
130/// A log only exists if the storage change that produced it took effect, so the receipt follows
131/// storage rollback: what the transaction has committed of its receipt is kept in
132/// [`ReceiptLogsCommitted`], and rolls back with any layer that rolls back, a contract frame or a
133/// `batch_all` around a plain call. The accumulator itself lives outside storage and is cut back
134/// to the committed state before every log, and once more in [`get_receipt_details`].
135///
136/// The receipt is bounded by [`limits::MAX_RECEIPT_LOGS_BYTES`]. A log that would take it past
137/// the limit is not kept and fails with [`Error::ReceiptTooLarge`]. It also sets
138/// [`CommittedReceiptLogs::exceeded`] in the current storage layer: every later log fails as
139/// well, and the frame or the transaction the log was emitted in fails, see
140/// [`ensure_receipt_within_limit`], even where the error is not propagated. The flag rolls back
141/// with that storage layer. If the layer rolls back while the frame or the transaction around it
142/// continues, for example a call that mirrored a log and then failed inside a `Utility::batch` or
143/// an XCM `Transact`, the overflow fails nothing outside that layer.
144pub fn capture_into_receipt<T: Config>(
145	contract: &H160,
146	data: &[u8],
147	topics: &[H256],
148) -> Option<Result<(), DispatchError>> {
149	receipt::with(|receipt| {
150		let committed = ReceiptLogsCommitted::<T>::get();
151		if committed.exceeded {
152			return Err(Error::<T>::ReceiptTooLarge.into());
153		}
154		let fits = (committed.encoded_len as usize)
155			.saturating_add(AccumulateReceipt::encoded_log_len(contract, data, topics)) <=
156			limits::MAX_RECEIPT_LOGS_BYTES as usize;
157		let next = if fits {
158			receipt.reset_to(&committed);
159			receipt.add_log(contract, data, topics);
160			receipt.committed()
161		} else {
162			CommittedReceiptLogs { exceeded: true, ..committed }
163		};
164		ReceiptLogsCommitted::<T>::put(next);
165		ensure!(fits, Error::<T>::ReceiptTooLarge);
166		Ok(())
167	})
168}

170/// Capture a log emitted by a contract frame: into the open ethereum transaction's receipt, or
171/// nowhere.
172///
173/// A frame outside an ethereum transaction does not use the block's outside-of-frame buffer. The
174/// drain of a buffered log is charged to the block, not to the frame, so one frame could fill the
175/// buffer for a few milliseconds of weight and drop every mirrored balance change behind it. Such
176/// a log stays a substrate-only event, as before the buffer existed. This keeps the `LOG` opcode's
177/// guarantee intact: a log takes effect or its frame reverts, and here it does take effect as a
178/// `ContractEmitted` event; there is no ethereum observer outside an ethereum transaction for it
179/// to be missing from.
180///
181/// So where a contract's log ends up depends on whether an ethereum transaction's receipt is open
182/// when the log is emitted, not on the dispatch path that led there. Inside `eth_transact` or
183/// `eth_substrate_call`, which run inside `with_ethereum_context`, or their dry run, the log goes
184/// on the receipt and counts against [`limits::MAX_RECEIPT_LOGS_BYTES`]. This includes a
185/// `Revive::call` that `eth_substrate_call` dispatches, directly or through a batch or an XCM
186/// `Transact`. Outside an ethereum transaction, such as in a `Revive::call` extrinsic or an XCM
187/// `Transact` executed by the message queue in `on_initialize` or `on_idle`, the log stays a
188/// substrate-only `ContractEmitted` event.
189///
190/// Fails with [`Error::ReceiptTooLarge`] when the log does not fit the open receipt, see
191/// [`capture_into_receipt`]. The emitting frame then fails like on any other error.
192pub fn capture_frame_log<T: Config>(
193	contract: &H160,
194	data: &[u8],
195	topics: &[H256],
196) -> Result<(), DispatchError> {
197	capture_into_receipt::<T>(contract, data, topics).unwrap_or(Ok(()))
198}

200/// Take the receipt details of the current transaction: the logs it committed, see
201/// [`capture_into_receipt`], and their bloom.
202///
203/// This method returns `None` if and only if the function is called
204/// from outside of the ethereum context.
205pub fn get_receipt_details<T: Config>() -> Option<(Vec<u8>, LogsBloom)> {
206	receipt::with(|receipt| {
207		receipt.reset_to(&ReceiptLogsCommitted::<T>::take());
208		let encoding = core::mem::take(&mut receipt.encoding);
209		let bloom = core::mem::take(&mut receipt.bloom);
210		(encoding, bloom)
211	})
212}

220/// Run `f` with a receipt open, as an ethereum transaction runs, and discard the receipt after.
221///
222/// For a dry run of an ethereum transaction: its logs are subject to the same
223/// [`limits::MAX_RECEIPT_LOGS_BYTES`] as when the transaction is included in a block.
224pub fn with_receipt_scope<T: Config, R>(f: impl FnOnce() -> R) -> R {
225	receipt::using(&mut AccumulateReceipt::new(), || {
226		let result = f();
227		ReceiptLogsCommitted::<T>::kill();
228		result
229	})

232/// Fail with [`Error::ReceiptTooLarge`] if a log did not fit the open receipt, see
233/// [`capture_into_receipt`]. Outside an ethereum transaction there is no receipt and nothing is
234/// read.
235///
236/// Every contract frame, precompiles included, checks this before it returns successfully, so a
237/// log that did not fit fails the frame it was emitted in, also when the runtime mirrored it or
238/// a precompile did not propagate the error. Logs emitted outside any frame are checked where
239/// they can be emitted: after `bare_call` and `bare_instantiate` executed their storage
240/// deposits, and after the runtime call dispatched by `eth_substrate_call` or by a dry run.
241/// Failing there leaves the fee to be computed from the failed result. A storage layer that
242/// recorded the overflow and was then rolled back took the flag with it, and nothing fails here,
243/// see [`capture_into_receipt`].
244pub fn ensure_receipt_within_limit<T: Config>() -> Result<(), DispatchError> {
245	let exceeded = receipt::with(|_| ReceiptLogsCommitted::<T>::get().exceeded);
246	ensure!(exceeded != Some(true), Error::<T>::ReceiptTooLarge);
247	Ok(())
248}

250/// Execute the Ethereum call, and write the block storage transaction details.
251///
252/// # Parameters
253/// - transaction_encoded: The RLP encoded transaction bytes.
254/// - call: A closure that executes the transaction logic and returns an `EthereumCallResult`.
255pub fn with_ethereum_context<T: Config>(
256	transaction_encoded: Vec<u8>,
257	call: impl FnOnce() -> EthereumCallResult,
258) -> DispatchResultWithPostInfo {
259	receipt::using(&mut AccumulateReceipt::new(), || {
260		let (err, receipt_gas_info, post_info) =
261			with_transaction(|| -> TransactionOutcome<Result<_, DispatchError>> {
262				let EthereumCallResult { receipt_gas_info, result } = call();
263				match result {
264					Ok(post_info) => {
265						TransactionOutcome::Commit(Ok((None, receipt_gas_info, post_info)))
266					},
267					Err(err) => TransactionOutcome::Rollback(Ok((
268						Some(err.error),
269						receipt_gas_info,
270						err.post_info,
271					))),
272				}
273			})?;
274
275		if let Some(dispatch_error) = err {
276			// A failed transaction's receipt carries no logs: the dispatch rolled back, and with
277			// it what the transaction had committed to its receipt, whether one of its frames
278			// failed or the transaction failed outside them, for example in the runtime call of
279			// `eth_substrate_call`, at the storage deposits, at a receipt check outside any frame
280			// (see `ensure_receipt_within_limit`) or at the fee settlement.
281			deposit_eth_extrinsic_revert_event::<T>(dispatch_error);
282			crate::block_storage::process_transaction::<T>(
283				transaction_encoded,
284				false,
285				receipt_gas_info,
286			);
287			Ok(post_info)
288		} else {
289			// deposit a dummy event in benchmark mode
290			#[cfg(feature = "runtime-benchmarks")]
291			deposit_eth_extrinsic_revert_event::<T>(crate::Error::<T>::BenchmarkingError.into());
292
293			crate::block_storage::process_transaction::<T>(
294				transaction_encoded,
295				true,
296				receipt_gas_info,
297			);
298			Ok(post_info)
299		}
300	})
301}

317/// The synthetic transaction that carries the block's outside-of-frame logs.
318///
319/// An unsigned legacy transaction whose only distinguishing field is the block number as `nonce`,
320/// so its hash is unique per block and deterministically reproducible offchain (the eth-rpc
321/// serving layer rebuilds the identical bytes from the block number and chain id). It is not a
322/// real, executable transaction — it exists only to give the aggregated logs a receipt so they
323/// enter the block's bloom, `receipts_root` and transaction trie.
324fn synthetic_transaction<T: Config>(block_number: BlockNumberFor<T>) -> Vec<u8> {
325	crate::evm::synthetic_log_transaction(block_number.into(), T::ChainId::get().into())
326}

328/// Build the ethereum block and store it into the pallet storage.
329///
330/// `outside_frame_logs` are the block's logs emitted outside any ethereum transaction, in emission
331/// order. They are flushed as one synthetic transaction after every real one, and what the block
332/// committed to it is returned for the caller to store.
333///
334/// The block hash this stores is also what tells a later emitter that the drain has run, so a log
335/// from an `on_finalize` that `construct_runtime!` orders after this pallet's is not buffered; see
336/// [`Pallet::emit_contract_log_outside_frame`].
337pub fn on_finalize_build_eth_block<T: Config>(
338	block_number: BlockNumberFor<T>,
339	outside_frame_logs: Vec<OutsideFrameLog>,
340) -> Option<SyntheticTransactionInfo> {
341	let block_builder_ir = EthBlockBuilderIR::<T>::get();
342	EthBlockBuilderIR::<T>::kill();
343
344	let mut block_builder = EthereumBlockBuilder::<T>::from_ir(block_builder_ir);
345
346	// The synthetic transaction is how the outside-of-frame logs enter the block bloom /
347	// receipts_root / transaction trie. It goes in after all real transactions, since the trie
348	// builders are order-sensitive.
349	let log_event_indices: Vec<u32> =
350		outside_frame_logs.iter().map(|log| log.event_index).collect();
351	if !outside_frame_logs.is_empty() {
352		let mut receipt = AccumulateReceipt::new();
353		for OutsideFrameLog { contract, topics, data, .. } in &outside_frame_logs {
354			receipt.add_log(contract, data, topics);
355		}
356		block_builder.process_transaction(
357			synthetic_transaction::<T>(block_number),
358			true,
359			ReceiptGasInfo { gas_used: U256::zero(), effective_gas_price: U256::zero() },
360			receipt.encoding,
361			receipt.bloom,
362		);
363	}
364
365	let (block, mut receipt_data) = block_builder.build_block(block_number);
366
367	// The synthetic transaction is processed after every real one, so its entry is the trailing
368	// one.
369	let synthetic = if !log_event_indices.is_empty() {
370		receipt_data
371			.pop()
372			.map(|gas_info| SyntheticTransactionInfo { gas_info, log_event_indices })
373	} else {
374		None
375	};
376
377	// Put the block hash into storage.
378	BlockHash::<T>::insert(block_number, block.hash);
379
380	// Prune older block hashes.
381	let block_hash_count = BLOCK_HASH_COUNT;
382	let to_remove = block_number.saturating_sub(block_hash_count.into()).saturating_sub(One::one());
383	if !Zero::is_zero(&to_remove) {
384		<BlockHash<T>>::remove(to_remove);
385	}
386	// Store the ETH block into the last block.
387	EthereumBlock::<T>::put(block);
388	// Store the receipt info data for offchain reconstruction.
389	ReceiptInfoData::<T>::put(receipt_data);
390
391	synthetic
392}

394/// Process a transaction payload with extra details.
395/// This stores the RLP encoded transaction and receipt details into storage.
396///
397/// The data is used during the `on_finalize` hook to reconstruct the ETH block.
398pub fn process_transaction<T: Config>(
399	transaction_encoded: Vec<u8>,
400	success: bool,
401	receipt_gas_info: ReceiptGasInfo,
402) {
403	// Method returns `None` only when called from outside of the ethereum context.
404	// This is not the case here, since this is called from within the
405	// ethereum context.
406	let (encoded_logs, bloom) = get_receipt_details::<T>().unwrap_or_default();
407
408	let block_builder_ir = EthBlockBuilderIR::<T>::get();
409	let mut block_builder = EthereumBlockBuilder::<T>::from_ir(block_builder_ir);
410
411	block_builder.process_transaction(
412		transaction_encoded,
413		success,
414		receipt_gas_info,
415		encoded_logs,
416		bloom,
417	);
418
419	EthBlockBuilderIR::<T>::put(block_builder.to_ir());
420}

422// The `EthereumBlockBuilder` builds the Ethereum-compatible block by maintaining
423// two incremental hash builders. Each builder accumulates entries until the trie
424// is finalized:
425//  1. `transactions_root` - builds the Merkle root of transaction payloads
426//  2. `receipts_root` - builds the Merkle root of transaction receipts (event logs)
427//
428// The `EthereumBlockBuilder` is serialized and deserialized to and from storage
429// on every transaction via the `EthereumBlockBuilderIR` object. This is needed until
430// the runtime exposes a better API to preserve the state between transactions (ie,
431// the global `environment!` is wiped because each transaction will instantiate a new
432// WASM instance).
433//
434// For this reason, we need to account for the memory used by the `EthereumBlockBuilder`
435// and for the pallet storage consumed by the `EthereumBlockBuilderIR`.
436//
437// ## Memory Usage Analysis
438//
439// The incremental hash builder accumulates entries until the trie is finalized.
440// The last added entry value is kept in memory until it can be hashed.
441// The keys are always ordered and the hashing happens when the next entry is added to
442// the trie. The common prefix of the current and previous keys forms the path into the
443// trie, and together with the value of the previous entry, a hash of 32 bytes is
444// computed.
445//
446// For this reason, the memory usage of the incremental hash builder is no greater
447// than two entries of maximum size, plus some marginal book-keeping overhead
448// (ignored to simplify calculations).
449//
450// `IncrementalHashBuilder = 2 * maximum size of the entry`
451//
452// Additionally, the block builder caches the first entry for each incremental hash.
453// The entry is loaded from storage into RAM when either:
454// - The block is finalized, OR
455// - After 127 transactions.
456// Therefore, an additional entry of maximum size is needed in memory.
457//
458// That gives us 3 items of maximum size per each hash builder.
459//
460// `EthereumBlockBuilder = 3 * (max size of transactions + max size of receipts)`
461// The maximum size of a transaction is limited by `limits::MAX_TRANSACTION_PAYLOAD_SIZE`,
462// enforced when an `eth_transact` is checked, while the maximum size of a receipt is
463// `MAX_RECEIPT_BYTES`.
464//
465// Similarly, this is the amount of pallet storage consumed by the
466// `EthereumBlockBuilderIR` object, plus a marginal book-keeping overhead.
467
468/// Upper bound on the memory and pallet storage the ethereum block builder uses.
469pub const BLOCK_BUILDER_BYTES: u32 = 3 * MAX_ENTRY_BYTES;
470
471/// The bytes a receipt's RLP encoding adds around its logs, at most.
472///
473/// The transaction type (1), the receipt's list header (4, as the receipt is below 16 MiB), the
474/// status (1), the cumulative gas as a `u64` (9), the bloom (259) and the logs' list header (4).
475const RECEIPT_ENVELOPE_BYTES: u32 = 278;
476
477/// Upper bound on the encoding of one transaction's receipt.
478const MAX_RECEIPT_BYTES: u32 = limits::MAX_RECEIPT_LOGS_BYTES + RECEIPT_ENVELOPE_BYTES;
479
480/// Upper bound on one transaction and its receipt, as the block builder holds them.
481const MAX_ENTRY_BYTES: u32 = limits::MAX_TRANSACTION_PAYLOAD_SIZE + MAX_RECEIPT_BYTES;
482
483// The block's first transaction and its receipt are stored as one value and loaded back as one.
484// With the few bytes its SCALE encoding adds, about 3.2 MiB, that value takes at most an eighth of
485// the largest buffer the runtime can allocate.
486const _: () = assert!(MAX_ENTRY_BYTES + 16 <= sp_core::MAX_POSSIBLE_ALLOCATION / 8);
