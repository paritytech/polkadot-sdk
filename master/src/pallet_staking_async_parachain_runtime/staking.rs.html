36pub(crate) fn enable_dot_preset(fast: bool) {
37	Pages::set(&32);
38	MinerPages::set(&4);
39	MaxElectingVoters::set(&22_500);
40	TargetSnapshotPerBlock::set(&2000);
41	if !fast {
42		SignedValidationPhase::set(&(8 * Pages::get()));
43		SignedPhase::set(&(20 * MINUTES));
44	}
45}

58// This macro contains all of the variable parameters that we intend to use for Polkadot and
59// Kusama.
60//
61// Note that this runtime has 3 broad presets:
62//
63// 1. dev: fast development preset.
64// 2. dot-size: as close to Polkadot as possible.
65// 3. ksm-size: as close to Kusama as possible.
66//
67// The default values here are related to `dev`. The above helper functions are used at launch (see
68// `build_state` runtime-api) to enable dot/ksm presets.
69parameter_types! {
70	/// Number of election pages that we operate upon.
71	///
72	/// * Polkadot: 32 (3.2m snapshot)
73	/// * Kusama: 16 (1.6m snapshot)
74	///
75	/// Reasoning: Both leads to around 700 nominators per-page, yielding the weights in
76	/// https://github.com/paritytech/polkadot-sdk/pull/8704, the maximum of which being around 1mb
77	/// compressed PoV and 2mb uncompressed.
78	///
79	/// NOTE: in principle, there is nothing preventing us from stretching these values further, it
80	/// will only reduce the per-page POVs. Although, some operations like the first snapshot, and
81	/// the last page of export (where we operate on `MaxValidatorSet` validators) will not get any
82	/// better.
83	pub storage Pages: u32 = 4;
84
85	/// * Polkadot: 16 * 32 (512 blocks, 51.2m).
86	/// * Kusama: 8 * 16 (12 blocks, 12.8m).
87	///
88	/// (MaxSubmissions * Pages) for both, enough to verify all solutions.
89	///
90	/// Reasoning: Less security needed in Kusama, to compensate for the shorter session duration.
91	pub storage SignedValidationPhase: u32 = Pages::get() * 2;
92
93	/// * Polkadot: 200 blocks, 20m.
94	/// * Kusama: 100 blocks, 10m.
95	///
96	/// Reasoning:
97	///
98	/// * Polkadot wishes at least 8 submitters to be able to submit. That is  8 * 32 = 256 pages
99	///   for all submitters. Weight of each submission page is roughly 0.0007 of block weight. 200
100	///   blocks is more than enough.
101	/// * Kusama wishes at least 4 submitters to be able to submit. That is 4 * 16 = 64 pages for
102	///   all submitters. Weight of each submission page is roughly 0.0007 of block weight. 100
103	///   blocks is more than enough.
104	///
105	/// See `signed_weight_ratios` test below for more info.
106	pub storage SignedPhase: u32 = 4 * MINUTES;
107
108	/// * Polkadot: 4
109	/// * Kusama: 4
110	///
111	/// Reasoning: with 4 pages, the `ElectionScore` computed in both Kusama and Polkadot is pretty
112	/// good. See and run `run_election_with_pages` below to see. With 4 pages, roughly 2800
113	/// nominators will be elected. This is not great for staking reward, but is good enough for
114	/// chain's economic security.
115	pub storage MinerPages: u32 = 4;
116
117	/// * Polkadot: 300 blocks, 30m
118	/// * Kusama: 150 blocks, 15m
119	///
120	/// Reasoning: The only criteria is for the phase to be long enough such that the OCW miner is
121	/// able to run the mining code at least twice. Note that `OffchainRepeat` limits execution of
122	/// the OCW to at most 4 times per round, for faster collators.
123	///
124	/// Benchmarks logs from tests below are:
125	///
126	/// * exec_time of polkadot miner in WASM with 4 pages is 27369ms
127	/// * exec_time of kusama miner in WASM with 4 pages is 23848ms
128	///
129	/// See `max_ocw_miner_pages_as_per_weights` test below.
130	pub storage UnsignedPhase: u32 = MINUTES;
131
132	/// * Polkadot: 22_500
133	/// * Kusama: 12_500
134	///
135	/// Reasoning: Yielding  703 nominators per page in both. See [`Pages`] for more info. Path to
136	/// Upgrade: We may wish to increase the number of "active nominators" in both networks by 1)
137	/// increasing the `Pages` and `MaxElectingVoters` in sync. This update needs to happen while an
138	/// election is NOT ongoing.
139	pub storage MaxElectingVoters: u32 = 1000;
140
141	/// * Polkadot: 2000 (always equal to `staking.maxValidatorCount`)
142	/// * Kusama: 4000 (always equal to `staking.maxValidatorCount`)
143	///
144	/// Reasoning: As of now, we don't have a way to sort validators, so we wish to select all of
145	/// them. In case this limit is reached, governance should introduce `minValidatorBond`, and
146	/// validators would have to compete with their self-stake to force-chill one another. More
147	/// info: SRL-417
148	pub storage TargetSnapshotPerBlock: u32 = 4000;
149
150	// NOTE: rest of the parameters are computed identically in both Kusama and Polkadot.
151
152	/// Allow OCW miner to at most run 4 times in the entirety of the 10m Unsigned Phase.
153	pub OffchainRepeat: u32 = UnsignedPhase::get() / 4;
154
155	/// Upper bound of `Staking.ValidatorCount`, which translates to
156	/// `ElectionProvider::DesiredTargets`. 1000 is the end-game for both Kusama and Polkadot for
157	/// the foreseeable future.
158	pub const MaxValidatorSet: u32 = 1000;
159
160	/// Number of nominators per page of the snapshot, and consequently number of backers in the
161	/// solution.
162	///
163	/// 704 in Polkadot (32 pages), 782 in Kusama (16 pages). Uses ceiling division so that
164	/// `VoterSnapshotPerBlock * Pages >= MaxElectingVoters` holds for any configured values.
165	pub VoterSnapshotPerBlock: u32 = MaxElectingVoters::get().div_ceil(Pages::get());
166
167	/// In each page, we may observe up to all of the validators.
168	pub const MaxWinnersPerPage: u32 = MaxValidatorSet::get();
169
170	/// In each page of the election, we allow up to all of the nominators of that page to be
171	/// present.
172	///
173	/// This in essence translates to "no limit on this as of now".
174	pub MaxBackersPerWinner: u32 = VoterSnapshotPerBlock::get();
175
176	/// Total number of backers per winner across all pages.
177	///
178	/// This in essence translates to "no limit on this as of now".
179	pub MaxBackersPerWinnerFinal: u32 = MaxElectingVoters::get();
180
181	/// Size of the exposures. This should be small enough to make the reward payouts cheap and
182	/// lightweight per-page.
183	// TODO: this is currently 512 in all networks, but 64 might yield better PoV, need to check logs.
184	pub const MaxExposurePageSize: u32 = 512;
185}

187// Signed phase parameters.
188parameter_types! {
189	/// * Polkadot: 16
190	/// * Kusama: 8
191	///
192	/// Reasoning: This is double the capacity of verification. There is no point for someone to be
193	/// a submitter if they cannot be verified, yet, it is beneficial to act as a "reserve", in case
194	/// someone bails out last minute.
195	pub MaxSubmissions: u32 = 8;
196
197	/// * Polkadot: Geometric progression with starting value 4 DOT, common factor 2. For 16
198	///   submissions, it will be [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384,
199	///   32768, 65536, 131072]. Sum is `262140 DOT` for all 16 submissions.
200	/// * Kusama: Geometric progression with with starting value 0.1 KSM, common factor 4. For 8
201	///   submissions, values will be: `[0.1, 0.4, 1.6, 6.4, 25.6, 102.4, 409.6, 1638.4]`. Sum is
202	///   `2184.5 KSM` for all 8 submissions.
203	pub DepositBase: Balance = 5 * UNITS;
204
205	/// * Polkadot: standard byte deposit configured in PAH.
206	/// * Kusama: standard byte deposit configured in KAH.
207	///
208	/// TODO: need a maximum solution length for each runtime.
209	pub DepositPerPage: Balance = 1 * UNITS;
210
211	/// * Polkadot: 20 DOT
212	/// * Kusama: 1 KSM
213	///
214	///
215	/// Fixed deposit for invulnerable accounts.
216	pub InvulnerableDeposit: Balance = UNITS;
217
218	/// * Polkadot: 10% (more restrictive, don't bail!)
219	/// * Kusama: 25%
220	///
221	/// Reasoning: The weight/fee of the `bail` transaction is already assuming you delete all pages
222	/// of your solution while bailing, and charges you accordingly. So the chain is being
223	/// compensated. The risk would be for an attacker to submit a lot of high score pages, and bail
224	/// at the end to avoid getting slashed.
225	pub BailoutGraceRatio: Perbill = Perbill::from_percent(5);
226
227	/// * Polkadot: 100%
228	/// * Kusama: 100%
229	///
230	/// The transaction fee of `register` takes into account the cost of possibly ejecting another
231	/// submission into account. In the scenario that the honest submitter is being ejected by an
232	/// attacker, the cost is on the attacker, and having 100% grace ratio here is only to the
233	/// benefit of the honest submitter.
234	pub EjectGraceRatio: Perbill = Perbill::from_percent(50);
235
236	/// * Polkadot: 5 DOTs per era/day
237	/// * Kusama: 1 KSM per era/6h
238	pub RewardBase: Balance = 10 * UNITS;
239}

391/// Polkadot inflation curve for DAP.
392///
393/// Same computation as the previous `EraPayout` but returns total emission
394/// (the staker/treasury split is now handled by DAP budget allocation).
395pub struct PolkadotIssuanceCurve;
396impl sp_staking::budget::IssuanceCurve<Balance> for PolkadotIssuanceCurve {
397	fn issue(_total_issuance: Balance, elapsed_millis: u64) -> Balance {
398		const MILLISECONDS_PER_YEAR: u64 = (1000 * 3600 * 24 * 36525) / 100;
399		let relative_period =
400			FixedU128::from_rational(elapsed_millis.into(), MILLISECONDS_PER_YEAR.into());
401
402		let fixed_total_issuance: i128 = 5_216_342_402_773_185_773;
403		let fixed_inflation_rate = FixedU128::from_rational(8, 100);
404		let yearly_emission = fixed_inflation_rate.saturating_mul_int(fixed_total_issuance);
405
406		let emission = relative_period.saturating_mul_int(yearly_emission);
407		emission.saturated_into()
408	}
409}

706/// Submits a transaction with the node's public and signature type. Adheres to the signed extension
707/// format of the chain.
708impl<LocalCall> frame_system::offchain::CreateSignedTransaction<LocalCall> for Runtime
709where
710	RuntimeCall: From<LocalCall>,
711{
712	fn create_signed_transaction<
713		C: frame_system::offchain::AppCrypto<Self::Public, Self::Signature>,
714	>(
715		call: RuntimeCall,
716		public: <Signature as Verify>::Signer,
717		account: AccountId,
718		nonce: <Runtime as frame_system::Config>::Nonce,
719	) -> Option<UncheckedExtrinsic> {
720		use sp_runtime::traits::StaticLookup;
721		// take the biggest period possible.
722		let period =
723			BlockHashCount::get().checked_next_power_of_two().map(|c| c / 2).unwrap_or(2) as u64;
724
725		let current_block = System::block_number()
726			.saturated_into::<u64>()
727			// The `System::block_number` is initialized with `n+1`,
728			// so the actual block number is `n`.
729			.saturating_sub(1);
730		let tip = 0;
731		let tx_ext = TxExtension::from((
732			frame_system::CheckNonZeroSender::<Runtime>::new(),
733			frame_system::CheckSpecVersion::<Runtime>::new(),
734			frame_system::CheckTxVersion::<Runtime>::new(),
735			frame_system::CheckGenesis::<Runtime>::new(),
736			frame_system::CheckEra::<Runtime>::from(generic::Era::mortal(period, current_block)),
737			frame_system::CheckNonce::<Runtime>::from(nonce),
738			frame_system::CheckWeight::<Runtime>::new(),
739			pallet_asset_conversion_tx_payment::ChargeAssetTxPayment::<Runtime>::from(tip, None),
740			frame_metadata_hash_extension::CheckMetadataHash::<Runtime>::new(true),
741		));
742		let raw_payload = SignedPayload::new(call, tx_ext)
743			.map_err(|e| {
744				log::warn!("Unable to create signed payload: {:?}", e);
745			})
746			.ok()?;
747		let signature = raw_payload.using_encoded(|payload| C::sign(payload, public))?;
748		let (call, tx_ext, _) = raw_payload.deconstruct();
749		let address = <Runtime as frame_system::Config>::Lookup::unlookup(account);
750		let transaction = UncheckedExtrinsic::new_signed(call, address, signature, tx_ext);
751		Some(transaction)
752	}
753}
