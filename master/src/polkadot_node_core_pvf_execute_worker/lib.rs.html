83/// Receives a handshake with information specific to the execute worker.
84fn recv_execute_handshake(stream: &mut UnixStream) -> io::Result<Handshake> {
85	let handshake_enc = framed_recv_blocking(stream)?;
86	let handshake = Handshake::decode(&mut &handshake_enc[..]).map_err(|_| {
87		io::Error::new(
88			io::ErrorKind::Other,
89			"execute pvf recv_execute_handshake: failed to decode Handshake".to_owned(),
90		)
91	})?;
92	Ok(handshake)
93}

95fn recv_request(stream: &mut UnixStream) -> io::Result<ExecuteRequest> {
96	let request_bytes = framed_recv_blocking(stream)?;
97	let request = ExecuteRequest::decode(&mut &request_bytes[..]).map_err(|_| {
98		io::Error::new(
99			io::ErrorKind::Other,
100			"execute pvf recv_request: failed to decode ExecuteRequest".to_string(),
101		)
102	})?;
103
104	Ok(request)
105}

107/// Sends an error to the host and returns the original error wrapped in `io::Error`.
108macro_rules! map_and_send_err {
109	($error:expr, $err_constructor:expr, $stream:expr, $worker_info:expr) => {{
110		let err: WorkerError = $err_constructor($error.to_string()).into();
111		let io_err = io::Error::new(io::ErrorKind::Other, err.to_string());
112		let _ = send_result::<WorkerResponse, WorkerError>($stream, Err(err), $worker_info);
113		io_err
114	}};
115}

117/// The entrypoint that the spawned execute worker should start with.
118///
119/// # Parameters
120///
121/// - `socket_path`: specifies the path to the socket used to communicate with the host.
122///
123/// - `worker_dir_path`: specifies the path to the worker-specific temporary directory.
124///
125/// - `node_version`: if `Some`, is checked against the `worker_version`. A mismatch results in
126///   immediate worker termination. `None` is used for tests and in other situations when version
127///   check is not necessary.
128///
129/// - `worker_version`: see above
130pub fn worker_entrypoint(
131	socket_path: PathBuf,
132	worker_dir_path: PathBuf,
133	node_version: Option<&str>,
134	worker_version: Option<&str>,
135) {
136	run_worker(
137		WorkerKind::Execute,
138		socket_path,
139		worker_dir_path,
140		node_version,
141		worker_version,
142		|mut stream, worker_info, security_status| {
143			let artifact_path = worker_dir::execute_artifact(&worker_info.worker_dir_path);
144
145			let Handshake { executor_params } =
146				recv_execute_handshake(&mut stream).map_err(|e| {
147					map_and_send_err!(
148						e,
149						InternalValidationError::HostCommunication,
150						&mut stream,
151						worker_info
152					)
153				})?;
154
155			let executor_params: Arc<ExecutorParams> = Arc::new(executor_params);
156			let execute_thread_stack_size = max_stack_size(&executor_params);
157
158			loop {
159				let request = recv_request(&mut stream).map_err(|e| {
160					map_and_send_err!(
161						e,
162						InternalValidationError::HostCommunication,
163						&mut stream,
164						worker_info
165					)
166				})?;
167
168				let pvd = request.pvd;
169				let pov = request.pov;
170				let execution_timeout = request.execution_timeout;
171				let artifact_checksum = request.artifact_checksum;
172				gum::debug!(
173					target: LOG_TARGET,
174					?worker_info,
175					?security_status,
176					"worker: validating artifact {}",
177					artifact_path.display(),
178				);
179
180				// Get the artifact bytes.
181				let compiled_artifact_blob = std::fs::read(&artifact_path).map_err(|e| {
182					map_and_send_err!(
183						e,
184						InternalValidationError::CouldNotOpenFile,
185						&mut stream,
186						worker_info
187					)
188				})?;
189
190				if artifact_checksum != compute_checksum(&compiled_artifact_blob) {
191					send_result::<WorkerResponse, WorkerError>(
192						&mut stream,
193						Ok(WorkerResponse {
194							job_response: JobResponse::CorruptedArtifact,
195							duration: Duration::ZERO,
196							pov_size: 0,
197						}),
198						worker_info,
199					)?;
200					continue;
201				}
202
203				let (pipe_read_fd, pipe_write_fd) = pipe2_cloexec().map_err(|e| {
204					map_and_send_err!(
205						e,
206						InternalValidationError::CouldNotCreatePipe,
207						&mut stream,
208						worker_info
209					)
210				})?;
211
212				let usage_before = nix::sys::resource::getrusage(UsageWho::RUSAGE_CHILDREN)
213					.map_err(|errno| {
214						let e = stringify_errno("getrusage before", errno);
215						map_and_send_err!(
216							e,
217							InternalValidationError::Kernel,
218							&mut stream,
219							worker_info
220						)
221					})?;
222				let stream_fd = stream.as_raw_fd();
223
224				let compiled_artifact_blob = Arc::new(compiled_artifact_blob);
225
226				let raw_block_data =
227					match sp_maybe_compressed_blob::decompress(&pov.block_data.0, POV_BOMB_LIMIT) {
228						Ok(data) => data,
229						Err(_) => {
230							send_result::<WorkerResponse, WorkerError>(
231								&mut stream,
232								Ok(WorkerResponse {
233									job_response: JobResponse::PoVDecompressionFailure,
234									duration: Duration::ZERO,
235									pov_size: 0,
236								}),
237								worker_info,
238							)?;
239							continue;
240						},
241					};
242
243				let pov_size = raw_block_data.len() as u32;
244
245				let params = ValidationParams {
246					parent_head: pvd.parent_head.clone(),
247					block_data: BlockData(raw_block_data.to_vec()),
248					relay_parent_number: pvd.relay_parent_number,
249					relay_parent_storage_root: pvd.relay_parent_storage_root,
250				};
251				let mut encoded_params = params.encode();
252
253				// Append V3+ extension based on descriptor version.
254				// SAFETY: ValidationParams is the complete message passed to the PVF.
255				// TrailingOption is safe here because:
256				// 1. ValidationParams is not embedded in any larger struct
257				// 2. The extension bytes are the ONLY thing after ValidationParams
258				// 3. The PVF will decode ValidationParams + optional extension as the entire input
259				let extension: TrailingOption<ValidationParamsExtension> =
260					match request.descriptor_version {
261						CandidateDescriptorVersion::V3 => {
262							// V3 candidate - append extension with both parent hashes
263							TrailingOption(Some(ValidationParamsExtension::V3 {
264								relay_parent: request.relay_parent,
265								scheduling_parent: request.scheduling_parent,
266							}))
267						},
268						CandidateDescriptorVersion::V1 |
269						CandidateDescriptorVersion::V2 |
270						CandidateDescriptorVersion::Unknown(_) => {
271							// V1/V2/Unknown - no extension appended
272							TrailingOption(None)
273						},
274					};
275				encoded_params.extend(extension.encode());
276
277				let params = Arc::new(encoded_params);
278
279				cfg_if::cfg_if! {
280					if #[cfg(target_os = "linux")] {
281						let result = if security_status.can_do_secure_clone {
282							handle_clone(
283								pipe_write_fd,
284								pipe_read_fd,
285								stream_fd,
286								&compiled_artifact_blob,
287								&executor_params,
288								&params,
289								execution_timeout,
290								execute_thread_stack_size,
291								worker_info,
292								security_status.can_unshare_user_namespace_and_change_root,
293								usage_before,
294								pov_size,
295							)?
296						} else {
297							// Fall back to using fork.
298							handle_fork(
299								pipe_write_fd,
300								pipe_read_fd,
301								stream_fd,
302								&compiled_artifact_blob,
303								&executor_params,
304								&params,
305								execution_timeout,
306								execute_thread_stack_size,
307								worker_info,
308								usage_before,
309								pov_size,
310							)?
311						};
312					} else {
313						let result = handle_fork(
314							pipe_write_fd,
315							pipe_read_fd,
316							stream_fd,
317							&compiled_artifact_blob,
318							&executor_params,
319							&params,
320							execution_timeout,
321							execute_thread_stack_size,
322							worker_info,
323							usage_before,
324							pov_size,
325						)?;
326					}
327				}
328
329				gum::trace!(
330					target: LOG_TARGET,
331					?worker_info,
332					"worker: sending result to host: {:?}",
333					result
334				);
335				send_result(&mut stream, result, worker_info)?;
336			}
337		},
338	);
339}

341fn validate_using_artifact(
342	compiled_artifact_blob: &[u8],
343	executor_params: &ExecutorParams,
344	params: &[u8],
345) -> JobResponse {
346	let descriptor_bytes = match unsafe {
347		// SAFETY: this should be safe since the compiled artifact passed here comes from the
348		//         file created by the prepare workers. These files are obtained by calling
349		//         [`executor_interface::prepare`].
350		execute_artifact(compiled_artifact_blob, executor_params, params)
351	} {
352		Err(ExecuteError::RuntimeConstruction(wasmerr)) => {
353			return JobResponse::runtime_construction("execute", &wasmerr.to_string())
354		},
355		Err(err) => return JobResponse::format_invalid("execute", &err.to_string()),
356		Ok(d) => d,
357	};
358
359	let result_descriptor = match ValidationResult::decode(&mut &descriptor_bytes[..]) {
360		Err(err) => {
361			return JobResponse::format_invalid(
362				"validation result decoding failed",
363				&err.to_string(),
364			)
365		},
366		Ok(r) => r,
367	};
368
369	JobResponse::Ok { result_descriptor }
370}

372#[cfg(target_os = "linux")]
373fn handle_clone(
374	pipe_write_fd: i32,
375	pipe_read_fd: i32,
376	stream_fd: i32,
377	compiled_artifact_blob: &Arc<Vec<u8>>,
378	executor_params: &Arc<ExecutorParams>,
379	params: &Arc<Vec<u8>>,
380	execution_timeout: Duration,
381	execute_stack_size: usize,
382	worker_info: &WorkerInfo,
383	have_unshare_newuser: bool,
384	usage_before: Usage,
385	pov_size: u32,
386) -> io::Result<Result<WorkerResponse, WorkerError>> {
387	use polkadot_node_core_pvf_common::worker::security;
388
389	// SAFETY: new process is spawned within a single threaded process. This invariant
390	// is enforced by tests. Stack size being specified to ensure child doesn't overflow
391	match unsafe {
392		security::clone::clone_on_worker(
393			worker_info,
394			have_unshare_newuser,
395			Box::new(|| {
396				handle_child_process(
397					pipe_write_fd,
398					pipe_read_fd,
399					stream_fd,
400					Arc::clone(compiled_artifact_blob),
401					Arc::clone(executor_params),
402					Arc::clone(params),
403					execution_timeout,
404					execute_stack_size,
405				)
406			}),
407		)
408	} {
409		Ok(child) => handle_parent_process(
410			pipe_read_fd,
411			pipe_write_fd,
412			worker_info,
413			child,
414			usage_before,
415			pov_size,
416			execution_timeout,
417		),
418		Err(security::clone::Error::Clone(errno)) => {
419			Ok(Err(internal_error_from_errno("clone", errno)))
420		},
421	}
422}

424fn handle_fork(
425	pipe_write_fd: i32,
426	pipe_read_fd: i32,
427	stream_fd: i32,
428	compiled_artifact_blob: &Arc<Vec<u8>>,
429	executor_params: &Arc<ExecutorParams>,
430	params: &Arc<Vec<u8>>,
431	execution_timeout: Duration,
432	execute_worker_stack_size: usize,
433	worker_info: &WorkerInfo,
434	usage_before: Usage,
435	pov_size: u32,
436) -> io::Result<Result<WorkerResponse, WorkerError>> {
437	// SAFETY: new process is spawned within a single threaded process. This invariant
438	// is enforced by tests.
439	match unsafe { nix::unistd::fork() } {
440		Ok(ForkResult::Child) => handle_child_process(
441			pipe_write_fd,
442			pipe_read_fd,
443			stream_fd,
444			Arc::clone(compiled_artifact_blob),
445			Arc::clone(executor_params),
446			Arc::clone(params),
447			execution_timeout,
448			execute_worker_stack_size,
449		),
450		Ok(ForkResult::Parent { child }) => handle_parent_process(
451			pipe_read_fd,
452			pipe_write_fd,
453			worker_info,
454			child,
455			usage_before,
456			pov_size,
457			execution_timeout,
458		),
459		Err(errno) => Ok(Err(internal_error_from_errno("fork", errno))),
460	}
461}

463/// This is used to handle child process during pvf execute worker.
464/// It executes the artifact and pipes back the response to the parent process.
465///
466/// # Returns
467///
468/// - pipe back `JobResponse` to the parent process.
469fn handle_child_process(
470	pipe_write_fd: i32,
471	pipe_read_fd: i32,
472	stream_fd: i32,
473	compiled_artifact_blob: Arc<Vec<u8>>,
474	executor_params: Arc<ExecutorParams>,
475	params: Arc<Vec<u8>>,
476	execution_timeout: Duration,
477	execute_thread_stack_size: usize,
478) -> ! {
479	// SAFETY: this is an open and owned file descriptor at this point.
480	let mut pipe_write = unsafe { PipeFd::from_raw_fd(pipe_write_fd) };
481
482	// Drop the read end so we don't have too many FDs open.
483	if let Err(errno) = nix::unistd::close(pipe_read_fd) {
484		send_child_response(&mut pipe_write, job_error_from_errno("closing pipe", errno));
485	}
486
487	// Dropping the stream closes the underlying socket. We want to make sure
488	// that the sandboxed child can't get any kind of information from the
489	// outside world. The only IPC it should be able to do is sending its
490	// response over the pipe.
491	if let Err(errno) = nix::unistd::close(stream_fd) {
492		send_child_response(&mut pipe_write, job_error_from_errno("closing stream", errno));
493	}
494
495	gum::debug!(
496		target: LOG_TARGET,
497		worker_job_pid = %process::id(),
498		"worker job: executing artifact",
499	);
500
501	// Conditional variable to notify us when a thread is done.
502	let condvar = thread::get_condvar();
503	let cpu_time_start = ProcessTime::now();
504
505	// Spawn a new thread that runs the CPU time monitor.
506	let (cpu_time_monitor_tx, cpu_time_monitor_rx) = channel::<()>();
507	let cpu_time_monitor_thread = thread::spawn_worker_thread(
508		"cpu time monitor thread",
509		move || cpu_time_monitor_loop(cpu_time_start, execution_timeout, cpu_time_monitor_rx),
510		Arc::clone(&condvar),
511		WaitOutcome::TimedOut,
512	)
513	.unwrap_or_else(|err| {
514		send_child_response(&mut pipe_write, Err(JobError::CouldNotSpawnThread(err.to_string())))
515	});
516
517	let execute_thread = thread::spawn_worker_thread_with_stack_size(
518		"execute thread",
519		move || validate_using_artifact(&compiled_artifact_blob, &executor_params, &params),
520		Arc::clone(&condvar),
521		WaitOutcome::Finished,
522		execute_thread_stack_size,
523	)
524	.unwrap_or_else(|err| {
525		send_child_response(&mut pipe_write, Err(JobError::CouldNotSpawnThread(err.to_string())))
526	});
527
528	let outcome = thread::wait_for_threads(condvar);
529
530	let response = match outcome {
531		WaitOutcome::Finished => {
532			let _ = cpu_time_monitor_tx.send(());
533			execute_thread.join().map_err(|e| JobError::Panic(stringify_panic_payload(e)))
534		},
535		// If the CPU thread is not selected, we signal it to end, the join handle is
536		// dropped and the thread will finish in the background.
537		WaitOutcome::TimedOut => match cpu_time_monitor_thread.join() {
538			Ok(Some(_cpu_time_elapsed)) => Err(JobError::TimedOut),
539			Ok(None) => Err(JobError::CpuTimeMonitorThread(
540				"error communicating over finished channel".into(),
541			)),
542			Err(e) => Err(JobError::CpuTimeMonitorThread(stringify_panic_payload(e))),
543		},
544		WaitOutcome::Pending => {
545			unreachable!("we run wait_while until the outcome is no longer pending; qed")
546		},
547	};
548
549	send_child_response(&mut pipe_write, response);
550}

552/// Returns stack size based on the number of threads.
553/// The stack size is represented by 2MiB * number_of_threads + native stack;
554///
555/// # Background
556///
557/// Wasmtime powers the Substrate Executor. It compiles the wasm bytecode into native code.
558/// That native code does not create any stacks and just reuses the stack of the thread that
559/// wasmtime was invoked from.
560///
561/// Also, we configure the executor to provide the deterministic stack and that requires
562/// supplying the amount of the native stack space that wasm is allowed to use. This is
563/// realized by supplying the limit into `wasmtime::Config::max_wasm_stack`.
564///
565/// There are quirks to that configuration knob:
566///
567/// 1. It only limits the amount of stack space consumed by wasm but does not ensure nor check that
568///    the stack space is actually available.
569///
570///    That means, if the calling thread has 1 MiB of stack space left and the wasm code consumes
571///    more, then the wasmtime limit will **not** trigger. Instead, the wasm code will hit the
572///    guard page and the Rust stack overflow handler will be triggered. That leads to an
573///    **abort**.
574///
575/// 2. It cannot and does not limit the stack space consumed by Rust code.
576///
577///    Meaning that if the wasm code leaves no stack space for Rust code, then the Rust code
578///    will abort and that will abort the process as well.
579///
580/// Typically on Linux the main thread gets the stack size specified by the `ulimit` and
581/// typically it's configured to 8 MiB. Rust's spawned threads are 2 MiB. OTOH, the
582/// DEFAULT_NATIVE_STACK_MAX is set to 256 MiB. Not nearly enough.
583///
584/// Hence we need to increase it. The simplest way to fix that is to spawn an execute thread with
585/// the desired stack limit. We must also make sure the job process has enough stack for *all* its
586/// threads. This function can be used to get the stack size of either the execute thread or execute
587/// job process.
588fn max_stack_size(executor_params: &ExecutorParams) -> usize {
589	let (_sem, deterministic_stack_limit) = params_to_wasmtime_semantics(executor_params);
590	return (2 * 1024 * 1024 + deterministic_stack_limit.native_stack_max) as usize;
591}

593/// Waits for child process to finish and handle child response from pipe.
594///
595/// # Returns
596///
597/// - The response, either `Ok` or some error state.
598fn handle_parent_process(
599	pipe_read_fd: i32,
600	pipe_write_fd: i32,
601	worker_info: &WorkerInfo,
602	job_pid: Pid,
603	usage_before: Usage,
604	pov_size: u32,
605	timeout: Duration,
606) -> io::Result<Result<WorkerResponse, WorkerError>> {
607	// the read end will wait until all write ends have been closed,
608	// this drop is necessary to avoid deadlock
609	if let Err(errno) = nix::unistd::close(pipe_write_fd) {
610		return Ok(Err(internal_error_from_errno("closing pipe write fd", errno)));
611	};
612
613	// SAFETY: pipe_read_fd is an open and owned file descriptor at this point.
614	let mut pipe_read = unsafe { PipeFd::from_raw_fd(pipe_read_fd) };
615
616	// Read from the child. Don't decode unless the process exited normally, which we check later.
617	let mut received_data = Vec::new();
618	pipe_read
619		.read_to_end(&mut received_data)
620		// Could not decode job response. There is either a bug or the job was hijacked.
621		// Should retry at any rate.
622		.map_err(|err| io::Error::new(io::ErrorKind::Other, err.to_string()))?;
623
624	let status = nix::sys::wait::waitpid(job_pid, None);
625	gum::trace!(
626		target: LOG_TARGET,
627		?worker_info,
628		%job_pid,
629		"execute worker received wait status from job: {:?}",
630		status,
631	);
632
633	let usage_after = match nix::sys::resource::getrusage(UsageWho::RUSAGE_CHILDREN) {
634		Ok(usage) => usage,
635		Err(errno) => return Ok(Err(internal_error_from_errno("getrusage after", errno))),
636	};
637
638	// Using `getrusage` is needed to check whether child has timedout since we cannot rely on
639	// child to report its own time.
640	// As `getrusage` returns resource usage from all terminated child processes,
641	// it is necessary to subtract the usage before the current child process to isolate its cpu
642	// time
643	let cpu_tv = get_total_cpu_usage(usage_after) - get_total_cpu_usage(usage_before);
644	if cpu_tv >= timeout {
645		gum::warn!(
646			target: LOG_TARGET,
647			?worker_info,
648			%job_pid,
649			"execute job took {}ms cpu time, exceeded execute timeout {}ms",
650			cpu_tv.as_millis(),
651			timeout.as_millis(),
652		);
653		return Ok(Err(WorkerError::JobTimedOut));
654	}
655
656	match status {
657		Ok(WaitStatus::Exited(_, exit_status)) => {
658			let mut reader = io::BufReader::new(received_data.as_slice());
659			let result = recv_child_response(&mut reader, "execute")?;
660
661			match result {
662				Ok(job_response) => {
663					// The exit status should have been zero if no error occurred.
664					if exit_status != 0 {
665						return Ok(Err(WorkerError::JobError(JobError::UnexpectedExitStatus(
666							exit_status,
667						))));
668					}
669
670					Ok(Ok(WorkerResponse { job_response, pov_size, duration: cpu_tv }))
671				},
672				Err(job_error) => {
673					gum::warn!(
674						target: LOG_TARGET,
675						?worker_info,
676						%job_pid,
677						"execute job error: {}",
678						job_error,
679					);
680					if matches!(job_error, JobError::TimedOut) {
681						Ok(Err(WorkerError::JobTimedOut))
682					} else {
683						Ok(Err(WorkerError::JobError(job_error.into())))
684					}
685				},
686			}
687		},
688		// The job was killed by the given signal.
689		//
690		// The job gets SIGSYS on seccomp violations, but this signal may have been sent for some
691		// other reason, so we still need to check for seccomp violations elsewhere.
692		Ok(WaitStatus::Signaled(_pid, signal, _core_dump)) => Ok(Err(WorkerError::JobDied {
693			err: format!("received signal: {signal:?}"),
694			job_pid: job_pid.as_raw(),
695		})),
696		Err(errno) => Ok(Err(internal_error_from_errno("waitpid", errno))),
697
698		// It is within an attacker's power to send an unexpected exit status. So we cannot treat
699		// this as an internal error (which would make us abstain), but must vote against.
700		Ok(unexpected_wait_status) => Ok(Err(WorkerError::JobDied {
701			err: format!("unexpected status from wait: {unexpected_wait_status:?}"),
702			job_pid: job_pid.as_raw(),
703		})),
704	}
705}

707/// Write a job response to the pipe and exit process after.
708///
709/// # Arguments
710///
711/// - `pipe_write`: A `PipeFd` structure, the writing end of a pipe.
712///
713/// - `response`: Child process response
714fn send_child_response(pipe_write: &mut PipeFd, response: JobResult) -> ! {
715	framed_send_blocking(pipe_write, response.encode().as_slice())
716		.unwrap_or_else(|_| process::exit(libc::EXIT_FAILURE));
717
718	if response.is_ok() {
719		process::exit(libc::EXIT_SUCCESS)
720	} else {
721		process::exit(libc::EXIT_FAILURE)
722	}
723}

725fn internal_error_from_errno(context: &'static str, errno: Errno) -> WorkerError {
726	WorkerError::InternalError(InternalValidationError::Kernel(stringify_errno(context, errno)))
727}

729fn job_error_from_errno(context: &'static str, errno: Errno) -> JobResult {
730	Err(JobError::Kernel(stringify_errno(context, errno)))
