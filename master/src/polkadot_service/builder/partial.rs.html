91pub(crate) fn new_partial_basics(
92	config: &mut Configuration,
93	telemetry_worker_handle: Option<TelemetryWorkerHandle>,
94) -> Result<Basics, Error> {
95	let telemetry = config
96		.telemetry_endpoints
97		.clone()
98		.filter(|x| !x.is_empty())
99		.map(move |endpoints| -> Result<_, sc_telemetry::Error> {
100			let (worker, mut worker_handle) = if let Some(worker_handle) = telemetry_worker_handle {
101				(None, worker_handle)
102			} else {
103				let worker = TelemetryWorker::new(16)?;
104				let worker_handle = worker.handle();
105				(Some(worker), worker_handle)
106			};
107			let telemetry = worker_handle.new_telemetry(endpoints);
108			Ok((worker, telemetry))
109		})
110		.transpose()?;
111
112	let heap_pages = config
113		.executor
114		.default_heap_pages
115		.map_or(DEFAULT_HEAP_ALLOC_STRATEGY, |h| HeapAllocStrategy::Static { extra_pages: h as _ });
116
117	let executor = WasmExecutor::builder()
118		.with_execution_method(config.executor.wasm_method)
119		.with_onchain_heap_alloc_strategy(heap_pages)
120		.with_offchain_heap_alloc_strategy(heap_pages)
121		.with_max_runtime_instances(config.executor.max_runtime_instances)
122		.with_runtime_cache_size(config.executor.runtime_cache_size)
123		.build();
124
125	// Use GrandpaPruningFilter to preserve blocks with GRANDPA justifications during
126	// pruning. This is required for warp sync to work on pruned nodes.
127	let (client, backend, keystore_container, task_manager) = sc_service::new_full_parts(
128		&config,
129		telemetry.as_ref().map(|(_, telemetry)| telemetry.handle()),
130		executor,
131		vec![Arc::new(GrandpaPruningFilter)],
132	)?;
133	let client = Arc::new(client);
134
135	let telemetry = telemetry.map(|(worker, telemetry)| {
136		if let Some(worker) = worker {
137			task_manager.spawn_handle().spawn(
138				"telemetry",
139				Some("telemetry"),
140				Box::pin(worker.run()),
141			);
142		}
143		telemetry
144	});
145
146	Ok(Basics { task_manager, client, backend, keystore_container, telemetry })
147}

149pub(crate) fn new_partial<ChainSelection>(
150	config: &mut Configuration,
151	Basics { task_manager, backend, client, keystore_container, telemetry }: Basics,
152	select_chain: ChainSelection,
153) -> Result<PolkadotPartialComponents<ChainSelection>, Error>
154where
155	ChainSelection: 'static + SelectChain<Block>,
156{
157	let transaction_pool = Arc::from(
158		sc_transaction_pool::Builder::new(
159			task_manager.spawn_essential_handle(),
160			client.clone(),
161			config.role.is_authority().into(),
162		)
163		.with_options(config.transaction_pool.clone())
164		.with_prometheus(config.prometheus_registry())
165		.build(),
166	);
167
168	let grandpa_hard_forks = if config.chain_spec.is_kusama() {
169		grandpa_support::kusama_hard_forks()
170	} else {
171		Vec::new()
172	};
173
174	let (grandpa_block_import, grandpa_link) =
175		sc_consensus_grandpa::block_import_with_authority_set_hard_forks(
176			client.clone(),
177			GRANDPA_JUSTIFICATION_PERIOD,
178			&client.clone(),
179			select_chain.clone(),
180			grandpa_hard_forks,
181			telemetry.as_ref().map(|x| x.handle()),
182		)?;
183	let justification_import = grandpa_block_import.clone();
184
185	let (beefy_block_import, beefy_voter_links, beefy_rpc_links) =
186		sc_consensus_beefy::beefy_block_import_and_links(
187			grandpa_block_import,
188			backend.clone(),
189			client.clone(),
190			config.prometheus_registry().cloned(),
191		);
192
193	let babe_config = sc_consensus_babe::configuration(&*client)?;
194	let slot_duration = babe_config.slot_duration();
195	let (block_import, babe_link) = sc_consensus_babe::block_import(
196		babe_config.clone(),
197		beefy_block_import,
198		client.clone(),
199		Arc::new(move |_, _| async move {
200			let timestamp = sp_timestamp::InherentDataProvider::from_system_time();
201			let slot = sp_consensus_babe::inherents::InherentDataProvider::from_timestamp_and_slot_duration(
202				*timestamp,
203				slot_duration,
204			);
205			Ok((slot, timestamp))
206		}) as BabeCreateInherentDataProviders<Block>,
207		select_chain.clone(),
208		OffchainTransactionPoolFactory::new(transaction_pool.clone()),
209	)?;
210
211	let (import_queue, babe_worker_handle) =
212		sc_consensus_babe::import_queue(sc_consensus_babe::ImportQueueParams {
213			link: babe_link.clone(),
214			block_import: block_import.clone(),
215			justification_import: Some(Box::new(justification_import)),
216			client: client.clone(),
217			slot_duration,
218			spawner: &task_manager.spawn_essential_handle(),
219			registry: config.prometheus_registry(),
220			telemetry: telemetry.as_ref().map(|x| x.handle()),
221		})?;
222
223	let justification_stream = grandpa_link.justification_stream();
224	let shared_authority_set = grandpa_link.shared_authority_set().clone();
225	let shared_voter_state = sc_consensus_grandpa::SharedVoterState::empty();
226	let finality_proof_provider = GrandpaFinalityProofProvider::new_for_service(
227		backend.clone(),
228		Some(shared_authority_set.clone()),
229	);
230
231	let import_setup = (block_import, grandpa_link, babe_link, beefy_voter_links);
232	let rpc_setup = shared_voter_state.clone();
233
234	let rpc_extensions_builder = {
235		let client = client.clone();
236		let keystore = keystore_container.keystore();
237		let transaction_pool = transaction_pool.clone();
238		let select_chain = select_chain.clone();
239		let chain_spec = config.chain_spec.cloned_box();
240		let backend = backend.clone();
241
242		move |subscription_executor: polkadot_rpc::SubscriptionTaskExecutor|
243		      -> Result<polkadot_rpc::RpcExtension, sc_service::Error> {
244			let deps = polkadot_rpc::FullDeps {
245				client: client.clone(),
246				pool: transaction_pool.clone(),
247				select_chain: select_chain.clone(),
248				chain_spec: chain_spec.cloned_box(),
249				babe: polkadot_rpc::BabeDeps {
250					babe_worker_handle: babe_worker_handle.clone(),
251					keystore: keystore.clone(),
252				},
253				grandpa: polkadot_rpc::GrandpaDeps {
254					shared_voter_state: shared_voter_state.clone(),
255					shared_authority_set: shared_authority_set.clone(),
256					justification_stream: justification_stream.clone(),
257					subscription_executor: subscription_executor.clone(),
258					finality_provider: finality_proof_provider.clone(),
259				},
260				beefy: polkadot_rpc::BeefyDeps::<ecdsa_crypto::AuthorityId> {
261					beefy_finality_proof_stream: beefy_rpc_links.from_voter_justif_stream.clone(),
262					beefy_best_block_stream: beefy_rpc_links.from_voter_best_beefy_stream.clone(),
263					subscription_executor,
264				},
265				backend: backend.clone(),
266			};
267
268			polkadot_rpc::create_full(deps).map_err(Into::into)
269		}
270	};
271
272	Ok(sc_service::PartialComponents {
273		client,
274		backend,
275		task_manager,
276		keystore_container,
277		select_chain,
278		import_queue,
279		transaction_pool,
280		other: (
281			Box::new(rpc_extensions_builder),
282			import_setup,
283			rpc_setup,
284			slot_duration,
285			telemetry,
286		),
287	})
