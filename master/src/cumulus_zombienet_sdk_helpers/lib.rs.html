36/// Specifies which block should occupy a full core.
37#[derive(Debug, Clone, PartialEq, Eq)]
38pub enum BlockToCheck {
39	/// The exact block hash provided should occupy a full core.
40	Exact(H256),
41	/// Wait for the next first bundle block.
42	NextFirstBundleBlock(H256),
43}

45// Maximum number of blocks to wait for a session change.
46// If it does not arrive for whatever reason, we should not wait forever.
47const WAIT_MAX_BLOCKS_FOR_SESSION: u32 = 50;
48
49// Maximum time to wait for PVF preparation to conclude on a validator before
50// starting throughput measurement. PVF preparation is a one-off ~20s wasm
51// compile per validator that contends for CPU. The clock starts at the session
52// change, but the first parachain candidate (which triggers PVF preparation)
53// can be delayed by several minutes of collator warm-up, which we must account for.
54const PVF_PREPARE_TIMEOUT_SECS: u64 = 300;

56/// Format a `sp_runtime::DispatchError` using runtime metadata for human-readable output.
57///
58/// For module errors this resolves the pallet index and error index to their names
59/// (e.g. `ParachainSystem::TooBig`) instead of showing raw bytes.
60fn format_dispatch_error(err: &sp_runtime::DispatchError, metadata: &Metadata) -> String {
61	match err {
62		sp_runtime::DispatchError::Module(module_err) => {
63			let pallet = metadata.pallet_by_index(module_err.index);
64			let pallet_name = pallet.as_ref().map(|p| p.name()).unwrap_or("UnknownPallet");
65			let error_name = pallet
66				.and_then(|p| p.error_variant_by_index(module_err.error[0]))
67				.map(|v| v.name.as_str())
68				.unwrap_or("UnknownError");
69			format!("{pallet_name}::{error_name}")
70		},
71		other => format!("{other:?}"),
72	}
73}

75/// Find an event in subxt `Events` and attempt to decode the fields of the event.
76fn find_event_and_decode_fields<T: Decode>(
77	events: &Events<PolkadotConfig>,
78	pallet: &str,
79	variant: &str,
80) -> Result<Vec<T>, anyhow::Error> {
81	let mut result = vec![];
82	for event in events.iter() {
83		let event = event?;
84		if event.pallet_name() == pallet && event.variant_name() == variant {
85			result.push(T::decode(&mut &event.field_bytes()[..])?);
86		}
87	}
88	Ok(result)
89}
90/// Returns `true` if the `block` is a session change.
91async fn is_session_change(
92	block: &Block<PolkadotConfig, OnlineClient<PolkadotConfig>>,
93) -> Result<bool, anyhow::Error> {
94	let events = block.events().await?;
95	Ok(events.iter().any(|event| {
96		event.as_ref().is_ok_and(|event| {
97			event.pallet_name() == "Session" && event.variant_name() == "NewSession"
98		})
99	}))
100}

102// Helper function for asserting the throughput of parachains, after the first session change.
103//
104// The throughput is measured as total number of backed candidates in a window of `stop_after` relay
105// chain blocks. The counting window starts from the relay chain block after the first one that
106// contains a backed candidate for a tracked para. Relay chain blocks with session changes are
107// generally ignored, but it is ensured that no blocks are build on top of these relay blocks.
108//
109// For tests where PVF preparation timing affects throughput (e.g. elastic scaling, runtime
110// upgrades), call [`wait_for_pvf_prepare`] before this helper to ensure all validators have
111// finished preparing the relevant PVFs.
112pub async fn assert_para_throughput(
113	relay_client: &OnlineClient<PolkadotConfig>,
114	stop_after: u32,
115	expected_candidate_ranges: impl Into<HashMap<ParaId, Range<u32>>>,
116	expected_number_of_blocks: impl Into<HashMap<ParaId, (OnlineClient<PolkadotConfig>, Range<u32>)>>,
117) -> Result<(), anyhow::Error> {
118	let ranges = expected_candidate_ranges.into();
119	let expected_number_of_blocks = expected_number_of_blocks.into();
120
121	let candidate_count =
122		collect_para_throughput(relay_client, stop_after, ranges, |_| Ok(true)).await?;
123
124	assert_expected_number_of_blocks(candidate_count, expected_number_of_blocks).await
125}

127/// Like [`assert_para_throughput`], but accepts a closure to validate each backed candidate
128/// receipt.
129///
130/// The closure receives each [`CandidateReceiptV2`] and should return:
131/// - `Ok(true)` to count the candidate,
132/// - `Ok(false)` to skip it,
133/// - `Err(e)` to fail immediately.
134///
135/// Only receipts for para IDs present in `expected_candidate_ranges` are passed to the closure.
136pub async fn assert_para_throughput_with<F>(
137	relay_client: &OnlineClient<PolkadotConfig>,
138	stop_after: u32,
139	expected_candidate_ranges: impl Into<HashMap<ParaId, Range<u32>>>,
140	validate: F,
141) -> Result<(), anyhow::Error>
142where
143	F: Fn(&CandidateReceiptV2<H256>) -> Result<bool, anyhow::Error>,
144{
145	collect_para_throughput(relay_client, stop_after, expected_candidate_ranges, validate)
146		.await
147		.map(|_| ())
148}

150/// Waits until every relaychain validator in `network` reports
151/// `polkadot_pvf_prepare_concluded >= min_total_prepares`.
152///
153/// Use this before [`assert_para_throughput`] in tests where PVF preparation timing demonstrably
154/// affects throughput — typically elastic-scaling tests (high core count, validators bound on
155/// PVF-compile CPU) and tests that measure throughput across a parachain runtime upgrade (which
156/// triggers re-preparation of the new PVF).
157///
158/// `min_total_prepares` is the absolute minimum value the metric must reach, in cumulative
159/// concluded prepare jobs since validator startup. For one round of preparation (one PVF per
160/// tracked parachain), pass `tracked_paras as u32`. For a measurement after a runtime upgrade,
161/// pass `2 * tracked_paras as u32`. Caller is responsible for tracking the round — we
162/// deliberately do not read a baseline from the metric and add a delta, because validators
163/// may already have started or finished preparing a new PVF before the baseline read, which
164/// makes the delta racy.
165pub async fn wait_for_pvf_prepare(
166	network: &Network<LocalFileSystem>,
167	min_total_prepares: u32,
168) -> Result<(), anyhow::Error> {
169	let validators = network.relaychain().nodes();
170	let target = min_total_prepares as f64;
171	log::info!(
172		"Waiting for PVF preparation to conclude on {} validator(s) (target {} concluded job(s) per validator).",
173		validators.len(),
174		target,
175	);
176	for node in &validators {
177		let node_name = node.name();
178		log::info!("Waiting for {node_name} PVF prep (target={target})...");
179		node.wait_metric_with_timeout(
180			"polkadot_pvf_prepare_concluded",
181			|c| c >= target,
182			PVF_PREPARE_TIMEOUT_SECS,
183		)
184		.await
185		.map_err(|e| anyhow!("{node_name}: PVF prepare did not conclude within timeout: {e}"))?;
186	}
187	log::info!(
188		"All {} validator(s) have prepared PVF artifacts (target {})",
189		validators.len(),
190		target,
191	);
192	Ok(())
193}

195async fn collect_para_throughput<F>(
196	relay_client: &OnlineClient<PolkadotConfig>,
197	stop_after: u32,
198	expected_candidate_ranges: impl Into<HashMap<ParaId, Range<u32>>>,
199	validate: F,
200) -> Result<HashMap<ParaId, Vec<CandidateReceiptV2<H256>>>, anyhow::Error>
201where
202	F: Fn(&CandidateReceiptV2<H256>) -> Result<bool, anyhow::Error>,
203{
204	let mut blocks_sub = relay_client.blocks().subscribe_finalized().await?;
205	let mut candidate_count: HashMap<ParaId, Vec<CandidateReceiptV2<H256>>> = HashMap::new();
206	let mut current_block_count = 0;
207
208	let expected_candidate_ranges = expected_candidate_ranges.into();
209	let valid_para_ids: Vec<ParaId> = expected_candidate_ranges.keys().cloned().collect();
210
211	log::info!(
212		"Asserting parachain throughput for para_ids: {:?}. Wait for the first session change",
213		valid_para_ids
214	);
215	// Wait for the first session, block production on the parachain will start after that.
216	wait_for_first_session_change(&mut blocks_sub).await?;
217	log::info!(
218		"First session change detected. Waiting for backed candidates from all tracked paras before counting."
219	);
220
221	let mut paras_seen = std::collections::HashSet::new();
222	while let Some(block) = blocks_sub.next().await {
223		let block = block?;
224		log::debug!("Finalized relay chain block {}", block.number());
225
226		// Do not count blocks with session changes, no backed blocks there.
227		if is_session_change(&block).await? {
228			continue;
229		}
230
231		let events = block.events().await?;
232		let receipts = find_event_and_decode_fields::<CandidateReceiptV2<H256>>(
233			&events,
234			"ParaInclusion",
235			"CandidateBacked",
236		)?;
237
238		// Skip relay chain blocks until every tracked para has had at least one backed candidate.
239		// This avoids counting the initial warm-up period where the backing pipeline (PVF
240		// compilation, first collation) hasn't reached steady state yet.
241		for receipt in &receipts {
242			let para_id = receipt.descriptor.para_id();
243			if valid_para_ids.contains(&para_id) {
244				paras_seen.insert(para_id);
245			}
246		}
247		if paras_seen.len() != valid_para_ids.len() {
248			log::info!(
249				"Not all tracked paras have produced candidates by relay block {}. \
250				Not counting blocks yet.",
251				block.number()
252			);
253			continue;
254		}
255
256		current_block_count += 1;
257
258		for receipt in receipts {
259			let para_id = receipt.descriptor.para_id();
260			log::debug!("Block backed for para_id {para_id}");
261
262			if !valid_para_ids.contains(&para_id) {
263				continue;
264			}
265
266			if !validate(&receipt)? {
267				continue;
268			}
269
270			candidate_count.entry(para_id).or_default().push(receipt);
271		}
272
273		if current_block_count == stop_after {
274			break;
275		}
276	}
277
278	log::info!(
279		"Reached {stop_after} finalized relay chain blocks that contain backed candidates. The per-parachain distribution is: {:#?}",
280		candidate_count.iter().map(|(para_id, receipts)| format!("{para_id} has {} backed candidates", receipts.len())).collect::<Vec<_>>()
281	);
282
283	for (para_id, expected_candidate_range) in expected_candidate_ranges {
284		let actual = candidate_count
285			.get(&para_id)
286			.ok_or_else(|| anyhow!("ParaId {para_id} did not have any backed candidates"))?
287			.len() as u32;
288
289		if !expected_candidate_range.contains(&actual) {
290			return Err(anyhow!(
291				"ParaId {para_id}: candidate count {actual} not within expected range {expected_candidate_range:?}"
292			));
293		}
294	}
295
296	Ok(candidate_count)
297}

299async fn assert_expected_number_of_blocks(
300	candidate_count: HashMap<ParaId, Vec<CandidateReceiptV2<H256>>>,
301	expected_number_of_blocks: HashMap<ParaId, (OnlineClient<PolkadotConfig>, Range<u32>)>,
302) -> Result<(), anyhow::Error> {
303	for (para_id, (para_client, expected_number_of_blocks)) in expected_number_of_blocks {
304		let receipts = candidate_count
305			.get(&para_id)
306			.ok_or_else(|| anyhow!("ParaId did not have any backed candidates"))?;
307
308		let mut num_blocks = 0;
309
310		for receipt in receipts {
311			// We "abuse" the fact that the parachain is using `BlakeTwo256` as hash and thus, the
312			// `para_head` hash and the hash of the `header` should be equal.
313			let mut next_para_block_hash = receipt.descriptor().para_head();
314
315			let mut relay_identifier = None;
316			let mut core_info = None;
317
318			loop {
319				let block: Block<PolkadotConfig, OnlineClient<PolkadotConfig>> =
320					para_client.blocks().at(next_para_block_hash).await?;
321
322				// Genesis block is not part of a candidate :)
323				if block.number() == 0 {
324					break;
325				}
326
327				let ri = find_relay_block_identifier(&block)?;
328				let ci = find_core_info(&block)?;
329
330				// If the core changes or the relay identifier, we found all blocks for the
331				// candidate.
332				if *relay_identifier.get_or_insert(ri.clone()) != ri ||
333					*core_info.get_or_insert(ci.clone()) != ci
334				{
335					break;
336				}
337
338				num_blocks += 1;
339				next_para_block_hash = block.header().parent_hash;
340			}
341		}
342
343		if !expected_number_of_blocks.contains(&num_blocks) {
344			return Err(anyhow!(
345				"Block number count {num_blocks} not within range {expected_number_of_blocks:?}",
346			));
347		}
348	}
349
350	Ok(())
351}

388/// Wait for the first block with the Nth session change.
389///
390/// The session change is detected by inspecting the events in the block.
391pub async fn wait_for_nth_session_change(
392	blocks_sub: &mut zombienet_sdk::subxt::backend::StreamOfResults<
393		Block<PolkadotConfig, OnlineClient<PolkadotConfig>>,
394	>,
395	mut sessions_to_wait: u32,
396) -> Result<(), anyhow::Error> {
397	let mut waited_block_num = 0;
398	while let Some(block) = blocks_sub.next().await {
399		let block = block?;
400		log::debug!("Finalized relay chain block {}", block.number());
401
402		if is_session_change(&block).await? {
403			sessions_to_wait -= 1;
404			if sessions_to_wait == 0 {
405				return Ok(());
406			}
407
408			waited_block_num = 0;
409		} else {
410			if waited_block_num >= WAIT_MAX_BLOCKS_FOR_SESSION {
411				return Err(anyhow::format_err!("Waited for {WAIT_MAX_BLOCKS_FOR_SESSION}, a new session should have been arrived by now."));
412			}
413
414			waited_block_num += 1;
415		}
416	}
417	Ok(())
418}

420// Helper function that asserts the maximum finality lag.
421pub async fn assert_finality_lag(
422	client: &OnlineClient<PolkadotConfig>,
423	maximum_lag: u32,
424) -> Result<(), anyhow::Error> {
425	let mut best_stream = client.blocks().subscribe_best().await?;
426	let mut fut_stream = client.blocks().subscribe_finalized().await?;
427	let (Some(Ok(best)), Some(Ok(finalized))) = join!(best_stream.next(), fut_stream.next()) else {
428		return Err(anyhow::format_err!("Unable to fetch best an finalized block!"));
429	};
430	let finality_lag = best.number() - finalized.number();
431
432	log::info!(
433		"Finality lagged by {finality_lag} blocks, maximum expected was {maximum_lag} blocks"
434	);
435
436	assert!(finality_lag <= maximum_lag, "Expected finality to lag by a maximum of {maximum_lag} blocks, but was lagging by {finality_lag} blocks.");
437	Ok(())
438}

440/// Assert that finality has not stalled.
441pub async fn assert_blocks_are_being_finalized(
442	client: &OnlineClient<PolkadotConfig>,
443) -> Result<(), anyhow::Error> {
444	let sleep_duration = Duration::from_secs(12);
445	let mut finalized_blocks = client.blocks().subscribe_finalized().await?;
446	let first_measurement = finalized_blocks
447		.next()
448		.await
449		.ok_or(anyhow::anyhow!("Can't get finalized block from stream"))??
450		.number();
451	sleep(sleep_duration).await;
452	let second_measurement = finalized_blocks
453		.next()
454		.await
455		.ok_or(anyhow::anyhow!("Can't get finalized block from stream"))??
456		.number();
457
458	log::info!(
459		"Finalized {} blocks within {sleep_duration:?}",
460		second_measurement - first_measurement
461	);
462	assert!(second_measurement > first_measurement);
463
464	Ok(())
465}

467/// Checks if the given `RelayBlockIdentifier` matches a relay chain header.
468fn identifier_matches_header(
469	identifier: &RelayBlockIdentifier,
470	header: &<PolkadotConfig as Config>::Header,
471) -> bool {
472	match identifier {
473		RelayBlockIdentifier::ByHash(hash) => {
474			let header_hash = BlakeTwo256::hash(&header.encode());
475			header_hash == *hash
476		},
477		RelayBlockIdentifier::ByStorageRoot { storage_root, .. } => {
478			header.state_root == *storage_root
479		},
480	}
481}

483/// Asserts that parachain blocks have the correct relay parent offset. This also checks that the
484/// relay chain descendants do not contain any session changes.
485///
486/// # Arguments
487///
488/// * `relay_client` - Client connected to a relay chain node
489/// * `para_client` - Client connected to a parachain node
490/// * `offset` - Expected minimum offset between relay parent and highest seen relay block
491/// * `block_limit` - Number of parachain blocks to verify before completing
492pub async fn assert_relay_parent_offset(
493	relay_client: &OnlineClient<PolkadotConfig>,
494	para_client: &OnlineClient<PolkadotConfig>,
495	offset: u32,
496	block_limit: u32,
497) -> Result<(), anyhow::Error> {
498	let mut relay_block_stream = relay_client.blocks().subscribe_all().await?;
499
500	// First parachain header #0 does not contain relay block identifier digest item.
501	let mut para_block_stream = para_client.blocks().subscribe_all().await?.skip(1);
502	let mut highest_relay_block_seen = 0;
503	let mut num_para_blocks_seen = 0;
504	let mut forbidden_parents = Vec::new();
505	let mut seen_relay_parents = HashMap::new();
506	loop {
507		tokio::select! {
508			Some(Ok(relay_block)) = relay_block_stream.next() => {
509				highest_relay_block_seen = max(relay_block.number(), highest_relay_block_seen);
510				if highest_relay_block_seen > 15 && num_para_blocks_seen == 0 {
511					return Err(anyhow!("No parachain blocks produced!"))
512				}
513				// When a relay chain block contains a session change, parachains shall not build on
514				// any ancestor of that block, if the session change block is part of the descendants.
515				// Example:
516				// RC Chain: A -> B -> C -> D*
517				// "*" denotes session change
518				// In this scenario, parachains with an offset of 2 should never build on relay chain
519				// blocks B or C. Both of them would include the session change block D* in their
520				// descendants, and we know that the candidate would span a session boundary.
521				if is_session_change(&relay_block).await? {
522					log::debug!("RC block #{} contains session change, adding {offset} parents to forbidden list.", relay_block.number());
523					let mut current_hash = relay_block.header().parent_hash;
524					for _ in 0..offset {
525						let block = relay_client.blocks().at(current_hash).await.map_err(|_| anyhow!("Unable to fetch RC header."))?;
526						forbidden_parents.push(block.header().clone());
527						current_hash = block.header().parent_hash;
528					}
529				}
530			},
531			Some(Ok(para_block)) = para_block_stream.next() => {
532				let relay_block_identifier = find_relay_block_identifier(&para_block)?;
533
534				let relay_parent_number = match &relay_block_identifier {
535					RelayBlockIdentifier::ByHash(block_hash) => relay_client.blocks().at(*block_hash).await?.number(),
536					RelayBlockIdentifier::ByStorageRoot { block_number, .. } => *block_number,
537				};
538
539				let para_block_number = para_block.number();
540				seen_relay_parents.insert(relay_block_identifier.clone(), para_block);
541				log::debug!("Parachain block #{para_block_number} was built on relay parent #{relay_parent_number}, highest seen was {highest_relay_block_seen}");
542				assert!(
543					highest_relay_block_seen < offset ||
544					relay_parent_number <= highest_relay_block_seen.saturating_sub(offset),
545					"Relay parent is not at the correct offset! relay_parent: #{relay_parent_number} highest_seen_relay_block: #{highest_relay_block_seen}",
546				);
547				// As per explanation above, we need to check that no parachain blocks are built
548				// on the forbidden parents.
549				for forbidden in &forbidden_parents {
550					for (identifier, para_block) in &seen_relay_parents {
551						if identifier_matches_header(identifier, forbidden) {
552							panic!(
553								"Parachain block {} was built on forbidden relay parent with session change descendants ({:?})",
554								para_block.hash(),
555								identifier
556							);
557						}
558					}
559				}
560				num_para_blocks_seen += 1;
561				if num_para_blocks_seen >= block_limit {
562					log::info!("Successfully verified relay parent offset of {offset} for {num_para_blocks_seen} parachain blocks.");
563					break;
564				}
565			}
566		}
567	}
568
569	Ok(())
570}

572/// Submits the given `call` as signed transaction and waits for its successful finalization.
573///
574/// The transaction is sent as immortal transaction.
575pub async fn submit_extrinsic_and_wait_for_finalization_success<S: Signer<PolkadotConfig>>(
576	client: &OnlineClient<PolkadotConfig>,
577	call: &DynamicPayload,
578	signer: &S,

587}

599/// Submit the given transaction and wait for its finalization.
600async fn submit_tx_and_wait_for_finalization(
601	tx: SubmittableTransaction<PolkadotConfig, OnlineClient<PolkadotConfig>>,
602) -> Result<H256, anyhow::Error> {
603	log::info!("Submitting transaction: {:?}", tx.hash());
604
605	let mut tx = tx.submit_and_watch().await?;
606
607	while let Some(status) = tx.next().await.transpose()? {
608		match status {
609			TxStatus::InBestBlock(tx_in_block) => {
610				tx_in_block.wait_for_success().await?;
611				log::info!("[Best] In block: {:#?}", tx_in_block.block_hash());
612			},
613			TxStatus::InFinalizedBlock(ref tx_in_block) => {
614				tx_in_block.wait_for_success().await?;
615				log::info!("[Finalized] In block: {:#?}", tx_in_block.block_hash());
616				return Ok(tx_in_block.block_hash());
617			},
618			TxStatus::Error { message } |
619			TxStatus::Invalid { message } |
620			TxStatus::Dropped { message } => {
621				return Err(anyhow!("Error submitting tx: {message}"));
622			},
623			_ => continue,
624		}
625	}
626
627	Err(anyhow!("Transaction event stream ended without reaching the finalized state"))
628}

630/// Submits the given `call` as transaction and waits `timeout_secs` for it successful finalization.
631///
632/// If the transaction does not reach the finalized state in `timeout_secs` an error is returned.
633/// The transaction is send as immortal transaction.
634pub async fn submit_extrinsic_and_wait_for_finalization_success_with_timeout<
635	S: Signer<PolkadotConfig>,
636>(
637	client: &OnlineClient<PolkadotConfig>,
638	call: &DynamicPayload,
639	signer: &S,
640	timeout_secs: impl Into<u64>,
641) -> Result<(), anyhow::Error> {
642	let secs = timeout_secs.into();
643	let res = tokio::time::timeout(
644		Duration::from_secs(secs),
645		submit_extrinsic_and_wait_for_finalization_success(client, call, signer),
646	)
647	.await;
648
649	match res {
650		Ok(Ok(_)) => Ok(()),
651		Ok(Err(e)) => Err(anyhow!("Error waiting for metric: {}", e)),
652		// timeout
653		Err(_) => Err(anyhow!("Timeout ({secs}), waiting for extrinsic finalization")),
654	}
655}

657/// Asserts that the given `para_id` is registered at the relay chain.
658pub async fn assert_para_is_registered(
659	relay_client: &OnlineClient<PolkadotConfig>,
660	para_id: ParaId,
661	blocks_to_wait: u32,
662) -> Result<(), anyhow::Error> {
663	let mut blocks_sub = relay_client.blocks().subscribe_all().await?;
664	let para_id: u32 = para_id.into();
665
666	let keys: Vec<Value> = vec![];
667	let query = subxt::dynamic::storage("Paras", "Parachains", keys);
668
669	let mut blocks_cnt = 0;
670	while let Some(block) = blocks_sub.next().await {
671		let block = block?;
672		log::debug!("Relay block #{}, checking if para_id {para_id} is registered", block.number(),);
673		let parachains = block.storage().fetch(&query).await?;
674
675		let parachains: Vec<u32> = match parachains {
676			Some(parachains) => parachains.as_type()?,
677			None => vec![],
678		};
679
680		log::debug!("Registered para_ids: {:?}", parachains);
681
682		if parachains.iter().any(|p| para_id.eq(p)) {
683			log::debug!("para_id {para_id} registered");
684			return Ok(());
685		}
686		if blocks_cnt >= blocks_to_wait {
687			return Err(anyhow!(
688				"Parachain {para_id} not registered within {blocks_to_wait} blocks"
689			));
690		}
691		blocks_cnt += 1;
692	}
693
694	Err(anyhow!("No more blocks to check"))
695}

709/// Validates that the given block is a "special" block in the core.
710///
711/// If `is_only_block_in_core` is true, it checks if the given block is the first block in the core
712/// and the only one. If this is `false`, it only checks if the block is the last block in the core.
713async fn ensure_is_block_in_core_impl(
714	para_client: &OnlineClient<PolkadotConfig>,
715	block_hash: H256,
716	is_only_block_in_core: bool,
717) -> Result<(), anyhow::Error> {
718	let blocks = para_client.blocks();
719	let block = blocks.at(block_hash).await?;
720	let block_core_info = find_core_info(&block)?;
721
722	if is_only_block_in_core {
723		let parent = blocks.at(block.header().parent_hash).await?;
724
725		// Genesis is for sure on a different core :)
726		if parent.number() != 0 {
727			let parent_core_info = find_core_info(&parent)?;
728
729			if parent_core_info == block_core_info {
730				return Err(anyhow::anyhow!(
731					"Not first block ({}) in core, at least the parent block is on the same core.",
732					block.header().number
733				));
734			}
735		}
736	}
737
738	let next_block = loop {
739		// Start with the latest best block.
740		let mut current_block = Arc::new(blocks.subscribe_best().await?.next().await.unwrap()?);
741
742		let mut next_block = None;
743
744		while current_block.hash() != block_hash {
745			next_block = Some(current_block.clone());
746			current_block = Arc::new(blocks.at(current_block.header().parent_hash).await?);
747
748			if current_block.number() == 0 {
749				return Err(anyhow::anyhow!(
750					"Did not found block while going backwards from the best block"
751				));
752			}
753		}
754
755		// It possible that the first block we got is the same as the transaction got finalized.
756		// So, we just retry again until we found some more blocks.
757		if let Some(next_block) = next_block {
758			break next_block;
759		}
760	};
761
762	let next_block_core_info = find_core_info(&next_block)?;
763
764	if next_block_core_info == block_core_info {
765		return Err(anyhow::anyhow!(
766			"Not {} block ({}) in core, at least the following block is on the same core.",
767			if is_only_block_in_core { "first" } else { "last" },
768			block.header().number
769		));
770	}
771
772	Ok(())
773}

775/// Checks if the specified block occupies a full core.
776pub async fn ensure_is_only_block_in_core(
777	para_client: &OnlineClient<PolkadotConfig>,
778	block_to_check: BlockToCheck,
779) -> Result<(), anyhow::Error> {
780	let blocks = para_client.blocks();
781
782	match block_to_check {
783		BlockToCheck::Exact(block_hash) => {
784			ensure_is_block_in_core_impl(para_client, block_hash, true).await
785		},
786		BlockToCheck::NextFirstBundleBlock(start_block_hash) => {
787			let start_block = blocks.at(start_block_hash).await?;
788
789			let mut best_block_stream = blocks.subscribe_best().await?;
790
791			let mut next_first_bundle_block = None;
792			while let Some(mut block) = best_block_stream.next().await.transpose()? {
793				while block.number() > start_block.number() {
794					if find_block_bundle_info(&block)?.index == 0 {
795						next_first_bundle_block = Some(block.hash());
796					}
797
798					block = blocks.at(block.header().parent_hash).await?;
799				}
800
801				if next_first_bundle_block.is_some() {
802					break;
803				}
804			}
805
806			if let Some(block) = next_first_bundle_block {
807				ensure_is_block_in_core_impl(para_client, block, true).await
808			} else {
809				Err(anyhow!("Could not find the next bundle after {}", start_block.number()))
810			}
811		},
812	}
813}

815/// Checks if the specified block is the last block in a core.
816///
817/// Also ensures that the last block is NOT the first block.
818pub async fn ensure_is_last_block_in_core(
819	para_client: &OnlineClient<PolkadotConfig>,
820	block_to_check: H256,
821) -> Result<(), anyhow::Error> {

829	// block.
830	if bundle_info.index == 0 {
831		Err(anyhow!("`{block_to_check:?}` is the first block of a core and not the last"))
832	} else {
833		Ok(())
834	}

837/// Assigns the given `cores` to the given `para_id`.
838///
839/// Zombienet by default adds extra core for each registered parachain additionally to the one
840/// requested by `num_cores`. It then assigns the parachains to the extra cores allocated at the
841/// end. So, the passed core indices should be counted from zero.
842///
843/// # Example
844///
845/// Genesis patch:
846/// ```json
847/// "configuration": {
848///   "config": {
849///     "scheduler_params": {
850///       "num_cores": 2,
851///     }
852///   }
853/// }
854/// ```
855///
856/// Runs the relay chain with `2` cores and we also add two parachains.
857/// To assign these extra `2` cores, the call would look like this:
858///
859/// ```ignore
860/// assign_cores(&relay_node, PARA_ID, vec![0, 1])
861/// ```
862///
863/// The cores `2` and `3` are assigned to the parachains by Zombienet.
864pub async fn assign_cores(
865	client: &OnlineClient<PolkadotConfig>,
866	para_id: u32,
867	cores: Vec<u32>,
868) -> Result<(), anyhow::Error> {
869	log::info!("Assigning {:?} cores to parachain {}", cores, para_id);
870
871	let assign_cores_call =
872		create_assign_core_call(&cores.into_iter().map(|core| (core, para_id)).collect::<Vec<_>>());
873
874	let res = submit_extrinsic_and_wait_for_finalization_success_with_timeout(
875		client,
876		&assign_cores_call,
877		&zombienet_sdk::subxt_signer::sr25519::dev::alice(),
878		60u64,
879	)
880	.await;
881	assert!(res.is_ok(), "Extrinsic failed to finalize: {:?}", res.unwrap_err());
882	log::info!("Cores assigned to the parachain");
883
884	Ok(())
885}

959/// Submit a runtime upgrade via sudo and verify it was scheduled.
960///
961/// This submits a `Sudo::sudo_unchecked_weight(System::set_code(wasm))` extrinsic,
962/// waits for finalization, then checks the `Sudid` event to verify the inner dispatch
963/// succeeded. Returns the hash of the finalized block containing the upgrade extrinsic.
964pub async fn submit_sudo_runtime_upgrade<S: Signer<PolkadotConfig>>(
965	client: &OnlineClient<PolkadotConfig>,
966	wasm: &[u8],
967	signer: &S,
968) -> Result<H256, anyhow::Error> {
969	log::info!("Submitting sudo runtime upgrade, wasm size: {} bytes", wasm.len());
970	let call = create_runtime_upgrade_call(wasm);
971	let block_hash =
972		submit_extrinsic_and_wait_for_finalization_success(client, &call, signer).await?;
973
974	// Verify the inner sudo dispatch succeeded by checking the Sudid event.
975	// sudo_unchecked_weight always returns Ok at the extrinsic level, even if the
976	// inner call fails — the actual result is only in the Sudid event.
977	let block = client.blocks().at(block_hash).await?;
978	let events = block.events().await?;
979	let sudid_results: Vec<Result<(), sp_runtime::DispatchError>> =
980		find_event_and_decode_fields(&events, "Sudo", "Sudid")?;
981
982	match sudid_results.first() {
983		Some(Ok(())) => {
984			log::info!("Sudo runtime upgrade dispatched successfully in block {block_hash:?}")
985		},
986		Some(Err(e)) => {
987			return Err(anyhow!(
988				"Sudo runtime upgrade inner dispatch failed in block {block_hash:?}: {}",
989				format_dispatch_error(e, &client.metadata()),
990			))
991		},
992		None => return Err(anyhow!("Sudid event not found in block {block_hash:?}")),
993	}
994
995	Ok(block_hash)
996}

998/// Wait until a runtime upgrade has happened.
999///
1000/// This checks all finalized blocks until it finds a block that sets the
1001/// `RuntimeEnvironmentUpdated` digest.
1002///
1003/// Returns the hash of the block at which the runtime upgrade was applied.
1004pub async fn wait_for_runtime_upgrade(
1005	client: &OnlineClient<PolkadotConfig>,

1012			.digest

1015			.any(|d| matches!(d, DigestItem::RuntimeEnvironmentUpdated))
1016		{

1019			return Ok(block.hash());
1020		}
1021	}

1026/// Poll a node's WebSocket endpoint until its subxt metadata reports the given pallet,
1027/// returning a fresh `OnlineClient` against that metadata, or fail on timeout.
1028///
1029/// After a runtime upgrade that introduces a new pallet, subxt's cached metadata can lag

1049		}
1050		log::debug!("`{pallet_name}` not in metadata yet, retrying");
1051	}

1054/// Read the latest finalized `AuthorityDiscovery.Keys` set as a `HashSet` of 32-byte arrays.
1055///
1056/// Returns an empty set if the storage value is absent (e.g. the AD pallet is not present
1057/// in the runtime's metadata).
1058pub async fn read_authority_discovery_authorities(
1059	para_client: &OnlineClient<PolkadotConfig>,
1060) -> anyhow::Result<HashSet<[u8; 32]>> {
1061	let query = subxt::dynamic::storage("AuthorityDiscovery", "Keys", Vec::<Value>::new());
1062	let Some(thunk) = para_client.storage().at_latest().await?.fetch(&query).await? else {
1063		log::warn!("AuthorityDiscovery.Keys storage returned None — treating as empty set");
1064		return Ok(HashSet::new());
1065	};
1066
1067	let raw = thunk.into_encoded();
1068	let keys: HashSet<[u8; 32]> = Vec::<[u8; 32]>::decode(&mut &raw[..])
1069		.map_err(|e| anyhow!("decode AD keys: {e}"))?
1070		.into_iter()
1071		.collect();
1072	log::info!("AuthorityDiscovery.Keys: {} entries decoded", keys.len());
1073	Ok(keys)
1074}

1076/// Poll `AuthorityDiscovery.Keys` until it equals `expected` or `timeout` fires.
1077///
1078/// Finalizing `set_keys` only schedules the change; session rotation activates it later.
1079pub async fn wait_for_authority_discovery_keys(
1080	para_client: &OnlineClient<PolkadotConfig>,
1081	expected: &HashSet<[u8; 32]>,
1082	timeout: Duration,
1083	poll_interval: Duration,
1084) -> anyhow::Result<()> {
1085	let deadline = Instant::now() + timeout;
1086	loop {
1087		let current = read_authority_discovery_authorities(para_client).await?;
1088		if current == *expected {
1089			log::info!("All {} expected authority-discovery keys are active", expected.len());
1090			return Ok(());
1091		}
1092		let missing = expected.difference(&current).count();
1093		let unexpected = current.difference(expected).count();
1094		if Instant::now() >= deadline {
1095			return Err(anyhow!(
1096				"AuthorityDiscovery keys did not match the expected set within {timeout:?}: \
1097				 {missing} missing, {unexpected} unexpected",
1098			));
1099		}
1100		log::info!("Waiting for AD key activation: {missing} missing, {unexpected} unexpected");
1101		sleep(poll_interval).await;
1102	}
1103}

1105/// Assert that every named collator has fully meshed with the others via reserved peers.
1106///
1107/// Checks, in order:
1108///   1. Each collator's `collator_discovery_resolved_peers` metric reaches `n - 1`.
1109///   2. Each collator's `substrate_sub_libp2p_peers_count` metric reaches `n - 1`.
1110///   3. Every directed `(i, j)` pair appears in each side's `system_peers` output.
1111///
1112/// `timeout` applies separately to the metric checks (per node) and to the `(i, j)` loop.
1113pub async fn assert_full_collator_mesh(
1114	network: &Network<LocalFileSystem>,
1115	names: &[&str],
1116	timeout: Duration,
1117	poll_interval: Duration,
1118) -> anyhow::Result<()> {
1119	use zombienet_sdk::subxt::{backend::rpc::RpcClient, ext::subxt_rpcs::rpc_params};
1120
1121	let expected = (names.len() - 1) as f64;
1122	let per_node_timeout = timeout.as_secs();
1123
1124	// Step 1: resolved reserved peers — all nodes checked in parallel.
1125	try_join_all(names.iter().map(|&name| {
1126		let collator = network.get_node(name);
1127		async move {
1128			let collator = collator?;
1129			log::info!("Asserting `{name}` has resolved >= {expected} reserved peers");
1130			collator
1131				.wait_metric_with_timeout(
1132					"collator_discovery_resolved_peers",
1133					|c| c >= expected,
1134					per_node_timeout,
1135				)
1136				.await
1137				.map_err(|e| {
1138					anyhow!(
1139						"`{name}` did not reach {expected} resolved reserved peers within the timeout: {e}"
1140					)
1141				})
1142		}
1143	}))
1144	.await?;
1145
1146	// Step 2: libp2p peer count — all nodes checked in parallel.
1147	try_join_all(names.iter().map(|&name| {
1148		let collator = network.get_node(name);
1149		async move {
1150			let collator = collator?;
1151			log::info!("Asserting `{name}` has >= {expected} connected libp2p peers");
1152			collator
1153				.wait_metric_with_timeout(
1154					"substrate_sub_libp2p_peers_count",
1155					|c| c >= expected,
1156					per_node_timeout,
1157				)
1158				.await
1159				.map_err(|e| {
1160					anyhow!(
1161						"`{name}` did not reach {expected} connected peers — the reserved-mesh bypass of \
1162						 the non-reserved budget is not delivering full collator-to-collator connectivity: {e}"
1163					)
1164				})
1165		}
1166	}))
1167	.await?;
1168
1169	// Step 3: full (i, j) peer-id mesh check via `system_peers`.
1170	let mut peer_ids: HashMap<&str, String> = HashMap::new();
1171	for &name in names {
1172		let rpc: RpcClient = network.get_node(name)?.rpc().await?;
1173		let id: String = rpc.request("system_localPeerId", rpc_params![]).await?;
1174		log::info!("`{name}` local peer id: {id}");
1175		peer_ids.insert(name, id);
1176	}
1177
1178	let deadline = Instant::now() + timeout;
1179	loop {
1180		let mut all_full = true;
1181		let mut last_err: Option<String> = None;
1182
1183		for &name in names {
1184			let rpc: RpcClient = network.get_node(name)?.rpc().await?;
1185			let peers: serde_json::Value = rpc.request("system_peers", rpc_params![]).await?;
1186			let connected: HashSet<String> = peers
1187				.as_array()
1188				.map(|arr| {
1189					arr.iter()
1190						.filter_map(|p| p.get("peerId").and_then(|v| v.as_str()))
1191						.map(String::from)
1192						.collect()
1193				})
1194				.unwrap_or_default();
1195
1196			let missing: Vec<&&str> = names
1197				.iter()
1198				.filter(|other| **other != name)
1199				.filter(|other| !connected.contains(&peer_ids[*other]))
1200				.collect();
1201
1202			if !missing.is_empty() {
1203				all_full = false;
1204				last_err = Some(format!(
1205					"`{name}` is missing collator peers: {missing:?} (connected to {} parachain peers)",
1206					connected.len(),
1207				));
1208				break;
1209			}
1210		}
1211
1212		if all_full {
1213			log::info!(
1214				"Full collator-to-collator mesh confirmed: every collator is connected to every other.",
1215			);
1216			return Ok(());
1217		}
1218		if Instant::now() >= deadline {
1219			return Err(anyhow!(
1220				"Full collator mesh did not converge within {timeout:?}: {}",
1221				last_err.unwrap_or_else(|| "(unknown)".into()),
1222			));
1223		}
1224		log::info!("Mesh not yet full; retrying in {poll_interval:?}");
1225		sleep(poll_interval).await;
1226	}
1227}
