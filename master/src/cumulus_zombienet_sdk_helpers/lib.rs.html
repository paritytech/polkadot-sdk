39// Maximum number of blocks to wait for a session change.
40// If it does not arrive for whatever reason, we should not wait forever.
41const WAIT_MAX_BLOCKS_FOR_SESSION: u32 = 50;
42
43// Maximum time to wait for PVF preparation to conclude on a validator before
44// starting throughput measurement. PVF preparation is a one-off ~20s wasm
45// compile per validator that contends for CPU.
46const PVF_PREPARE_TIMEOUT_SECS: u64 = 180;

48/// Format a `sp_runtime::DispatchError` using runtime metadata for human-readable output.
49///
50/// For module errors this resolves the pallet index and error index to their names
51/// (e.g. `ParachainSystem::TooBig`) instead of showing raw bytes.
52fn format_dispatch_error(err: &sp_runtime::DispatchError, metadata: &Metadata) -> String {
53	match err {
54		sp_runtime::DispatchError::Module(module_err) => {
55			let pallet = metadata.pallet_by_index(module_err.index);
56			let pallet_name = pallet.as_ref().map(|p| p.name()).unwrap_or("UnknownPallet");
57			let error_name = pallet
58				.and_then(|p| p.error_variant_by_index(module_err.error[0]))
59				.map(|v| v.name.as_str())
60				.unwrap_or("UnknownError");
61			format!("{pallet_name}::{error_name}")
62		},
63		other => format!("{other:?}"),
64	}
65}

67/// Find an event in subxt `Events` and attempt to decode the fields of the event.
68fn find_event_and_decode_fields<T: Decode>(
69	events: &Events<PolkadotConfig>,
70	pallet: &str,
71	variant: &str,
72) -> Result<Vec<T>, anyhow::Error> {
73	let mut result = vec![];
74	for event in events.iter() {
75		let event = event?;
76		if event.pallet_name() == pallet && event.variant_name() == variant {
77			result.push(T::decode(&mut &event.field_bytes()[..])?);
78		}
79	}
80	Ok(result)
81}
82/// Returns `true` if the `block` is a session change.
83async fn is_session_change(
84	block: &Block<PolkadotConfig, OnlineClient<PolkadotConfig>>,
85) -> Result<bool, anyhow::Error> {
86	let events = block.events().await?;
87	Ok(events.iter().any(|event| {
88		event.as_ref().is_ok_and(|event| {
89			event.pallet_name() == "Session" && event.variant_name() == "NewSession"
90		})
91	}))
92}

94// Helper function for asserting the throughput of parachains, after the first session change.
95//
96// The throughput is measured as total number of backed candidates in a window of `stop_after` relay
97// chain blocks. The counting window starts from the relay chain block after the first one that
98// contains a backed candidate for a tracked para. Relay chain blocks with session changes are
99// generally ignored, but it is ensured that no blocks are build on top of these relay blocks.
100//
101// For tests where PVF preparation timing affects throughput (e.g. elastic scaling, runtime
102// upgrades), call [`wait_for_pvf_prepare`] before this helper to ensure all validators have
103// finished preparing the relevant PVFs.
104pub async fn assert_para_throughput(
105	relay_client: &OnlineClient<PolkadotConfig>,
106	stop_after: u32,
107	expected_candidate_ranges: impl Into<HashMap<ParaId, Range<u32>>>,
108	expected_number_of_blocks: impl Into<HashMap<ParaId, (OnlineClient<PolkadotConfig>, Range<u32>)>>,
109) -> Result<(), anyhow::Error> {
110	let ranges = expected_candidate_ranges.into();
111	let expected_number_of_blocks = expected_number_of_blocks.into();
112
113	let candidate_count =
114		collect_para_throughput(relay_client, stop_after, ranges, |_| Ok(true)).await?;
115
116	assert_expected_number_of_blocks(candidate_count, expected_number_of_blocks).await
117}

119/// Like [`assert_para_throughput`], but accepts a closure to validate each backed candidate
120/// receipt.
121///
122/// The closure receives each [`CandidateReceiptV2`] and should return:
123/// - `Ok(true)` to count the candidate,
124/// - `Ok(false)` to skip it,
125/// - `Err(e)` to fail immediately.
126///
127/// Only receipts for para IDs present in `expected_candidate_ranges` are passed to the closure.
128pub async fn assert_para_throughput_with<F>(
129	relay_client: &OnlineClient<PolkadotConfig>,
130	stop_after: u32,
131	expected_candidate_ranges: impl Into<HashMap<ParaId, Range<u32>>>,
132	validate: F,
133) -> Result<(), anyhow::Error>
134where
135	F: Fn(&CandidateReceiptV2<H256>) -> Result<bool, anyhow::Error>,
136{
137	collect_para_throughput(relay_client, stop_after, expected_candidate_ranges, validate)
138		.await
139		.map(|_| ())
140}

142/// Waits until every relaychain validator in `network` reports
143/// `polkadot_pvf_prepare_concluded >= min_total_prepares`.
144///
145/// Use this before [`assert_para_throughput`] in tests where PVF preparation timing demonstrably
146/// affects throughput — typically elastic-scaling tests (high core count, validators bound on
147/// PVF-compile CPU) and tests that measure throughput across a parachain runtime upgrade (which
148/// triggers re-preparation of the new PVF).
149///
150/// `min_total_prepares` is the absolute minimum value the metric must reach, in cumulative
151/// concluded prepare jobs since validator startup. For one round of preparation (one PVF per
152/// tracked parachain), pass `tracked_paras as u32`. For a measurement after a runtime upgrade,
153/// pass `2 * tracked_paras as u32`. Caller is responsible for tracking the round — we
154/// deliberately do not read a baseline from the metric and add a delta, because validators
155/// may already have started or finished preparing a new PVF before the baseline read, which
156/// makes the delta racy.
157pub async fn wait_for_pvf_prepare(
158	network: &Network<LocalFileSystem>,
159	min_total_prepares: u32,
160) -> Result<(), anyhow::Error> {
161	let validators = network.relaychain().nodes();
162	let target = min_total_prepares as f64;
163	log::info!(
164		"Waiting for PVF preparation to conclude on {} validator(s) (target {} concluded job(s) per validator).",
165		validators.len(),
166		target,
167	);
168	for node in &validators {
169		let node_name = node.name();
170		log::info!("Waiting for {node_name} PVF prep (target={target})...");
171		node.wait_metric_with_timeout(
172			"polkadot_pvf_prepare_concluded",
173			|c| c >= target,
174			PVF_PREPARE_TIMEOUT_SECS,
175		)
176		.await
177		.map_err(|e| anyhow!("{node_name}: PVF prepare did not conclude within timeout: {e}"))?;
178	}
179	log::info!(
180		"All {} validator(s) have prepared PVF artifacts (target {})",
181		validators.len(),
182		target,
183	);
184	Ok(())
185}

187async fn collect_para_throughput<F>(
188	relay_client: &OnlineClient<PolkadotConfig>,
189	stop_after: u32,
190	expected_candidate_ranges: impl Into<HashMap<ParaId, Range<u32>>>,
191	validate: F,
192) -> Result<HashMap<ParaId, Vec<CandidateReceiptV2<H256>>>, anyhow::Error>
193where
194	F: Fn(&CandidateReceiptV2<H256>) -> Result<bool, anyhow::Error>,
195{
196	let mut blocks_sub = relay_client.blocks().subscribe_finalized().await?;
197	let mut candidate_count: HashMap<ParaId, Vec<CandidateReceiptV2<H256>>> = HashMap::new();
198	let mut current_block_count = 0;
199
200	let expected_candidate_ranges = expected_candidate_ranges.into();
201	let valid_para_ids: Vec<ParaId> = expected_candidate_ranges.keys().cloned().collect();
202
203	log::info!(
204		"Asserting parachain throughput for para_ids: {:?}. Wait for the first session change",
205		valid_para_ids
206	);
207	// Wait for the first session, block production on the parachain will start after that.
208	wait_for_first_session_change(&mut blocks_sub).await?;
209	log::info!(
210		"First session change detected. Waiting for backed candidates from all tracked paras before counting."
211	);
212
213	// Skip relay chain blocks until every tracked para has had at least one backed candidate.
214	// This avoids counting the initial warm-up period where the backing pipeline (PVF
215	// compilation, first collation) hasn't reached steady state yet.
216	let mut paras_seen = std::collections::HashSet::new();
217	loop {
218		let block = blocks_sub
219			.next()
220			.await
221			.ok_or_else(|| anyhow!("Block stream ended while waiting for first candidate"))??;
222
223		if is_session_change(&block).await? {
224			continue;
225		}
226
227		let events = block.events().await?;
228		let receipts = find_event_and_decode_fields::<CandidateReceiptV2<H256>>(
229			&events,
230			"ParaInclusion",
231			"CandidateBacked",
232		)?;
233
234		for receipt in &receipts {
235			let para_id = receipt.descriptor.para_id();
236			if valid_para_ids.contains(&para_id) {
237				paras_seen.insert(para_id);
238			}
239		}
240
241		if paras_seen.len() == valid_para_ids.len() {
242			log::info!(
243				"All tracked paras have produced candidates by relay block {}. Counting {stop_after} blocks from the next one.",
244				block.number()
245			);
246			break;
247		}
248	}
249
250	while let Some(block) = blocks_sub.next().await {
251		let block = block?;
252		log::debug!("Finalized relay chain block {}", block.number());
253		let events = block.events().await?;
254
255		// Do not count blocks with session changes, no backed blocks there.
256		if is_session_change(&block).await? {
257			continue;
258		}
259
260		current_block_count += 1;
261
262		let receipts = find_event_and_decode_fields::<CandidateReceiptV2<H256>>(
263			&events,
264			"ParaInclusion",
265			"CandidateBacked",
266		)?;
267
268		for receipt in receipts {
269			let para_id = receipt.descriptor.para_id();
270			log::debug!("Block backed for para_id {para_id}");
271
272			if !valid_para_ids.contains(&para_id) {
273				continue;
274			}
275
276			if !validate(&receipt)? {
277				continue;
278			}
279
280			candidate_count.entry(para_id).or_default().push(receipt);
281		}
282
283		if current_block_count == stop_after {
284			break;
285		}
286	}
287
288	log::info!(
289		"Reached {stop_after} finalized relay chain blocks that contain backed candidates. The per-parachain distribution is: {:#?}",
290		candidate_count.iter().map(|(para_id, receipts)| format!("{para_id} has {} backed candidates", receipts.len())).collect::<Vec<_>>()
291	);
292
293	for (para_id, expected_candidate_range) in expected_candidate_ranges {
294		let actual = candidate_count
295			.get(&para_id)
296			.ok_or_else(|| anyhow!("ParaId {para_id} did not have any backed candidates"))?
297			.len() as u32;
298
299		if !expected_candidate_range.contains(&actual) {
300			return Err(anyhow!(
301				"ParaId {para_id}: candidate count {actual} not within expected range {expected_candidate_range:?}"
302			));
303		}
304	}
305
306	Ok(candidate_count)
307}

309async fn assert_expected_number_of_blocks(
310	candidate_count: HashMap<ParaId, Vec<CandidateReceiptV2<H256>>>,
311	expected_number_of_blocks: HashMap<ParaId, (OnlineClient<PolkadotConfig>, Range<u32>)>,
312) -> Result<(), anyhow::Error> {
313	for (para_id, (para_client, expected_number_of_blocks)) in expected_number_of_blocks {
314		let receipts = candidate_count
315			.get(&para_id)
316			.ok_or_else(|| anyhow!("ParaId did not have any backed candidates"))?;
317
318		let mut num_blocks = 0;
319
320		for receipt in receipts {
321			// We "abuse" the fact that the parachain is using `BlakeTwo256` as hash and thus, the
322			// `para_head` hash and the hash of the `header` should be equal.
323			let mut next_para_block_hash = receipt.descriptor().para_head();
324
325			let mut relay_identifier = None;
326			let mut core_info = None;
327
328			loop {
329				let block: Block<PolkadotConfig, OnlineClient<PolkadotConfig>> =
330					para_client.blocks().at(next_para_block_hash).await?;
331
332				// Genesis block is not part of a candidate :)
333				if block.number() == 0 {
334					break;
335				}
336
337				let ri = find_relay_block_identifier(&block)?;
338				let ci = find_core_info(&block)?;
339
340				// If the core changes or the relay identifier, we found all blocks for the
341				// candidate.
342				if *relay_identifier.get_or_insert(ri.clone()) != ri ||
343					*core_info.get_or_insert(ci.clone()) != ci
344				{
345					break;
346				}
347
348				num_blocks += 1;
349				next_para_block_hash = block.header().parent_hash;
350			}
351		}
352
353		if !expected_number_of_blocks.contains(&num_blocks) {
354			return Err(anyhow!(
355				"Block number count {num_blocks} not within range {expected_number_of_blocks:?}",
356			));
357		}
358	}
359
360	Ok(())
361}

388///
389/// The session change is detected by inspecting the events in the block.

391	blocks_sub: &mut zombienet_sdk::subxt::backend::StreamOfResults<
392		Block<PolkadotConfig, OnlineClient<PolkadotConfig>>,

396}

398/// Wait for the first block with the Nth session change.
399///
400/// The session change is detected by inspecting the events in the block.
401pub async fn wait_for_nth_session_change(
402	blocks_sub: &mut zombienet_sdk::subxt::backend::StreamOfResults<
403		Block<PolkadotConfig, OnlineClient<PolkadotConfig>>,
404	>,
405	mut sessions_to_wait: u32,
406) -> Result<(), anyhow::Error> {
407	let mut waited_block_num = 0;
408	while let Some(block) = blocks_sub.next().await {
409		let block = block?;
410		log::debug!("Finalized relay chain block {}", block.number());
411
412		if is_session_change(&block).await? {
413			sessions_to_wait -= 1;
414			if sessions_to_wait == 0 {
415				return Ok(());
416			}
417
418			waited_block_num = 0;
419		} else {
420			if waited_block_num >= WAIT_MAX_BLOCKS_FOR_SESSION {
421				return Err(anyhow::format_err!("Waited for {WAIT_MAX_BLOCKS_FOR_SESSION}, a new session should have been arrived by now."));
422			}
423
424			waited_block_num += 1;
425		}
426	}
427	Ok(())
428}

430// Helper function that asserts the maximum finality lag.
431pub async fn assert_finality_lag(
432	client: &OnlineClient<PolkadotConfig>,
433	maximum_lag: u32,
434) -> Result<(), anyhow::Error> {
435	let mut best_stream = client.blocks().subscribe_best().await?;
436	let mut fut_stream = client.blocks().subscribe_finalized().await?;
437	let (Some(Ok(best)), Some(Ok(finalized))) = join!(best_stream.next(), fut_stream.next()) else {
438		return Err(anyhow::format_err!("Unable to fetch best an finalized block!"));
439	};
440	let finality_lag = best.number() - finalized.number();
441
442	log::info!(
443		"Finality lagged by {finality_lag} blocks, maximum expected was {maximum_lag} blocks"
444	);
445
446	assert!(finality_lag <= maximum_lag, "Expected finality to lag by a maximum of {maximum_lag} blocks, but was lagging by {finality_lag} blocks.");
447	Ok(())
448}

450/// Assert that finality has not stalled.
451pub async fn assert_blocks_are_being_finalized(
452	client: &OnlineClient<PolkadotConfig>,
453) -> Result<(), anyhow::Error> {
454	let sleep_duration = Duration::from_secs(12);
455	let mut finalized_blocks = client.blocks().subscribe_finalized().await?;
456	let first_measurement = finalized_blocks
457		.next()
458		.await
459		.ok_or(anyhow::anyhow!("Can't get finalized block from stream"))??
460		.number();
461	sleep(sleep_duration).await;
462	let second_measurement = finalized_blocks
463		.next()
464		.await
465		.ok_or(anyhow::anyhow!("Can't get finalized block from stream"))??
466		.number();
467
468	log::info!(
469		"Finalized {} blocks within {sleep_duration:?}",
470		second_measurement - first_measurement
471	);
472	assert!(second_measurement > first_measurement);
473
474	Ok(())
475}

477/// Checks if the given `RelayBlockIdentifier` matches a relay chain header.
478fn identifier_matches_header(
479	identifier: &RelayBlockIdentifier,
480	header: &<PolkadotConfig as Config>::Header,
481) -> bool {
482	match identifier {
483		RelayBlockIdentifier::ByHash(hash) => {
484			let header_hash = BlakeTwo256::hash(&header.encode());
485			header_hash == *hash
486		},
487		RelayBlockIdentifier::ByStorageRoot { storage_root, .. } => {
488			header.state_root == *storage_root
489		},
490	}
491}

493/// Asserts that parachain blocks have the correct relay parent offset. This also checks that the
494/// relay chain descendants do not contain any session changes.
495///
496/// # Arguments
497///
498/// * `relay_client` - Client connected to a relay chain node
499/// * `para_client` - Client connected to a parachain node
500/// * `offset` - Expected minimum offset between relay parent and highest seen relay block
501/// * `block_limit` - Number of parachain blocks to verify before completing
502pub async fn assert_relay_parent_offset(
503	relay_client: &OnlineClient<PolkadotConfig>,
504	para_client: &OnlineClient<PolkadotConfig>,
505	offset: u32,
506	block_limit: u32,
507) -> Result<(), anyhow::Error> {
508	let mut relay_block_stream = relay_client.blocks().subscribe_all().await?;
509
510	// First parachain header #0 does not contain relay block identifier digest item.
511	let mut para_block_stream = para_client.blocks().subscribe_all().await?.skip(1);
512	let mut highest_relay_block_seen = 0;
513	let mut num_para_blocks_seen = 0;
514	let mut forbidden_parents = Vec::new();
515	let mut seen_relay_parents = HashMap::new();
516	loop {
517		tokio::select! {
518			Some(Ok(relay_block)) = relay_block_stream.next() => {
519				highest_relay_block_seen = max(relay_block.number(), highest_relay_block_seen);
520				if highest_relay_block_seen > 15 && num_para_blocks_seen == 0 {
521					return Err(anyhow!("No parachain blocks produced!"))
522				}
523				// When a relay chain block contains a session change, parachains shall not build on
524				// any ancestor of that block, if the session change block is part of the descendants.
525				// Example:
526				// RC Chain: A -> B -> C -> D*
527				// "*" denotes session change
528				// In this scenario, parachains with an offset of 2 should never build on relay chain
529				// blocks B or C. Both of them would include the session change block D* in their
530				// descendants, and we know that the candidate would span a session boundary.
531				if is_session_change(&relay_block).await? {
532					log::debug!("RC block #{} contains session change, adding {offset} parents to forbidden list.", relay_block.number());
533					let mut current_hash = relay_block.header().parent_hash;
534					for _ in 0..offset {
535						let block = relay_client.blocks().at(current_hash).await.map_err(|_| anyhow!("Unable to fetch RC header."))?;
536						forbidden_parents.push(block.header().clone());
537						current_hash = block.header().parent_hash;
538					}
539				}
540			},
541			Some(Ok(para_block)) = para_block_stream.next() => {
542				let relay_block_identifier = find_relay_block_identifier(&para_block)?;
543
544				let relay_parent_number = match &relay_block_identifier {
545					RelayBlockIdentifier::ByHash(block_hash) => relay_client.blocks().at(*block_hash).await?.number(),
546					RelayBlockIdentifier::ByStorageRoot { block_number, .. } => *block_number,
547				};
548
549				let para_block_number = para_block.number();
550				seen_relay_parents.insert(relay_block_identifier.clone(), para_block);
551				log::debug!("Parachain block #{para_block_number} was built on relay parent #{relay_parent_number}, highest seen was {highest_relay_block_seen}");
552				assert!(
553					highest_relay_block_seen < offset ||
554					relay_parent_number <= highest_relay_block_seen.saturating_sub(offset),
555					"Relay parent is not at the correct offset! relay_parent: #{relay_parent_number} highest_seen_relay_block: #{highest_relay_block_seen}",
556				);
557				// As per explanation above, we need to check that no parachain blocks are built
558				// on the forbidden parents.
559				for forbidden in &forbidden_parents {
560					for (identifier, para_block) in &seen_relay_parents {
561						if identifier_matches_header(identifier, forbidden) {
562							panic!(
563								"Parachain block {} was built on forbidden relay parent with session change descendants ({:?})",
564								para_block.hash(),
565								identifier
566							);
567						}
568					}
569				}
570				num_para_blocks_seen += 1;
571				if num_para_blocks_seen >= block_limit {
572					log::info!("Successfully verified relay parent offset of {offset} for {num_para_blocks_seen} parachain blocks.");
573					break;
574				}
575			}
576		}
577	}
578
579	Ok(())
580}

582/// Submits the given `call` as signed transaction and waits for its successful finalization.

609/// Submit the given transaction and wait for its finalization.
610async fn submit_tx_and_wait_for_finalization(
611	tx: SubmittableTransaction<PolkadotConfig, OnlineClient<PolkadotConfig>>,
612) -> Result<H256, anyhow::Error> {
613	log::info!("Submitting transaction: {:?}", tx.hash());
614
615	let mut tx = tx.submit_and_watch().await?;
616
617	while let Some(status) = tx.next().await.transpose()? {
618		match status {
619			TxStatus::InBestBlock(tx_in_block) => {
620				tx_in_block.wait_for_success().await?;
621				log::info!("[Best] In block: {:#?}", tx_in_block.block_hash());
622			},
623			TxStatus::InFinalizedBlock(ref tx_in_block) => {
624				tx_in_block.wait_for_success().await?;
625				log::info!("[Finalized] In block: {:#?}", tx_in_block.block_hash());
626				return Ok(tx_in_block.block_hash());
627			},
628			TxStatus::Error { message } |
629			TxStatus::Invalid { message } |
630			TxStatus::Dropped { message } => {
631				return Err(anyhow!("Error submitting tx: {message}"));
632			},
633			_ => continue,
634		}
635	}
636
637	Err(anyhow!("Transaction event stream ended without reaching the finalized state"))
638}

640/// Submits the given `call` as transaction and waits `timeout_secs` for it successful finalization.
641///
642/// If the transaction does not reach the finalized state in `timeout_secs` an error is returned.
643/// The transaction is send as immortal transaction.
644pub async fn submit_extrinsic_and_wait_for_finalization_success_with_timeout<
645	S: Signer<PolkadotConfig>,
646>(
647	client: &OnlineClient<PolkadotConfig>,
648	call: &DynamicPayload,
649	signer: &S,
650	timeout_secs: impl Into<u64>,
651) -> Result<(), anyhow::Error> {
652	let secs = timeout_secs.into();
653	let res = tokio::time::timeout(
654		Duration::from_secs(secs),
655		submit_extrinsic_and_wait_for_finalization_success(client, call, signer),
656	)
657	.await;
658
659	match res {
660		Ok(Ok(_)) => Ok(()),
661		Ok(Err(e)) => Err(anyhow!("Error waiting for metric: {}", e)),
662		// timeout
663		Err(_) => Err(anyhow!("Timeout ({secs}), waiting for extrinsic finalization")),
664	}
665}

667/// Asserts that the given `para_id` is registered at the relay chain.
668pub async fn assert_para_is_registered(
669	relay_client: &OnlineClient<PolkadotConfig>,
670	para_id: ParaId,
671	blocks_to_wait: u32,
672) -> Result<(), anyhow::Error> {
673	let mut blocks_sub = relay_client.blocks().subscribe_all().await?;
674	let para_id: u32 = para_id.into();
675
676	let keys: Vec<Value> = vec![];
677	let query = subxt::dynamic::storage("Paras", "Parachains", keys);
678
679	let mut blocks_cnt = 0;
680	while let Some(block) = blocks_sub.next().await {
681		let block = block?;
682		log::debug!("Relay block #{}, checking if para_id {para_id} is registered", block.number(),);
683		let parachains = block.storage().fetch(&query).await?;
684
685		let parachains: Vec<u32> = match parachains {
686			Some(parachains) => parachains.as_type()?,
687			None => vec![],
688		};
689
690		log::debug!("Registered para_ids: {:?}", parachains);
691
692		if parachains.iter().any(|p| para_id.eq(p)) {
693			log::debug!("para_id {para_id} registered");
694			return Ok(());
695		}
696		if blocks_cnt >= blocks_to_wait {
697			return Err(anyhow!(
698				"Parachain {para_id} not registered within {blocks_to_wait} blocks"
699			));
700		}
701		blocks_cnt += 1;
702	}
703
704	Err(anyhow!("No more blocks to check"))
705}

707/// Returns [`BlockBundleInfo`] for the given parachain block.
708fn find_block_bundle_info(
709	block: &Block<PolkadotConfig, OnlineClient<PolkadotConfig>>,
710) -> Result<BlockBundleInfo, anyhow::Error> {
711	let substrate_digest =
712		sp_runtime::generic::Digest::decode(&mut &block.header().digest.encode()[..])
713			.expect("`subxt::Digest` and `substrate::Digest` should encode and decode; qed");
714
715	CumulusDigestItem::find_block_bundle_info(&substrate_digest)
716		.ok_or_else(|| anyhow!("Failed to find `BlockBundleInfo` digest"))
717}

719/// Validates that the given block is a "special" block in the core.
720///
721/// If `is_only_block_in_core` is true, it checks if the given block is the first block in the core
722/// and the only one. If this is `false`, it only checks if the block is the last block in the core.
723async fn ensure_is_block_in_core_impl(
724	para_client: &OnlineClient<PolkadotConfig>,
725	block_hash: H256,
726	is_only_block_in_core: bool,
727) -> Result<(), anyhow::Error> {
728	let blocks = para_client.blocks();
729	let block = blocks.at(block_hash).await?;
730	let block_core_info = find_core_info(&block)?;
731
732	if is_only_block_in_core {
733		let parent = blocks.at(block.header().parent_hash).await?;
734
735		// Genesis is for sure on a different core :)
736		if parent.number() != 0 {
737			let parent_core_info = find_core_info(&parent)?;
738
739			if parent_core_info == block_core_info {
740				return Err(anyhow::anyhow!(
741					"Not first block ({}) in core, at least the parent block is on the same core.",
742					block.header().number
743				));
744			}
745		}
746	}
747
748	let next_block = loop {
749		// Start with the latest best block.
750		let mut current_block = Arc::new(blocks.subscribe_best().await?.next().await.unwrap()?);
751
752		let mut next_block = None;
753
754		while current_block.hash() != block_hash {
755			next_block = Some(current_block.clone());
756			current_block = Arc::new(blocks.at(current_block.header().parent_hash).await?);
757
758			if current_block.number() == 0 {
759				return Err(anyhow::anyhow!(
760					"Did not found block while going backwards from the best block"
761				));
762			}
763		}
764
765		// It possible that the first block we got is the same as the transaction got finalized.
766		// So, we just retry again until we found some more blocks.
767		if let Some(next_block) = next_block {
768			break next_block;
769		}
770	};
771
772	let next_block_core_info = find_core_info(&next_block)?;
773
774	if next_block_core_info == block_core_info {
775		return Err(anyhow::anyhow!(
776			"Not {} block ({}) in core, at least the following block is on the same core.",
777			if is_only_block_in_core { "first" } else { "last" },
778			block.header().number
779		));
780	}
781
782	Ok(())
783}

785/// Checks if the specified block occupies a full core.
786pub async fn ensure_is_only_block_in_core(
787	para_client: &OnlineClient<PolkadotConfig>,
788	block_to_check: BlockToCheck,
789) -> Result<(), anyhow::Error> {
790	let blocks = para_client.blocks();
791
792	match block_to_check {
793		BlockToCheck::Exact(block_hash) => {
794			ensure_is_block_in_core_impl(para_client, block_hash, true).await
795		},
796		BlockToCheck::NextFirstBundleBlock(start_block_hash) => {
797			let start_block = blocks.at(start_block_hash).await?;
798
799			let mut best_block_stream = blocks.subscribe_best().await?;
800
801			let mut next_first_bundle_block = None;
802			while let Some(mut block) = best_block_stream.next().await.transpose()? {
803				while block.number() > start_block.number() {
804					if find_block_bundle_info(&block)?.index == 0 {
805						next_first_bundle_block = Some(block.hash());
806					}
807
808					block = blocks.at(block.header().parent_hash).await?;
809				}
810
811				if next_first_bundle_block.is_some() {
812					break;
813				}
814			}
815
816			if let Some(block) = next_first_bundle_block {
817				ensure_is_block_in_core_impl(para_client, block, true).await
818			} else {
819				Err(anyhow!("Could not find the next bundle after {}", start_block.number()))
820			}
821		},
822	}
823}

825/// Checks if the specified block is the last block in a core.
826///
827/// Also ensures that the last block is NOT the first block.
828pub async fn ensure_is_last_block_in_core(
829	para_client: &OnlineClient<PolkadotConfig>,
830	block_to_check: H256,
831) -> Result<(), anyhow::Error> {
832	ensure_is_block_in_core_impl(para_client, block_to_check, false).await?;
833
834	let blocks = para_client.blocks();
835	let block = blocks.at(block_to_check).await?;
836	let bundle_info = find_block_bundle_info(&block)?;
837
838	// Above we ensure it is the last block in the core and now we want to ensure it isn't the first
839	// block.
840	if bundle_info.index == 0 {
841		Err(anyhow!("`{block_to_check:?}` is the first block of a core and not the last"))
842	} else {
843		Ok(())
844	}
845}

847/// Assigns the given `cores` to the given `para_id`.
848///
849/// Zombienet by default adds extra core for each registered parachain additionally to the one
850/// requested by `num_cores`. It then assigns the parachains to the extra cores allocated at the
851/// end. So, the passed core indices should be counted from zero.
852///
853/// # Example
854///
855/// Genesis patch:
856/// ```json
857/// "configuration": {
858///   "config": {
859///     "scheduler_params": {
860///       "num_cores": 2,
861///     }
862///   }
863/// }
864/// ```
865///
866/// Runs the relay chain with `2` cores and we also add two parachains.
867/// To assign these extra `2` cores, the call would look like this:
868///
869/// ```ignore
870/// assign_cores(&relay_node, PARA_ID, vec![0, 1])
871/// ```
872///
873/// The cores `2` and `3` are assigned to the parachains by Zombienet.
874pub async fn assign_cores(
875	client: &OnlineClient<PolkadotConfig>,
876	para_id: u32,
877	cores: Vec<u32>,
878) -> Result<(), anyhow::Error> {
879	log::info!("Assigning {:?} cores to parachain {}", cores, para_id);
880
881	let assign_cores_call =
882		create_assign_core_call(&cores.into_iter().map(|core| (core, para_id)).collect::<Vec<_>>());
883
884	let res = submit_extrinsic_and_wait_for_finalization_success_with_timeout(
885		client,
886		&assign_cores_call,
887		&zombienet_sdk::subxt_signer::sr25519::dev::alice(),
888		60u64,
889	)
890	.await;
891	assert!(res.is_ok(), "Extrinsic failed to finalize: {:?}", res.unwrap_err());
892	log::info!("Cores assigned to the parachain");
893
894	Ok(())
895}

897fn create_assign_core_call(core_and_para: &[(u32, u32)]) -> DynamicPayload {
898	let mut assign_cores = vec![];
899	for (core, para_id) in core_and_para.iter() {
900		assign_cores.push(value! {
901			Coretime(assign_core { core : *core, begin: 0, assignment: ((Task(*para_id), 57600)), end_hint: None() })
902		});
903	}
904
905	zombienet_sdk::subxt::tx::dynamic(
906		"Sudo",
907		"sudo",
908		vec![value! {
909			Utility(batch { calls: assign_cores })
910		}],
911	)
912}

935/// Submit a runtime upgrade via sudo and verify it was scheduled.
936///
937/// This submits a `Sudo::sudo_unchecked_weight(System::set_code(wasm))` extrinsic,
938/// waits for finalization, then checks the `Sudid` event to verify the inner dispatch
939/// succeeded. Returns the hash of the finalized block containing the upgrade extrinsic.
940pub async fn submit_sudo_runtime_upgrade<S: Signer<PolkadotConfig>>(
941	client: &OnlineClient<PolkadotConfig>,
942	wasm: &[u8],
943	signer: &S,
944) -> Result<H256, anyhow::Error> {
945	log::info!("Submitting sudo runtime upgrade, wasm size: {} bytes", wasm.len());
946	let call = create_runtime_upgrade_call(wasm);
947	let block_hash =
948		submit_extrinsic_and_wait_for_finalization_success(client, &call, signer).await?;
949
950	// Verify the inner sudo dispatch succeeded by checking the Sudid event.
951	// sudo_unchecked_weight always returns Ok at the extrinsic level, even if the
952	// inner call fails — the actual result is only in the Sudid event.
953	let block = client.blocks().at(block_hash).await?;
954	let events = block.events().await?;
955	let sudid_results: Vec<Result<(), sp_runtime::DispatchError>> =
956		find_event_and_decode_fields(&events, "Sudo", "Sudid")?;
957
958	match sudid_results.first() {
959		Some(Ok(())) => {
960			log::info!("Sudo runtime upgrade dispatched successfully in block {block_hash:?}")
961		},
962		Some(Err(e)) => {
963			return Err(anyhow!(
964				"Sudo runtime upgrade inner dispatch failed in block {block_hash:?}: {}",
965				format_dispatch_error(e, &client.metadata()),
966			))
967		},
968		None => return Err(anyhow!("Sudid event not found in block {block_hash:?}")),
969	}
970
971	Ok(block_hash)
972}

974/// Wait until a runtime upgrade has happened.
975///
976/// This checks all finalized blocks until it finds a block that sets the
977/// `RuntimeEnvironmentUpdated` digest.
978///
979/// Returns the hash of the block at which the runtime upgrade was applied.
980pub async fn wait_for_runtime_upgrade(
981	client: &OnlineClient<PolkadotConfig>,
982) -> Result<H256, anyhow::Error> {
983	let mut finalized_blocks = client.blocks().subscribe_finalized().await?;
984
985	while let Some(Ok(block)) = finalized_blocks.next().await {
986		if block
987			.header()
988			.digest
989			.logs
990			.iter()
991			.any(|d| matches!(d, DigestItem::RuntimeEnvironmentUpdated))
992		{
993			log::info!("Runtime upgraded in block {:?}", block.hash());
994
995			return Ok(block.hash());
996		}
997	}
998
999	Err(anyhow!("Did not find a runtime upgrade"))
1000}
