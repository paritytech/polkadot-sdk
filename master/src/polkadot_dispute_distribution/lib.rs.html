17//! # Sending and receiving of `DisputeRequest`s.
18//!
19//! This subsystem essentially consists of two parts:
20//!
21//! - a sender
22//! - and a receiver
23//!
24//! The sender is responsible for getting our vote out, see `sender`. The receiver handles
25//! incoming [`DisputeRequest`](v1::DisputeRequest)s and offers spam protection, see `receiver`.

50/// ## The sender [`DisputeSender`]
51///
52/// The sender (`DisputeSender`) keeps track of live disputes and makes sure our vote gets out for
53/// each one of those. The sender is responsible for sending our vote to each validator
54/// participating in the dispute and to each authority currently authoring blocks. The sending can
55/// be initiated by sending `DisputeDistributionMessage::SendDispute` message to this subsystem.
56///
57/// In addition the `DisputeSender` will query the coordinator for active disputes on each
58/// [`DisputeSender::update_leaves`] call and will initiate sending (start a `SendTask`) for every,
59/// to this subsystem, unknown dispute. This is to make sure, we get our vote out, even on
60/// restarts.
61///
62/// The actual work of sending and keeping track of transmission attempts to each validator for a
63/// particular dispute are done by [`SendTask`].  The purpose of the `DisputeSender` is to keep
64/// track of all ongoing disputes and start and clean up `SendTask`s accordingly.
65mod sender;
66use self::sender::{DisputeSender, DisputeSenderMessage};
67
68/// ## The receiver [`DisputesReceiver`]
69///
70/// The receiving side is implemented as `DisputesReceiver` and is run as a separate long running
71/// task within this subsystem ([`DisputesReceiver::run`]).
72///
73/// Conceptually all the receiver has to do, is waiting for incoming requests which are passed in
74/// via a dedicated channel and forwarding them to the dispute coordinator via
75/// `DisputeCoordinatorMessage::ImportStatements`. Being the interface to the network and untrusted
76/// nodes, the reality is not that simple of course. Before importing statements the receiver will
77/// batch up imports as well as possible for efficient imports while maintaining timely dispute
78/// resolution and handling of spamming validators:
79///
80/// - Drop all messages from non validator nodes, for this it requires the [`AuthorityDiscovery`]
81/// service.
82/// - Drop messages from a node, if it sends at a too high rate.
83/// - Filter out duplicate messages (over some period of time).
84/// - Drop any obviously invalid votes (invalid signatures for example).
85/// - Ban peers whose votes were deemed invalid.
86///
87/// In general dispute-distribution works on limiting the work the dispute-coordinator will have to
88/// do, while at the same time making it aware of new disputes as fast as possible.
89///
90/// For successfully imported votes, we will confirm the receipt of the message back to the sender.
91/// This way a received confirmation guarantees, that the vote has been stored to disk by the
92/// receiver.
93mod receiver;
94use self::receiver::DisputesReceiver;

109/// Rate limit on the `receiver` side.
110///
111/// If messages from one peer come in at a higher rate than every `RECEIVE_RATE_LIMIT` on average,
112/// we start dropping messages from that peer to enforce that limit.
113pub const RECEIVE_RATE_LIMIT: Duration = Duration::from_millis(100);
114
115/// Rate limit on the `sender` side.
116///
117/// In order to not hit the `RECEIVE_RATE_LIMIT` on the receiving side, we limit out sending rate as
118/// well.
119///
120/// We add 50ms extra, just to have some save margin to the `RECEIVE_RATE_LIMIT`.
121pub const SEND_RATE_LIMIT: Duration = RECEIVE_RATE_LIMIT.saturating_add(Duration::from_millis(50));

123/// The dispute distribution subsystem.
124pub struct DisputeDistributionSubsystem<AD> {
125	/// Easy and efficient runtime access for this subsystem.
126	runtime: RuntimeInfo,
127
128	/// Sender for our dispute requests.
129	disputes_sender: DisputeSender<DisputeSenderMessage>,
130
131	/// Receive messages from `DisputeSender` background tasks.
132	sender_rx: mpsc::Receiver<DisputeSenderMessage>,
133
134	/// Receiver for incoming requests.
135	req_receiver: Option<IncomingRequestReceiver<v1::DisputeRequest>>,
136
137	/// Authority discovery service.
138	authority_discovery: AD,
139
140	/// Metrics for this subsystem.
141	metrics: Metrics,
142
143	/// Monotonic flag: set to `true` once any activated leaf has the `CandidateReceiptV3`
144	/// node feature enabled. Shared with the receiver task via `Arc<AtomicBool>`.
145	/// See `CandidateDescriptorV2::version_for_candidate_validation` for the safety argument.
146	v3_ever_seen: Arc<AtomicBool>,
147}
148
149#[overseer::subsystem(DisputeDistribution, error = SubsystemError, prefix = self::overseer)]
150impl<Context, AD> DisputeDistributionSubsystem<AD>
151where
152	<Context as overseer::DisputeDistributionContextTrait>::Sender:
153		overseer::DisputeDistributionSenderTrait + Sync + Send,
154	AD: AuthorityDiscovery + Clone,
155{
156	fn start(self, ctx: Context) -> SpawnedSubsystem {
157		let future = self
158			.run(ctx)
159			.map_err(|e| SubsystemError::with_origin("dispute-distribution", e))
160			.boxed();
161
162		SpawnedSubsystem { name: "dispute-distribution-subsystem", future }
163	}
164}
165
166#[overseer::contextbounds(DisputeDistribution, prefix = self::overseer)]
167impl<AD> DisputeDistributionSubsystem<AD>
168where
169	AD: AuthorityDiscovery + Clone,
170{
171	/// Create a new instance of the dispute distribution.
172	pub fn new(
173		keystore: KeystorePtr,
174		req_receiver: IncomingRequestReceiver<v1::DisputeRequest>,
175		authority_discovery: AD,
176		metrics: Metrics,
177	) -> Self {
178		let runtime = RuntimeInfo::new_with_config(runtime::Config {
179			keystore: Some(keystore),
180			session_cache_lru_size: DISPUTE_WINDOW.get(),
181		});
182		let (tx, sender_rx) = NestingSender::new_root(1);
183		let disputes_sender = DisputeSender::new(tx, metrics.clone());
184		Self {
185			runtime,
186			disputes_sender,
187			sender_rx,
188			req_receiver: Some(req_receiver),
189			authority_discovery,
190			metrics,
191			v3_ever_seen: Arc::new(AtomicBool::new(false)),
192		}
193	}
194
195	/// Start processing work as passed on from the Overseer.
196	async fn run<Context>(mut self, mut ctx: Context) -> std::result::Result<(), FatalError> {
197		let receiver = DisputesReceiver::new(
198			ctx.sender().clone(),
199			self.req_receiver
200				.take()
201				.expect("Must be provided on `new` and we take ownership here. qed."),
202			self.authority_discovery.clone(),
203			self.metrics.clone(),
204			self.v3_ever_seen.clone(),
205		);
206		ctx.spawn("disputes-receiver", receiver.run().boxed())
207			.map_err(FatalError::SpawnTask)?;
208
209		// Process messages for sending side.
210		//
211		// Note: We want the sender to be rate limited and we are currently taking advantage of the
212		// fact that the root task of this subsystem is only concerned with sending: Functions of
213		// `DisputeSender` might back pressure if the rate limit is hit, which will slow down this
214		// loop. If this fact ever changes, we will likely need another task.
215		loop {
216			let message = MuxedMessage::receive(&mut ctx, &mut self.sender_rx).await;
217			match message {
218				MuxedMessage::Subsystem(result) => {
219					let result = match result? {
220						FromOrchestra::Signal(signal) => {
221							match self.handle_signals(&mut ctx, signal).await {
222								Ok(SignalResult::Conclude) => return Ok(()),
223								Ok(SignalResult::Continue) => Ok(()),
224								Err(f) => Err(f),
225							}
226						},
227						FromOrchestra::Communication { msg } => {
228							self.handle_subsystem_message(&mut ctx, msg).await
229						},
230					};
231					log_error(result, "on FromOrchestra")?;
232				},
233				MuxedMessage::Sender(result) => {
234					let result = self
235						.disputes_sender
236						.on_message(
237							&mut ctx,
238							&mut self.runtime,
239							result.ok_or(FatalError::SenderExhausted)?,
240						)
241						.await
242						.map_err(Error::Sender);
243					log_error(result, "on_message")?;
244				},
245			}
246		}
247	}
248
249	/// Handle overseer signals.
250	async fn handle_signals<Context>(
251		&mut self,
252		ctx: &mut Context,
253		signal: OverseerSignal,
254	) -> Result<SignalResult> {
255		match signal {
256			OverseerSignal::Conclude => return Ok(SignalResult::Conclude),
257			OverseerSignal::ActiveLeaves(ref update) => {
258				// Detect V3 node feature on activated leaves (same approach as
259				// dispute-coordinator and candidate-validation).
260				if !self.v3_ever_seen.load(Ordering::Relaxed) {
261					if let Some(ref activated) = update.activated {
262						if let Ok(session_index) = self
263							.runtime
264							.get_session_index_for_child(ctx.sender(), activated.hash)
265							.await
266						{
267							if let Ok(Ok(features)) =
268								request_node_features(activated.hash, session_index, ctx.sender())
269									.await
270									.await
271							{
272								if FeatureIndex::CandidateReceiptV3.is_set(&features) {
273									gum::info!(
274										target: LOG_TARGET,
275										?session_index,
276										"CandidateReceiptV3 node feature detected in \
277										 dispute-distribution",
278									);
279									self.v3_ever_seen.store(true, Ordering::Relaxed);
280								}
281							}
282						}
283					}
284				}
285
286				self.disputes_sender
287					.update_leaves(ctx, &mut self.runtime, update.clone())
288					.await?;
289			},
290			OverseerSignal::BlockFinalized(_, _) => {},
291		};
292		Ok(SignalResult::Continue)
293	}
294
295	/// Handle `DisputeDistributionMessage`s.
296	async fn handle_subsystem_message<Context>(
297		&mut self,
298		ctx: &mut Context,
299		msg: DisputeDistributionMessage,
300	) -> Result<()> {
301		match msg {
302			DisputeDistributionMessage::SendDispute(dispute_msg) => {
303				let v3_ever_seen = self.v3_ever_seen.load(Ordering::Relaxed);
304				self.disputes_sender
305					.start_sender(ctx, &mut self.runtime, dispute_msg, v3_ever_seen)
306					.await?
307			},
308		}
309		Ok(())
310	}
311}

313/// Messages to be handled in this subsystem.
314#[derive(Debug)]
315enum MuxedMessage {
316	/// Messages from other subsystems.
317	Subsystem(FatalResult<FromOrchestra<DisputeDistributionMessage>>),
318	/// Messages from spawned sender background tasks.
319	Sender(Option<DisputeSenderMessage>),
320}
321
322#[overseer::contextbounds(DisputeDistribution, prefix = self::overseer)]
323impl MuxedMessage {
324	async fn receive<Context>(
325		ctx: &mut Context,
326		from_sender: &mut mpsc::Receiver<DisputeSenderMessage>,
327	) -> Self {
328		// We are only fusing here to make `select` happy, in reality we will quit if the stream
329		// ends.
330		let from_overseer = ctx.recv().fuse();
331		futures::pin_mut!(from_overseer, from_sender);
332		// We select biased to make sure we finish up loose ends, before starting new work.
333		futures::select_biased!(
334			msg = from_sender.next() => MuxedMessage::Sender(msg),
335			msg = from_overseer => MuxedMessage::Subsystem(msg.map_err(FatalError::SubsystemReceive)),
336		)
337	}
338}
339
340/// Result of handling signal from overseer.
341enum SignalResult {
342	/// Overseer asked us to conclude.
343	Conclude,
344	/// We can continue processing events.
345	Continue,
