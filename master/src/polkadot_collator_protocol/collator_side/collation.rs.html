35/// The status of a collation as seen from the collator.
36#[derive(Clone, Debug, PartialEq)]
37pub enum CollationStatus {
38	/// The collation was created, but we did not advertise it to any validator.
39	Created,
40	/// The collation was advertised to at least one validator.
41	Advertised,
42	/// The collation was requested by at least one validator.
43	Requested,
44}
45
46impl CollationStatus {
47	/// Advance to the [`Self::Advertised`] status.
48	///
49	/// This ensures that `self` isn't already [`Self::Requested`].
50	pub fn advance_to_advertised(&mut self) {
51		if !matches!(self, Self::Requested) {
52			*self = Self::Advertised;
53		}
54	}
55
56	/// Advance to the [`Self::Requested`] status.
57	pub fn advance_to_requested(&mut self) {
58		*self = Self::Requested;
59	}
60
61	/// Return label for metrics.
62	pub fn label(&self) -> &'static str {
63		match self {
64			CollationStatus::Created => "created",
65			CollationStatus::Advertised => "advertised",
66			CollationStatus::Requested => "requested",
67		}
68	}
69}

71/// A collation built by the collator.
72pub struct Collation {
73	/// Candidate receipt.
74	pub receipt: CandidateReceipt,
75	/// Proof to verify the state transition of the parachain.
76	pub pov: PoV,
77	/// Parent head-data
78	pub parent_head_data: HeadData,
79	/// Collation status.
80	pub status: CollationStatus,
81}

83/// Stores the state for waiting collation fetches per relay parent.
84#[derive(Default)]
85pub struct WaitingCollationFetches {
86	/// A flag indicating that we have an ongoing request.
87	/// This limits the number of collations being sent at any moment
88	/// of time to 1 for each relay parent.
89	///
90	/// If set to `true`, any new request will be queued.
91	pub collation_fetch_active: bool,
92	/// The collation fetches waiting to be fulfilled.
93	pub req_queue: VecDeque<VersionedCollationRequest>,
94	/// All peers that are waiting or actively uploading.
95	///
96	/// We will not accept multiple requests from the same peer, otherwise our DoS protection of
97	/// moving on to the next peer after `MAX_UNSHARED_UPLOAD_TIME` would be pointless.
98	pub waiting_peers: HashSet<(PeerId, CandidateHash)>,
99}

101/// Backwards-compatible wrapper for incoming collations requests.
102pub enum VersionedCollationRequest {
103	V2(IncomingRequest<protocol_v2::CollationFetchingRequest>),
104	V3(IncomingRequest<protocol_v3::CollationFetchingRequest>),
105}
106
107impl From<IncomingRequest<protocol_v2::CollationFetchingRequest>> for VersionedCollationRequest {
108	fn from(req: IncomingRequest<protocol_v2::CollationFetchingRequest>) -> Self {
109		Self::V2(req)
110	}
111}
112
113impl From<IncomingRequest<protocol_v3::CollationFetchingRequest>> for VersionedCollationRequest {
114	fn from(req: IncomingRequest<protocol_v3::CollationFetchingRequest>) -> Self {
115		Self::V3(req)
116	}
117}
118
119impl VersionedCollationRequest {
120	/// Returns parachain id from the request payload.
121	pub fn para_id(&self) -> ParaId {
122		match self {
123			VersionedCollationRequest::V2(req) => req.payload.para_id,
124			VersionedCollationRequest::V3(req) => req.payload.para_id,
125		}
126	}
127
128	/// Returns relay parent from the request payload.
129	pub fn scheduling_parent(&self) -> Hash {
130		match self {
131			VersionedCollationRequest::V2(req) => req.payload.scheduling_parent,
132			VersionedCollationRequest::V3(req) => req.payload.scheduling_parent,
133		}
134	}
135
136	/// Returns id of the peer the request was received from.
137	pub fn peer_id(&self) -> PeerId {
138		match self {
139			VersionedCollationRequest::V2(req) => req.peer,
140			VersionedCollationRequest::V3(req) => req.peer,
141		}
142	}
143
144	/// Builds and sends the versioned response carrying the collation.
145	pub fn send_collation_response(
146		self,
147		receipt: CandidateReceipt,
148		pov: PoV,
149		parent_head_data: HeadData,
150		sent_feedback: oneshot::Sender<()>,
151	) -> Result<(), ()> {
152		match self {
153			Self::V2(req) => req.send_outgoing_response(OutgoingResponse {
154				result: Ok(protocol_v2::CollationFetchingResponse::CollationWithParentHeadData {
155					receipt,
156					pov,
157					parent_head_data,
158				}),
159				reputation_changes: Vec::new(),
160				sent_feedback: Some(sent_feedback),
161			}),
162			Self::V3(req) => req.send_outgoing_response(OutgoingResponse {
163				result: Ok(protocol_v3::CollationFetchingResponse::Collation {
164					receipt,
165					pov,
166					parent_head_data,
167				}),
168				reputation_changes: Vec::new(),
169				sent_feedback: Some(sent_feedback),
170			}),
171		}
172	}
173}

175/// Result of the finished background send-collation task.
176///
177/// Note that if the timeout was hit the request doesn't get
178/// aborted, it only indicates that we should start processing
179/// the next one from the queue.
180pub struct CollationSendResult {
181	/// Candidate's relay parent.
182	pub relay_parent: Hash,
183	/// Candidate hash.
184	pub candidate_hash: CandidateHash,
185	/// Peer id.
186	pub peer_id: PeerId,
187	/// Whether the max unshared timeout was hit.
188	pub timed_out: bool,
189}
190
191pub type ActiveCollationFetches = FuturesUnordered<BoxFuture<'static, CollationSendResult>>;
