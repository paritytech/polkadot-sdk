118enum ShouldAdvertiseTo {
119	Yes,
120	NotAuthority,
121	AlreadyAdvertised,
122}

124/// Info about validators we are currently connected to.
125///
126/// It keeps track to which validators we advertised our collation.
127#[derive(Debug)]
128struct ValidatorGroup {
129	/// Validators discovery ids.
130	validators: Vec<AuthorityDiscoveryId>,
131
132	/// Bits indicating which validators have already seen the announcement
133	/// per core. Always the same length as `validators`.
134	segment_advertised_to: BitVec,
135}
136
137impl ValidatorGroup {
138	/// Create a group for `validators`, with nothing advertised yet.
139	fn new(validators: Vec<AuthorityDiscoveryId>) -> Self {
140		Self { segment_advertised_to: bitvec![0; validators.len()], validators }
141	}
142
143	/// Returns `true` if we should advertise the segment to the given peer.
144	fn should_advertise_segment(
145		&self,
146		peer_ids: &HashMap<PeerId, HashSet<AuthorityDiscoveryId>>,
147		peer: &PeerId,
148	) -> ShouldAdvertiseTo {
149		let authority_ids = match peer_ids.get(peer) {
150			Some(authority_ids) => authority_ids,
151			None => return ShouldAdvertiseTo::NotAuthority,
152		};
153
154		for id in authority_ids {
155			// One peer id may correspond to different discovery ids across sessions,
156			// having a non-empty intersection is sufficient to assume that this peer
157			// belongs to this particular validator group.
158			let validator_index = match self.validators.iter().position(|v| v == id) {
159				Some(idx) => idx,
160				None => continue,
161			};
162			// Either the segment is unseen by this validator group
163			// or the corresponding bit is not set.
164			if !self.segment_advertised_to.get(validator_index).map(|b| *b).unwrap_or(false) {
165				return ShouldAdvertiseTo::Yes;
166			} else {
167				return ShouldAdvertiseTo::AlreadyAdvertised;
168			}
169		}
170		ShouldAdvertiseTo::NotAuthority
171	}
172
173	/// Should be called after we advertised our segment to the given `peer` to keep track of it.
174	fn segment_advertised_to_peer(
175		&mut self,
176		peer_ids: &HashMap<PeerId, HashSet<AuthorityDiscoveryId>>,
177		peer: &PeerId,
178	) {
179		if let Some(authority_ids) = peer_ids.get(peer) {
180			for id in authority_ids {
181				let validator_index = match self.validators.iter().position(|v| v == id) {
182					Some(idx) => idx,
183					None => continue,
184				};
185				self.segment_advertised_to.set(validator_index, true)
186			}
187		}
188	}
189
190	/// Resets the advertised state for the given peer. Should be called when the peer disconnects
191	/// so that we re-advertise the segment when they reconnect.
192	fn reset_segment_advertised_to_peer(&mut self, authority_ids: &HashSet<AuthorityDiscoveryId>) {
193		for id in authority_ids {
194			let validator_index = match self.validators.iter().position(|v| v == id) {
195				Some(idx) => idx,
196				None => continue,
197			};
198			self.segment_advertised_to.set(validator_index, false);
199		}
200	}
201}

246/// A segment distributed by our collation-generation for a `(scheduling_parent, core)`, kept so
247/// we can (re-)advertise it to validators. All candidates in a segment share one descriptor
248/// version, hoisted here out of the per-candidate [`CandidateFingerprint`].
249struct StoredSegment {
250	/// Descriptor version shared by all candidates in the segment.
251	descriptor_version: CandidateDescriptorVersion,
252	/// Ordered candidate fingerprints advertised to validators.
253	fingerprints: BoundedVec<CandidateFingerprint, ConstU32<MAX_SEGMENT_LEN>>,
254}

256struct PerSchedulingParent {
257	/// Per core index validators group responsible for backing candidates built
258	/// on top of this relay parent.
259	validator_group: HashMap<CoreIndex, ValidatorGroup>,
260	/// Distributed collations keyed by output head hash.
261	collations: HashMap<Hash, CollationData>,
262	/// Reverse index over materialized collations.
263	by_candidate_hash: HashMap<CandidateHash, Hash>,
264	/// Cores our para is assigned to at this scheduling parent.
265	assignments: HashSet<CoreIndex>,
266	/// The relay parent block number
267	block_number: Option<BlockNumber>,
268	/// The session index of this relay parent.
269	session_index: SessionIndex,
270	/// Current segment per core, used for validator advertisements.
271	segments: HashMap<CoreIndex, StoredSegment>,
272}
273
274impl PerSchedulingParent {
275	#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
276	async fn new<Context>(
277		ctx: &mut Context,
278		runtime: &mut RuntimeInfo,
279		para_id: ParaId,
280		claim_queue: ClaimQueueSnapshot,
281		block_number: Option<BlockNumber>,
282		block_hash: Hash,
283		session_index: SessionIndex,
284	) -> Result<Self> {
285		let assignments =
286			claim_queue.iter_all_claims().fold(HashSet::new(), |mut acc, (core, claims)| {
287				let n_claims = claims.iter().filter(|para| para == &&para_id).count();
288				if n_claims > 0 {
289					acc.insert(*core);
290				}
291				acc
292			});
293
294		let mut validator_groups = HashMap::default();
295
296		for core in &assignments {
297			let GroupValidators { validators } =
298				determine_our_validators(ctx, runtime, *core, block_hash).await?;
299			validator_groups.insert(*core, ValidatorGroup::new(validators));
300		}
301		let segments: HashMap<CoreIndex, StoredSegment> = HashMap::new();
302
303		Ok(Self {
304			validator_group: validator_groups,
305			collations: HashMap::new(),
306			by_candidate_hash: HashMap::new(),
307			assignments,
308			block_number,
309			session_index,
310			segments,
311		})
312	}
313
314	/// Resolve a collation by candidate hash via the reverse index.
315	fn collation_by_hash(&self, hash: &CandidateHash) -> Option<&CollationData> {
316		self.by_candidate_hash.get(hash).and_then(|oh| self.collations.get(oh))
317	}
318
319	fn collation_by_hash_mut(&mut self, hash: &CandidateHash) -> Option<&mut CollationData> {
320		let output_head = *self.by_candidate_hash.get(hash)?;
321		self.collations.get_mut(&output_head)
322	}
323
324	fn collation_for_request(&self, req: &VersionedCollationRequest) -> Option<&CollationData> {
325		match req {
326			VersionedCollationRequest::V2(req) => {
327				self.collation_by_hash(&req.payload.candidate_hash)
328			},
329			VersionedCollationRequest::V3(req) => {
330				self.collations.get(&req.payload.output_head_data_hash)
331			},
332		}
333	}
334
335	fn collation_for_request_mut(
336		&mut self,
337		req: &VersionedCollationRequest,
338	) -> Option<&mut CollationData> {
339		match req {
340			VersionedCollationRequest::V2(req) => {
341				self.collation_by_hash_mut(&req.payload.candidate_hash)
342			},
343			VersionedCollationRequest::V3(req) => {
344				self.collations.get_mut(&req.payload.output_head_data_hash)
345			},
346		}
347	}

440/// Distribute a segment.
441///
442/// Figure out the core our para is assgined to and the relevant validators.
443/// Issues a connection request to these validators.
444/// If the para is not scheduled or next up on any core, at the scheduling-parent,
445/// or the scheduling-parent isn't in the implicit ancestry, we ignore the message
446/// as it must be invalid in that case - although this indicates a logic error
447/// elsewhere in the node.
448///
449/// A segment is write-once per `(scheduling_parent, core_index)`: the first one we accept
450/// wins and any later segment for that key is dropped. Consequently the "already advertised"
451/// bookkeeping (`ValidatorGroup::segment_advertised_to`) is a single bitvec per key rather
452/// than per candidate — there is never a second segment to advertise, so a validator that
453/// has seen this key's segment has seen everything we will ever offer for it.
454#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
455async fn distribute_segment<Context>(
456	ctx: &mut Context,
457	state: &mut State,
458	id: ParaId,
459	core_index: CoreIndex,
460	segment: Segment,
461) -> Result<()> {
462	// We should already be connected to the validators, but if we aren't, we will try to connect to
463	// them now. Do this BEFORE checking if parents are in view to ensure connections are properly
464	// updated even when rejecting out-of-view collations.
465	update_validator_connections(
466		ctx,
467		&state.peer_ids,
468		&state.implicit_view,
469		&state.per_scheduling_parent,
470		id,
471		true,
472	)
473	.await;
474
475	let (scheduling_parent, descriptor_version, sp_session, entries) = match segment {
476		Segment::V2(entry) => {
477			(entry.relay_parent, CandidateDescriptorVersion::V2, None, vec![entry])
478		},
479		Segment::V3 { scheduling_parent, scheduling_session, candidates } => (
480			scheduling_parent,
481			CandidateDescriptorVersion::V3,
482			Some(scheduling_session),
483			candidates.into_inner(),
484		),
485	};
486
487	let per_scheduling_parent = match state.per_scheduling_parent.get_mut(&scheduling_parent) {
488		Some(per_scheduling_parent) => per_scheduling_parent,
489		None => {
490			gum::debug!(
491				target: LOG_TARGET,
492				para_id = %id,
493				?scheduling_parent,
494				"Scheduling parent is out of our view",
495			);
496			return Ok(());
497		},
498	};
499
500	if entries.is_empty() {
501		gum::warn!(
502				target: LOG_TARGET,
503				para_id = %id,
504				?scheduling_parent,
505				?core_index,
506				"Received an empty segment, ignoring",
507		);
508		return Ok(());
509	}
510
511	let Some(_) = per_scheduling_parent.assignments.get(&core_index) else {
512		gum::warn!(
513			target: LOG_TARGET,
514			para_id = %id,
515			?scheduling_parent,
516			cores = ?per_scheduling_parent.assignments,
517			?core_index,
518			"Attempting to distribute collation for a core we are not assigned to ",
519		);
520
521		return Ok(());
522	};
523
524	let elastic_scaling = per_scheduling_parent.assignments.len() > 1;
525	if elastic_scaling {
526		gum::debug!(
527			target: LOG_TARGET,
528			para_id = %id,
529			cores = ?per_scheduling_parent.assignments,
530			"{} is assigned to {} cores at {}", id, per_scheduling_parent.assignments.len(), scheduling_parent,
531		);
532	}
533
534	// The group was determined for every assigned core in `PerSchedulingParent::new`, so a
535	// missing or empty one means we have no backers to advertise to.
536	let validators = match per_scheduling_parent.validator_group.get(&core_index) {
537		Some(group) if !group.validators.is_empty() => &group.validators,
538		_ => {
539			gum::warn!(
540				target: LOG_TARGET,
541				core = ?core_index,
542				"there are no validators assigned to core",
543			);
544			return Ok(());
545		},
546	};
547
548	// Segments are write-once per `(scheduling_parent, core_index)` — see the doc comment above.
549	if per_scheduling_parent.segments.contains_key(&core_index) {
550		gum::debug!(target: LOG_TARGET, "Received a new segment at core {:?}/sp:{}. Dropping it...", core_index, scheduling_parent);
551		return Ok(());
552	}
553
554	gum::debug!(
555		target: LOG_TARGET,
556		para_id = %id,
557		scheduling_parent = %scheduling_parent,
558		?core_index,
559		current_validators = ?validators,
560		"Accepted segment, connecting to validators."
561	);
562
563	let mut segment_fingerprint = vec![];
564	for entry in entries {
565		let pov_hash = entry.pov.hash();
566		let receipt =
567			assemble_receipt(id, core_index, scheduling_parent, sp_session, &entry, pov_hash);
568		let candidate_hash = receipt.hash();
569
570		let para_head = entry.output_head_data_hash;
571		segment_fingerprint.push(CandidateFingerprint {
572			output_head_data_hash: para_head,
573			parent_head_data_hash: entry.parent_head_data.hash(),
574			claim_queue_offset: 0,
575		});
576		// We have already seen collation for this scheduling parent.
577		if per_scheduling_parent.collations.contains_key(&para_head) {
578			if per_scheduling_parent.by_candidate_hash.contains_key(&candidate_hash) {
579				gum::warn!(target: LOG_TARGET, ?scheduling_parent, ?candidate_hash, "Already seen this candidate.");
580			} else {
581				gum::warn!(target: LOG_TARGET, ?scheduling_parent, ?candidate_hash, output_head = ?para_head, "Received a candidate with the same output head at this scheduling parent.");
582			}
583			continue;
584		}
585		// Store collation data
586		per_scheduling_parent.by_candidate_hash.insert(candidate_hash, para_head);
587		per_scheduling_parent.collations.insert(
588			para_head,
589			CollationData {
590				collation: Collation {
591					receipt,
592					pov: entry.pov,
593					parent_head_data: entry.parent_head_data,
594					status: CollationStatus::Created,
595				},
596				core_index,
597				stats: per_scheduling_parent.block_number.map(|n| {
598					CollationStats::new(
599						&*state.clock,
600						para_head,
601						n,
602						scheduling_parent,
603						&state.metrics,
604						*candidate_hash,
605						pov_hash,
606					)
607				}),
608				session_index: per_scheduling_parent.session_index,
609			},
610		);
611	}
612
613	let new_segment = StoredSegment {
614		descriptor_version,
615		fingerprints: BoundedVec::try_from(segment_fingerprint)
616			.expect("exactly one fingerprint per entry, entries bounded by the message; qed"),
617	};
618
619	per_scheduling_parent.segments.insert(core_index, new_segment);
620
621	// The leaf should be present in the allowed ancestry of some leaf.
622	//
623	// It's collation-producer responsibility to verify that there exists
624	// a hypothetical membership in a fragment chain for the candidate.
625	let interested = state
626		.peer_data
627		.iter()
628		.filter(|(_, PeerData { view: v, .. })| {
629			v.iter().any(|block_hash| {
630				state.implicit_view.as_ref().map(|implicit_view| {
631					implicit_view
632						.known_allowed_relay_parents_under(block_hash)
633						.unwrap_or_default()
634						.contains(&scheduling_parent)
635				}) == Some(true)
636			})
637		})
638		.map(|(id, _)| id);
639
640	for peer_id in interested {
641		// Get the peer's protocol version. The peer should exist in peer_data
642		// since we iterated over it to build `interested`.
643		let peer_version =
644			state.peer_data.get(peer_id).expect("peer from peer_data should exist").version;
645		advertise_segment(
646			ctx,
647			scheduling_parent,
648			core_index,
649			per_scheduling_parent,
650			peer_id,
651			peer_version,
652			&state.peer_ids,
653			&state.metrics,
654			id,
655		)
656		.await;
657	}
658	Ok(())
659}

708/// Figure out current group of validators assigned to the para being collated on.
709///
710/// Returns [`ValidatorId`]'s of current group as determined based on the `relay_parent`.
711#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
712async fn determine_our_validators<Context>(
713	ctx: &mut Context,
714	runtime: &mut RuntimeInfo,
715	core_index: CoreIndex,
716	relay_parent: Hash,
717) -> Result<GroupValidators> {
718	let session_index = runtime.get_session_index_for_child(ctx.sender(), relay_parent).await?;
719	let info = &runtime
720		.get_session_info_by_index(ctx.sender(), relay_parent, session_index)
721		.await?
722		.session_info;
723	gum::debug!(target: LOG_TARGET, ?session_index, "Received session info");
724	let groups = &info.validator_groups;
725	let num_cores = groups.len();
726	let rotation_info = get_group_rotation_info(ctx.sender(), relay_parent).await?;
727
728	let current_group_index = rotation_info.group_for_core(core_index, num_cores);
729	let current_validators =
730		groups.get(current_group_index).map(|v| v.as_slice()).unwrap_or_default();
731
732	let validators = &info.discovery_keys;
733
734	let current_validators =
735		current_validators.iter().map(|i| validators[i.0 as usize].clone()).collect();
736
737	let current_validators = GroupValidators { validators: current_validators };
738
739	Ok(current_validators)
740}

742/// Construct the declare message to be sent to validator.
743fn declare_message(
744	state: &mut State,
745	version: CollationVersion,
746) -> Option<
747	CollationProtocols<
748		protocol_v1::CollationProtocol,
749		protocol_v2::CollationProtocol,
750		protocol_v3::CollationProtocol,
751		protocol_v4::CollationProtocol,
752	>,
753> {
754	let para_id = state.collating_on?;
755	match version {
756		CollationVersion::V2 => {
757			let declare_signature_payload =
758				protocol_v2::declare_signature_payload(&state.local_peer_id);
759			let wire_message = protocol_v2::CollatorProtocolMessage::Declare(
760				state.collator_pair.public(),
761				para_id,
762				state.collator_pair.sign(&declare_signature_payload),
763			);
764			Some(CollationProtocols::V2(protocol_v2::CollationProtocol::CollatorProtocol(
765				wire_message,
766			)))
767		},
768		CollationVersion::V3 => {
769			let declare_signature_payload =
770				protocol_v3::declare_signature_payload(&state.local_peer_id);
771			let wire_message = protocol_v3::CollatorProtocolMessage::Declare(
772				state.collator_pair.public(),
773				para_id,
774				state.collator_pair.sign(&declare_signature_payload),
775			);
776			Some(CollationProtocols::V3(protocol_v3::CollationProtocol::CollatorProtocol(
777				wire_message,
778			)))
779		},
780		// V4 has no Declare; the para is carried in AdvertiseSegment.
781		CollationVersion::V4 => None,
782		_ => {
783			gum::warn!(target: LOG_TARGET, ?version, "Attempting to declare with an unsupported collation version");
784			None
785		},
786	}
787}

803/// Checks whether there are any core assignments for our para on any active relay chain leaves.
804fn has_assigned_cores(
805	implicit_view: &Option<ImplicitView>,
806	per_scheduling_parent: &HashMap<Hash, PerSchedulingParent>,
807) -> bool {
808	let Some(implicit_view) = implicit_view else { return false };
809
810	for leaf in implicit_view.leaves() {
811		if let Some(scheduling_parent) = per_scheduling_parent.get(leaf) {
812			if !scheduling_parent.assignments.is_empty() {
813				return true;
814			}
815		}
816	}
817
818	false
819}

821/// Get a list of backing validators at all allowed relay parents.
822/// If `pending_collation` is true, we will only return the validators
823/// that have a collation pending.
824fn list_of_backing_validators_in_view(
825	implicit_view: &Option<ImplicitView>,
826	per_scheduling_parent: &HashMap<Hash, PerSchedulingParent>,
827	pending_collation: bool,
828) -> Vec<AuthorityDiscoveryId> {
829	let mut backing_validators = HashSet::new();
830	let Some(implicit_view) = implicit_view else { return vec![] };
831
832	for leaf in implicit_view.leaves() {
833		let allowed_ancestry =
834			implicit_view.known_allowed_relay_parents_under(leaf).unwrap_or_default();
835
836		for allowed_scheduling_parent in allowed_ancestry {
837			let Some(scheduling_parent) = per_scheduling_parent.get(allowed_scheduling_parent)
838			else {
839				continue;
840			};
841
842			if pending_collation {
843				// Check if there is any collation for this scheduling parent.
844				for collation_data in scheduling_parent.collations.values() {
845					let core_index = collation_data.core_index();
846					if let Some(group) = scheduling_parent.validator_group.get(core_index) {
847						backing_validators.extend(group.validators.iter().cloned());
848					}
849				}
850			} else {
851				for group in scheduling_parent.validator_group.values() {
852					backing_validators.extend(group.validators.iter().cloned());
853				}
854			}
855		}
856	}
857
858	backing_validators.into_iter().collect()
859}

861/// Connect or disconnect to/from all backers at all viable relay parents.
862#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
863async fn update_validator_connections<Context>(
864	ctx: &mut Context,
865	peer_ids: &HashMap<PeerId, HashSet<AuthorityDiscoveryId>>,
866	implicit_view: &Option<ImplicitView>,
867	per_scheduling_parent: &HashMap<Hash, PerSchedulingParent>,
868	para_id: ParaId,
869	connect: bool,
870) {
871	gum::trace!(target: LOG_TARGET, ?para_id, ?connect, "update_validator_connections");
872
873	// Ignore address resolution failure, will reissue a new request on new collation.
874	let (failed, _) = oneshot::channel();
875
876	let msg = if connect {
877		let cores_assigned = has_assigned_cores(implicit_view, per_scheduling_parent);
878		// If no cores are assigned to the para, we still need to send a ConnectToValidators request
879		// to the network bridge passing an empty list of validator ids. Otherwise, it will keep
880		// connecting to the last requested validators until a new request is issued.
881		let validator_ids = if cores_assigned {
882			list_of_backing_validators_in_view(implicit_view, per_scheduling_parent, false)
883		} else {
884			Vec::new()
885		};
886
887		gum::trace!(
888			target: LOG_TARGET,
889			?cores_assigned,
890			"Sending connection request to validators: {:?}",
891			validator_ids,
892		);
893		NetworkBridgeTxMessage::ConnectToValidators {
894			validator_ids,
895			peer_set: PeerSet::Collation,
896			failed,
897		}
898	} else {
899		if peer_ids.is_empty() {
900			return;
901		}
902
903		let validator_ids =
904			list_of_backing_validators_in_view(implicit_view, per_scheduling_parent, true);
905
906		gum::trace!(
907			target: LOG_TARGET,
908			"Keeping connections to validators with pending collations: {:?}",
909			validator_ids,
910		);
911
912		// Disconnect from all validators with no pending collations.
913		NetworkBridgeTxMessage::ConnectToValidators {
914			validator_ids,
915			peer_set: PeerSet::Collation,
916			failed,
917		}
918	};
919
920	ctx.send_message(msg).await;
921}

923/// Advertise segment to the given `peer`.
924///
925/// This will only advertise a segment if there exists at least one for the given
926/// `scheduling_parent` and `core_index` and the giver `peer` is set as a validator
927/// for our para at the given `scheduling_parent`.
928///
929/// We also make sure not to advertise a segment multiple times to the same validator.
930#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
931async fn advertise_segment<Context>(
932	ctx: &mut Context,
933	scheduling_parent: Hash,
934	core_index: CoreIndex,
935	per_scheduling_parent: &mut PerSchedulingParent,
936	peer: &PeerId,
937	peer_version: CollationVersion,
938	peer_ids: &HashMap<PeerId, HashSet<AuthorityDiscoveryId>>,
939	metrics: &Metrics,
940	para_id: ParaId,
941) {
942	let Some(stored) = per_scheduling_parent.segments.get(&core_index) else {
943		gum::debug!(
944			target: LOG_TARGET,
945			?scheduling_parent,
946			?core_index,
947			"Skipping advertising, no segments on the sp/core key.",
948		);
949		return;
950	};
951	let candidates_descriptor_version = stored.descriptor_version;
952	let core_segment = &stored.fingerprints;
953	let Some(validator_group) = per_scheduling_parent.validator_group.get_mut(&core_index) else {
954		gum::debug!(
955			target: LOG_TARGET,
956			?scheduling_parent,
957			?core_index,
958			"Skipping advertising to validator, validator group for core not found",
959		);
960		return;
961	};
962
963	if core_segment.is_empty() {
964		gum::debug!(
965			target: LOG_TARGET,
966			?scheduling_parent,
967			?core_index,
968			"Skipping advertising to validator, segment is empty.",
969		);
970		return;
971	}
972
973	let should_advertise = validator_group.should_advertise_segment(peer_ids, &peer);
974	match should_advertise {
975		ShouldAdvertiseTo::Yes => {},
976		ShouldAdvertiseTo::NotAuthority | ShouldAdvertiseTo::AlreadyAdvertised => {
977			gum::trace!(
978				target: LOG_TARGET,
979				?scheduling_parent,
980				peer_id = %peer,
981				reason = ?should_advertise,
982				"Not advertising collation"
983			);
984			return;
985		},
986	};
987
988	let message = match peer_version {
989		CollationVersion::V4 => CollationProtocols::V4(protocol_v4::AdvertiseSegment {
990			scheduling_parent,
991			candidates_descriptor_version,
992			candidates: core_segment.clone(),
993			para_id,
994		}),
995		CollationVersion::V3 => {
996			// Send V3 protocol message with the actual descriptor version
997			let newest_candidate = core_segment.last().expect("Segment is not empty; qed");
998			let Some(tip) =
999				per_scheduling_parent.collations.get(&newest_candidate.output_head_data_hash)
1000			else {
1001				gum::debug!(
1002						target: LOG_TARGET,
1003						?scheduling_parent,
1004						?core_index,
1005						"Tip collation not stored, not advertising to old-protocol peer",
1006				);
1007				return;
1008			};
1009			let tip_candidate_hash = tip.collation().receipt.hash();
1010			CollationProtocols::V3(protocol_v3::CollationProtocol::CollatorProtocol(
1011				protocol_v3::CollatorProtocolMessage::AdvertiseCollation {
1012					scheduling_parent,
1013					candidate_hash: tip_candidate_hash,
1014					parent_head_data_hash: newest_candidate.parent_head_data_hash,
1015					candidate_descriptor_version: candidates_descriptor_version,
1016					relay_parent: tip.collation().receipt.descriptor.relay_parent(),
1017				},
1018			))
1019		},
1020		CollationVersion::V2 | CollationVersion::V1 => {
1021			// Fall back to V2 protocol for older peers
1022			let newest_candidate = core_segment.last().expect("Segment is not empty; qed");
1023			let Some(tip) =
1024				per_scheduling_parent.collations.get(&newest_candidate.output_head_data_hash)
1025			else {
1026				gum::debug!(
1027						target: LOG_TARGET,
1028						?scheduling_parent,
1029						?core_index,
1030						"Tip collation not stored, not advertising to old-protocol peer",
1031				);
1032				return;
1033			};
1034			let tip_candidate_hash = tip.collation().receipt.hash();
1035			CollationProtocols::V2(protocol_v2::CollationProtocol::CollatorProtocol(
1036				protocol_v2::CollatorProtocolMessage::AdvertiseCollation {
1037					scheduling_parent,
1038					candidate_hash: tip_candidate_hash,
1039					parent_head_data_hash: newest_candidate.parent_head_data_hash,
1040				},
1041			))
1042		},
1043	};
1044
1045	ctx.send_message(NetworkBridgeTxMessage::SendCollationMessage(vec![*peer], message))
1046		.await;
1047
1048	// Only mark the segment's collations advertised once the message is actually out. The
1049	// per-version arms above can bail without sending; advancing the status before that would
1050	// report a collation as advertised when nothing was sent, hiding the very bug that made
1051	// us bail.
1052	for fingerprint in core_segment {
1053		let Some(collation_and_core) =
1054			per_scheduling_parent.collations.get_mut(&fingerprint.output_head_data_hash)
1055		else {
1056			// Should not happen
1057			gum::warn!(
1058					target: LOG_TARGET,
1059					?scheduling_parent,
1060					output_head = ?fingerprint.output_head_data_hash,
1061					"Segment entry has no stored collation",
1062			);
1063			continue;
1064		};
1065		let collation = collation_and_core.collation_mut();
1066		collation.status.advance_to_advertised();
1067	}
1068
1069	validator_group.segment_advertised_to_peer(peer_ids, peer);
1070	metrics.on_advertisement_made();
1071}

1073/// The main incoming message dispatching switch.
1074#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
1075async fn process_msg<Context>(
1076	ctx: &mut Context,
1077	runtime: &mut RuntimeInfo,
1078	state: &mut State,
1079	msg: CollatorProtocolMessage,
1080) -> Result<()> {
1081	use CollatorProtocolMessage::*;
1082
1083	match msg {
1084		ConnectToBackingGroups => {
1085			gum::debug!(
1086				target: LOG_TARGET,
1087				"Received PreConnectToBackingGroups message."
1088			);
1089			state.connect_to_backers = true;
1090
1091			if let Some(para_id) = state.collating_on {
1092				update_validator_connections(
1093					ctx,
1094					&state.peer_ids,
1095					&state.implicit_view,
1096					&state.per_scheduling_parent,
1097					para_id,
1098					state.connect_to_backers,
1099				)
1100				.await;
1101			}
1102		},
1103		DisconnectFromBackingGroups => {
1104			gum::debug!(
1105				target: LOG_TARGET,
1106				"Received DisconnectFromBackingGroups message."
1107			);
1108			state.connect_to_backers = false;
1109
1110			if let Some(para_id) = state.collating_on {
1111				update_validator_connections(
1112					ctx,
1113					&state.peer_ids,
1114					&state.implicit_view,
1115					&state.per_scheduling_parent,
1116					para_id,
1117					state.connect_to_backers,
1118				)
1119				.await;
1120			}
1121		},
1122		CollateOn(id) => {
1123			state.collating_on = Some(id);
1124			state.implicit_view = Some(ImplicitView::new());
1125		},
1126		DistributeSegment { core_index, para_id, segment } => {
1127			match state.collating_on {
1128				Some(id) if para_id != id => {
1129					// If the ParaId of a collation requested to be distributed does not match
1130					// the one we expect, we ignore the message.
1131					gum::warn!(
1132						target: LOG_TARGET,
1133						%para_id,
1134						collating_on = %id,
1135						"DistributeSegment for unexpected para_id",
1136					);
1137				},
1138				Some(id) => {
1139					gum::trace!(target: LOG_TARGET, "DistributeSegment for para_id: {}", id);
1140					let _ = state.metrics.time_collation_distribution("distribute");
1141					distribute_segment(ctx, state, id, core_index, segment).await?;
1142				},
1143				None => {
1144					gum::warn!(
1145						target: LOG_TARGET,
1146						%para_id,
1147						"DistributeSegment message while not collating on any",
1148					);
1149				},
1150			}
1151		},
1152		NetworkBridgeUpdate(event) => {
1153			// We should count only this shoulder in the histogram, as other shoulders are just
1154			// introducing noise
1155			let _ = state.metrics.time_process_msg();
1156
1157			if let Err(e) = handle_network_msg(ctx, runtime, state, event).await {
1158				gum::warn!(
1159					target: LOG_TARGET,
1160					err = ?e,
1161					"Failed to handle incoming network message",
1162				);
1163			}
1164		},
1165		msg @ (Invalid(..) | Seconded(..)) => {
1166			gum::warn!(
1167				target: LOG_TARGET,
1168				"{:?} message is not expected on the collator side of the protocol",
1169				msg,
1170			);
1171		},
1172	}
1173
1174	Ok(())
1175}

1213/// A networking messages switch.
1214#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
1215async fn handle_incoming_peer_message<Context>(
1216	ctx: &mut Context,
1217	state: &mut State,
1218	origin: PeerId,
1219	msg: CollationProtocols<
1220		protocol_v1::CollatorProtocolMessage,
1221		protocol_v2::CollatorProtocolMessage,
1222		protocol_v3::CollatorProtocolMessage,
1223		protocol_v4::AdvertiseSegment,
1224	>,
1225) -> Result<()> {
1226	use protocol_v1::CollatorProtocolMessage as V1;
1227	use protocol_v2::CollatorProtocolMessage as V2;
1228	use protocol_v3::CollatorProtocolMessage as V3;
1229
1230	match msg {
1231		CollationProtocols::V1(V1::Declare(..)) |
1232		CollationProtocols::V2(V2::Declare(..)) |
1233		CollationProtocols::V3(V3::Declare(..)) => {
1234			gum::trace!(
1235				target: LOG_TARGET,
1236				?origin,
1237				"Declare message is not expected on the collator side of the protocol",
1238			);
1239
1240			// If we are declared to, this is another collator, and we should disconnect.
1241			ctx.send_message(NetworkBridgeTxMessage::DisconnectPeers(
1242				vec![origin],
1243				PeerSet::Collation,
1244			))
1245			.await;
1246		},
1247		CollationProtocols::V1(V1::AdvertiseCollation(_)) |
1248		CollationProtocols::V2(V2::AdvertiseCollation { .. }) |
1249		CollationProtocols::V3(V3::AdvertiseCollation { .. }) => {
1250			gum::trace!(
1251				target: LOG_TARGET,
1252				?origin,
1253				"AdvertiseCollation message is not expected on the collator side of the protocol",
1254			);
1255
1256			modify_reputation(&mut state.reputation, ctx.sender(), origin, COST_UNEXPECTED_MESSAGE)
1257				.await;
1258
1259			// If we are advertised to, this is another collator, and we should disconnect.
1260			ctx.send_message(NetworkBridgeTxMessage::DisconnectPeers(
1261				vec![origin],
1262				PeerSet::Collation,
1263			))
1264			.await;
1265		},
1266		// `AdvertiseSegment` is the only V4 message, so this arm covers the whole
1267		// version rather than one variant of it.
1268		CollationProtocols::V4(_) => {
1269			gum::trace!(
1270				target: LOG_TARGET,
1271				?origin,
1272				"AdvertiseSegment message is not expected on the collator side of the protocol",
1273			);
1274
1275			modify_reputation(&mut state.reputation, ctx.sender(), origin, COST_UNEXPECTED_MESSAGE)
1276				.await;
1277
1278			// If we are advertised to, this is another collator, and we should disconnect.
1279			ctx.send_message(NetworkBridgeTxMessage::DisconnectPeers(
1280				vec![origin],
1281				PeerSet::Collation,
1282			))
1283			.await;
1284		},
1285		CollationProtocols::V1(V1::CollationSeconded(..)) |
1286		CollationProtocols::V2(V2::CollationSeconded(..)) |
1287		CollationProtocols::V3(V3::CollationSeconded(..)) => {
1288			// Validators running older versions still send this. We no longer
1289			// track seconding, so the message carries no information for us.
1290			gum::trace!(
1291					target: LOG_TARGET,
1292					?origin,
1293					"Ignoring `CollationSeconded` message",
1294			);
1295		},
1296	}
1297
1298	Ok(())
1299}

1301/// Process an incoming network request for a collation.
1302#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
1303async fn handle_incoming_request<Context>(
1304	ctx: &mut Context,
1305	state: &mut State,
1306	req: std::result::Result<VersionedCollationRequest, incoming::Error>,
1307) -> Result<()> {
1308	let req = req?;
1309	let scheduling_parent = req.scheduling_parent();
1310	let peer_id = req.peer_id();
1311	let para_id = req.para_id();
1312
1313	match state.collating_on {
1314		Some(our_para_id) if our_para_id == para_id => {
1315			let per_scheduling_parent =
1316				match state.per_scheduling_parent.get_mut(&scheduling_parent) {
1317					Some(per_scheduling_parent) => per_scheduling_parent,
1318					None => {
1319						gum::debug!(
1320							target: LOG_TARGET,
1321							%scheduling_parent,
1322							"received a `RequestCollation` for a scheduling parent out of our view",
1323						);
1324
1325						return Ok(());
1326					},
1327				};
1328
1329			let collation_with_core = per_scheduling_parent.collation_for_request_mut(&req);
1330			let (receipt, pov, parent_head_data) =
1331				if let Some(collation_with_core) = collation_with_core {
1332					let collation = collation_with_core.collation_mut();
1333					collation.status.advance_to_requested();
1334					(
1335						collation.receipt.clone(),
1336						collation.pov.clone(),
1337						collation.parent_head_data.clone(),
1338					)
1339				} else {
1340					gum::warn!(
1341						target: LOG_TARGET,
1342						scheduling_parent = %scheduling_parent,
1343						"received a `RequestCollation` for a relay parent we don't have collation stored.",
1344					);
1345
1346					return Ok(());
1347				};
1348
1349			state.metrics.on_collation_sent_requested();
1350
1351			let waiting = state.waiting_collation_fetches.entry(scheduling_parent).or_default();
1352			let candidate_hash = receipt.hash();
1353
1354			if !waiting.waiting_peers.insert((peer_id, candidate_hash)) {
1355				gum::debug!(
1356					target: LOG_TARGET,
1357					"Dropping incoming request as peer has a request in flight already."
1358				);
1359				modify_reputation(
1360					&mut state.reputation,
1361					ctx.sender(),
1362					peer_id,
1363					COST_APPARENT_FLOOD.into(),
1364				)
1365				.await;
1366				return Ok(());
1367			}
1368
1369			if waiting.collation_fetch_active {
1370				waiting.req_queue.push_back(req);
1371			} else {
1372				waiting.collation_fetch_active = true;
1373				// Obtain a timer for sending collation
1374				let _ = state.metrics.time_collation_distribution("send");
1375
1376				send_collation(state, req, receipt, pov, parent_head_data).await;
1377			}
1378		},
1379		Some(our_para_id) => {
1380			gum::warn!(
1381				target: LOG_TARGET,
1382				for_para_id = %para_id,
1383				our_para_id = %our_para_id,
1384				"received a `CollationFetchingRequest` for unexpected para_id",
1385			);
1386		},
1387		None => {
1388			gum::warn!(
1389				target: LOG_TARGET,
1390				for_para_id = %para_id,
1391				"received a `RequestCollation` while not collating on any para",
1392			);
1393		},
1394	}
1395	Ok(())
1396}

1398/// Advertises collations for the given scheduling parents to the specified peer.
1399///
1400/// Returns a list of unknown scheduling parents.
1401#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
1402async fn advertise_collations_for_scheduling_parents<Context>(
1403	ctx: &mut Context,
1404	state: &mut State,
1405	peer_id: &PeerId,
1406	scheduling_parents: impl IntoIterator<Item = Hash>,
1407) -> Vec<Hash> {
1408	let mut unknown_scheduling_parents = Vec::new();
1409
1410	let peer_version = match state.peer_data.get(peer_id) {
1411		Some(peer) => peer.version,
1412		None => return unknown_scheduling_parents,
1413	};
1414
1415	let Some(para_id) = state.collating_on else {
1416		gum::debug!(target: LOG_TARGET, "Not collating on a para, skipping advertisements.");
1417		return unknown_scheduling_parents;
1418	};
1419
1420	for scheduling_parent in scheduling_parents {
1421		let block_hashes = match state.per_scheduling_parent.contains_key(&scheduling_parent) {
1422			true => state
1423				.implicit_view
1424				.as_ref()
1425				.and_then(|implicit_view| {
1426					implicit_view.known_allowed_relay_parents_under(&scheduling_parent)
1427				})
1428				.unwrap_or_default(),
1429			false => {
1430				unknown_scheduling_parents.push(scheduling_parent);
1431				continue;
1432			},
1433		};
1434
1435		for block_hash in block_hashes {
1436			if let Some(per_scheduling_parent) = state.per_scheduling_parent.get_mut(block_hash) {
1437				let cores: Vec<CoreIndex> =
1438					per_scheduling_parent.segments.keys().copied().collect();
1439				for core_index in cores {
1440					advertise_segment(
1441						ctx,
1442						*block_hash,
1443						core_index,
1444						per_scheduling_parent,
1445						peer_id,
1446						peer_version,
1447						&state.peer_ids,
1448						&state.metrics,
1449						para_id,
1450					)
1451					.await;
1452				}
1453			}
1454		}
1455	}
1456
1457	unknown_scheduling_parents
1458}

1460/// Peer's view has changed. Send advertisements for new scheduling parents
1461/// if there're any.
1462#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
1463async fn handle_peer_view_change<Context>(
1464	ctx: &mut Context,
1465	state: &mut State,
1466	peer_id: PeerId,
1467	view: View,
1468) {
1469	let Some(added) = state.peer_data.get_mut(&peer_id).map(|peer| {
1470		let diff: Vec<Hash> = view.difference(&peer.view).cloned().collect();
1471		peer.view = view;
1472		diff
1473	}) else {
1474		return;
1475	};
1476
1477	let unknown_scheduling_parents =
1478		advertise_collations_for_scheduling_parents(ctx, state, &peer_id, added).await;
1479
1480	if !unknown_scheduling_parents.is_empty() {
1481		gum::trace!(
1482			target: LOG_TARGET,
1483			?peer_id,
1484			new_leaves = ?unknown_scheduling_parents,
1485			"New leaves in peer's view are unknown",
1486		);
1487
1488		if let Some(PeerData { unknown_heads, .. }) = state.peer_data.get_mut(&peer_id) {
1489			for unknown in unknown_scheduling_parents {
1490				unknown_heads.insert(unknown, ());
1491			}
1492		}
1493	}
1494}

1496/// Bridge messages switch.
1497#[overseer::contextbounds(CollatorProtocol, prefix = self::overseer)]
1498async fn handle_network_msg<Context>(
1499	ctx: &mut Context,
1500	runtime: &mut RuntimeInfo,
1501	state: &mut State,
1502	bridge_message: NetworkBridgeEvent<net_protocol::CollatorProtocolMessage>,
1503) -> Result<()> {
1504	use NetworkBridgeEvent::*;
1505
1506	match bridge_message {
1507		PeerConnected(peer_id, observed_role, protocol_version, maybe_authority) => {
1508			// If it is possible that a disconnected validator would attempt a reconnect
1509			// it should be handled here.
1510			gum::trace!(target: LOG_TARGET, ?peer_id, ?observed_role, ?maybe_authority, "Peer connected");
1511
1512			let version: CollationVersion = match protocol_version.try_into() {
1513				Ok(version) => version,
1514				Err(err) => {
1515					// Network bridge is expected to handle this.
1516					gum::error!(
1517						target: LOG_TARGET,
1518						?peer_id,
1519						?observed_role,
1520						?err,
1521						"Unsupported protocol version"
1522					);
1523					return Ok(());
1524				},
1525			};
1526			if version == CollationVersion::V1 {
1527				gum::warn!(
1528					target: LOG_TARGET,
1529					?peer_id,
1530					?observed_role,
1531					"Unsupported protocol version v1"
1532				);
1533
1534				// V1 no longer supported, we should disconnect.
1535				ctx.send_message(NetworkBridgeTxMessage::DisconnectPeers(
1536					vec![peer_id],
1537					PeerSet::Collation,
1538				))
1539				.await;
1540				return Ok(());
1541			}
1542
1543			state.peer_data.entry(peer_id).or_insert_with(|| PeerData {
1544				view: View::default(),
1545				// Unlikely that the collator is falling 10 blocks behind and if so, it probably is
1546				// not able to keep up any way.
1547				unknown_heads: LruMap::new(ByLength::new(10)),
1548				version,
1549			});
1550
1551			if let Some(authority_ids) = maybe_authority {
1552				gum::trace!(
1553					target: LOG_TARGET,
1554					?authority_ids,
1555					?peer_id,
1556					"Connected to requested validator"
1557				);
1558				state.peer_ids.insert(peer_id, authority_ids);
1559
1560				declare(ctx, state, &peer_id, version).await;
1561			}
1562		},
1563		PeerViewChange(peer_id, view) => {
1564			gum::trace!(target: LOG_TARGET, ?peer_id, ?view, "Peer view change");
1565			handle_peer_view_change(ctx, state, peer_id, view).await;
1566		},
1567		PeerDisconnected(peer_id) => {
1568			gum::trace!(target: LOG_TARGET, ?peer_id, "Peer disconnected");
1569
1570			// Reset advertised_to bits for this peer so that collations will be
1571			// re-advertised when they reconnect.
1572			if let Some(authority_ids) = state.peer_ids.get(&peer_id) {
1573				for per_scheduling_parent in state.per_scheduling_parent.values_mut() {
1574					for validator_group in per_scheduling_parent.validator_group.values_mut() {
1575						validator_group.reset_segment_advertised_to_peer(authority_ids);
1576					}
1577				}
1578			}
1579
1580			state.peer_data.remove(&peer_id);
1581			state.peer_ids.remove(&peer_id);
1582		},
1583		OurViewChange(view) => {
1584			gum::trace!(target: LOG_TARGET, ?view, "Own view change");
1585			handle_our_view_change(ctx, runtime, state, view).await?;
1586			// Connect only if we are collating on a para.
1587			if let Some(para_id) = state.collating_on {
1588				update_validator_connections(
1589					ctx,
1590					&state.peer_ids,
1591					&state.implicit_view,
1592					&state.per_scheduling_parent,
1593					para_id,
1594					state.connect_to_backers,
1595				)
1596				.await;
1597			}
1598		},
1599		PeerMessage(remote, msg) => {
1600			handle_incoming_peer_message(ctx, state, remote, msg).await?;
1601		},
1602		UpdatedAuthorityIds(peer_id, authority_ids) => {
1603			gum::trace!(target: LOG_TARGET, ?peer_id, ?authority_ids, "Updated authority ids");
1604			if let Some(version) = state.peer_data.get(&peer_id).map(|data| data.version) {
1605				let is_new_peer = state.peer_ids.insert(peer_id, authority_ids).is_none();
1606
1607				if is_new_peer {
1608					// The peer connected before its authority ids were known; declare
1609					// now, using its negotiated protocol version.
1610					declare(ctx, state, &peer_id, version).await;
1611				} else {
1612					// Authority IDs changed for an existing peer. Re-advertise collations
1613					// for scheduling parents already in their view, as the previous
1614					// authority IDs may not have matched our validator groups.
1615					let scheduling_parents_in_view: Vec<_> = state
1616						.peer_data
1617						.get(&peer_id)
1618						.map(|data| data.view.iter().cloned().collect())
1619						.unwrap_or_default();
1620
1621					// Unknown scheduling parents are ignored because they were
1622					// handled when the peer's view was first processed.
1623					let _ = advertise_collations_for_scheduling_parents(
1624						ctx,
1625						state,
1626						&peer_id,
1627						scheduling_parents_in_view,
1628					)
1629					.await;
1630				}
1631			}
1632		},
1633		NewGossipTopology { .. } => {
1634			// impossible!
1635		},
1636	}
1637
1638	Ok(())
1639}

1641/// Update collation tracker with the backed and included candidates.
1642#[overseer::contextbounds(CollatorProtocol, prefix = crate::overseer)]
1643async fn process_block_events<Context>(
1644	ctx: &mut Context,
1645	collation_tracker: &mut CollationTracker,
1646	leaf: Hash,
1647	maybe_block_number: Option<BlockNumber>,
1648	para_id: ParaId,
1649) {
1650	if let Ok(events) = get_candidate_events(ctx.sender(), leaf).await {
1651		let Some(block_number) = maybe_block_number else {
1652			// This should not happen. If it does this log message explains why
1653			// metrics and logs are missing for the candidates under this block.
1654			gum::debug!(
1655				target: crate::LOG_TARGET_STATS,
1656				relay_block = ?leaf,
1657				?para_id,
1658				"Failed to get relay chain block number",
1659			);
1660			return;
1661		};
1662
1663		for ev in events {
1664			match ev {
1665				CandidateEvent::CandidateIncluded(receipt, _, _, _) => {
1666					if receipt.descriptor.para_id() != para_id {
1667						continue;
1668					}
1669					collation_tracker.collation_included(block_number, leaf, receipt);
1670				},
1671				CandidateEvent::CandidateBacked(receipt, _, _, _) => {
1672					if receipt.descriptor.para_id() != para_id {
1673						continue;
1674					}
1675
1676					let Some(block_number) = maybe_block_number else { continue };
1677					collation_tracker.collation_backed(block_number, leaf, receipt);
1678				},
1679				_ => {
1680					// do not care about other events
1681				},
1682			}
1683		}
1684	}
1685}

1687/// Handles our view changes.
1688#[overseer::contextbounds(CollatorProtocol, prefix = crate::overseer)]
1689async fn handle_our_view_change<Context>(
1690	ctx: &mut Context,
1691	runtime: &mut RuntimeInfo,
1692	state: &mut State,
1693	view: OurView,
1694) -> Result<()> {
1695	let Some(implicit_view) = &mut state.implicit_view else { return Ok(()) };
1696	let Some(para_id) = state.collating_on else { return Ok(()) };
1697
1698	let removed: Vec<_> = implicit_view.leaves().filter(|h| !view.contains(h)).copied().collect();
1699	let added: Vec<_> = view.iter().filter(|h| !implicit_view.contains_leaf(h)).collect();
1700
1701	for leaf in added {
1702		let session_index = runtime.get_session_index_for_child(ctx.sender(), *leaf).await?;
1703
1704		let claim_queue = fetch_claim_queue(ctx.sender(), *leaf).await?;
1705
1706		implicit_view
1707			.activate_leaf(ctx.sender(), *leaf)
1708			.await
1709			.map_err(Error::ImplicitViewFetchError)?;
1710
1711		let block_number = implicit_view.block_number(leaf);
1712
1713		state.per_scheduling_parent.insert(
1714			*leaf,
1715			PerSchedulingParent::new(
1716				ctx,
1717				runtime,
1718				para_id,
1719				claim_queue,
1720				block_number,
1721				*leaf,
1722				session_index,
1723			)
1724			.await?,
1725		);
1726
1727		process_block_events(ctx, &mut state.collation_tracker, *leaf, block_number, para_id).await;
1728		let allowed_ancestry =
1729			implicit_view.known_allowed_relay_parents_under(leaf).unwrap_or_default();
1730
1731		// Get the peers that already reported us this head, but we didn't know it at this
1732		// point.
1733		let peers = state
1734			.peer_data
1735			.iter_mut()
1736			.filter_map(|(id, data)| data.unknown_heads.remove(leaf).map(|_| *id))
1737			.collect::<Vec<_>>();
1738
1739		for block_hash in allowed_ancestry {
1740			let block_number = implicit_view.block_number(block_hash);
1741
1742			let per_relay_parent = match state.per_scheduling_parent.entry(*block_hash) {
1743				Entry::Vacant(entry) => {
1744					let claim_queue = match fetch_claim_queue(ctx.sender(), *block_hash).await {
1745						Ok(cq) => cq,
1746						Err(error) => {
1747							gum::debug!(
1748								target: LOG_TARGET,
1749								?block_hash,
1750								?error,
1751								"Failed to fetch claim queue while iterating allowed ancestry",
1752							);
1753							continue;
1754						},
1755					};
1756					let session_index =
1757						match runtime.get_session_index_for_child(ctx.sender(), *leaf).await {
1758							Ok(si) => si,
1759							Err(error) => {
1760								gum::debug!(
1761									target: LOG_TARGET,
1762									?block_hash,
1763									?error,
1764									"Failed to fetch session index while iterating allowed ancestry",
1765								);
1766								continue;
1767							},
1768						};
1769
1770					entry.insert(
1771						PerSchedulingParent::new(
1772							ctx,
1773							runtime,
1774							para_id,
1775							claim_queue,
1776							block_number,
1777							*block_hash,
1778							session_index,
1779						)
1780						.await?,
1781					)
1782				},
1783				Entry::Occupied(entry) => entry.into_mut(),
1784			};
1785
1786			// Announce relevant collations to these peers.
1787			for peer_id in &peers {
1788				// Get the peer's protocol version, skip if peer disconnected
1789				let Some(peer_version) = state.peer_data.get(peer_id).map(|data| data.version)
1790				else {
1791					gum::debug!(
1792						target: LOG_TARGET,
1793						?peer_id,
1794						?block_hash,
1795						"Peer not found in peer_data, likely disconnected. Skipping advertisement.",
1796					);
1797					continue;
1798				};
1799
1800				let cores: Vec<CoreIndex> = per_relay_parent.segments.keys().copied().collect();
1801				for core_index in cores {
1802					advertise_segment(
1803						ctx,
1804						*block_hash,
1805						core_index,
1806						per_relay_parent,
1807						peer_id,
1808						peer_version,
1809						&state.peer_ids,
1810						&state.metrics,
1811						para_id,
1812					)
1813					.await;
1814				}
1815			}
1816		}
1817	}
1818
1819	let highest_session_index =
1820		state.per_scheduling_parent.values().map(|pr| pr.session_index).max();
1821
1822	for leaf in removed {
1823		// If the leaf is deactivated it still may stay in the view as a part
1824		// of implicit ancestry. Only update the state after the hash is actually
1825		// pruned from the block info storage.
1826		let maybe_block_number = implicit_view.block_number(&leaf);
1827		let pruned = implicit_view.deactivate_leaf(leaf);
1828
1829		for removed in &pruned {
1830			gum::debug!(
1831				target: LOG_TARGET,
1832				relay_parent = ?removed,
1833				"Removing relay parent because our view changed.",
1834			);
1835
1836			if let Some(block_number) = maybe_block_number {
1837				let expired_collations = state.collation_tracker.drain_expired(block_number);
1838				process_expired_collations(expired_collations, para_id, &state.metrics);
1839			}
1840
1841			// Get all the collations built on top of the removed leaf.
1842			let collations = state
1843				.per_scheduling_parent
1844				.remove(removed)
1845				.map(|per_relay_parent| per_relay_parent.collations)
1846				.unwrap_or_default();
1847
1848			for collation_with_core in collations.into_values() {
1849				process_out_of_view_collation(
1850					&mut state.collation_tracker,
1851					collation_with_core,
1852					highest_session_index,
1853				);
1854			}
1855
1856			state.waiting_collation_fetches.remove(removed);
1857		}
1858	}
1859	Ok(())
1860}

1862fn process_out_of_view_collation(
1863	collation_tracker: &mut CollationTracker,
1864	mut collation_with_core: CollationData,
1865	highest_session_index: Option<SessionIndex>,
1866) {
1867	let is_same_session =
1868		highest_session_index.map_or(true, |hs| hs == collation_with_core.session_index);
1869	let collation = collation_with_core.collation_mut();
1870	let candidate_hash = collation.receipt.hash();
1871
1872	match collation.status {
1873		CollationStatus::Created => {
1874			if is_same_session {
1875				gum::warn!(
1876					target: LOG_TARGET,
1877					?candidate_hash,
1878					pov_hash = ?collation.pov.hash(),
1879					"Collation wasn't advertised to any validator.",
1880				)
1881			} else {
1882				gum::debug!(
1883					target: LOG_TARGET,
1884					?candidate_hash,
1885					pov_hash = ?collation.pov.hash(),
1886					"Collation wasn't advertised because it was built on a relay chain block that is now part of an old session.",
1887				)
1888			}
1889		},
1890		CollationStatus::Advertised => gum::debug!(
1891			target: LOG_TARGET,
1892			?candidate_hash,
1893			pov_hash = ?collation.pov.hash(),
1894			"Collation was advertised but not requested by any validator.",
1895		),
1896		CollationStatus::Requested => {
1897			gum::debug!(
1898				target: LOG_TARGET,
1899				?candidate_hash,
1900				pov_hash = ?collation.pov.hash(),
1901				"Collation was requested.",
1902			);
1903		},
1904	}
1905
1906	let collation_status = collation.status.clone();
1907	let Some(mut stats) = collation_with_core.take_stats() else { return };
1908
1909	// If the collation stats are still available, it means it was never
1910	// successfully fetched, even if a fetch request was received, but not succeed.
1911	//
1912	// Will expire in it's current state at the next block import.
1913	stats.set_pre_backing_status(collation_status);
1914	collation_tracker.track(stats);
1915}

1952/// Process collations that may have been finalized on the relay chain.
1953///
1954/// Collations are no longer tracked after this call.
1955///
1956/// ## Limitations
1957///
1958/// We only track the inclusion block number, not the inclusion block hash.
1959/// This means a collation included in a fork that was later dropped could still be counted as
1960/// finalized. This is acceptable for metrics because such cases are rare.
1961async fn process_possibly_finalized_collations(
1962	collations: Vec<CollationStats>,
1963	last_finalized: (Hash, BlockNumber),
1964	sender: &mut impl overseer::SubsystemSender<ChainApiMessage>,
1965	metrics: &Metrics,
1966) {
1967	if collations.is_empty() {
1968		return;
1969	}
1970
1971	let (last_hash, last_number) = last_finalized;
1972	let mut blocks_to_request = collations
1973		.iter()
1974		.map(|stats| stats.relay_parent().1)
1975		.filter(|n| *n != last_number) // No need to request `last_hash` for `last_number`
1976		.collect::<Vec<_>>();
1977	blocks_to_request.sort_unstable();
1978	blocks_to_request.dedup();
1979
1980	gum::debug!(target: LOG_TARGET_STATS, ?blocks_to_request, "Collations possibly finalized on blocks");
1981
1982	let mut finalized = vec![(last_hash, last_number)];
1983
1984	for chunk in blocks_to_request.as_slice().chunks(MAX_PARALLEL_CHAIN_API_REQUESTS) {
1985		let futures = chunk.iter().map(|&bn| {
1986				let mut sender = sender.clone();
1987				async move {
1988					let (tx, rx) = oneshot::channel();
1989					sender.send_message(ChainApiMessage::FinalizedBlockHash(bn, tx)).await;
1990					match rx.await {
1991						Ok(Ok(Some(bh))) => Some((bh, bn)),
1992						_ => {
1993							gum::warn!(target: LOG_TARGET_STATS, block_number = ?bn, "Can't request hash for the finalized block from ChainApi");
1994							None
1995						},
1996					}
1997				}
1998			});
1999		finalized.extend(futures::future::join_all(futures).await.into_iter().flatten());
2000	}
2001
2002	for collation in collations {
2003		if !finalized.contains(&collation.relay_parent()) {
2004			// Omit collations built on forks: they're dropped but it's expected
2005			gum::debug!(
2006				target: crate::LOG_TARGET_STATS,
2007				relay_parent = ?collation.relay_parent(),
2008				head = ?collation.head(),
2009				"Collation is built on a fork, skipping",
2010			);
2011			continue;
2012		}
2013
2014		// Report metrics for finalized collations
2015		if let Some(latency) = collation.backed() {
2016			metrics.on_collation_backed(latency as f64);
2017		}
2018		if let Some(latency) = collation.included() {
2019			metrics.on_collation_included(latency as f64);
2020		}
2021		gum::debug!(
2022			target: crate::LOG_TARGET_STATS,
2023			relay_parent = ?collation.relay_parent(),
2024			head = ?collation.head(),
2025			"Collation finalized, stop tracking",
2026		);
2027	}
2028}
