75	/// Handle a new peer connection.
76	pub async fn handle_peer_connected<Sender: CollatorProtocolSenderTrait>(
77		&mut self,
78		sender: &mut Sender,
79		peer_id: PeerId,
80		version: CollationVersion,
81	) {
82		let _timer = self.metrics.time_handler(TimedHandler::PeerConnected);
83
84		let outcome = self
85			.peer_manager
86			.try_accept_connection(
87				sender,
88				peer_id,
89				PeerInfo { version, state: PeerState::Connected },
90			)
91			.await;
92		match outcome {
93			TryAcceptOutcome::Added => {
94				gum::trace!(
95					target: LOG_TARGET,
96					?peer_id,
97					?version,
98					"Peer connected",
99				);
100			},
101			TryAcceptOutcome::Replaced(other_peers) => {
102				gum::trace!(
103					target: LOG_TARGET,
104					?peer_id,
105					?version,
106					?other_peers,
107					"Peer connected and replaced the connection slots of other peers",
108				);
109				self.collation_manager.remove_peers(other_peers.iter());
110			},
111			TryAcceptOutcome::Rejected => {
112				gum::debug!(
113					target: LOG_TARGET,
114					?peer_id,
115					?version,
116					"Peer connection was rejected. Going to disconnect",
117				);
118			},
119		}
120
121		self.metrics.note_collator_peer_count(self.peer_manager.connected_peer_count());
122	}

158	/// Handle our view update.
159	pub async fn handle_our_view_change<Sender: CollatorProtocolSenderTrait>(
160		&mut self,
161		sender: &mut Sender,
162		new_view: OurView,
163	) -> FatalResult<()> {
164		let _timer = self.metrics.time_handler(TimedHandler::OurViewChange);
165
166		gum::trace!(
167			target: LOG_TARGET,
168			?new_view,
169			"Handling our view change",
170		);
171		let old_assignments = self.collation_manager.assignments();
172
173		self.collation_manager.update_view(sender, new_view).await?;
174
175		let new_assignments = self.collation_manager.assignments();
176		gum::trace!(
177			target: LOG_TARGET,
178			?old_assignments,
179			?new_assignments,
180			"Old assignments vs new assignments",
181		);
182
183		if old_assignments != new_assignments {
184			gum::debug!(
185				target: LOG_TARGET,
186				?old_assignments,
187				?new_assignments,
188				"Collator protocol assignments changed",
189			);
190		}
191
192		self.metrics.note_assigned_paras(new_assignments.len());
193
194		let maybe_disconnected_peers =
195			self.peer_manager.scheduled_paras_update(sender, new_assignments).await;
196
197		if !maybe_disconnected_peers.is_empty() {
198			gum::trace!(
199				target: LOG_TARGET,
200				?maybe_disconnected_peers,
201				"Disconnecting peers due to our view change",
202			);
203		}
204
205		self.collation_manager.remove_peers(maybe_disconnected_peers.iter());
206		self.metrics.note_collator_peer_count(self.peer_manager.connected_peer_count());
207
208		Ok(())
209	}

211	/// Handle a finalized block notification.
212	pub async fn handle_finalized_block<Sender: CollatorProtocolSenderTrait>(
213		&mut self,
214		sender: &mut Sender,
215		hash: Hash,
216		number: BlockNumber,
217	) -> FatalResult<()> {
218		let _timer = self.metrics.time_handler(TimedHandler::FinalizedBlock);
219
220		gum::trace!(
221			target: LOG_TARGET,
222			?hash,
223			number,
224			"Processing new block finality notification",
225		);
226
227		// Process reputation bumps
228		if let Err(err) = self
229			.peer_manager
230			.update_reputations_on_new_finalized_block(sender, (hash, number))
231			.await
232		{
233			err.split()?.log();
234		}
235
236		// Refresh the per-para reputation score distribution metric
237		self.metrics.note_score_distribution(&self.peer_manager.score_distribution());
238
239		// Process potential changes in the registered paras set.
240		let session_index = match recv_runtime(request_session_index_for_child(hash, sender).await)
241			.await
242			.map_err(Error::Runtime)
243		{
244			Ok(session_index) => session_index,
245			Err(err) => {
246				err.split()?.log();
247				return Ok(());
248			},
249		};
250
251		self.peer_manager.prune_registered_paras(sender, session_index, hash).await;
252
253		Ok(())
254	}

256	/// Handle a new advertisement.
257	pub async fn handle_advertisement<Sender: CollatorProtocolSenderTrait>(
258		&mut self,
259		sender: &mut Sender,
260		peer_id: PeerId,
261		scheduling_parent: Hash,
262		entries: Vec<ProspectiveCandidate>,
263		descriptor_version: Option<CandidateDescriptorVersion>,
264		// Some for V4 self declaring ad.
265		advertised_para_id: Option<ParaId>,
266	) {
267		let advertisement_log = if advertised_para_id.is_some() {
268			"Received a segment advertisement"
269		} else {
270			"Received advertisement"
271		};
272		let _timer = self.metrics.time_handler(TimedHandler::Advertisement);
273
274		gum::debug!(
275			target: LOG_TARGET,
276			?scheduling_parent,
277			?peer_id,
278			advertisement_log,
279		);
280
281		// V4 has no `Declare`: a peer's first advertisement carries its para and binds it.
282		// Until then a V4 peer holds a reserved slot on every scheduled para; binding here
283		// releases the slots it held on all the other paras.
284		if let Some(para_id) = advertised_para_id {
285			if !self.peer_manager.declared(sender, peer_id, para_id).await {
286				self.collation_manager.remove_peer(&peer_id);
287				return;
288			}
289		}
290
291		let Some(PeerInfo { state, .. }) = self.peer_manager.peer_info(&peer_id) else {
292			self.metrics.on_advertisement_rejected_unconnected_peer();
293			gum::warn!(
294				target: LOG_TARGET,
295				?scheduling_parent,
296				?peer_id,
297				"Received an advertisement from an unconnected peer"
298			);
299			return;
300		};
301
302		// Advertised without being declared. Not a big waste of our time, so ignore it.
303		let PeerState::Collating(para_id) = state else {
304			self.metrics.on_advertisement_rejected_undeclared_peer();
305			gum::debug!(
306				target: LOG_TARGET,
307				?scheduling_parent,
308				?peer_id,
309				"Received advertisement for undeclared peer",
310			);
311			return;
312		};
313
314		// We have a result here, but it's not worth affecting reputations because advertisements
315		// are cheap.
316		// Note: `try_accept_segment` involves two other subsystems, so it's not super cheap,
317		// actually, but cheap enough.
318		match self
319			.collation_manager
320			.try_accept_segment(
321				sender,
322				peer_id,
323				*para_id,
324				scheduling_parent,
325				descriptor_version,
326				entries,
327			)
328			.await
329		{
330			Err(err) => {
331				match err {
332					AdvertisementError::Duplicate => {
333						self.metrics.on_advertisement_rejected_duplicate(para_id)
334					},
335					AdvertisementError::OutOfOurView => {
336						self.metrics.on_advertisement_rejected_out_of_view(para_id)
337					},
338					AdvertisementError::PeerLimitReached => {
339						self.metrics.on_advertisement_rejected_peer_limit_reached(para_id)
340					},
341					AdvertisementError::BlockedByBacking => {
342						self.metrics.on_advertisement_rejected_blocked_by_backing(para_id)
343					},
344					AdvertisementError::V1AdvertisementForImplicitParent => {
345						self.metrics.on_advertisement_rejected_v1_for_implicit_parent(para_id)
346					},
347					AdvertisementError::SchedulingParentNotValid => {
348						self.metrics.on_advertisement_rejected_scheduling_parent_invalid(para_id)
349					},
350				}
351				gum::debug!(
352					target: LOG_TARGET,
353					?scheduling_parent,
354					?peer_id,
355					?para_id,
356					?err,
357					"Advertisement rejected",
358				);
359			},
360			Ok(()) => {
361				self.metrics.on_advertisement_accepted(para_id);
362				gum::debug!(
363					target: LOG_TARGET,
364					?scheduling_parent,
365					?peer_id,
366					?para_id,
367					"Advertisement accepted",
368				);
369			},
370		}
371	}

379	pub async fn handle_fetched_collation<Sender: CollatorProtocolSenderTrait>(
380		&mut self,
381		sender: &mut Sender,
382		res: CollationFetchResponse,
383	) {
384		let _timer = self.metrics.time_handler(TimedHandler::FetchedCollation);
385
386		let fetch_result = res.1.is_ok();
387		let advertisement = res.0;
388
389		if let Err(err) = &res.1 {
390			gum::debug!(
391				target: LOG_TARGET,
392				?advertisement,
393				"Collation fetch attempt failed: {}",
394				err
395			);
396		} else {
397			gum::debug!(
398				target: LOG_TARGET,
399				?advertisement,
400				"Collation fetch attempt succeeded",
401			);
402		}
403
404		let collation_version = self.peer_manager.get_peer_protocol_version(&advertisement.peer_id);
405		let can_second = self.collation_manager.note_fetched(sender, res, collation_version).await;
406
407		// To be consistent with the old implementation, if the fetch is successful we count the
408		// request as successful, despite we might not be able to second it.
409		let collation_request_metrics_result = if fetch_result { Ok(()) } else { Err(()) };
410		match can_second {
411			CanSecond::Yes(candidate_receipt, pov, pvd) => {
412				let para_id = candidate_receipt.descriptor.para_id();
413				sender
414					.send_message(CandidateBackingMessage::Second {
415						scheduling_parent: candidate_receipt.descriptor().scheduling_parent(),
416						candidate: candidate_receipt,
417						pvd,
418						pov,
419					})
420					.await;
421
422				self.metrics.on_collation_seconded(&para_id);
423
424				gum::debug!(
425					target: LOG_TARGET,
426					?advertisement,
427					"Started seconding"
428				);
429			},
430			CanSecond::No(maybe_slash, reject_info) => {
431				gum::debug!(
432					target: LOG_TARGET,
433					?maybe_slash,
434					?reject_info,
435					"Cannot second collation",
436				);
437
438				if let Some(slash) = maybe_slash {
439					self.metrics.on_slash_failed_fetch(&reject_info.para_id);
440					self.peer_manager
441						.slash_reputation(&reject_info.peer_id, &reject_info.para_id, slash)
442						.await;
443				}
444
445				self.collation_manager.release_slot(
446					&reject_info.scheduling_parent,
447					reject_info.para_id,
448					reject_info.maybe_candidate_hash.as_ref(),
449					reject_info.maybe_output_head_hash,
450				);
451			},
452			CanSecond::BlockedOnParent(parent_hash, reject_info) => {
453				self.metrics.on_collation_blocked_on_parent(&reject_info.para_id);
454				gum::debug!(
455					target: LOG_TARGET,
456					?parent_hash,
457					?reject_info,
458					"Collation blocked on parent, waiting for parent to be validated",
459				);
460			},
461		};
462
463		self.metrics.on_request(collation_request_metrics_result);
464	}

466	pub async fn handle_invalid_collation(
467		&mut self,
468		receipt: CandidateReceipt,
469		scheduling_parent: Hash,
470	) {
471		let _timer = self.metrics.time_handler(TimedHandler::InvalidCollation);
472
473		let candidate_hash = receipt.hash();
474
475		gum::debug!(
476			target: LOG_TARGET,
477			para_id = ?receipt.descriptor.para_id(),
478			?scheduling_parent,
479			?candidate_hash,
480			"Invalid collation",
481		);
482
483		let maybe_peer_id = self.collation_manager.release_slot(
484			&scheduling_parent,
485			receipt.descriptor.para_id(),
486			Some(&candidate_hash),
487			Some(receipt.descriptor.para_head()),
488		);
489
490		let Some(peer_id) = maybe_peer_id else {
491			gum::warn!(
492				target: LOG_TARGET,
493				para_id = ?receipt.descriptor.para_id(),
494				?scheduling_parent,
495				?candidate_hash,
496				"Could not find the peer id of the invalid collation",
497			);
498			return;
499		};
500
501		gum::debug!(
502			target: LOG_TARGET,
503			?scheduling_parent,
504			?candidate_hash,
505			?peer_id,
506			"Invalid collation reported, slashing peer reputation",
507		);
508
509		self.metrics.on_slash_invalid_collation(&receipt.descriptor.para_id());
510		self.peer_manager
511			.slash_reputation(&peer_id, &receipt.descriptor.para_id(), INVALID_COLLATION_SLASH)
512			.await;
513	}

515	pub async fn handle_seconded_collation<Sender: CollatorProtocolSenderTrait>(
516		&mut self,
517		sender: &mut Sender,
518		statement: SignedFullStatement,
519		scheduling_parent: Hash,
520	) {
521		let _timer = self.metrics.time_handler(TimedHandler::Seconded);
522
523		let receipt = match statement.payload() {
524			Statement::Seconded(receipt) => receipt,
525			Statement::Valid(_) => {
526				gum::warn!(
527					target: LOG_TARGET,
528					?statement,
529					"Seconded message received with a `Valid` statement",
530				);
531				return;
532			},
533		};
534
535		let candidate_hash = receipt.hash();
536		let para_id = receipt.descriptor.para_id();
537
538		gum::debug!(
539			target: LOG_TARGET,
540			?para_id,
541			?scheduling_parent,
542			?candidate_hash,
543			"Collation seconded",
544		);
545
546		let (peer_id, unblocked_collations) = self
547			.collation_manager
548			.note_seconded(
549				sender,
550				&scheduling_parent,
551				&para_id,
552				&candidate_hash,
553				receipt.descriptor.para_head(),
554			)
555			.await;
556
557		match peer_id {
558			Some(peer_id) => match self.peer_manager.peer_info(&peer_id) {
559				Some(PeerInfo { version, .. }) => {
560					gum::debug!(
561						target: LOG_TARGET,
562						?para_id,
563						?scheduling_parent,
564						?candidate_hash,
565						?peer_id,
566						"Notifying collator about seconded collation",
567					);
568					notify_collation_seconded(
569						sender,
570						peer_id,
571						*version,
572						scheduling_parent,
573						statement,
574					)
575					.await;
576				},
577				// We know who fetched it, but they disconnected before we could ack the second.
578				None => {
579					gum::trace!(
580						target: LOG_TARGET,
581						?para_id,
582						?scheduling_parent,
583						?candidate_hash,
584						?peer_id,
585						"Not notifying collator about seconded collation: peer no longer connected",
586					);
587				},
588			},
589			// No tracked fetcher for this candidate (e.g. its slot was already released).
590			None => {
591				gum::trace!(
592					target: LOG_TARGET,
593					?para_id,
594					?scheduling_parent,
595					?candidate_hash,
596					"Not notifying any collator about seconded collation: fetcher unknown",
597				);
598			},
599		}
600
601		if !unblocked_collations.is_empty() {
602			gum::debug!(
603				target: LOG_TARGET,
604				?scheduling_parent,
605				?candidate_hash,
606				?para_id,
607				"Seconded candidate unblocked {} collations",
608				unblocked_collations.len(),
609			);
610
611			self.try_second_unblocked_collations(sender, unblocked_collations).await;
612		}
613	}

615	pub async fn try_launch_new_fetch_requests<Sender: CollatorProtocolSenderTrait>(
616		&mut self,
617		sender: &mut Sender,
618	) -> Option<Duration> {
619		let _timer = self.metrics.time_handler(TimedHandler::LaunchFetchRequests);
620
621		let peer_manager = &self.peer_manager;
622		let connected_rep_query_fn = move |peer_id: &PeerId, para_id: &ParaId| {
623			peer_manager.connected_peer_score(peer_id, para_id)
624		};
625		let max_reps = self
626			.peer_manager
627			.max_scores_for_paras(self.collation_manager.assignments())
628			.await;
629
630		let metrics = &self.metrics;
631		let create_timer_fn = || metrics.time_collation_request_duration();
632
633		let (requests, maybe_delay) = self.collation_manager.try_make_new_fetch_requests(
634			connected_rep_query_fn,
635			max_reps,
636			create_timer_fn,
637		);
638
639		if !requests.is_empty() {
640			gum::debug!(
641				target: LOG_TARGET,
642				?requests,
643				"Sending {} collation fetch requests",
644				requests.len()
645			);
646
647			sender
648				.send_message(NetworkBridgeTxMessage::SendRequests(
649					requests,
650					IfDisconnected::ImmediateError,
651				))
652				.await;
653		}
654
655		maybe_delay
656	}

658	async fn try_second_unblocked_collations<Sender: CollatorProtocolSenderTrait>(
659		&mut self,
660		sender: &mut Sender,
661		unblocked_collations: Vec<CanSecond>,
662	) {
663		for can_second_unblocked in unblocked_collations {
664			match can_second_unblocked {
665				CanSecond::Yes(candidate_receipt, pov, pvd) => {
666					let candidate_hash = candidate_receipt.hash();
667					let para_id = candidate_receipt.descriptor.para_id();
668					let scheduling_parent = candidate_receipt.descriptor().scheduling_parent();
669
670					sender
671						.send_message(CandidateBackingMessage::Second {
672							scheduling_parent,
673							candidate: candidate_receipt,
674							pvd,
675							pov,
676						})
677						.await;
678
679					self.metrics.on_collation_seconded(&para_id);
680
681					gum::debug!(
682						target: LOG_TARGET,
683						?scheduling_parent,
684						?candidate_hash,
685						?para_id,
686						"Started seconding unblocked collation"
687					);
688				},
689				CanSecond::No(maybe_slash, reject_info) => {
690					gum::debug!(
691						target: LOG_TARGET,
692						scheduling_parent = ?reject_info.scheduling_parent,
693						maybe_candidate_hash = ?reject_info.maybe_candidate_hash,
694						para_id = ?reject_info.para_id,
695						"Cannot second unblocked collation"
696					);
697
698					if let Some(slash) = maybe_slash {
699						self.metrics.on_slash_failed_fetch(&reject_info.para_id);
700						self.peer_manager
701							.slash_reputation(&reject_info.peer_id, &reject_info.para_id, slash)
702							.await;
703					}
704
705					self.collation_manager.release_slot(
706						&reject_info.scheduling_parent,
707						reject_info.para_id,
708						reject_info.maybe_candidate_hash.as_ref(),
709						reject_info.maybe_output_head_hash,
710					);
711				},
712				CanSecond::BlockedOnParent(parent, reject_info) => {
713					gum::warn!(
714						target: LOG_TARGET,
715						scheduling_parent = ?reject_info.scheduling_parent,
716						maybe_candidate_hash = ?reject_info.maybe_candidate_hash,
717						?parent,
718						para_id = ?reject_info.para_id,
719						"Cannot second unblocked collation even though its parent was just seconded"
720					);
721
722					self.collation_manager.release_slot(
723						&reject_info.scheduling_parent,
724						reject_info.para_id,
725						reject_info.maybe_candidate_hash.as_ref(),
726						reject_info.maybe_output_head_hash,
727					);
728				},
729			}
730		}
731	}
