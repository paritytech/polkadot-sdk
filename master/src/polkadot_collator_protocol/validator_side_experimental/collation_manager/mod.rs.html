68/// Reason for rejecting an advertisement.
69#[derive(Debug, thiserror::Error)]
70pub enum AdvertisementError {
71	#[error("Duplicate advertisement")]
72	Duplicate,
73	#[error("Advertised scheduling parent is out of our view")]
74	OutOfOurView,
75	#[error("Peer reached the candidate limit (or para is not schedulable from this SP)")]
76	PeerLimitReached,
77	#[error("Seconding not allowed by backing subsystem")]
78	BlockedByBacking,
79	#[error("V1 advertisements are only allowed on active leaves")]
80	V1AdvertisementForImplicitParent,
81	#[error("For V3 candidate descriptors, scheduling_parent does not match any expected scheduling parent.")]
82	SchedulingParentNotValid,
83}

145	pub async fn update_view<Sender: CollatorProtocolSenderTrait>(
146		&mut self,
147		sender: &mut Sender,
148		new_view: OurView,
149	) -> FatalResult<()> {
150		let removed = self
151			.implicit_view
152			.leaves()
153			.filter(|h| !new_view.contains(h))
154			.cloned()
155			.collect::<Vec<_>>();
156		let added = new_view
157			.iter()
158			.filter(|h| !self.implicit_view.contains_leaf(h))
159			.cloned()
160			.collect::<Vec<_>>();
161
162		gum::trace!(
163			target: LOG_TARGET,
164			?added,
165			?removed,
166			"CollationManager: Processing view update"
167		);
168
169		for leaf in added.iter() {
170			match extract_leaf_scheduling_info(sender, *leaf).await {
171				Some(info) => {
172					self.leaf_scheduling_info.insert(*leaf, info);
173				},
174				None => {
175					gum::warn!(
176						target: LOG_TARGET,
177						?leaf,
178						"Could not extract BABE slot from leaf header; \
179						 V3 scheduling parent validation will reject advertisements \
180						 referencing this leaf",
181					);
182				},
183			}
184
185			if let Err(err) = self
186				.implicit_view
187				.activate_leaf(sender, *leaf)
188				.await
189				.map_err(Error::FailedToActivateLeafInImplicitView)
190			{
191				err.split()?.log();
192				continue;
193			}
194		}
195
196		for leaf in removed {
197			self.implicit_view.deactivate_leaf(leaf);
198			self.leaf_scheduling_info.remove(&leaf);
199			self.leaf_claim_queues.remove(&leaf);
200		}
201
202		// Rebuild `per_scheduling_parent`, dropping entries no longer reachable from any
203		// current leaf and cancelling their in-flight fetches.
204		self.per_scheduling_parent = std::mem::take(&mut self.per_scheduling_parent)
205			.into_iter()
206			.filter_map(|(sp, per_sp)| {
207				if !self.implicit_view.paths_via_relay_parent(&sp).is_empty() {
208					return Some((sp, per_sp));
209				}
210
211				gum::trace!(
212					target: LOG_TARGET,
213					scheduling_parent = ?sp,
214					"Scheduling parent no longer reachable from any leaf; dropping it and cancelling its in-flight fetches",
215				);
216
217				for (advertisement, _) in per_sp.all_advertisements() {
218					self.fetching.cancel(advertisement);
219				}
220				None
221			})
222			.collect();
223
224		// Remove blocked seconding requests whose scheduling parent is no longer tracked.
225		self.blocked_from_seconding.retain(|_, collations| {
226			collations.retain(|c| self.per_scheduling_parent.contains_key(&c.scheduling_parent()));
227			!collations.is_empty()
228		});
229
230		for leaf in added.iter() {
231			let Some(allowed_ancestry) =
232				self.implicit_view.known_allowed_relay_parents_under(leaf).map(|v| v.to_vec())
233			else {
234				continue;
235			};
236			let session_index =
237				match recv_runtime(request_session_index_for_child(*leaf, sender).await)
238					.await
239					.map_err(Error::Runtime)
240				{
241					Ok(session_index) => session_index,
242					Err(err) => {
243						err.split()?.log();
244						continue;
245					},
246				};
247
248			// Register every newly-known scheduling parent (the leaf and any of its allowed
249			// ancestors not yet in our view) with the core our group is assigned to *at that
250			// block*. This is what determines which core's view of the leaf's CQ applies to
251			// advertisements rooted at that scheduling parent.
252			for ancestor in allowed_ancestry.iter() {
253				if self.per_scheduling_parent.contains_key(ancestor) {
254					continue;
255				}
256
257				let core = match self.get_our_core(sender, ancestor, session_index).await {
258					Ok(core) => core,
259					Err(err) => {
260						err.split()?.log();
261						continue;
262					},
263				};
264				gum::trace!(
265					target: LOG_TARGET,
266					scheduling_parent = ?ancestor,
267					?core,
268					session_index,
269					"Registered scheduling parent on our assigned core",
270				);
271				self.per_scheduling_parent
272					.insert(*ancestor, PerSchedulingParent::new(session_index, core, &*self.clock));
273			}
274
275			// Fetch and store the leaf's per-core claim queues and scheduling lookahead. Capacity
276			// at every scheduling parent on a path to this leaf is computed from these via offset
277			// arithmetic — the leaf is authoritative because it's closest to what the runtime will
278			// see when candidates get backed.
279			match LeafClaimQueues::fetch(*leaf, session_index, sender)
280				.await
281				.map_err(Error::Runtime)
282			{
283				Ok(leaf_claim_queues) => {
284					self.leaf_claim_queues.insert(*leaf, leaf_claim_queues);
285				},
286				Err(err) => {
287					err.split()?.log();
288					continue;
289				},
290			}
291		}
292
293		Ok(())
294	}

300	/// All paras our group will back at *some* scheduling parent in our view. Used to decide
301	/// which collators we should be willing to talk to. We take the union across all
302	/// scheduling parents of `our_window(sp)` — the slice of the leaf's CQ visible from that
303	/// SP for our core.
304	pub fn assignments(&self) -> BTreeSet<ParaId> {
305		self.per_scheduling_parent
306			.iter()
307			.flat_map(|(sp, per_sp)| self.our_window(sp, per_sp.core_index))
308			.collect()
309	}
310
311	/// Number of CQ positions assigned to `para_id` in the SP's visible window of our core.
312	///
313	/// Returns `0` if the SP isn't in view.
314	///
315	/// Note: this is *not* a capacity check. Capacity (which slots are still unfulfilled) is
316	/// enforced separately in `try_make_new_fetch_requests` via
317	/// `unfulfilled_claim_queue_entries`. Accepting an advertisement that won't be fetchable
318	/// right away is fine — it stays parked in `peer_advertisements` until a slot opens up.
319	fn slots_available(&self, scheduling_parent: &Hash, para_id: ParaId) -> usize {
320		let Some(per_sp) = self.per_scheduling_parent.get(scheduling_parent) else {
321			return 0;
322		};
323		self.our_window(scheduling_parent, per_sp.core_index)
324			.iter()
325			.filter(|p| **p == para_id)
326			.count()
327	}

329	pub async fn try_accept_advertisement<Sender: CollatorProtocolSenderTrait>(
330		&mut self,
331		sender: &mut Sender,
332		advertisement: Advertisement,
333	) -> std::result::Result<(), AdvertisementError> {
334		let Advertisement {
335			scheduling_parent,
336			para_id,
337			prospective_candidate,
338			advertised_descriptor_version,
339			..
340		} = advertisement;
341
342		// V1 advertisements are only allowed on active leaves.
343		if prospective_candidate.is_none() && !self.implicit_view.contains_leaf(&scheduling_parent)
344		{
345			return Err(AdvertisementError::V1AdvertisementForImplicitParent);
346		}
347
348		// V3 candidate descriptors require scheduling_parent to be the block from the last
349		// finished relay chain slot.
350		if advertised_descriptor_version == Some(CandidateDescriptorVersion::V3) &&
351			!is_scheduling_parent_valid(
352				&*self.clock,
353				&scheduling_parent,
354				&self.leaf_scheduling_info,
355			) {
356			return Err(AdvertisementError::SchedulingParentNotValid);
357		}
358
359		let available_slots = self.slots_available(&scheduling_parent, para_id);
360
361		let Some(per_sp) = self.per_scheduling_parent.get_mut(&scheduling_parent) else {
362			return Err(AdvertisementError::OutOfOurView);
363		};
364
365		if let Some(ProspectiveCandidate { candidate_hash, .. }) = prospective_candidate {
366			if per_sp.fetched_collations.contains_key(&candidate_hash) {
367				return Err(AdvertisementError::Duplicate);
368			}
369		}
370
371		if self.fetching.contains(&advertisement) {
372			return Err(AdvertisementError::Duplicate);
373		}
374
375		per_sp.can_keep_advertisement(advertisement, available_slots)?;
376
377		if !backing_allows_seconding(sender, &advertisement).await {
378			return Err(AdvertisementError::BlockedByBacking);
379		}
380
381		per_sp.add_advertisement(advertisement, self.clock.now());
382
383		Ok(())
384	}

386	/// CQ positions at `core` schedulable by an advertisement made at `scheduling_parent`.
387	///
388	/// We use the *leaf's* CQ rather than the SP's: the SP's original CQ predicted slots
389	/// SP+1…SP+L, but `d` of those have already been filled by the blocks from SP up to
390	/// and including the leaf. The leaf's CQ is what remains unconsumed.
391	///
392	/// Example: leaf-CQ = [A, B, C, D] (L=4), SP at depth d=2:
393	///
394	///   blocks:   SP ──── b₁ ──── leaf ──── s₁ ──── s₂ ──── s₃ ──── s₄
395	///                  ╰─ 2 of SP's ─╯       ▲       ▲       ▲       ▲
396	///                  ╰─ CQ filled  ─╯      A       B       C       D
397	///                                        ╰── usable ──╯╰── trimmed ──╯
398	///
399	/// `s₃, s₄` would land after SP exits view (its lifetime is bounded by L), so we
400	/// keep [A, B] = leaf-CQ[0 .. L-d).
401	///
402	/// Across forks the same SP sits under multiple leaves with different `d` and
403	/// prefix-nested windows, so longest = union.
404	fn our_window(&self, scheduling_parent: &Hash, core: CoreIndex) -> Vec<ParaId> {
405		self.implicit_view
406			.paths_via_relay_parent(scheduling_parent)
407			.into_iter()
408			.filter_map(|path| {
409				let leaf = path.last()?;
410				let depth = path
411					.iter()
412					.rev()
413					.position(|h| h == scheduling_parent)
414					.expect("paths_via_relay_parent only returns paths containing the SP; qed");
415				Some(self.leaf_claim_queues.get(leaf)?.window(core, depth))
416			})
417			.max_by_key(Vec::len)
418			.unwrap_or_default()
419	}

421	pub fn try_make_new_fetch_requests<
422		RepQueryFn: Fn(&PeerId, &ParaId) -> Option<Score>,
423		TimerFn: FnMut() -> Option<HistogramTimer>,
424	>(
425		&mut self,
426		connected_rep_query_fn: RepQueryFn,
427		max_scores: HashMap<ParaId, Score>,
428		mut create_timer_fn: TimerFn,
429	) -> (Vec<Requests>, Option<Duration>) {
430		let now = self.clock.now();
431		let mut requests = vec![];
432		let mut maybe_min_delay = None;
433
434		// Build per-(leaf, core) capacity views once, with all current consumers already
435		// allocated. Each `LeafCoreCq` is a self-contained answer to "what's still free on
436		// this core's CQ at this leaf?".
437		let mut leaf_core_cqs = self.build_leaf_core_cqs();
438
439		// Fill claim queue positions for each (leaf, core), starting at the back for best
440		// utilization.
441		for lc_idx in 0..leaf_core_cqs.len() {
442			let cq_len = leaf_core_cqs[lc_idx].cq.len();
443			for idx in (0..cq_len).rev() {
444				let Some(para_id) = leaf_core_cqs[lc_idx].cq[idx] else { continue };
445
446				let candidate_sps = leaf_core_cqs[lc_idx].sps_reaching(idx);
447				let highest_rep_of_para = max_scores.get(&para_id).copied().unwrap_or_default();
448
449				let outcome = self.pick_best_advertisement(
450					now,
451					para_id,
452					candidate_sps,
453					highest_rep_of_para,
454					&connected_rep_query_fn,
455				);
456
457				let advertisement = match outcome {
458					Either::Left(Some(adv)) => adv,
459					Either::Left(None) => continue,
460					Either::Right(delay) => {
461						maybe_min_delay = Some(
462							maybe_min_delay
463								.map_or(delay, |min: Duration| std::cmp::min(min, delay)),
464						);
465						continue;
466					},
467				};
468
469				gum::trace!(
470					target: LOG_TARGET,
471					peer_id = ?advertisement.peer_id,
472					?para_id,
473					scheduling_parent = ?advertisement.scheduling_parent,
474					maybe_candidate_hash = ?advertisement.candidate_hash(),
475					"Requesting collation",
476				);
477				let req = self.fetching.launch(&advertisement, create_timer_fn());
478				requests.push(req);
479
480				// Reserve on _all_ reachable leaf-core views. `reserve_slot` is a no-op for views
481				// whose `path` doesn't contain this SP — including cross-core views.
482				for lc in leaf_core_cqs.iter_mut() {
483					lc.reserve_slot(&advertisement.scheduling_parent, para_id);
484				}
485			}
486		}
487
488		(requests, maybe_min_delay)
489	}

491	/// One LeafCoreCq per (leaf, core) pair we need to reason about. After rotation a single
492	/// chain may yield LeafCoreCqs under two different cores.
493	///
494	/// Each LeafCoreCq comes back with all current consumers (in-flight + fetched candidates
495	/// whose SP lies on its chain *and* uses its core) already allocated into the CQ via
496	/// greedy matching: narrowest window first, latest still-free position in window —
497	/// pushing wide-window consumers to later positions so narrower SPs keep access to their
498	/// (only) reachable positions.
499	fn build_leaf_core_cqs(&self) -> Vec<LeafCoreCq> {
500		// One LeafCoreCq per (leaf, core) pair where some tracked SP lives on `core`.
501		let cores: BTreeSet<CoreIndex> =
502			self.per_scheduling_parent.values().map(|p| p.core_index).collect();
503		let leaves: BTreeSet<Hash> = self.implicit_view.leaves().copied().collect();
504
505		let mut out: Vec<LeafCoreCq> = Vec::new();
506		for leaf in leaves {
507			for &core in &cores {
508				let Some(leaf_cqs) = self.leaf_claim_queues.get(&leaf) else { continue };
509				let Some(path) = self.implicit_view.known_allowed_relay_parents_under(&leaf) else {
510					continue;
511				};
512				let Some(mut cq) = leaf_cqs.slots(core) else { continue };
513				// SPs by depth from the leaf (leaf = 0). Cross-core ancestors are masked as
514				// `None` so `sps_reaching` and `reserve_slot` automatically skip them.
515				let sps_by_depth: Vec<Option<Hash>> = path
516					.iter()
517					.map(|sp_hash| {
518						self.per_scheduling_parent
519							.get(sp_hash)
520							.is_some_and(|per_sp| per_sp.core_index == core)
521							.then_some(*sp_hash)
522					})
523					.collect();
524
525				// Collect consumers as `(para, valid_len)` for every same-core SP on the path.
526				let mut consumers: Vec<(ParaId, usize)> = Vec::new();
527				for (depth, sp_hash) in
528					sps_by_depth.iter().enumerate().filter_map(|(i, x)| x.map(|h| (i, h)))
529				{
530					let Some(per_sp) = self.per_scheduling_parent.get(&sp_hash) else { continue };
531					let valid_len = cq.len().saturating_sub(depth);
532					let in_flight = self
533						.fetching
534						.iter()
535						.filter(|adv| adv.scheduling_parent == sp_hash)
536						.map(|adv| adv.para_id);
537					let fetched = per_sp.fetched_collations.values().map(|info| info.para_id);
538					for para in in_flight.chain(fetched) {
539						consumers.push((para, valid_len));
540					}
541				}
542
543				// Allocate narrowest-first, latest-position-in-window. Overflow (no free
544				// position in window — typically a stale claim from a CQ change at an older
545				// ancestor) is tolerated quietly.
546				consumers.sort_by_key(|(_, valid_len)| *valid_len);
547				for (para, valid_len) in consumers {
548					if let Some(latest) =
549						cq[..valid_len].iter().rposition(|slot| *slot == Some(para))
550					{
551						cq[latest] = None;
552					}
553				}
554
555				out.push(LeafCoreCq { sps_by_depth, cq });
556			}
557		}
558		out
559	}

561	pub fn remove_peer(&mut self, peer: &PeerId) {
562		for per_sp in self.per_scheduling_parent.values_mut() {
563			// No need to reset now the statuses of claims that were pending fetch for these
564			// candidates, or even cancel the futures as the requests will soon conclude with a
565			// network error.
566			per_sp.remove_peer_advertisements(peer);
567		}
568	}

577	pub async fn note_fetched<Sender: CollatorProtocolSenderTrait>(
578		&mut self,
579		sender: &mut Sender,
580		res: CollationFetchResponse,
581		maybe_collation_version: Option<CollationVersion>,
582	) -> CanSecond {
583		let advertisement = res.0;
584		let mut reject_info = SecondingRejectionInfo::from(&advertisement);
585
586		self.fetching.note_completed(&advertisement);
587
588		let Some(per_sp) = self.per_scheduling_parent.get_mut(&advertisement.scheduling_parent)
589		else {
590			gum::debug!(
591				target: LOG_TARGET,
592				hash = ?advertisement.scheduling_parent,
593				para_id = ?advertisement.para_id,
594				peer_id = ?advertisement.peer_id,
595				"Collation fetch concluded for scheduling parent out of view"
596			);
597			return CanSecond::No(None, reject_info);
598		};
599
600		per_sp.remove_advertisement(&advertisement);
601
602		let Some(collation_version) = maybe_collation_version else {
603			gum::debug!(
604				target: LOG_TARGET,
605				?advertisement,
606				"Peer may not be connected."
607			);
608			return CanSecond::No(None, reject_info);
609		};
610
611		match process_collation_fetch_result(
612			res,
613			&mut self.network_error_freq,
614			&mut self.canceled_freq,
615		) {
616			Ok(fetched_collation) => {
617				let candidate_hash = fetched_collation.candidate_receipt.hash();
618				// It can't be a duplicate, because we check before initiating fetch. For the old
619				// protocol version, we anyway only fetch one per scheduling parent.
620				per_sp.fetched_collations.insert(
621					candidate_hash,
622					FetchedCollationInfo {
623						peer_id: advertisement.peer_id,
624						para_id: advertisement.para_id,
625					},
626				);
627
628				// Now that the candidate hash is known, populate it on the rejection info so
629				// V1 release paths can clean up the right entry too.
630				reject_info.maybe_candidate_hash = Some(candidate_hash);
631				reject_info.maybe_output_head_hash =
632					Some(fetched_collation.candidate_receipt.descriptor.para_head());
633
634				// Some initial sanity checks on the fetched collation, based on the advertisement.
635				if let Err(err) = fetched_collation.ensure_matches_advertisement(&advertisement) {
636					gum::warn!(
637						target: LOG_TARGET,
638						?advertisement,
639						"Invalid fetched collation: {}",
640						err
641					);
642					return CanSecond::No(Some(FAILED_FETCH_SLASH), reject_info);
643				}
644
645				let scheduling_session = per_sp.session_index;
646
647				// Sanity check of the candidate receipt version.
648				if let Err(err) = descriptor_version_sanity_check_with_params(
649					fetched_collation.candidate_receipt.descriptor(),
650					per_sp.core_index,
651					scheduling_session,
652					collation_version,
653				) {
654					gum::warn!(
655						target: LOG_TARGET,
656						?advertisement,
657						"Failed descriptor version sanity check for fetched collation: {}",
658						err
659					);
660					return CanSecond::No(Some(FAILED_FETCH_SLASH), reject_info);
661				}
662
663				self.can_begin_seconding(
664					sender,
665					scheduling_session,
666					fetched_collation,
667					true,
668					reject_info,
669				)
670				.await
671			},
672			Err(rep_change) => CanSecond::No(rep_change, reject_info),
673		}
674	}

677	/// (validation rejected, blocked-on-parent gave up, etc.). After this, capacity at
678	/// `scheduling_parent` for `para_id` increases by one. Returns the peer id of the fetcher

686		scheduling_parent: &Hash,
687		para_id: ParaId,
688		maybe_candidate_hash: Option<&CandidateHash>,
689		maybe_output_head_hash: Option<Hash>,

692			let info = self
693				.per_scheduling_parent
694				.get_mut(scheduling_parent)?
695				.fetched_collations

698				gum::debug!(

782	/// Advertisements at `sp` for `para_id` that are *fetchable right now* — i.e. all dedup
783	/// checks (already-fetched, in-flight, V1 single-shot per `(sp, para)`) have been applied.
784	fn eligible_advertisements<'a>(
785		&'a self,
786		sp: Hash,
787		para_id: ParaId,
788	) -> impl Iterator<Item = (&'a Advertisement, &'a Instant)> {
789		// `Either` unifies the two iterator types into one `impl Iterator`: empty for an
790		// untracked SP, the filter chain otherwise.
791		let per_sp = match self.per_scheduling_parent.get(&sp) {
792			Some(p) => p,
793			None => return Either::Left(std::iter::empty()),
794		};
795
796		// V1 ads have no candidate hash and are only meaningful at the block they were
797		// advertised against — they require their SP to be an active leaf.
798		let is_active_leaf = self.implicit_view.contains_leaf(&sp);
799
800		// V1 has no candidate hash to dedup by, so at most one V1 fetch may be in-flight or
801		// already fetched per (sp, para). Multiple peers may hold V1 ads for the same
802		// (sp, para); we must filter out *all* V1 ads for that (sp, para) once one is taken.
803		let v1_blocked = per_sp.fetched_collations.values().any(|info| info.para_id == para_id) ||
804			self.fetching.iter().any(|adv| {
805				adv.scheduling_parent == sp &&
806					adv.para_id == para_id &&
807					adv.prospective_candidate.is_none()
808			});
809
810		let fetching = &self.fetching;
811		Either::Right(per_sp.all_advertisements().filter(move |(adv, _)| {
812			if adv.para_id != para_id {
813				return false;
814			}
815			if fetching.contains(adv) {
816				return false;
817			}
818			match adv.prospective_candidate {
819				None => is_active_leaf && !v1_blocked,
820				Some(p) => !per_sp.fetched_collations.contains_key(&p.candidate_hash),
821			}
822		}))
823	}

825	/// Picks the best (= highest-scored, earliest, in that order) advertisement for `para_id`
826	/// among `candidate_sps`, with delay arithmetic relative to each SP's activation.
827	///
828	/// Returns:
829	/// - `Either::Left(Some(adv))` if a fetchable advertisement was found,
830	/// - `Either::Left(None)` if there are no eligible advertisements,
831	/// - `Either::Right(delay)` if the best advertisement still has remaining fetch delay relative
832	///   to its scheduling parent's activation time.
833	fn pick_best_advertisement<RepQueryFn: Fn(&PeerId, &ParaId) -> Option<Score>>(
834		&self,
835		now: Instant,
836		para_id: ParaId,
837		candidate_sps: impl Iterator<Item = Hash>,
838		highest_rep_of_para: Score,
839		connected_rep_query_fn: &RepQueryFn,
840	) -> Either<Option<Advertisement>, Duration> {
841		let advertisements: BTreeSet<AcceptedAdvertisement> = candidate_sps
842			.filter_map(|sp| {
843				let activated_at = self.per_scheduling_parent.get(&sp)?.activated_at;
844				Some(self.eligible_advertisements(sp, para_id).filter_map(
845					move |(adv, timestamp)| {
846						Some(AcceptedAdvertisement {
847							adv,
848							score: connected_rep_query_fn(&adv.peer_id, &adv.para_id)?,
849							timestamp,
850							activated_at,
851						})
852					},
853				))
854			})
855			.flatten()
856			.collect();
857
858		// `Ord` is custom: descending by score, so first = best.
859		let Some(best) = advertisements.first() else {
860			gum::trace!(
861				target: LOG_TARGET,
862				?para_id,
863				"No fetchable advertisement for a free claim-queue slot",
864			);
865			return Either::Left(None);
866		};
867
868		let delay = Self::calculate_delay(best.score, highest_rep_of_para);
869
870		// Delay is relative to the chosen SP's activation, not advertisement arrival — once
871		// the SP has been active long enough, even unknown peers' delays elapse and we fetch
872		// immediately.
873		let elapsed = now.duration_since(best.activated_at);
874		let remaining = delay.saturating_sub(elapsed);
875
876		if remaining.is_zero() {
877			gum::debug!(
878				target: LOG_TARGET,
879				peer_id = ?best.adv.peer_id,
880				scheduling_parent = ?best.adv.scheduling_parent,
881				para_id = ?best.adv.para_id,
882				?elapsed,
883				?delay,
884				"Delay elapsed; initiating fetch."
885			);
886			Either::Left(Some(*best.adv))
887		} else {
888			gum::trace!(
889				target: LOG_TARGET,
890				peer_id = ?best.adv.peer_id,
891				scheduling_parent = ?best.adv.scheduling_parent,
892				?para_id,
893				?remaining,
894				"Best advertisement is fetch-delayed; will fetch once the delay elapses",
895			);
896			Either::Right(remaining)
897		}
898	}

954		&mut self,
955		sender: &mut Sender,
956		scheduling_session: SessionIndex,
957		fetched_collation: FetchedCollation,
958		queue_blocked_collations: bool,
959		reject_info: SecondingRejectionInfo,
960	) -> CanSecond {
961		let scheduling_parent = fetched_collation.scheduling_parent();
962		let candidate_hash = fetched_collation.candidate_receipt.hash();
963		let para_id = fetched_collation.candidate_receipt.descriptor.para_id();
964
965		match fetch_pvd(
966			sender,
967			&fetched_collation.candidate_receipt,
968			scheduling_session,
969			fetched_collation.maybe_parent_head_data_hash,
970			fetched_collation.maybe_parent_head_data.clone(),
971		)
972		.await
973		{
974			Ok(pvd) => {

977			Err(SecondingError::BlockedOnParent(parent)) => {
978				gum::debug!(
979					target: LOG_TARGET,
980					?candidate_hash,
981					?scheduling_parent,
982					?para_id,
983					"Collation with parent head data hash {} is blocked from seconding. \
984					 Waiting on its parent to be validated.",
985					parent,
986				);
987
988				if queue_blocked_collations {
989					self.blocked_from_seconding
990						.entry(BlockedCollationId { para_id, parent_head_data_hash: parent })
991						.or_default()
992						.push(fetched_collation);
993				}
994
995				CanSecond::BlockedOnParent(parent, reject_info)
996			},
997			Err(err) => {
998				gum::warn!(
999					target: LOG_TARGET,
1000					?candidate_hash,
1001					?scheduling_parent,
1002					?para_id,
1003					"Failed persisted validation data checks: {}",
1004					err,
1005				);
1006
1007				let slash = err.is_malicious().then_some(FAILED_FETCH_SLASH);
1008				CanSecond::No(slash, reject_info)
1009			},
1010		}
1011	}

1084	/// Performs a sanity check between advertised and fetched collations.
1085	fn ensure_matches_advertisement(
1086		&self,
1087		advertised: &Advertisement,
1088	) -> std::result::Result<(), SecondingError> {
1089		let candidate_receipt = &self.candidate_receipt;
1090
1091		match advertised.prospective_candidate {
1092			// This implies a check on the declared para if this was a v2 advertisement
1093			Some(ProspectiveCandidate { candidate_hash, .. }) => {
1094				if candidate_hash != candidate_receipt.hash() {
1095					return Err(SecondingError::CandidateHashMismatch);
1096				}
1097			},
1098			// Otherwise, do the explicit check for the para_id.
1099			None => {
1100				if advertised.para_id != candidate_receipt.descriptor.para_id() {
1101					return Err(SecondingError::ParaIdMismatch);
1102				}
1103			},
1104		}
1105
1106		if advertised.scheduling_parent != candidate_receipt.descriptor.scheduling_parent() {
1107			return Err(SecondingError::SchedulingParentMismatch);
1108		}
1109		if let Some(advertised_version) = &advertised.advertised_descriptor_version {
1110			let fetched_version = candidate_receipt.descriptor().version();
1111			if advertised_version != &fetched_version {
1112				return Err(SecondingError::DescriptorVersionMismatch(
1113					*advertised_version,
1114					fetched_version,
1115				));
1116			}
1117		}
1118
1119		Ok(())
1120	}

1123/// Represents an advertisement which we have accepted. Supports ordering of the advertisements.
1124///
1125/// Ordering priority: score (descending), then timestamp (ascending), then advertisement as
1126/// tiebreaker. Higher scores come first so that `BTreeSet::first()` returns the best advertisement.

1129	adv: &'a Advertisement,
1130	score: Score,
1131	timestamp: &'a Instant,

1139			.score
1140			.cmp(&self.score) // Descending: higher score comes first
1141			.then_with(|| self.timestamp.cmp(other.timestamp)) // Ascending: earlier timestamp comes first
1142			.then_with(|| self.adv.cmp(other.adv))
1143	}
1144}

1157/// Per-(leaf, core) capacity view used by the fetch planner.
1158///
1159/// `cq[i]` is `Some(para)` if leaf-CQ position `i` is still free for `para`, or `None` if
1160/// already consumed (or if the runtime CQ didn't schedule a para there — see padding below).
1161/// The build pass allocates existing consumers into `cq` so what remains `Some` is residual
1162/// capacity SPs can fetch into.
1163///
1164/// `sps_by_depth[i]` is `Some(sp)` if the chain block at depth `i` from the leaf (leaf at 0)
1165/// is a scheduling parent on *this* core; cross-core ancestors are `None`. This implicitly
1166/// scopes both `sps_reaching` and `reserve_slot` to our core: cross-core SPs never appear as
1167/// candidates for our slots, and cross-core reservations are no-ops because the SP isn't
1168/// found in `sps_by_depth`.
1169///
1170/// `cq` is padded to the scheduling lookahead (`build_leaf_core_cqs`), so the SP-window
1171/// arithmetic (`cq.len() - depth`) is bounded by the lookahead, not the runtime CQ length
1172/// (which may be shorter for e.g. on-demand cores).
1173struct LeafCoreCq {
1174	sps_by_depth: Vec<Option<Hash>>,
1175	cq: Vec<Option<ParaId>>,
1176}
1177
1178impl LeafCoreCq {
1179	/// Same-core SPs whose window includes leaf-CQ position `idx`.
1180	///
1181	/// An SP at depth `d` has a lookahead window covering leaf-CQ positions `0..lookahead - d`,
1182	/// so position `idx` is reachable from SPs with `d < lookahead - idx`. With `cq` padded to
1183	/// the lookahead, that's the first `cq.len() - idx` entries of `sps_by_depth`.
1184	fn sps_reaching(&self, idx: usize) -> impl Iterator<Item = Hash> + '_ {
1185		self.sps_by_depth.iter().take(self.cq.len() - idx).filter_map(|x| *x)
1186	}
1187
1188	/// Mark one CQ position as consumed for `para` reachable from `sp`. Clears the latest
1189	/// still-free position for `para` in `sp`'s window — same rule the build pass uses for
1190	/// existing consumers, so newly-launched fetches and prior consumers stay consistently
1191	/// allocated. No-op if `sp` isn't on this chain *for this core*.
1192	fn reserve_slot(&mut self, sp: &Hash, para: ParaId) {
1193		let Some(depth) = self.sps_by_depth.iter().position(|x| x.as_ref() == Some(sp)) else {
1194			return;
1195		};
1196		let valid_len = self.cq.len().saturating_sub(depth);
1197		if let Some(latest) = self.cq[..valid_len].iter().rposition(|slot| *slot == Some(para)) {
1198			self.cq[latest] = None;
1199		}
1200	}
1201}

1203struct PerSchedulingParent {

1214	// activation (group rotation is per-block) and keep it for the lifetime of this SP.

1217	// relative to leaf activation.

1225			core_index,

1232	/// Every advertisement at this scheduling parent paired with the time it arrived.
1233	fn all_advertisements<'a>(&'a self) -> impl Iterator<Item = (&'a Advertisement, &'a Instant)> {
1234		self.peer_advertisements.values().flat_map(|list| &list.advertisements)
1235	}
1236
1237	/// Whether `advertisement` may be kept; pair with `add_advertisement` after the caller's
1238	/// async backing check. Bumps the rate-limit counter (`PeerAdvertisements::total`) even
1239	/// on rejection — by design, so a peer can't spam past their cap with bad advertisements.
1240	fn can_keep_advertisement(
1241		&mut self,
1242		advertisement: Advertisement,
1243		max_assignments: usize,
1244	) -> std::result::Result<(), AdvertisementError> {
1245		let peer_advertisements =
1246			self.peer_advertisements.entry(advertisement.peer_id).or_default();
1247
1248		peer_advertisements.total += 1;
1249
1250		if peer_advertisements.total > max_assignments {
1251			return Err(AdvertisementError::PeerLimitReached);
1252		}
1253
1254		if peer_advertisements.advertisements.contains_key(&advertisement) {
1255			return Err(AdvertisementError::Duplicate);
1256		}
1257
1258		Ok(())
1259	}
1260
1261	fn add_advertisement(&mut self, advertisement: Advertisement, now: Instant) {
1262		self.peer_advertisements
1263			.entry(advertisement.peer_id)
1264			.or_default()
1265			.advertisements
1266			.insert(advertisement, now);
1267	}
1268
1269	fn remove_advertisement(&mut self, advertisement: &Advertisement) {
1270		if let Some(advertisements) = self.peer_advertisements.get_mut(&advertisement.peer_id) {
1271			advertisements.advertisements.remove(&advertisement);
1272		}
1273	}
1274
1275	fn remove_peer_advertisements(&mut self, peer_id: &PeerId) {
1276		self.peer_advertisements.remove(peer_id);
1277	}

1332async fn fetch_pvd<Sender: CollatorProtocolSenderTrait>(
1333	sender: &mut Sender,
1334	receipt: &CandidateReceipt,
1335	scheduling_session: SessionIndex,
1336	maybe_parent_head_data_hash: Option<Hash>,
1337	maybe_parent_head_data: Option<HeadData>,
1338) -> std::result::Result<PersistedValidationData, SecondingError> {
1339	let para_id = receipt.descriptor.para_id();
1340
1341	let pvd = match maybe_parent_head_data_hash {
1342		Some(parent_head_data_hash) => {
1343			let maybe_pvd = request_prospective_validation_data(
1344				sender,
1345				receipt.descriptor.relay_parent(),
1346				receipt.descriptor.session_index().unwrap_or(scheduling_session),
1347				parent_head_data_hash,
1348				para_id,
1349				maybe_parent_head_data.clone(),
1350			)
1351			.await?;
1352
1353			let (expected_hash, pvd) = match (maybe_pvd, &maybe_parent_head_data) {
1354				(Some(pvd), Some(parent_head)) => (parent_head.hash(), pvd),
1355				(Some(pvd), None) => (pvd.parent_head.hash(), pvd),
1356				(None, None) => return Err(SecondingError::BlockedOnParent(parent_head_data_hash)),
1357				(None, _) => return Err(SecondingError::PersistedValidationDataNotFound),
1358			};
1359			if parent_head_data_hash != expected_hash {
1360				return Err(SecondingError::ParentHeadDataMismatch);
1361			}
1362			pvd
1363		},
1364		None => {
1365			let pvd = request_persisted_validation_data(
1366				sender,
1367				receipt.descriptor.relay_parent(),
1368				para_id,
1369			)
1370			.await?;
1371			pvd.ok_or(SecondingError::PersistedValidationDataNotFound)?
1372		},
1373	};
1374
1375	if pvd.hash() != receipt.descriptor.persisted_validation_data_hash() {
1376		return Err(SecondingError::PersistedValidationDataMismatch);
1377	}
1378
1379	Ok(pvd)
1380}

1382fn process_collation_fetch_result(
1383	(advertisement, res): CollationFetchResponse,
1384	network_error_freq: &mut gum::Freq,
1385	canceled_freq: &mut gum::Freq,
1386) -> std::result::Result<FetchedCollation, Option<Score>> {
1387	match res {
1388		Err(CollationFetchError::Cancelled) => {
1389			// Was cancelled by the subsystem.
1390			Err(None)
1391		},
1392		Err(CollationFetchError::Request(RequestError::InvalidResponse(err))) => {
1393			gum::warn!(
1394				target: LOG_TARGET,
1395				?advertisement,
1396				err = ?err,
1397				"Collator provided response that could not be decoded"
1398			);
1399			Err(Some(FAILED_FETCH_SLASH))

1412				max_rate: gum::Times::PerHour(100),
1413				target: LOG_TARGET,
1414				?advertisement,
1415				err = ?err,
1416				"Fetching collation failed due to network error"
1417			);
1418			Err(None)
1419		},
1420		Err(CollationFetchError::Request(RequestError::Canceled(err))) => {

1423				max_rate: gum::Times::PerHour(100),

1425				?advertisement,

1445		},
1446		Ok(request_v2::CollationFetchingResponse::CollationWithParentHeadData {

1450		}) => {
1451			gum::debug!(

1454				"Received collation with parent head data",
1455			);
1456
1457			Ok(FetchedCollation::new(
1458				receipt,
1459				pov,
1460				Some(parent_head_data),
1461				advertisement.prospective_candidate.map(|p| p.parent_head_data_hash),
1462				advertisement.peer_id,
1463			))
1464		},
1465	}

1788		// Multiple advertisements - picks highest score.
1789		{
1790			let mut collation_manager = new_collation_manager_instance();
1791			let peer_a_clone = peer_a;
1792			let peer_b_clone = peer_b;
1793			let peer_c_clone = peer_c;
1794			let get_rep = move |peer: &PeerId, _: &ParaId| {
1795				if *peer == peer_a_clone {
1796					Some(score(50))
1797				} else if *peer == peer_b_clone {
1798					Some(score(100))
1799				} else if *peer == peer_c_clone {
1800					Some(score(75))
1801				} else {
1802					None
1803				}
1804			};
1805
1806			let per_sp =
1807				collation_manager.per_scheduling_parent.get_mut(&scheduling_parent).unwrap();
1808			per_sp.add_advertisement(make_adv(peer_a), old_timestamp);
1809			per_sp.add_advertisement(make_adv(peer_b), old_timestamp);
1810			per_sp.add_advertisement(make_adv(peer_c), old_timestamp);
1811
1812			// All have old timestamps, so delay has passed. Should pick peer_b (highest score).
1813			assert_eq!(
1814				collation_manager.pick_best_advertisement(
1815					now,
1816					para_id,
1817					std::iter::once(scheduling_parent),
1818					score(100),
1819					&get_rep,
1820				),
1821				Either::Left(Some(make_adv(peer_b)))
1822			);
1823		}

1892		// Delay passed because leaf has been active long enough, even though advertisement arrived
1893		// recently. Tests that the delay is relative to activation time, not advertisement
1894		// arrival time. When the scheduling parent (leaf) has been active longer than the full
1895		// delay, the remaining delay should be zero and the advertisement should be fetched
1896		// immediately.
1897		{
1898			let mut collation_manager = new_collation_manager_instance();
1899			let get_rep = |_: &PeerId, _: &ParaId| Some(score(0));
1900
1901			// Set activated_at far enough in the past that any delay has elapsed.
1902			let per_sp =
1903				collation_manager.per_scheduling_parent.get_mut(&scheduling_parent).unwrap();
1904			per_sp.activated_at = now.checked_sub(MAX_FETCH_DELAY * 2).unwrap();
1905
1906			// Advertisement arrives now (recent), but the leaf has been active long enough.
1907			per_sp.add_advertisement(make_adv(peer_a), recent_timestamp);
1908
1909			// highest_rep = 100, peer's score = 0 (< INSTANT_FETCH_REP_THRESHOLD), so delay =
1910			// MAX_FETCH_DELAY. But activated_at is 2*MAX_FETCH_DELAY ago, so remaining_delay = 0.
1911			assert_eq!(
1912				collation_manager.pick_best_advertisement(
1913					now,
1914					para_id,
1915					std::iter::once(scheduling_parent),
1916					score(100),
1917					&get_rep,
1918				),
1919				Either::Left(Some(make_adv(peer_a)))
1920			);
1921		}

1924		{
1925			let mut collation_manager = new_collation_manager_instance();
1926			let get_rep = |_: &PeerId, _: &ParaId| Some(score(0));
1927
1928			// Set activated_at so that only part of the delay has elapsed.
1929			// score(0) < INSTANT_FETCH_REP_THRESHOLD and < highest_rep => delay = MAX_FETCH_DELAY
1930			// activated_at = MAX_FETCH_DELAY / 4 ago => remaining = MAX_FETCH_DELAY * 3/4
1931			let per_sp =
1932				collation_manager.per_scheduling_parent.get_mut(&scheduling_parent).unwrap();
1933			per_sp.activated_at = now.checked_sub(MAX_FETCH_DELAY / 4).unwrap();
1934
1935			per_sp.add_advertisement(make_adv(peer_a), recent_timestamp);
1936
1937			let result = collation_manager.pick_best_advertisement(
1938				now,
1939				para_id,
1940				std::iter::once(scheduling_parent),
1941				score(100),
1942				&get_rep,
1943			);
1944
1945			assert_eq!(result, Either::Right(MAX_FETCH_DELAY / 4 * 3));
1946		}
