29/// Represents the state of a claim.
30#[derive(Debug, PartialEq, Clone)]
31enum ClaimState {
32	/// Unclaimed
33	Free,
34	/// The candidate is pending fetching or validation. The candidate hash is optional because
35	/// collator protocol v1 doesn't care about specific candidate hashes, but only about their
36	/// number.
37	Pending(Option<CandidateHash>),
38	/// The candidate is seconded.
39	Seconded(CandidateHash),
40}
41
42impl ClaimState {
43	fn candidate_hash(&self) -> Option<&CandidateHash> {
44		match self {
45			ClaimState::Pending(Some(candidate)) => Some(candidate),
46			ClaimState::Seconded(candidate) => Some(candidate),
47			_ => None,
48		}
49	}
50
51	fn clone_or_default(&self, known_candidates: &HashSet<CandidateHash>) -> Self {
52		match self {
53			ClaimState::Pending(Some(candidate)) | ClaimState::Seconded(candidate)
54				if !known_candidates.contains(candidate) =>
55			{
56				ClaimState::Free
57			},
58			_ => self.clone(),
59		}
60	}
61}

63/// Represents a single claim from the claim queue, mapped to the relay chain block where it could
64/// be backed on-chain.
65#[derive(Debug, PartialEq, Clone)]
66struct ClaimInfo {
67	// Hash of the relay chain block. Can be `None` if it is still not known (a future block).
68	hash: Option<Hash>,
69	/// Represents the `ParaId` scheduled for the block. Can be `None` if nothing is scheduled.
70	claim: Option<ParaId>,
71	/// The length of the claim queue at the block. It is used to determine the 'block window'
72	/// where a claim can be made.
73	claim_queue_len: usize,
74	/// The claim state.
75	claimed: ClaimState,
76}
77
78impl ClaimInfo {
79	fn hash_equals(&self, hash: &Hash) -> bool {
80		self.hash.as_ref() == Some(hash)
81	}
82}
83
84trait ClaimInfoRef {
85	fn hash_equals(&self, hash: &Hash) -> bool;
86
87	fn claim_queue_len(&self) -> usize;
88}
89
90impl<'a> ClaimInfoRef for &'a ClaimInfo {
91	fn hash_equals(&self, hash: &Hash) -> bool {
92		ClaimInfo::hash_equals(self, hash)
93	}
94
95	fn claim_queue_len(&self) -> usize {
96		self.claim_queue_len
97	}
98}
99
100impl<'a> ClaimInfoRef for &'a mut ClaimInfo {
101	fn hash_equals(&self, hash: &Hash) -> bool {
102		ClaimInfo::hash_equals(self, hash)
103	}
104
105	fn claim_queue_len(&self) -> usize {
106		self.claim_queue_len
107	}
108}
