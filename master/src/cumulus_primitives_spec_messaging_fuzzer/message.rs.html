17//! The off-chain response verifiers in `message`.
18//!
19//! [`MessageOracle`] commits `STREAMS` streams under `N` successive `StreamsRoot`s. At height `h`
20//! stream 0 has `h` leaves, stream 1 `h / 2` and stream 2 `h / 3`; a stream with none is left out
21//! of the tree. Streams 0 and 1 share leaves (leaf version 0), stream 2 hashes the same payloads as
22//! version 1, so material for one stream can be offered for another. A fourth stream is never
23//! committed.
24//!
25//! Responses start honest, then take arbitrary edits. Whenever a verifier accepts one under a real
26//! `StreamsRoot`, the claim behind it must hold:
27//! - `FORGERY`: payloads, a position or a frontier the committed stream does not have.
28//! - `MALLEABLE`: the right answer, from material other than the honest proof.
29//!
30//! An accepted response must also honour its request.
31//!
32//! - `message_fetch`: [`verify_messages`].
33//! - `message_event`: [`verify_event`], head and positional reads.
34//! - `message_decode`: wire bytes through [`verify_exchange`]. Every decoded value must be
35//!   canonical (`NONCANONICAL`). Seed it with honest encodings first, from this crate:
36//!   `SEED_DIR=hfuzz_workspace/message_decode/input cargo test write_decode_seeds -- --ignored`
37

55/// Committed streams. Index `STREAMS` is a stream that is never committed.
56pub const STREAMS: usize = 3;
57const RECIPIENT: u32 = 2000;
58
59pub fn stream_id(s: usize) -> StreamId {
60	let recipient = ParaId::from(RECIPIENT);
61	match s {
62		0 => StreamId::Channel { recipient, domain: 0, num: 0 },
63		1 => StreamId::Channel { recipient, domain: 0, num: 1 },
64		2 => StreamId::Broadcast { domain: 0, subdomain: 0, num: 0 },
65		_ => StreamId::Channel { recipient, domain: 0, num: 2 },
66	}
67}
68
69/// Leaves of stream `s` under the `StreamsRoot` of height `h`.
70fn committed_len(s: usize, h: u64) -> u64 {
71	match s {
72		0 => h,
73		1 => h / 2,
74		2 => h / 3,
75		_ => 0,
76	}
77}
78
79fn leaf_version(s: usize) -> u8 {
80	if s == 2 {
81		LEAF_VERSION + 1
82	} else {
83		LEAF_VERSION
84	}
85}

87struct Commit {
88	under: StreamsRoot,
89	/// The honest membership proof of each committed stream.
90	proofs: [Option<StreamProof>; STREAMS],
91}

93pub struct MessageOracle {
94	/// Leaf versions 0 and 1.
95	mmrs: [Oracle; 2],
96	/// Index `h - 1`.
97	commits: Vec<Commit>,
98	/// `StreamsRoot` to height.
99	heights: HashMap<Hash, u64>,
100	/// Both MMR palettes, plus every `StreamsRoot` and tree sibling.
101	palette: Vec<Hash>,
102}
103
104impl Default for MessageOracle {
105	fn default() -> Self {
106		Self::new()
107	}
108}
109
110impl MessageOracle {
111	pub fn new() -> Self {
112		let mmrs =
113			[Oracle::with_leaf_version(LEAF_VERSION), Oracle::with_leaf_version(LEAF_VERSION + 1)];
114		let commits: Vec<Commit> = (1..=N as u64)
115			.map(|h| {
116				let entries: BTreeMap<StreamId, Hash> = (0..STREAMS)
117					.filter(|s| committed_len(*s, h) > 0)
118					.map(|s| {
119						let m = committed_len(s, h);
120						(stream_id(s), mmrs[(s == 2) as usize].frontiers[m as usize].root().0)
121					})
122					.collect();
123				let under = streams_root(&entries).expect("stream 0 is always committed; qed");
124				let proofs = core::array::from_fn(|s| {
125					gen_stream_proof(&entries, stream_id(s)).map(|(root, proof)| {
126						assert_eq!(root, under, "gen_stream_proof disagrees with streams_root");
127						proof
128					})
129				});
130				Commit { under, proofs }
131			})
132			.collect();
133		let heights = commits.iter().zip(1..).map(|(c, h)| (c.under.0, h)).collect();
134		let mut palette: BTreeSet<Hash> =
135			mmrs.iter().flat_map(|o| o.palette.iter().copied()).collect();
136		for commit in &commits {
137			palette.insert(commit.under.0);
138			for proof in commit.proofs.iter().flatten() {
139				palette.extend(proof.steps.iter().map(|step| step.sibling));
140			}
141		}
142		Self { mmrs, commits, heights, palette: palette.into_iter().collect() }
143	}
144
145	fn mmr(&self, s: usize) -> &Oracle {
146		&self.mmrs[(s == 2) as usize]
147	}
148
149	fn commit(&self, h: u64) -> &Commit {
150		&self.commits[h as usize - 1]
151	}
152
153	fn height(byte: u8) -> u64 {
154		byte as u64 % N as u64 + 1
155	}
156
157	fn committed(stream: StreamId, h: u64) -> Option<usize> {
158		(0..STREAMS).find(|s| stream_id(*s) == stream && committed_len(*s, h) > 0)
159	}
160
161	/// The stream whose honest material a response starts from: `material`, or the requested one.
162	fn material(stream: StreamId, material: Option<u8>) -> usize {
163		match material {
164			Some(s) => s as usize % (STREAMS + 1),
165			None => (0..STREAMS).find(|s| stream_id(*s) == stream).unwrap_or(STREAMS),
166		}
167	}
168
169	fn under(&self, choice: UnderChoice) -> StreamsRoot {
170		match choice {
171			UnderChoice::Commit(h) => self.commit(Self::height(h)).under,
172			UnderChoice::Hash(hash) => StreamsRoot(hash.resolve(&self.palette)),
173		}
174	}
175
176	fn tree_proof(&self, h: u64, s: usize) -> StreamProof {
177		self.commit(h)
178			.proofs
179			.get(s)
180			.cloned()
181			.flatten()
182			.unwrap_or(StreamProof { steps: BoundedVec::new() })
183	}

185	/// The honest fetch of stream `s` under height `h`: payloads `base..base + len` with both
186	/// reduced into range, the extension from there to the committed count.
187	pub fn honest_fetch(
188		&self,
189		h: u64,
190		s: usize,
191		base: u64,
192		len: u64,
193	) -> (MessagesRequest, MessagesResponse) {
194		let m = committed_len(s, h);
195		let base = base % (m + 1);
196		let end = base + len % (m - base + 1);
197		let mmr = self.mmr(s);
198		let payloads: Vec<Vec<u8>> = (base..end).map(payload).collect();
199		let max_bytes = payloads.iter().map(|p| p.len() as u32).sum();
200		let resp = MessagesResponse {
201			base: MessagePosition(base),
202			leaf_version: leaf_version(s),
203			payloads: payloads.try_into().expect("N < MAX_MESSAGES_PER_RESPONSE; qed"),
204			start_peaks: BoundedVec::try_from(mmr.frontiers[base as usize].peaks().to_vec())
205				.expect("a frontier fits the bound; qed"),
206			extension: mmr.extension(end, m).cloned().unwrap_or_default(),
207			tree_proof: self.tree_proof(h, s),
208		};
209		let req = MessagesRequest {
210			stream: stream_id(s),
211			start: MessagePosition(base),
212			under: self.commit(h).under,
213			max_bytes,
214		};
215		(req, resp)
216	}

218	/// The honest event read of stream `s` under height `h`: the head, or leaf `at` reduced into
219	/// range.
220	pub fn honest_event(&self, h: u64, s: usize, at: Option<u64>) -> (EventRequest, EventResponse) {
221		let m = committed_len(s, h);
222		let at = at.map(|i| i % m.max(1));
223		let leaf = at.unwrap_or(m.saturating_sub(1));
224		let inclusion = self
225			.mmr(s)
226			.inclusion(leaf, m)
227			.cloned()
228			.unwrap_or(MmrInclusionProof { mmr_size: 0, items: BoundedVec::new() });
229		let resp = EventResponse {
230			payload: payload(leaf),
231			leaf_version: leaf_version(s),
232			inclusion,
233			tree_proof: self.tree_proof(h, s),
234		};
235		let req = EventRequest {
236			stream: stream_id(s),
237			under: self.commit(h).under,
238			at: at.map(MessagePosition),
239		};
240		(req, resp)
241	}

243	/// What an accepted fetch claims: the requested range of the committed stream, from the honest
244	/// proof.
245	fn fetch_claim(&self, req: &MessagesRequest, resp: &MessagesResponse) -> Outcome {
246		let payloads = &resp.payloads[..];
247		assert_eq!(resp.base, req.start, "accepted a response that does not start at the request");
248		let total: u64 = payloads.iter().map(|p| p.len() as u64).sum();
249		assert!(
250			total <= req.max_bytes as u64 && (req.max_bytes != 0 || payloads.is_empty()),
251			"accepted {total} bytes against a budget of {}",
252			req.max_bytes
253		);
254		assert!(
255			payloads.iter().all(|p| p.len() <= MAX_SPECULATIVE_MESSAGE_LEN as usize),
256			"accepted a payload over the single-message bound"
257		);
258		let Some(&h) = self.heights.get(&req.under.0) else { return Outcome::Unbound };
259		let Some(s) = Self::committed(req.stream, h) else {
260			panic!("FORGERY: {:?} is not committed at height {h}, yet a fetch verified", req.stream)
261		};
262		let m = committed_len(s, h);
263		let base = resp.base.0;
264		let end = base + payloads.len() as u64;
265		assert!(end <= m, "FORGERY: positions {base}..{end} of stream {s}, which has {m} leaves");
266		// With no payloads the version is never hashed, and nothing returned depends on it.
267		if !payloads.is_empty() {
268			assert_eq!(resp.leaf_version, leaf_version(s), "FORGERY: leaf version of stream {s}");
269		}
270		for (j, p) in payloads.iter().enumerate() {
271			let at = base + j as u64;
272			assert_eq!(p, &payload(at), "FORGERY: wrong payload at position {at} of stream {s}");
273		}
274		let mmr = self.mmr(s);
275		assert_eq!(
276			&resp.start_peaks[..],
277			mmr.frontiers[base as usize].peaks(),
278			"MALLEABLE: non-honest start peaks at {base} for stream {s}"
279		);
280		assert_eq!(
281			resp.extension,
282			mmr.extension(end, m).cloned().unwrap_or_default(),
283			"MALLEABLE: non-honest extension from {end} to {m} for stream {s}"
284		);
285		assert_eq!(
286			Some(&resp.tree_proof),
287			self.commit(h).proofs[s].as_ref(),
288			"MALLEABLE: non-honest tree proof for stream {s} at height {h}"
289		);
290		Outcome::Real
291	}

293	/// What an accepted event read claims: the head, or the requested leaf, of the committed
294	/// stream, from the honest proof.
295	fn event_claim(
296		&self,
297		req: &EventRequest,
298		resp: &EventResponse,
299		verified: &VerifiedEvent,
300	) -> Outcome {
301		let position = match verified {
302			VerifiedEvent::Head { position, frontier } => {
303				assert!(req.at.is_none(), "a positional request verified as a head read");
304				assert_eq!(
305					frontier.leaf_count(),
306					position.0 + 1,
307					"a head read's frontier does not end at its position"
308				);
309				*position
310			},
311			VerifiedEvent::Positional { position } => {
312				assert_eq!(req.at, Some(*position), "a positional read at another position");
313				*position
314			},
315		};
316		assert!(
317			resp.payload.len() <= MAX_SPECULATIVE_MESSAGE_LEN as usize,
318			"accepted a payload over the single-message bound"
319		);
320		let Some(&h) = self.heights.get(&req.under.0) else { return Outcome::Unbound };
321		let Some(s) = Self::committed(req.stream, h) else {
322			panic!(
323				"FORGERY: {:?} is not committed at height {h}, yet an event verified",
324				req.stream
325			)
326		};
327		let m = committed_len(s, h);
328		let mmr = self.mmr(s);
329		assert!(
330			position.0 < m,
331			"FORGERY: position {} of stream {s}, which has {m} leaves",
332			position.0
333		);
334		if let VerifiedEvent::Head { frontier, .. } = verified {
335			assert_eq!(position.0 + 1, m, "FORGERY: leaf {} served as the head of {m}", position.0);
336			assert_eq!(Some(frontier), mmr.frontier(m), "FORGERY: a head frontier of stream {s}");
337		}
338		assert_eq!(resp.leaf_version, leaf_version(s), "FORGERY: leaf version of stream {s}");
339		assert_eq!(
340			resp.payload,
341			payload(position.0),
342			"FORGERY: wrong payload at position {} of stream {s}",
343			position.0
344		);
345		assert_eq!(
346			Some(&resp.inclusion),
347			mmr.inclusion(position.0, m),
348			"MALLEABLE: non-honest inclusion proof of leaf {} in {m} for stream {s}",
349			position.0
350		);
351		assert_eq!(
352			Some(&resp.tree_proof),
353			self.commit(h).proofs[s].as_ref(),
354			"MALLEABLE: non-honest tree proof for stream {s} at height {h}"
355		);
356		Outcome::Real
357	}

359	/// Verify a fetch, and check the claim behind it if accepted.
360	pub fn check_fetch(&self, req: &MessagesRequest, resp: &MessagesResponse) -> Outcome {
361		match verify_messages(req, resp) {
362			Ok(()) => self.fetch_claim(req, resp),
363			Err(_) => Outcome::Rejected,
364		}
365	}

367	/// Verify an event read, and check the claim behind it if accepted.
368	pub fn check_event(&self, req: &EventRequest, resp: &EventResponse) -> Outcome {
369		match verify_event(req, resp) {
370			Ok(verified) => self.event_claim(req, resp, &verified),
371			Err(_) => Outcome::Rejected,
372		}
373	}

375	/// [`Self::check_fetch`] or [`Self::check_event`], by variant. `verify_exchange` must agree
376	/// with the verifier the variants select.
377	pub fn check_exchange(&self, req: &ExchangeRequest, resp: &ExchangeResponse) -> Outcome {
378		let result = verify_exchange(req, resp);
379		match (req, resp) {
380			(ExchangeRequest::Messages(req), ExchangeResponse::Messages(resp)) => {
381				let direct = verify_messages(req, resp).map(|()| ExchangeVerified::Messages);
382				assert_eq!(result, direct, "verify_exchange disagrees with verify_messages");
383				self.check_fetch(req, resp)
384			},
385			(ExchangeRequest::Event(req), ExchangeResponse::Event(resp)) => {
386				let direct = verify_event(req, resp).map(ExchangeVerified::Event);
387				assert_eq!(result, direct, "verify_exchange disagrees with verify_event");
388				self.check_event(req, resp)
389			},
390			_ => {
391				assert_eq!(
392					result,
393					Err(VerifyError::VariantMismatch),
394					"accepted a cross-variant answer"
395				);
396				Outcome::Rejected
397			},
398		}
399	}

401	/// Fuzz entry point for [`verify_messages`].
402	pub fn check_fetch_input(&self, input: &FetchInput) -> Outcome {
403		let Some(stream) = input.stream.get() else { return Outcome::Skipped };
404		let h = Self::height(input.commit);
405		let material = Self::material(stream, input.material);
406		let (mut req, mut resp) =
407			self.honest_fetch(h, material, input.base as u64, input.len as u64);
408		req.stream = stream;
409		if let Some(target) = input.target {
410			let end = resp.base.0 + resp.payloads.len() as u64;
411			resp.extension = self
412				.mmr(material)
413				.extension(end, target as u64 % (N as u64 + 1))
414				.cloned()
415				.unwrap_or_default();
416		}
417		if let Some(base) = input.resp_base {
418			resp.base = MessagePosition(base.get());
419		}
420		if let Some(version) = input.leaf_version {
421			resp.leaf_version = version;
422		}
423		let mut payloads = resp.payloads.to_vec();
424		apply_payload_edits(&mut payloads, &input.payload_edits);
425		if let Ok(payloads) = payloads.try_into() {
426			resp.payloads = payloads;
427		}
428		apply_bounded(&self.palette, &mut resp.start_peaks, &input.peak_edits);
429		if let Some(count) = input.ext_count {
430			resp.extension.leaf_count = count.get();
431		}
432		apply_bounded(&self.palette, &mut resp.extension.connecting_nodes, &input.ext_edits);
433		apply_tree_edits(&self.palette, &mut resp.tree_proof, &input.tree_edits);
434		req.start = input.start.map_or(resp.base, |c| MessagePosition(c.get()));
435		req.max_bytes = input.max_bytes.unwrap_or_else(|| {
436			let total = resp.payloads.iter().map(|p| p.len() as u32).sum::<u32>();
437			if resp.payloads.is_empty() {
438				0
439			} else {
440				total.max(1)
441			}
442		});
443		if let Some(under) = input.under {
444			req.under = self.under(under);
445		}
446		self.check_fetch(&req, &resp)
447	}

449	/// Fuzz entry point for [`verify_event`].
450	pub fn check_event_input(&self, input: &EventInput) -> Outcome {
451		let Some(stream) = input.stream.get() else { return Outcome::Skipped };
452		let h = Self::height(input.commit);
453		let material = Self::material(stream, input.material);
454		let at = input.at.map(|c| c.get());
455		let (mut req, mut resp) = self.honest_event(h, material, at);
456		req.stream = stream;
457		// `honest_event` reduced the position into range; the request keeps it as is.
458		req.at = at.map(MessagePosition);
459		if let Some((i, n)) = input.template {
460			let (i, n) = (i as u64 % N as u64, n as u64 % (N as u64 + 1));
461			if let Some(proof) = self.mmr(material).inclusion(i, n) {
462				resp.inclusion = proof.clone();
463				resp.payload = payload(i);
464			}
465		}
466		if let Some(size) = input.mmr_size {
467			resp.inclusion.mmr_size = size.get();
468		}
469		apply_bounded(&self.palette, &mut resp.inclusion.items, &input.item_edits);
470		let mut payloads = vec![resp.payload];
471		apply_payload_edits(&mut payloads, &input.payload_edits);
472		resp.payload = payloads.into_iter().next().unwrap_or_default();
473		if let Some(version) = input.leaf_version {
474			resp.leaf_version = version;
475		}
476		apply_tree_edits(&self.palette, &mut resp.tree_proof, &input.tree_edits);
477		if let Some(under) = input.under {
478			req.under = self.under(under);
479		}
480		self.check_event(&req, &resp)
481	}

483	/// Fuzz entry point for raw wire bytes. The first byte picks the layout:
484	/// - `0`: `[h, s, max_bytes[4]]` then an `ExchangeResponse`, answering a fetch of known stream
485	///   `s` under height `h` from the response's own base.
486	/// - `1`: `[h, s, positional, at[8]]` then an `ExchangeResponse`, answering an event read.
487	/// - `2`: an `(ExchangeRequest, ExchangeResponse)` pair.
488	pub fn check_message_decode(&self, data: &[u8]) -> Outcome {
489		let Some((&selector, rest)) = data.split_first() else { return Outcome::Skipped };
490		match selector % 3 {
491			0 => {
492				let Some(([h, s, b0, b1, b2, b3], body)) = split_header(rest) else {
493					return Outcome::Skipped;
494				};
495				let Some(resp) = canonical::<ExchangeResponse>(body) else {
496					return Outcome::Skipped;
497				};
498				let start = match &resp {
499					ExchangeResponse::Messages(m) => m.base,
500					ExchangeResponse::Event(_) => MessagePosition(0),
501				};
502				let req = ExchangeRequest::Messages(MessagesRequest {
503					stream: stream_id(s as usize % (STREAMS + 1)),
504					start,
505					under: self.commit(Self::height(h)).under,
506					max_bytes: u32::from_le_bytes([b0, b1, b2, b3]),
507				});
508				self.check_exchange(&req, &resp)
509			},
510			1 => {
511				let Some(([h, s, positional, a0, a1, a2, a3, a4, a5, a6, a7], body)) =
512					split_header(rest)
513				else {
514					return Outcome::Skipped;
515				};
516				let Some(resp) = canonical::<ExchangeResponse>(body) else {
517					return Outcome::Skipped;
518				};
519				let at = u64::from_le_bytes([a0, a1, a2, a3, a4, a5, a6, a7]);
520				let req = ExchangeRequest::Event(EventRequest {
521					stream: stream_id(s as usize % (STREAMS + 1)),
522					under: self.commit(Self::height(h)).under,
523					at: (positional & 1 == 1).then_some(MessagePosition(at)),
524				});
525				self.check_exchange(&req, &resp)
526			},
527			_ => {
528				let Some((req, resp)) = canonical::<(ExchangeRequest, ExchangeResponse)>(rest)
529				else {
530					return Outcome::Skipped;
531				};
532				self.check_exchange(&req, &resp)
533			},
534		}
535	}

537	/// Honest wire inputs for `message_decode`, so it starts from bytes that verify.
538	pub fn decode_seeds(&self) -> Vec<Vec<u8>> {
539		let mut seeds = Vec::new();
540		for h in [1u64, 2, 3, 6, 7, 13, 31, 32, 33, 63, 64] {
541			for s in 0..=STREAMS {
542				for (base, len) in [(0, 0), (0, 64), (1, 1), (5, 3), (64, 0)] {
543					let (req, resp) = self.honest_fetch(h, s, base, len);
544					let mut seed = vec![0, h as u8 - 1, s as u8];
545					seed.extend_from_slice(&req.max_bytes.to_le_bytes());
546					seed.extend(ExchangeResponse::Messages(resp.clone()).encode());
547					seeds.push(seed);
548					let mut seed = vec![2];
549					seed.extend(
550						(ExchangeRequest::Messages(req), ExchangeResponse::Messages(resp)).encode(),
551					);
552					seeds.push(seed);
553				}
554				for at in [None, Some(0), Some(5), Some(63)] {
555					let (req, resp) = self.honest_event(h, s, at);
556					let mut seed = vec![1, h as u8 - 1, s as u8, req.at.is_some() as u8];
557					seed.extend_from_slice(&req.at.map_or(0, |p| p.0).to_le_bytes());
558					seed.extend(ExchangeResponse::Event(resp.clone()).encode());
559					seeds.push(seed);
560					let mut seed = vec![2];
561					seed.extend(
562						(ExchangeRequest::Event(req), ExchangeResponse::Event(resp)).encode(),
563					);
564					seeds.push(seed);
565				}
566			}
567		}
568		seeds
569	}
570}

572fn split_header<const L: usize>(data: &[u8]) -> Option<([u8; L], &[u8])> {
573	(data.len() >= L).then(|| (data[..L].try_into().expect("length checked; qed"), &data[L..]))
574}

576/// Decode a `T` from the front of `data`. Its encoding must be exactly the bytes consumed, and
577/// decode back to itself.
578fn canonical<T: Decode + Encode + PartialEq + Debug>(data: &[u8]) -> Option<T> {
579	let mut input = data;
580	let value = T::decode(&mut input).ok()?;
581	let consumed = data.len() - input.len();
582	let encoded = value.encode();
583	assert_eq!(&data[..consumed], &encoded[..], "NONCANONICAL: {value:?}");
584	assert_eq!(
585		T::decode(&mut &encoded[..]).ok().as_ref(),
586		Some(&value),
587		"NONCANONICAL: round trip"
588	);
589	Some(value)
590}

592fn apply_payload_edits(payloads: &mut Vec<Vec<u8>>, edits: &[PayloadEdit]) {
593	for edit in edits {
594		let len = payloads.len();
595		match edit {
596			PayloadEdit::Set { slot, payload } if len > 0 => {
597				payloads[*slot as usize % len] = payload.get()
598			},
599			PayloadEdit::Insert { slot, payload } => {
600				payloads.insert(*slot as usize % (len + 1), payload.get())
601			},
602			PayloadEdit::Remove { slot } if len > 0 => {
603				payloads.remove(*slot as usize % len);
604			},
605			PayloadEdit::Swap { a, b } if len > 0 => {
606				payloads.swap(*a as usize % len, *b as usize % len)
607			},
608			PayloadEdit::Flip { slot, byte, mask } if len > 0 => {
609				let p = &mut payloads[*slot as usize % len];
610				if !p.is_empty() {
611					let i = *byte as usize % p.len();
612					p[i] ^= mask;
613				}
614			},
615			PayloadEdit::Truncate { slot, len: to } if len > 0 => {
616				payloads[*slot as usize % len].truncate(*to as usize)
617			},
618			PayloadEdit::Push { slot, byte } if len > 0 => {
619				payloads[*slot as usize % len].push(*byte)
620			},
621			_ => {},
622		}
623	}
624}

626fn apply_tree_edits(palette: &[Hash], proof: &mut StreamProof, edits: &[TreeEdit]) {
627	let mut steps = proof.steps.to_vec();
628	for edit in edits {
629		let len = steps.len();
630		match *edit {
631			TreeEdit::Split { step, bit } if len > 0 => steps[step as usize % len].split_bit = bit,
632			TreeEdit::Sibling { step, hash } if len > 0 => {
633				steps[step as usize % len].sibling = hash.resolve(palette)
634			},
635			TreeEdit::Insert { slot, bit, hash } => steps.insert(
636				slot as usize % (len + 1),
637				TreeStep { split_bit: bit, sibling: hash.resolve(palette) },
638			),
639			TreeEdit::Remove { slot } if len > 0 => {
640				steps.remove(slot as usize % len);
641			},
642			TreeEdit::Swap { a, b } if len > 0 => steps.swap(a as usize % len, b as usize % len),
643			_ => {},
644		}
645	}
646	if let Ok(steps) = steps.try_into() {
647		proof.steps = steps;
648	}
649}

651#[derive(Arbitrary, Clone, Copy, Debug)]
652pub enum StreamChoice {
653	/// `stream_id(s % (STREAMS + 1))`.
654	Known(u8),
655	/// Skipped unless it decodes.
656	Raw([u8; 8]),
657}
658
659impl StreamChoice {
660	fn get(self) -> Option<StreamId> {
661		match self {
662			StreamChoice::Known(s) => Some(stream_id(s as usize % (STREAMS + 1))),
663			StreamChoice::Raw(bytes) => StreamId::from_bytes(bytes).ok(),
664		}
665	}
666}

675#[derive(Arbitrary, Clone, Debug)]
676pub enum PayloadChoice {
677	/// The payload of leaf `i % N`, the same bytes in every stream.
678	Real(u8),
679	Raw(Vec<u8>),
680	/// `MAX_SPECULATIVE_MESSAGE_LEN - 2 + x % 4` zero bytes: either side of the bound.
681	NearBound(u8),
682}
683
684impl PayloadChoice {
685	fn get(&self) -> Vec<u8> {
686		match self {
687			PayloadChoice::Real(i) => payload(*i as u64 % N as u64),
688			PayloadChoice::Raw(bytes) => bytes.clone(),
689			PayloadChoice::NearBound(x) => {
690				vec![0; MAX_SPECULATIVE_MESSAGE_LEN as usize - 2 + *x as usize % 4]
691			},
692		}
693	}
694}

719	},
720	Truncate {
721		slot: u8,

724	Push {

727	},
