17//! The off-chain response verifier in `message`.
18//!
19//! [`MessageOracle`] commits `STREAMS` streams under `N` successive `StreamsRoot`s. At height `h`
20//! stream 0 has `h` leaves, stream 1 `h / 2` and stream 2 `h / 3`; a stream with none is left out
21//! of the tree. Streams 0, 1 and 2 share payload shapes, so material for one stream can be offered
22//! for another. A fourth stream is never committed.
23//!
24//! Responses start honest, then take arbitrary edits. Whenever `verify_messages` accepts one under
25//! a real `StreamsRoot`, the claim behind it must hold:
26//! - `FORGERY`: payloads, a position or a frontier the committed stream does not have.
27//! - `MALLEABLE`: the right answer, from material other than the honest proof.
28//!
29//! An accepted response must also honour its request.
30//!
31//! - `message_fetch`: [`verify_messages`] (range and head reads).
32//! - `message_decode`: wire bytes through [`verify_messages`]. Every decoded value must be
33//!   canonical (`NONCANONICAL`). Seed it with honest encodings first, from this crate:
34//!   `SEED_DIR=hfuzz_workspace/message_decode/input cargo test write_decode_seeds -- --ignored`
35

52/// Committed streams. Index `STREAMS` is a stream that is never committed.
53pub const STREAMS: usize = 3;
54const RECIPIENT: u32 = 2000;
55
56pub fn stream_id(s: usize) -> StreamId {
57	let recipient = ParaId::from(RECIPIENT);
58	match s {
59		0 => StreamId::Channel { recipient, domain: 0, num: 0 },
60		1 => StreamId::Channel { recipient, domain: 0, num: 1 },
61		2 => StreamId::Broadcast { domain: 0, subdomain: 0, num: 0 },
62		_ => StreamId::Channel { recipient, domain: 0, num: 2 },
63	}
64}
65
66/// Leaves of stream `s` under the `StreamsRoot` of height `h`.
67fn committed_len(s: usize, h: u64) -> u64 {
68	match s {
69		0 => h,
70		1 => h / 2,
71		2 => h / 3,
72		_ => 0,
73	}
74}

76struct Commit {
77	under: StreamsRoot,
78	/// The honest membership proof of each committed stream.
79	proofs: [Option<StreamProof>; STREAMS],
80}

82pub struct MessageOracle {
83	mmr: Oracle,
84	/// Index `h - 1`.
85	commits: Vec<Commit>,
86	/// `StreamsRoot` to height.
87	heights: HashMap<Hash, u64>,
88	/// MMR palette, plus every `StreamsRoot` and tree sibling.
89	palette: Vec<Hash>,
90}

92impl Default for MessageOracle {

94		Self::new()
95	}

98impl MessageOracle {
99	pub fn new() -> Self {
100		let mmr = Oracle::with_leaf_version(LEAF_VERSION);
101		let commits: Vec<Commit> = (1..=N as u64)
102			.map(|h| {
103				let entries: BTreeMap<StreamId, Hash> = (0..STREAMS)
104					.filter(|s| committed_len(*s, h) > 0)
105					.map(|s| {
106						let m = committed_len(s, h);
107						(stream_id(s), mmr.frontiers[m as usize].root().0)
108					})
109					.collect();
110				let under = streams_root(&entries).expect("stream 0 is always committed; qed");
111				let proofs = core::array::from_fn(|s| {
112					gen_stream_proof(&entries, stream_id(s)).map(|(root, proof)| {
113						assert_eq!(root, under, "gen_stream_proof disagrees with streams_root");
114						proof
115					})
116				});
117				Commit { under, proofs }
118			})
119			.collect();
120		let heights = commits.iter().zip(1..).map(|(c, h)| (c.under.0, h)).collect();
121		let mut palette: BTreeSet<Hash> = mmr.palette.iter().copied().collect();
122		for commit in &commits {
123			palette.insert(commit.under.0);
124			for proof in commit.proofs.iter().flatten() {
125				palette.extend(proof.steps.iter().map(|step| step.sibling));
126			}
127		}
128		Self { mmr, commits, heights, palette: palette.into_iter().collect() }
129	}

131	fn commit(&self, h: u64) -> &Commit {
132		&self.commits[h as usize - 1]
133	}
134
135	fn height(byte: u8) -> u64 {
136		byte as u64 % N as u64 + 1
137	}
138
139	fn committed(stream: StreamId, h: u64) -> Option<usize> {
140		(0..STREAMS).find(|s| stream_id(*s) == stream && committed_len(*s, h) > 0)
141	}

143	/// The stream whose honest material a response starts from: `material`, or the requested one.
144	fn material(stream: StreamId, material: Option<u8>) -> usize {
145		match material {
146			Some(s) => s as usize % (STREAMS + 1),
147			None => (0..STREAMS).find(|s| stream_id(*s) == stream).unwrap_or(STREAMS),
148		}
149	}
150
151	fn under(&self, choice: UnderChoice) -> StreamsRoot {
152		match choice {
153			UnderChoice::Commit(h) => self.commit(Self::height(h)).under,
154			UnderChoice::Hash(hash) => StreamsRoot(hash.resolve(&self.palette)),
155		}
156	}
157
158	fn tree_proof(&self, h: u64, s: usize) -> StreamProof {
159		self.commit(h)
160			.proofs
161			.get(s)
162			.cloned()
163			.flatten()
164			.unwrap_or(StreamProof { steps: BoundedVec::new() })
165	}

167	/// The honest fetch of stream `s` under height `h`:
168	/// - If `start` is `Some(base)`: payloads `base..base + len` with both reduced into range, the
169	///   extension from there to the committed count.
170	/// - If `start` is `None`: the head at `m - 1` (if `m > 0`) with either 1 payload and identity
171	///   extension (`len % 2 == 1`) or 0 payloads and the `(m - 1) -> m` extension (`len % 2 == 0`,
172	///   position-only head when budget was too small).
173	pub fn honest_fetch(
174		&self,
175		h: u64,
176		s: usize,
177		start: Option<u64>,
178		len: u64,
179	) -> (MessagesRequest, MessagesResponse) {
180		let m = committed_len(s, h);
181		let (req_start, base, end) = match start {
182			Some(b) => {
183				let base = b % (m + 1);
184				let end = base + len % (m - base + 1);
185				(Some(MessagePosition(base)), base, end)
186			},
187			None => {
188				let base = m.saturating_sub(1);
189				let end = (base + len % 2).min(m);
190				(None, base, end)
191			},
192		};
193		let payloads: Vec<Vec<u8>> = (base..end).map(payload).collect();
194		let max_bytes = payloads.iter().map(|p| p.len() as u32).sum();
195		let resp = MessagesResponse {
196			frontier: self.mmr.frontiers[base as usize].clone(),
197			payloads: payloads.try_into().expect("N < MAX_MESSAGES_PER_RESPONSE; qed"),
198			extension: self.mmr.extension(end, m).cloned().expect("end <= m <= N; qed"),
199			tree_proof: self.tree_proof(h, s),
200		};
201		let req = MessagesRequest {
202			stream: stream_id(s),
203			start: req_start,
204			under: self.commit(h).under,
205			max_bytes,
206		};
207		(req, resp)
208	}

210	/// What an accepted fetch claims: the requested range (or head) of the committed stream, from
211	/// the honest proof.
212	fn fetch_claim(&self, req: &MessagesRequest, resp: &MessagesResponse) -> Outcome {
213		let payloads = &resp.payloads[..];
214		let base = resp.frontier.leaf_count();
215		if let Some(start) = req.start {
216			assert_eq!(
217				MessagePosition(base),
218				start,
219				"accepted a response that does not start at the request"
220			);
221		} else {
222			let end_count = resp.extension.leaf_count;
223			assert_eq!(
224				end_count,
225				base + 1,
226				"accepted a head response ending at {end_count} instead of {}",
227				base + 1
228			);
229			assert!(
230				payloads.len() <= 1,
231				"accepted a head response with {} payloads",
232				payloads.len()
233			);
234		}
235		let total: u64 = payloads.iter().map(|p| p.len() as u64).sum();
236		assert!(
237			total <= req.max_bytes as u64 && (req.max_bytes != 0 || payloads.is_empty()),
238			"accepted {total} bytes against a budget of {}",
239			req.max_bytes
240		);
241		assert!(
242			payloads.iter().all(|p| p.len() <= MAX_SPECULATIVE_MESSAGE_LEN as usize),
243			"accepted a payload over the single-message bound"
244		);
245		let Some(&h) = self.heights.get(&req.under.0) else { return Outcome::Unbound };
246		let Some(s) = Self::committed(req.stream, h) else {
247			panic!("FORGERY: {:?} is not committed at height {h}, yet a fetch verified", req.stream)
248		};
249		let m = committed_len(s, h);
250		let end = base + payloads.len() as u64;
251		assert!(end <= m, "FORGERY: positions {base}..{end} of stream {s}, which has {m} leaves");
252		if req.start.is_none() {
253			assert_eq!(
254				base + 1,
255				m,
256				"FORGERY: head response for stream {s} placed head at {base} instead of {}",
257				m.saturating_sub(1)
258			);
259		}
260		for (j, p) in payloads.iter().enumerate() {
261			let at = base + j as u64;
262			assert_eq!(p, &payload(at), "FORGERY: wrong payload at position {at} of stream {s}");
263		}
264		assert_eq!(
265			&resp.frontier, &self.mmr.frontiers[base as usize],
266			"MALLEABLE: non-honest frontier at {base} for stream {s}"
267		);
268		assert_eq!(
269			resp.extension,
270			self.mmr.extension(end, m).cloned().expect("end <= m <= N; qed"),
271			"MALLEABLE: non-honest extension from {end} to {m} for stream {s}"
272		);
273		assert_eq!(
274			Some(&resp.tree_proof),
275			self.commit(h).proofs[s].as_ref(),
276			"MALLEABLE: non-honest tree proof for stream {s} at height {h}"
277		);
278		Outcome::Real
279	}

281	/// Verify a fetch, and check the claim behind it if accepted.
282	pub fn check_fetch(&self, req: &MessagesRequest, resp: &MessagesResponse) -> Outcome {
283		match verify_messages(req, resp) {
284			Ok(()) => self.fetch_claim(req, resp),
285			Err(_) => Outcome::Rejected,
286		}
287	}

289	/// Fuzz entry point for [`verify_messages`].
290	pub fn check_fetch_input(&self, input: &FetchInput) -> Outcome {
291		let Some(stream) = input.stream.get() else { return Outcome::Skipped };
292		let h = Self::height(input.commit);
293		let material = Self::material(stream, input.material);
294		let (mut req, mut resp) =
295			self.honest_fetch(h, material, input.base.map(u64::from), input.len as u64);
296		req.stream = stream;
297		if let Some(target) = input.target {
298			let end = resp.frontier.leaf_count() + resp.payloads.len() as u64;
299			resp.extension =
300				self.mmr.extension(end, target as u64 % (N as u64 + 1)).cloned().unwrap_or_else(
301					|| MmrExtensionProof::identity(&self.mmr.frontiers[end as usize]),
302				);
303		}
304		let leaf_count = input.resp_base.map_or(resp.frontier.leaf_count(), Count::get);
305		let mut peaks = resp.frontier.peaks().to_vec();
306		apply(&self.palette, &mut peaks, &input.peak_edits);
307		if input.resp_base.is_some() || !input.peak_edits.is_empty() {
308			let Some(frontier) = MmrFrontier::from_parts(peaks, leaf_count) else {
309				return Outcome::Skipped;
310			};
311			resp.frontier = frontier;
312		}
313		let mut payloads = resp.payloads.to_vec();
314		apply_payload_edits(&mut payloads, &input.payload_edits);
315		if let Ok(payloads) = payloads.try_into() {
316			resp.payloads = payloads;
317		}
318		if let Some(count) = input.ext_count {
319			resp.extension.leaf_count = count.get();
320		}
321		apply_bounded(&self.palette, &mut resp.extension.connecting_nodes, &input.ext_edits);
322		apply_tree_edits(&self.palette, &mut resp.tree_proof, &input.tree_edits);
323		req.start = match input.start {
324			Some(Some(c)) => Some(MessagePosition(c.get())),
325			Some(None) => None,
326			None => req.start.map(|_| MessagePosition(resp.frontier.leaf_count())),
327		};
328		req.max_bytes = input.max_bytes.unwrap_or_else(|| {
329			let total = resp.payloads.iter().map(|p| p.len() as u32).sum::<u32>();
330			if resp.payloads.is_empty() {
331				0
332			} else {
333				total.max(1)
334			}
335		});
336		if let Some(under) = input.under {
337			req.under = self.under(under);
338		}
339		self.check_fetch(&req, &resp)
340	}

342	/// Fuzz entry point for raw wire bytes. The first byte picks the layout:
343	/// - `0`: `[h, s, is_range, max_bytes[4]]` then a `MessagesResponse`, answering a fetch of
344	///   known stream `s` under height `h` from the response's own frontier count (or head if
345	///   `!is_range`).
346	/// - `1`: a `(MessagesRequest, MessagesResponse)` pair.
347	pub fn check_message_decode(&self, data: &[u8]) -> Outcome {
348		let Some((&selector, rest)) = data.split_first() else { return Outcome::Skipped };
349		match selector % 2 {
350			0 => {
351				let Some(([h, s, is_range, b0, b1, b2, b3], body)) = split_header(rest) else {
352					return Outcome::Skipped;
353				};
354				let Some(resp) = canonical::<MessagesResponse>(body) else {
355					return Outcome::Skipped;
356				};
357				let start =
358					(is_range & 1 == 1).then_some(MessagePosition(resp.frontier.leaf_count()));
359				let req = MessagesRequest {
360					stream: stream_id(s as usize % (STREAMS + 1)),
361					start,
362					under: self.commit(Self::height(h)).under,
363					max_bytes: u32::from_le_bytes([b0, b1, b2, b3]),
364				};
365				self.check_fetch(&req, &resp)
366			},
367			_ => {
368				let Some((req, resp)) = canonical::<(MessagesRequest, MessagesResponse)>(rest)
369				else {
370					return Outcome::Skipped;
371				};
372				self.check_fetch(&req, &resp)
373			},
374		}
375	}

377	/// Honest wire inputs for `message_decode`, so it starts from bytes that verify.
378	pub fn decode_seeds(&self) -> Vec<Vec<u8>> {
379		let mut seeds = Vec::new();
380		for h in [1u64, 2, 3, 6, 7, 13, 31, 32, 33, 63, 64] {
381			for s in 0..=STREAMS {
382				for (start, len) in [
383					(Some(0), 0),
384					(Some(0), 64),
385					(Some(1), 1),
386					(Some(5), 3),
387					(Some(64), 0),
388					(None, 0),
389					(None, 1),
390				] {
391					let (req, resp) = self.honest_fetch(h, s, start, len);
392					let mut seed = vec![0, h as u8 - 1, s as u8, req.start.is_some() as u8];
393					seed.extend_from_slice(&req.max_bytes.to_le_bytes());
394					seed.extend(resp.encode());
395					seeds.push(seed);
396					let mut seed = vec![1];
397					seed.extend((req, resp).encode());
398					seeds.push(seed);
399				}
400			}
401		}
402		seeds
403	}

406fn split_header<const L: usize>(data: &[u8]) -> Option<([u8; L], &[u8])> {
407	(data.len() >= L).then(|| (data[..L].try_into().expect("length checked; qed"), &data[L..]))
408}

410/// Decode a `T` from the front of `data`. Its encoding must be exactly the bytes consumed, and
411/// decode back to itself.
412fn canonical<T: Decode + Encode + PartialEq + Debug>(data: &[u8]) -> Option<T> {
413	let mut input = data;
414	let value = T::decode(&mut input).ok()?;
415	let consumed = data.len() - input.len();
416	let encoded = value.encode();
417	assert_eq!(&data[..consumed], &encoded[..], "NONCANONICAL: {value:?}");
418	assert_eq!(
419		T::decode(&mut &encoded[..]).ok().as_ref(),
420		Some(&value),
421		"NONCANONICAL: round trip"
422	);
423	Some(value)
424}

426/// Fuzz entry point for the inherent and the channel leaf formats. The first byte picks the type:
427/// [`MessagingInherentData`], [`ChannelMessage`] or [`Register`]. Each must accept only canonical
428/// bytes, so no two encodings mean the same value. For the inherent, this includes strictly
429/// increasing map keys.
430pub fn check_inherent_decode(data: &[u8]) {
431	let Some((&selector, rest)) = data.split_first() else { return };
432	match selector % 3 {
433		0 => drop(canonical::<MessagingInherentData>(rest)),
434		1 => drop(canonical::<ChannelMessage>(rest)),
435		_ => drop(canonical::<Register>(rest)),
436	}
437}

439fn apply_payload_edits(payloads: &mut Vec<Vec<u8>>, edits: &[PayloadEdit]) {
440	for edit in edits {
441		let len = payloads.len();
442		match edit {
443			PayloadEdit::Set { slot, payload } if len > 0 => {
444				payloads[*slot as usize % len] = payload.get()
445			},
446			PayloadEdit::Insert { slot, payload } => {
447				payloads.insert(*slot as usize % (len + 1), payload.get())
448			},
449			PayloadEdit::Remove { slot } if len > 0 => {
450				payloads.remove(*slot as usize % len);
451			},
452			PayloadEdit::Swap { a, b } if len > 0 => {
453				payloads.swap(*a as usize % len, *b as usize % len)
454			},
455			PayloadEdit::Flip { slot, byte, mask } if len > 0 => {
456				let p = &mut payloads[*slot as usize % len];
457				if !p.is_empty() {
458					let i = *byte as usize % p.len();
459					p[i] ^= mask;
460				}
461			},
462			PayloadEdit::Truncate { slot, len: to } if len > 0 => {
463				payloads[*slot as usize % len].truncate(*to as usize)
464			},
465			PayloadEdit::Push { slot, byte } if len > 0 => {
466				payloads[*slot as usize % len].push(*byte)
467			},
468			_ => {},
469		}
470	}
471}
472
473fn apply_tree_edits(palette: &[Hash], proof: &mut StreamProof, edits: &[TreeEdit]) {
474	let mut steps = proof.steps.to_vec();
475	for edit in edits {
476		let len = steps.len();
477		match *edit {
478			TreeEdit::Split { step, bit } if len > 0 => steps[step as usize % len].split_bit = bit,
479			TreeEdit::Sibling { step, hash } if len > 0 => {
480				steps[step as usize % len].sibling = hash.resolve(palette)
481			},
482			TreeEdit::Insert { slot, bit, hash } => steps.insert(
483				slot as usize % (len + 1),
484				TreeStep { split_bit: bit, sibling: hash.resolve(palette) },
485			),
486			TreeEdit::Remove { slot } if len > 0 => {
487				steps.remove(slot as usize % len);
488			},
489			TreeEdit::Swap { a, b } if len > 0 => steps.swap(a as usize % len, b as usize % len),
490			_ => {},
491		}
492	}
493	if let Ok(steps) = steps.try_into() {
494		proof.steps = steps;
495	}
496}

498#[derive(Arbitrary, Clone, Copy, Debug)]
499pub enum StreamChoice {
500	/// `stream_id(s % (STREAMS + 1))`.
501	Known(u8),
502	/// Skipped unless it decodes.
503	Raw([u8; 8]),
504}
505
506impl StreamChoice {
507	fn get(self) -> Option<StreamId> {
508		match self {
509			StreamChoice::Known(s) => Some(stream_id(s as usize % (STREAMS + 1))),
510			StreamChoice::Raw(bytes) => StreamId::from_bytes(bytes).ok(),
511		}
512	}
513}

515#[derive(Arbitrary, Clone, Copy, Debug)]
516pub enum UnderChoice {
517	/// A real `StreamsRoot`, by height.
518	Commit(u8),
519	Hash(HashChoice),
520}

522#[derive(Arbitrary, Clone, Debug)]
523pub enum PayloadChoice {
524	/// The payload of leaf `i % N`, the same bytes in every stream.
525	Real(u8),
526	Raw(Vec<u8>),
527	/// `MAX_SPECULATIVE_MESSAGE_LEN - 2 + x % 4` zero bytes: either side of the bound.
528	NearBound(u8),
529}
530
531impl PayloadChoice {
532	fn get(&self) -> Vec<u8> {
533		match self {
534			PayloadChoice::Real(i) => payload(*i as u64 % N as u64),
535			PayloadChoice::Raw(bytes) => bytes.clone(),
536			PayloadChoice::NearBound(x) => {
537				vec![0; MAX_SPECULATIVE_MESSAGE_LEN as usize - 2 + *x as usize % 4]
538			},
539		}
540	}
541}

549	},
550	Insert {

552		payload: PayloadChoice,

554	Remove {

557	Swap {
558		a: u8,

561	/// XOR `mask` into byte `byte % len`.

564		byte: u8,

566	},
567	Truncate {

569		len: u8,
570	},
571	Push {

573		byte: u8,
574	},

577/// Steps wrap around the proof's length.
578#[derive(Arbitrary, Clone, Copy, Debug)]
579pub enum TreeEdit {
580	Split { step: u8, bit: u8 },
581	Sibling { step: u8, hash: HashChoice },
582	Insert { slot: u8, bit: u8, hash: HashChoice },
583	Remove { slot: u8 },
584	Swap { a: u8, b: u8 },
585}
586
587#[derive(Arbitrary, Debug)]
588pub struct FetchInput {
589	/// Height `commit % N + 1`.
590	pub commit: u8,
591	pub stream: StreamChoice,
592	/// Build the honest material for stream `s % (STREAMS + 1)` instead of the requested one.
593	pub material: Option<u8>,
594	/// `None` starts from an honest head read; `Some` starts from an honest range read reduced
595	/// into the committed count.
596	pub base: Option<u8>,
597	pub len: u8,
598	/// Extend to `target % (N + 1)` leaves instead of the committed count.
599	pub target: Option<u8>,
600	pub resp_base: Option<Count>,
601	/// `None` defaults to matching the honest template's mode (`Some(frontier.leaf_count())` for
602	/// range, `None` for head).
603	pub start: Option<Option<Count>>,
604	/// Defaults to the payloads' total, at least 1 if there are any.
605	pub max_bytes: Option<u32>,
606	pub payload_edits: Vec<PayloadEdit>,
607	pub peak_edits: Vec<Edit>,
608	pub ext_count: Option<Count>,
609	pub ext_edits: Vec<Edit>,
610	pub tree_edits: Vec<TreeEdit>,
611	/// Defaults to the real `StreamsRoot` at the height.
612	pub under: Option<UnderChoice>,
613}

619	#[test]
620	fn honest_fetches_are_real() {
621		let oracle = MessageOracle::new();
622		for h in 1..=N as u64 {
623			for s in 0..=STREAMS {
624				let m = committed_len(s, h);
625				for base in 0..=m {
626					for len in [0, 1, m - base] {
627						let (req, resp) = oracle.honest_fetch(h, s, Some(base), len);
628						let outcome = oracle.check_fetch(&req, &resp);
629						assert_eq!(outcome, if m > 0 { Outcome::Real } else { Outcome::Rejected });
630					}
631				}
632				for len in [0, 1] {
633					let (req, resp) = oracle.honest_fetch(h, s, None, len);
634					let outcome = oracle.check_fetch(&req, &resp);
635					assert_eq!(outcome, if m > 0 { Outcome::Real } else { Outcome::Rejected });
636				}
637			}
638		}
639	}

643		use cumulus_primitives_spec_messaging::{InChannelId, RegisterStatus};
644		let a = InChannelId { sender: ParaId::from(1000), domain: 0, num: 0 };
645		let b = InChannelId { sender: ParaId::from(2000), domain: 0, num: 0 };
646		let data = MessagingInherentData {

649		};
650		check_inherent_decode(&[&[0][..], &data.encode()].concat());

652		check_inherent_decode(&[&[2][..], &register.encode()].concat());
653		check_inherent_decode(&[&[1][..], &ChannelMessage::Data(vec![7]).encode()].concat());

657		let unsorted =

659		assert!(MessagingInherentData::decode(&mut &unsorted[..]).is_err());
660		check_inherent_decode(&[&[0][..], &unsorted].concat());
661	}
