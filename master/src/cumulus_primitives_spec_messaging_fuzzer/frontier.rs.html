17//! `mmr_frontier` and `mmr_decode`: `MmrFrontier`, `bag_peaks` and `MmrRoot` from `mmr`.
18//!
19//! `Model` is an independent peaks-with-heights MMR. It hashes with plain blake2-256 and the
20//! domain tags directly and bags by the documented right fold, so it shares no code with the crate.
21//! Every frontier is checked against it after every step, at any leaf count up to the ceiling. When
22//! the leaves are known (grown from empty or an oracle prefix), `mmr_lib` is a second reference.
23//!
24//! Any mismatch is a `DIVERGENCE` panic.
25

41/// Above this many known leaves the `mmr_lib` rebuild is skipped.
42const MMR_LIB_LEAVES: usize = 512;
43
44/// Each append is checked against the model in full, so this bounds an input's run time.
45const MAX_APPENDS: usize = 2048;
46
47fn tagged(tag: u8, left: &Hash, right: &Hash) -> Hash {
48	let mut preimage = vec![tag];
49	preimage.extend_from_slice(left.as_bytes());
50	preimage.extend_from_slice(right.as_bytes());
51	Blake2Hasher::hash(&preimage)
52}
53
54/// `bag([p1..pn]) = H(PEAK_TAG ++ bag([p2..pn]) ++ p1)`, `bag([p]) = p`, `bag([]) = 0`.
55fn model_bag(peaks: &[Hash]) -> Hash {
56	match peaks.split_first() {
57		None => Hash::zero(),
58		Some((first, [])) => *first,
59		Some((first, rest)) => tagged(PEAK_TAG, &model_bag(rest), first),
60	}
61}
62
63/// `H(ROOT_TAG ++ leaf_count[8, LE] ++ bag)`.
64fn model_root(leaf_count: u64, bag: Hash) -> Hash {
65	let mut preimage = vec![ROOT_TAG];
66	preimage.extend_from_slice(&leaf_count.to_le_bytes());
67	preimage.extend_from_slice(bag.as_bytes());
68	Blake2Hasher::hash(&preimage)
69}

71/// Peaks with their heights, highest first. Appends merge while the two lowest peaks are equal.
72#[derive(Default)]
73struct Model {
74	leaf_count: u64,
75	peaks: Vec<(u32, Hash)>,
76}
77
78impl Model {
79	fn from_parts(peaks: &[Hash], leaf_count: u64) -> Self {
80		let heights = (0..u64::BITS).rev().filter(|h| leaf_count >> h & 1 == 1);
81		Self { leaf_count, peaks: heights.zip(peaks.iter().copied()).collect() }
82	}
83
84	fn append(&mut self, leaf: Hash) {
85		self.peaks.push((0, leaf));
86		while let [.., (hl, l), (hr, r)] = self.peaks[..] {
87			if hl != hr {
88				break;
89			}
90			self.peaks.truncate(self.peaks.len() - 2);
91			self.peaks.push((hl + 1, tagged(INNER_TAG, &l, &r)));
92		}
93		self.leaf_count += 1;
94	}
95
96	fn hashes(&self) -> Vec<Hash> {
97		self.peaks.iter().map(|(_, h)| *h).collect()
98	}
99
100	fn root(&self) -> Hash {
101		model_root(self.leaf_count, model_bag(&self.hashes()))
102	}
103}
104
105fn check_against_model(frontier: &MmrFrontier, model: &Model) {
106	let count = frontier.leaf_count();
107	assert!(count <= MAX_MMR_LEAF_COUNT, "DIVERGENCE: a frontier holds {count} leaves");
108	assert_eq!(
109		frontier.peaks().len(),
110		count.count_ones() as usize,
111		"DIVERGENCE: {count} leaves but {} peaks",
112		frontier.peaks().len()
113	);
114	assert_eq!(count, model.leaf_count, "DIVERGENCE: leaf count");
115	assert_eq!(frontier.peaks(), &model.hashes()[..], "DIVERGENCE: peaks at {count} leaves");
116	assert_eq!(frontier.root().0, model.root(), "DIVERGENCE: root at {count} leaves");
117}
118
119fn check_against_mmr_lib(frontier: &MmrFrontier, leaves: &[Hash]) {
120	let store = MemStore::<Hash>::default();
121	let mut mmr = MemMMR::<Hash, SpecMerge>::new(0, &store);
122	for leaf in leaves {
123		mmr.push(*leaf).expect("MemStore never fails; qed");
124	}
125	let n = leaves.len() as u64;
126	assert_eq!(frontier.leaf_count(), n, "DIVERGENCE: leaf count against mmr_lib");
127	if n == 0 {
128		assert_eq!(frontier.root(), empty_root(), "DIVERGENCE: empty root against mmr_lib");
129		return;
130	}
131	mmr.commit().expect("MemStore never fails; qed");
132	let peaks: Vec<Hash> = get_peaks(mmr.mmr_size())
133		.into_iter()
134		.map(|pos| (&store).get_elem(pos).expect("MemStore never fails; qed").expect("a peak; qed"))
135		.collect();
136	assert_eq!(frontier.peaks(), &peaks[..], "DIVERGENCE: peaks against mmr_lib at {n} leaves");
137	let bag = MemMMR::<Hash, SpecMerge>::new(mmr.mmr_size(), &store)
138		.get_root()
139		.expect("a non-empty MMR has a root; qed");
140	assert_eq!(
141		frontier.root(),
142		MmrRoot::from_bag(n, bag),
143		"DIVERGENCE: root against mmr_lib at {n} leaves"
144	);
145}

147fn check_round_trip(frontier: &MmrFrontier) {
148	let encoded = frontier.encode();
149	assert_eq!(
150		encoded,
151		(frontier.leaf_count(), frontier.peaks().to_vec()).encode(),
152		"DIVERGENCE: wire layout"
153	);
154	let decoded = MmrFrontier::decode(&mut &encoded[..])
155		.expect("DIVERGENCE: a frontier fails to decode its own encoding");
156	assert_eq!(&decoded, frontier, "DIVERGENCE: round trip");
157}

159/// `from_parts` accepts exactly the shapes the docs promise, and `Decode` agrees with it.
160fn check_from_parts(peaks: &[Hash], leaf_count: u64) -> Option<MmrFrontier> {
161	let valid = leaf_count <= MAX_MMR_LEAF_COUNT && peaks.len() == leaf_count.count_ones() as usize;
162	let built = MmrFrontier::from_parts(peaks.to_vec(), leaf_count);
163	assert_eq!(
164		built.is_some(),
165		valid,
166		"DIVERGENCE: from_parts({leaf_count}, {} peaks)",
167		peaks.len()
168	);
169	let decoded = MmrFrontier::decode(&mut &(leaf_count, peaks.to_vec()).encode()[..]).ok();
170	assert_eq!(decoded, built, "DIVERGENCE: decode disagrees with from_parts");
171	built
172}

174/// Small counts, either side of the ceiling, and around powers of two.
175#[derive(Arbitrary, Clone, Copy, Debug)]
176pub enum LeafCount {
177	Small(Count),
178	/// `MAX_MMR_LEAF_COUNT - 128 + x`.
179	AroundCeiling(u8),
180	/// `2^(exp % 64) + offset`, wrapping.
181	PowerOfTwo {
182		exp: u8,
183		offset: i8,
184	},
185}
186
187impl LeafCount {
188	fn get(self) -> u64 {
189		match self {
190			LeafCount::Small(c) => c.get(),
191			LeafCount::AroundCeiling(x) => MAX_MMR_LEAF_COUNT - 128 + x as u64,
192			LeafCount::PowerOfTwo { exp, offset } => {
193				(1u64 << (exp % 64)).wrapping_add(offset as i64 as u64)
194			},
195		}
196	}
197}

199#[derive(Arbitrary, Debug)]
200pub enum Start {
201	Empty,
202	/// The oracle's frontier after `k % (N + 1)` leaves.
203	Real(u8),
204	/// Skipped unless `from_parts` accepts it. `fit` cycles or truncates the peaks to one per set
205	/// bit of the count.
206	Parts {
207		leaf_count: LeafCount,
208		peaks: Vec<HashChoice>,
209		fit: bool,
210	},
211}
212
213#[derive(Arbitrary, Debug)]
214pub enum FrontierOp {
215	Append(HashChoice),
216	/// `n` leaves `leaf_hash(seed, [j])`.
217	AppendMany {
218		n: u8,
219		seed: u8,
220	},
221	RoundTrip,
222	Bag(Vec<HashChoice>),
223}
224
225#[derive(Arbitrary, Debug)]
226pub struct FrontierInput {
227	pub start: Start,
228	pub ops: Vec<FrontierOp>,
229}

231/// `leaves` are the frontier's leaves, when known.
232fn run_ops(
233	palette: &[Hash],
234	mut frontier: MmrFrontier,
235	mut leaves: Option<Vec<Hash>>,
236	ops: &[FrontierOp],
237) {
238	let mut model = Model::from_parts(frontier.peaks(), frontier.leaf_count());
239	check_against_model(&frontier, &model);
240	let mut appends = 0;
241	let mut append = |frontier: &mut MmrFrontier, leaf: Hash| {
242		if appends == MAX_APPENDS {
243			return;
244		}
245		appends += 1;
246		let before = frontier.clone();
247		match frontier.try_append(leaf) {
248			Ok(()) => {
249				assert!(
250					before.leaf_count() < MAX_MMR_LEAF_COUNT,
251					"DIVERGENCE: an append carried a frontier past the ceiling"
252				);
253				model.append(leaf);
254				if let Some(known) = leaves.as_mut() {
255					known.push(leaf);
256				}
257			},
258			Err(FrontierFull) => {
259				assert_eq!(
260					before.leaf_count(),
261					MAX_MMR_LEAF_COUNT,
262					"DIVERGENCE: an append refused below the ceiling"
263				);
264				assert_eq!(&before, frontier, "DIVERGENCE: a refused append changed the frontier");
265			},
266		}
267		check_against_model(frontier, &model);
268	};
269	for op in ops {
270		match op {
271			FrontierOp::Append(choice) => append(&mut frontier, choice.resolve(palette)),
272			FrontierOp::AppendMany { n, seed } => {
273				for j in 0..*n {
274					append(&mut frontier, leaf_hash(*seed, &[j]));
275				}
276			},
277			FrontierOp::RoundTrip => check_round_trip(&frontier),
278			FrontierOp::Bag(choices) => {
279				let peaks: Vec<Hash> = choices.iter().map(|c| c.resolve(palette)).collect();
280				assert_eq!(bag_peaks(&peaks), model_bag(&peaks), "DIVERGENCE: bag_peaks");
281			},
282		}
283	}
284	check_round_trip(&frontier);
285	if let Some(known) = leaves.filter(|l| l.len() <= MMR_LIB_LEAVES) {
286		check_against_mmr_lib(&frontier, &known);
287	}
288}

290impl Oracle {
291	/// Fuzz entry point for `MmrFrontier` and the root and bagging in `mmr`.
292	pub fn check_frontier_input(&self, input: &FrontierInput) -> Outcome {
293		let (frontier, leaves) = match &input.start {
294			Start::Empty => (MmrFrontier::new(), Some(Vec::new())),
295			Start::Real(k) => {
296				let k = *k as usize % (N + 1);
297				(self.frontiers[k].clone(), Some(self.leaves[..k].to_vec()))
298			},
299			Start::Parts { leaf_count, peaks, fit } => {
300				let leaf_count = leaf_count.get();
301				let mut peaks: Vec<Hash> = peaks.iter().map(|c| c.resolve(&self.palette)).collect();
302				if *fit {
303					let have = peaks.len().max(1);
304					peaks = (0..leaf_count.count_ones() as usize)
305						.map(|i| peaks.get(i % have).copied().unwrap_or_default())
306						.collect();
307				}
308				let Some(frontier) = check_from_parts(&peaks, leaf_count) else {
309					return Outcome::Skipped;
310				};
311				(frontier, None)
312			},
313		};
314		run_ops(&self.palette, frontier, leaves, &input.ops);
315		Outcome::Real
316	}
317
318	/// Fuzz entry point for raw `MmrFrontier` bytes: decoding must accept exactly the valid
319	/// `(leaf_count, peaks)` pairs, in their canonical encoding. The rest of the input is appended
320	/// as 32-byte leaves.
321	pub fn check_frontier_decode(&self, data: &[u8]) -> Outcome {
322		let mut input = data;
323		let decoded = MmrFrontier::decode(&mut input);
324
325		let mut reference = data;
326		let valid = <(u64, Vec<Hash>)>::decode(&mut reference).is_ok_and(|(count, peaks)| {
327			count <= MAX_MMR_LEAF_COUNT && peaks.len() == count.count_ones() as usize
328		});
329		let frontier = match decoded {
330			Ok(frontier) => {
331				assert!(valid, "DIVERGENCE: decoded an invalid frontier {frontier:?}");
332				assert_eq!(input.len(), reference.len(), "DIVERGENCE: decode length");
333				// The reference read these bytes as `(leaf_count, peaks)`: equal bytes, equal
334				// fields.
335				let consumed = &data[..data.len() - input.len()];
336				assert_eq!(consumed, &frontier.encode()[..], "DIVERGENCE: non-canonical");
337				frontier
338			},
339			Err(e) => {
340				assert!(!valid, "DIVERGENCE: refused a valid frontier: {e:?}");
341				return Outcome::Rejected;
342			},
343		};
344		let ops: Vec<FrontierOp> = input
345			.chunks_exact(32)
346			.map(|chunk| {
347				FrontierOp::Append(HashChoice::Raw(chunk.try_into().expect("32 bytes; qed")))
348			})
349			.chain(core::iter::once(FrontierOp::RoundTrip))
350			.collect();
351		run_ops(&self.palette, frontier, None, &ops);
352		Outcome::Real
353	}
354}

360	/// The crate's frozen vectors, so the model agrees with the protocol, not only with the crate.
361	#[test]
362	fn model_matches_frozen_vector() {
363		let mut model = Model::default();
364		for i in 1..=5u8 {
365			model.append(Hash::repeat_byte(i));
366		}
367		assert_eq!(
368			model.root(),
369			Hash(hex_literal::hex!(
370				"1190cb02577a8bafec9f05c38e64ca9155b65e9d862604c4f3d920dff4e5a85f"
371			))
372		);
373		assert_eq!(model_root(0, Hash::zero()), MmrFrontier::new().root().0);
374	}
375
376	#[test]
377	fn honest_growth_and_ceiling_pass() {
378		let oracle = Oracle::new();
379		for k in 0..=N as u8 {
380			let input = FrontierInput {
381				start: Start::Real(k),
382				ops: vec![FrontierOp::AppendMany { n: 70, seed: 3 }, FrontierOp::RoundTrip],
383			};
384			assert_eq!(oracle.check_frontier_input(&input), Outcome::Real);
385		}
386		let input = FrontierInput {
387			start: Start::Parts {
388				leaf_count: LeafCount::AroundCeiling(120),
389				peaks: vec![],
390				fit: true,
391			},
392			ops: vec![FrontierOp::AppendMany { n: 20, seed: 0 }],
393		};
394		assert_eq!(oracle.check_frontier_input(&input), Outcome::Real);
395		let raw = (MAX_MMR_LEAF_COUNT, vec![Hash::zero()]).encode();
396		assert_eq!(oracle.check_frontier_decode(&raw), Outcome::Real);
397		assert_eq!(
398			oracle.check_frontier_decode(&(3u64, vec![Hash::zero()]).encode()),
399			Outcome::Rejected
400		);
401	}
