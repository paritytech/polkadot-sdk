51/// Domain-separation tag used when deriving P-256 key from ed25519 key via HMAC-SHA256.
52const CERTIFICATE_KEY_DST: &[u8] = b"substrate-webrtc-p256-v1";
53
54/// Common name used for both subject and issuer, the certificate being self-signed.
55const SUBJECT_COMMON_NAME: &str = "polkadot-sdk-webrtc";
56
57/// Deterministically generate a WebRTC DTLS certificate from the node's secret key.
58pub fn derive_certificate(
59	node_secret_key: Ed25519SecretKey,
60) -> Result<DtlsCertificate, litep2p::Error> {
61	// NOTE: none of the expects in this function are input-dependent.
62	let signing_key = derive_keys(node_secret_key);
63
64	// RFC 5758 §3.2: `parameters` MUST be absent for ECDSA.
65	let signature_algorithm =
66		AlgorithmIdentifierOwned { oid: ECDSA_WITH_SHA_256, parameters: None };
67	let name = common_name();
68	// Uncompressed SEC1 form, requested explicitly: RFC 5480 permits several point forms.
69	let point = signing_key.verifying_key().as_affine().to_encoded_point(false);
70
71	let tbs_certificate = TbsCertificate {
72		// Every other WebRTC stack emits V3, let's do the same, even though it can be V1.
73		version: Version::V3,
74		serial_number: derive_serial(&point),
75		signature: signature_algorithm.clone(),
76		issuer: name.clone(),
77		validity: validity(),
78		subject: name,
79		subject_public_key_info: SubjectPublicKeyInfoOwned {
80			// RFC 5480 §2.1.1: `namedCurve`, rather than `implicitCurve` or `specifiedCurve`.
81			algorithm: AlgorithmIdentifierOwned {
82				oid: ID_EC_PUBLIC_KEY,
83				parameters: Some(Any::from(SECP_256_R_1)),
84			},
85			subject_public_key: BitString::new(0, point.as_bytes())
86				.expect("a SEC1 point is a whole number of octets; qed"),
87		},
88		// RFC 5280 §4.1.2.8: conforming CAs MUST NOT generate unique identifiers.
89		issuer_unique_id: None,
90		subject_unique_id: None,
91		extensions: None,
92	};
93
94	let tbs_der = tbs_certificate.to_der().expect("the certificate is well-formed; qed");
95
96	// `PrehashSigner` is RFC 6979 deterministic ECDSA; the module's determinism rests on it. If
97	// an upgrade ever changes its output (caught by the tests), the old implementation must be
98	// vendored into `sc-network`.
99	//
100	// Signing fails only if the RFC 6979 nonce or a signature scalar is zero (each 2^-256).
101	let signature: DerSignature = signing_key
102		.sign_prehash(&Sha256::digest(&tbs_der))
103		.expect("the signing key is valid; qed");
104
105	let certificate_der = Certificate {
106		tbs_certificate,
107		signature_algorithm,
108		// RFC 3279 §2.2.3: the DER-encoded `Ecdsa-Sig-Value`, with no unused bits.
109		signature: BitString::new(0, signature.to_vec())
110			.expect("an ECDSA signature is a whole number of octets; qed"),
111	}
112	.to_der()
113	.expect("a well-formed certificate is DER-encodable; qed");
114
115	let pk_pkcs8_der = signing_key
116		.to_pkcs8_der()
117		.expect("a P-256 signing key is PKCS#8-encodable; qed")
118		.as_bytes()
119		.to_vec();
120
121	DtlsCertificate::load(certificate_der, pk_pkcs8_der)
122}

124/// Derive the P-256 key by rejection-sampling HMAC-SHA256(node key, `DST || counter`).
125fn derive_keys(node_secret_key: Ed25519SecretKey) -> SigningKey {
126	(0u8..255u8)
127		.find_map(|counter| {
128			let okm = Hmac::<Sha256>::new_from_slice(node_secret_key.as_ref())
129				.expect("HMAC accepts keys of any length; qed")
130				.chain_update(CERTIFICATE_KEY_DST)
131				.chain_update([counter])
132				.finalize()
133				.into_bytes();
134
135			// Range-checked here rather than by letting `SigningKey::from_slice` reject: which
136			// counter wins must not depend on whether out-of-range values are rejected or reduced.
137			let scalar = U256::from_be_slice(&okm);
138			(scalar != U256::ZERO && scalar < NistP256::ORDER)
139				.then(|| SigningKey::from_slice(&okm).expect("checked to be in range; qed"))
140		})
141		.expect("each iteration succeeds with probability 1 - 2^-32, and we have 256 of them; qed")
142}

144/// Serial derived from the public key, only to avoid one hardcoded serial for all certificates.
145fn derive_serial(point: &EncodedPoint) -> SerialNumber {
146	let digest = Sha256::digest(point.as_bytes());
147
148	// A `u64` names exactly one integer, fixing the DER by spec; a byte string would leave
149	// signedness to `SerialNumber`.
150	SerialNumber::from(u64::from_be_bytes(digest[..8].try_into().expect("8 of 32 bytes; qed")))
151}

153/// `CN=polkadot-sdk-webrtc`, built explicitly so the attribute's string type is pinned.
154fn common_name() -> Name {
155	let attribute = AttributeTypeAndValue {
156		oid: COMMON_NAME,
157		value: Any::new(Tag::Utf8String, SUBJECT_COMMON_NAME.as_bytes())
158			.expect("the common name is a valid `UTF8String`; qed"),
159	};
160	let rdn = SetOfVec::try_from(vec![attribute]).expect("a single-element set is sorted; qed");
161
162	Name::from(vec![RelativeDistinguishedName::from(rdn)])
163}
164
165/// Fixed validity dates, required for determinism. WebRTC peers pin the certificate by
166/// certhash and ignore its lifetime.
167fn validity() -> Validity {
168	// RFC 5280 §4.1.2.5: dates through 2049 MUST be encoded as `UTCTime`.
169	let not_before = UtcTime::from_date_time(
170		DateTime::new(2000, 1, 1, 0, 0, 0).expect("2000-01-01 00:00:00 is a valid date; qed"),
171	)
172	.expect("2000 is within the `UTCTime` range; qed");
173	// RFC 5280 §4.1.2.5: the encoding for certificates with no well-defined expiration date.
174	let not_after = GeneralizedTime::from_date_time(
175		DateTime::new(9999, 12, 31, 23, 59, 59).expect("9999-12-31 23:59:59 is a valid date; qed"),
176	);
177
178	Validity { not_before: Time::UtcTime(not_before), not_after: Time::GeneralTime(not_after) }
179}

200	/// Compute the `/certhash/<hash>` multiaddress component of a certificate.
201	fn certhash(certificate: &DtlsCertificate) -> String {
202		let hash = Code::Sha2_256.digest(certificate.as_parts().0);
203		Multiaddr::empty().with(Protocol::Certhash(hash)).to_string()
204	}

241	#[test]
242	fn stable_certhash() {
243		// Pins the node key -> certificate derivation. If this test ever fails, the derivation
244		// changed and the certhash published by every node relying on it breaks.
245		//
246		// Three vectors, because the serial's INTEGER encoding splits on its first byte:
247		// `node_key(7)` (serial `0x87..`) gets a `0x00` sign byte, `node_key(42)` (`0x6e..`)
248		// encodes plain, and the third key (`0x002b..`) has its leading zero stripped.
249		assert_eq!(
250			certhash(&derive_certificate(node_key(7)).unwrap()),
251			"/certhash/uEiAXqXtF_3QIfMcgXwMgneoB4EuSE_EcpGvKhY4yz7HfcA"
252		);
253		assert_eq!(
254			certhash(&derive_certificate(node_key(42)).unwrap()),
255			"/certhash/uEiAWsH8V-_VMveqodSJYiAhW5FikqSzBNLV0FyeEb_oetA"
256		);
257		let mut stripped_serial_key = [3u8; 32];
258		stripped_serial_key[31] = 26;
259		assert_eq!(
260			certhash(&derive_certificate(node_key_from(stripped_serial_key)).unwrap()),
261			"/certhash/uEiAfSKLRHZTkoALez2X0jqB0Yyh6T4DYQGM3wLpbR1u7dQ"
262		);
263	}

265	#[test]
266	fn rfc6979_signing_is_pinned() {
267		use p256::ecdsa::Signature;
268
269		// Everything else in this module states its own encoding; the signature cannot: ECDSA
270		// needs a nonce, and reproducibility rests on `sign_prehash` deriving it via RFC 6979.
271		// The RFC leaves inputs that `ecdsa` fills in for us, so pin the signature itself — an
272		// upgrade changing any of them fails here, naming the cause.
273		//
274		// Pinned only for prehashes below the group order: `ecdsa` 0.16 feeds `h1` to the nonce
275		// DRBG unreduced, 0.17 reduces it per RFC 6979 §2.3.4. Bumping past 0.16 thus changes
276		// the certhash of nodes whose TBS digest is >= `n` — accepted, as that is ~2^-32 of them.
277		let key = SigningKey::from_slice(&array_bytes::hex2bytes_unchecked(
278			"c9afa9d845ba75166b5c215767b1d6934e50c3db36e89b127b8a622b120f6721",
279		))
280		.unwrap();
281
282		// RFC 6979 A.2.5: P-256 with SHA-256, message "sample". Pins the HMAC hash
283		// (`NistP256::Digest`) and the empty additional-data argument.
284		let signature: Signature = key.sign_prehash(&Sha256::digest(b"sample")).unwrap();
285		assert_eq!(
286			array_bytes::bytes2hex("", signature.to_bytes()),
287			"efd48b2aacb6a8fd1140dd9cd45e81d69d2c877b56aaf991c34d0ea84eaf3716\
288			 f7cb1c942d657c41d436c7a1b6e29f65f3e900dbb9aff4064dc4ab2f843acda8"
289		);
290	}

307	#[test]
308	fn generated_certificate_is_valid() {
309		use p256::ecdsa::{signature::Verifier, VerifyingKey};
310		use x509_cert::der::Decode;
311
312		let certificate = derive_certificate(node_key(8)).unwrap();
313		let (certificate_der, _) = certificate.as_parts();
314
315		let parsed = Certificate::from_der(certificate_der).unwrap();
316		let validity = parsed.tbs_certificate.validity;
317		assert_eq!(validity.not_before.to_date_time(), DateTime::new(2000, 1, 1, 0, 0, 0).unwrap());
318		assert_eq!(
319			validity.not_after.to_date_time(),
320			DateTime::new(9999, 12, 31, 23, 59, 59).unwrap()
321		);
322		// The choice of time type is part of the encoding, not just the instant it denotes.
323		assert!(matches!(validity.not_before, Time::UtcTime(_)));
324		assert!(matches!(validity.not_after, Time::GeneralTime(_)));
325
326		// The self-signature verifies with the certificate's own public key.
327		let public_key = VerifyingKey::from_sec1_bytes(
328			parsed
329				.tbs_certificate
330				.subject_public_key_info
331				.subject_public_key
332				.as_bytes()
333				.unwrap(),
334		)
335		.unwrap();
336		let tbs = parsed.tbs_certificate.to_der().unwrap();
337		let signature = DerSignature::try_from(parsed.signature.as_bytes().unwrap()).unwrap();
338		public_key.verify(&tbs, &signature).unwrap();
339	}
