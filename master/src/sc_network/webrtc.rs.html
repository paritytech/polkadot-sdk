57/// Domain-separation tag used when deriving P-256 key from ed25519 key via HMAC-SHA256.
58const CERTIFICATE_KEY_DST: &[u8] = b"substrate-webrtc-p256-v1";
59
60/// Common name used for both subject and issuer, the certificate being self-signed.
61const SUBJECT_COMMON_NAME: &str = "polkadot-sdk-webrtc";

63/// Deterministically generate a WebRTC DTLS certificate from the node's secret key.
64pub fn derive_certificate(
65	node_secret_key: Ed25519SecretKey,
66) -> Result<DtlsCertificate, litep2p::Error> {
67	// NOTE: none of the expects in this function are input-dependent.
68	let signing_key = derive_keys(node_secret_key);
69
70	// RFC 5758 §3.2: `parameters` MUST be absent for ECDSA.
71	let signature_algorithm =
72		AlgorithmIdentifierOwned { oid: ECDSA_WITH_SHA_256, parameters: None };
73	let name = common_name();
74	// Uncompressed SEC1 form, requested explicitly: RFC 5480 permits several point forms.
75	let point = signing_key.verifying_key().as_affine().to_encoded_point(false);
76
77	let tbs_certificate = TbsCertificate {
78		// Every other WebRTC stack emits V3, let's do the same, even though it can be V1.
79		version: Version::V3,
80		serial_number: derive_serial(&point),
81		signature: signature_algorithm.clone(),
82		issuer: name.clone(),
83		validity: validity(),
84		subject: name,
85		subject_public_key_info: SubjectPublicKeyInfoOwned {
86			// RFC 5480 §2.1.1: `namedCurve`, rather than `implicitCurve` or `specifiedCurve`.
87			algorithm: AlgorithmIdentifierOwned {
88				oid: ID_EC_PUBLIC_KEY,
89				parameters: Some(Any::from(SECP_256_R_1)),
90			},
91			subject_public_key: BitString::new(0, point.as_bytes())
92				.expect("a SEC1 point is a whole number of octets; qed"),
93		},
94		// RFC 5280 §4.1.2.8: conforming CAs MUST NOT generate unique identifiers.
95		issuer_unique_id: None,
96		subject_unique_id: None,
97		extensions: None,
98	};
99
100	let tbs_der = tbs_certificate.to_der().expect("the certificate is well-formed; qed");
101
102	// `PrehashSigner` is RFC 6979 deterministic ECDSA; the module's determinism rests on it. If
103	// an upgrade ever changes its output (caught by the tests), the old implementation must be
104	// vendored into `sc-network`.
105	//
106	// Signing fails only if the RFC 6979 nonce or a signature scalar is zero (each 2^-256).
107	let signature: DerSignature = signing_key
108		.sign_prehash(&Sha256::digest(&tbs_der))
109		.expect("the signing key is valid; qed");
110
111	let certificate_der = Certificate {
112		tbs_certificate,
113		signature_algorithm,
114		// RFC 3279 §2.2.3: the DER-encoded `Ecdsa-Sig-Value`, with no unused bits.
115		signature: BitString::new(0, signature.to_vec())
116			.expect("an ECDSA signature is a whole number of octets; qed"),
117	}
118	.to_der()
119	.expect("a well-formed certificate is DER-encodable; qed");
120
121	let pk_pkcs8_der = signing_key
122		.to_pkcs8_der()
123		.expect("a P-256 signing key is PKCS#8-encodable; qed")
124		.as_bytes()
125		.to_vec();
126
127	DtlsCertificate::load(certificate_der, pk_pkcs8_der)
128}

130/// Derive the P-256 key by rejection-sampling HMAC-SHA256(node key, `DST || counter`).
131fn derive_keys(node_secret_key: Ed25519SecretKey) -> SigningKey {
132	(0u8..255u8)
133		.find_map(|counter| {
134			let okm = Hmac::<Sha256>::new_from_slice(node_secret_key.as_ref())
135				.expect("HMAC accepts keys of any length; qed")
136				.chain_update(CERTIFICATE_KEY_DST)
137				.chain_update([counter])
138				.finalize()
139				.into_bytes();
140
141			// Range-checked here rather than by letting `SigningKey::from_slice` reject: which
142			// counter wins must not depend on whether out-of-range values are rejected or reduced.
143			let scalar = U256::from_be_slice(&okm);
144			(scalar != U256::ZERO && scalar < NistP256::ORDER)
145				.then(|| SigningKey::from_slice(&okm).expect("checked to be in range; qed"))
146		})
147		.expect("each iteration succeeds with probability 1 - 2^-32, and we have 256 of them; qed")
148}

150/// Serial derived from the public key, only to avoid one hardcoded serial for all certificates.
151fn derive_serial(point: &EncodedPoint) -> SerialNumber {
152	let digest = Sha256::digest(point.as_bytes());
153
154	// A `u64` names exactly one integer, fixing the DER by spec; a byte string would leave
155	// signedness to `SerialNumber`.
156	SerialNumber::from(u64::from_be_bytes(digest[..8].try_into().expect("8 of 32 bytes; qed")))
157}

159/// `CN=polkadot-sdk-webrtc`, built explicitly so the attribute's string type is pinned.
160fn common_name() -> Name {
161	let attribute = AttributeTypeAndValue {
162		oid: COMMON_NAME,
163		value: Any::new(Tag::Utf8String, SUBJECT_COMMON_NAME.as_bytes())
164			.expect("the common name is a valid `UTF8String`; qed"),
165	};
166	let rdn = SetOfVec::try_from(vec![attribute]).expect("a single-element set is sorted; qed");
167
168	Name::from(vec![RelativeDistinguishedName::from(rdn)])
169}

171/// Fixed validity dates, required for determinism. WebRTC peers pin the certificate by
172/// certhash and ignore its lifetime.
173fn validity() -> Validity {
174	// RFC 5280 §4.1.2.5: dates through 2049 MUST be encoded as `UTCTime`.
175	let not_before = UtcTime::from_date_time(
176		DateTime::new(2000, 1, 1, 0, 0, 0).expect("2000-01-01 00:00:00 is a valid date; qed"),
177	)
178	.expect("2000 is within the `UTCTime` range; qed");
179	// RFC 5280 §4.1.2.5: the encoding for certificates with no well-defined expiration date.
180	let not_after = GeneralizedTime::from_date_time(
181		DateTime::new(9999, 12, 31, 23, 59, 59).expect("9999-12-31 23:59:59 is a valid date; qed"),
182	);
183
184	Validity { not_before: Time::UtcTime(not_before), not_after: Time::GeneralTime(not_after) }
185}

187/// Whether `address` contains the `webrtc-direct` transport
188pub(crate) fn is_webrtc_address(address: &Multiaddr) -> bool {
189	address.iter().any(|protocol| matches!(protocol, Protocol::WebRTCDirect))
190}
191
192/// Check that `address` is a bare `webrtc-direct` address.
193///
194/// Dns is accepted as host only if the validation is applied to a public address.
195fn validate(address: &Multiaddr, public_addr: bool) -> Result<(), Error> {
196	let mut iter = address.iter();
197
198	let host_is_valid = match iter.next() {
199		Some(Protocol::Ip4(_) | Protocol::Ip6(_)) => true,
200		Some(Protocol::Dns(_) | Protocol::Dns4(_) | Protocol::Dns6(_)) => public_addr,
201		_ => false,
202	};
203
204	// `/udp/<port>/webrtc-direct` and nothing after it.
205	let is_valid = host_is_valid &&
206		matches!(
207			(iter.next(), iter.next(), iter.next()),
208			(Some(Protocol::Udp(_)), Some(Protocol::WebRTCDirect), None)
209		);
210
211	is_valid
212		.then_some(())
213		.ok_or_else(|| Error::InvalidWebRtcAddress { address: address.clone() })
214}
215
216pub(crate) fn validate_listen_address(address: &Multiaddr) -> Result<(), Error> {
217	validate(address, false)
218}
219
220pub(crate) fn validate_public_address(address: &Multiaddr) -> Result<(), Error> {
221	validate(address, true)
222}
223
224/// Append the node's `certhash` to a public `webrtc-direct` address.
225pub fn complete_public_address(address: &mut Multiaddr, certhash: Multihash) -> Result<(), Error> {
226	validate_public_address(address)?;
227	address.push(Protocol::Certhash(certhash));
228	Ok(())
229}
230
231/// Check the shape of every `webrtc-direct` address, then append `certhash` to the public ones.
232pub(crate) fn validate_and_complete_addresses(
233	listen_addresses: &[Multiaddr],
234	public_addresses: &mut [Multiaddr],
235	certhash: Multihash,
236) -> Result<(), Error> {
237	// Listen addresses are validated but never completed, unlike the public ones below.
238	//
239	// A listen address names a socket to bind, and `/certhash` is no part of binding one.
240	// litep2p appends it itself when it reports the address it actually bound.
241	//
242	// A public address is the opposite: it is handed to peers to dial, and a `webrtc-direct` dialer
243	// verifies the DTLS handshake against that hash, so it has to be there before anything reads
244	// `--public-addr`.
245	for address in listen_addresses {
246		if is_webrtc_address(address) {
247			validate_listen_address(address)?;
248		}
249	}
250
251	for address in public_addresses.iter_mut() {
252		if is_webrtc_address(address) {
253			complete_public_address(address, certhash)?;
254		}
255	}
256
257	Ok(())
258}

277	/// Compute the `/certhash/<hash>` multiaddress component of a certificate.
278	fn certhash_component(certificate: &DtlsCertificate) -> String {
279		let hash = Code::Sha2_256.digest(certificate.as_parts().0);
280		Multiaddr::empty().with(Protocol::Certhash(hash)).to_string()
281	}

318	#[test]
319	fn stable_certhash() {
320		// Pins the node key -> certificate derivation. If this test ever fails, the derivation
321		// changed and the certhash published by every node relying on it breaks.
322		//
323		// Three vectors, because the serial's INTEGER encoding splits on its first byte:
324		// `node_key(7)` (serial `0x87..`) gets a `0x00` sign byte, `node_key(42)` (`0x6e..`)
325		// encodes plain, and the third key (`0x002b..`) has its leading zero stripped.
326		assert_eq!(
327			certhash_component(&derive_certificate(node_key(7)).unwrap()),
328			"/certhash/uEiAXqXtF_3QIfMcgXwMgneoB4EuSE_EcpGvKhY4yz7HfcA"
329		);
330		assert_eq!(
331			certhash_component(&derive_certificate(node_key(42)).unwrap()),
332			"/certhash/uEiAWsH8V-_VMveqodSJYiAhW5FikqSzBNLV0FyeEb_oetA"
333		);
334		let mut stripped_serial_key = [3u8; 32];
335		stripped_serial_key[31] = 26;
336		assert_eq!(
337			certhash_component(&derive_certificate(node_key_from(stripped_serial_key)).unwrap()),

342	#[test]
343	fn rfc6979_signing_is_pinned() {
344		use p256::ecdsa::Signature;
345
346		// Everything else in this module states its own encoding; the signature cannot: ECDSA
347		// needs a nonce, and reproducibility rests on `sign_prehash` deriving it via RFC 6979.
348		// The RFC leaves inputs that `ecdsa` fills in for us, so pin the signature itself — an
349		// upgrade changing any of them fails here, naming the cause.
350		//
351		// Pinned only for prehashes below the group order: `ecdsa` 0.16 feeds `h1` to the nonce
352		// DRBG unreduced, 0.17 reduces it per RFC 6979 §2.3.4. Bumping past 0.16 thus changes
353		// the certhash of nodes whose TBS digest is >= `n` — accepted, as that is ~2^-32 of them.
354		let key = SigningKey::from_slice(&array_bytes::hex2bytes_unchecked(
355			"c9afa9d845ba75166b5c215767b1d6934e50c3db36e89b127b8a622b120f6721",
356		))
357		.unwrap();
358
359		// RFC 6979 A.2.5: P-256 with SHA-256, message "sample". Pins the HMAC hash
360		// (`NistP256::Digest`) and the empty additional-data argument.
361		let signature: Signature = key.sign_prehash(&Sha256::digest(b"sample")).unwrap();
362		assert_eq!(
363			array_bytes::bytes2hex("", signature.to_bytes()),
364			"efd48b2aacb6a8fd1140dd9cd45e81d69d2c877b56aaf991c34d0ea84eaf3716\
365			 f7cb1c942d657c41d436c7a1b6e29f65f3e900dbb9aff4064dc4ab2f843acda8"
366		);
367	}
