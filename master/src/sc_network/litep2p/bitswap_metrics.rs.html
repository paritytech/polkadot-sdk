19//! Prometheus metrics for the litep2p bitswap server.
20
21use litep2p::protocol::libp2p::bitswap::{BlockPresenceType, ResponseType};
22use prometheus_endpoint::{
23	exponential_buckets, register, Counter, CounterVec, Histogram, HistogramOpts, Opts,
24	PrometheusError, Registry, U64,
25};
26use std::time::Duration;
27
28/// `outcome` label values for `substrate_sub_libp2p_bitswap_entries_total`.
29pub mod outcomes {
30	pub const BLOCK_SERVED: &str = "block_served";
31	pub const HAVE: &str = "have";
32	pub const DONT_HAVE: &str = "dont_have";
33	pub const UNSUPPORTED_CID: &str = "unsupported_cid";
34}
35
36/// `reason` label values for `substrate_sub_libp2p_bitswap_request_errors_total`.
37pub mod errors {
38	pub const TOO_MANY_ENTRIES: &str = "too_many_entries";
39	pub const CLIENT: &str = "client";
40}
41
42struct Inner {
43	entries_total: CounterVec<U64>,
44	request_errors_total: CounterVec<U64>,
45	inbound_request_duration_seconds: Histogram,
46	response_bytes_total: Counter<U64>,
47}
48
49impl Inner {
50	fn register(registry: &Registry) -> Result<Self, PrometheusError> {
51		Ok(Self {
52			entries_total: register(
53				CounterVec::new(
54					Opts::new(
55						"substrate_sub_libp2p_bitswap_entries_total",
56						"Total number of bitswap wantlist entries processed, by outcome",
57					),
58					&["outcome"],
59				)?,
60				registry,
61			)?,
62			request_errors_total: register(
63				CounterVec::new(
64					Opts::new(
65						"substrate_sub_libp2p_bitswap_request_errors_total",
66						"Total number of bitswap inbound requests rejected, by reason",
67					),
68					&["reason"],
69				)?,
70				registry,
71			)?,
72			inbound_request_duration_seconds: register(
73				Histogram::with_opts(HistogramOpts {
74					common_opts: Opts::new(
75						"substrate_sub_libp2p_bitswap_inbound_request_duration_seconds",
76						"Duration of handling an inbound bitswap wantlist, in seconds",
77					),
78					buckets: exponential_buckets(0.001, 2.0, 16)
79						.expect("parameters are always valid values; qed"),
80				})?,
81				registry,
82			)?,
83			response_bytes_total: register(
84				Counter::new(
85					"substrate_sub_libp2p_bitswap_response_bytes_total",
86					"Total bytes sent in bitswap responses to inbound wantlists",
87				)?,
88				registry,
89			)?,
90		})
91	}
92}
93
94/// Helper wrapper around the bitswap server metrics.
95///
96/// When constructed without a `Registry`, all recording methods become no-ops.
97pub struct BitswapMetrics {
98	inner: Option<Inner>,
99}
100
101impl BitswapMetrics {
102	/// Register the metrics with the given Prometheus registry, if any.
103	pub fn new(registry: Option<&Registry>) -> Result<Self, PrometheusError> {
104		Ok(Self { inner: registry.map(Inner::register).transpose()? })
105	}
106
107	/// Record one wantlist entry processed with the given outcome.
108	pub fn record_entry(&self, outcome: &str) {
109		if let Some(inner) = &self.inner {
110			inner.entries_total.with_label_values(&[outcome]).inc();
111		}
112	}
113
114	/// Record one outbound response variant under the matching outcome label.
115	pub fn record_response(&self, response: &ResponseType) {
116		let outcome = match response {
117			ResponseType::Block { .. } => outcomes::BLOCK_SERVED,
118			ResponseType::Presence { presence: BlockPresenceType::Have, .. } => outcomes::HAVE,
119			ResponseType::Presence { presence: BlockPresenceType::DontHave, .. } => {
120				outcomes::DONT_HAVE
121			},
122		};
123		self.record_entry(outcome);
124	}
125
126	/// Record one request-level error with the given reason.
127	pub fn record_error(&self, reason: &str) {
128		if let Some(inner) = &self.inner {
129			inner.request_errors_total.with_label_values(&[reason]).inc();
130		}
131	}
132
133	/// Observe the duration of an inbound wantlist handling.
134	pub fn record_duration(&self, duration: Duration) {
135		if let Some(inner) = &self.inner {
136			inner.inbound_request_duration_seconds.observe(duration.as_secs_f64());
137		}
138	}
139
140	/// Add to the running total of bitswap response bytes sent.
141	pub fn add_response_bytes(&self, bytes: u64) {
142		if let Some(inner) = &self.inner {
143			inner.response_bytes_total.inc_by(bytes);
144		}
145	}
146}

148#[cfg(test)]
149mod tests {
150	use super::*;
151	use cid::{multihash::Multihash as CidMultihash, Cid};
152
153	fn make_cid() -> Cid {
154		let mh = CidMultihash::<64>::wrap(0xb220, &[0u8; 32]).unwrap();
155		Cid::new_v1(0x55, mh)
156	}
157
158	#[test]
159	fn disabled_metrics_are_no_ops() {
160		let metrics = BitswapMetrics::new(None).unwrap();
161		metrics.record_entry(outcomes::BLOCK_SERVED);
162		metrics.record_error(errors::CLIENT);
163		metrics.record_duration(Duration::from_millis(1));
164		metrics.add_response_bytes(42);
165	}
166
167	#[test]
168	fn record_response_maps_variants_to_outcomes() {
169		let registry = Registry::new();
170		let metrics = BitswapMetrics::new(Some(&registry)).unwrap();
171		let cid = make_cid();
172
173		metrics.record_response(&ResponseType::Block { cid, block: vec![1, 2, 3] });
174		metrics.record_response(&ResponseType::Block { cid, block: vec![4] });
175		metrics.record_response(&ResponseType::Presence { cid, presence: BlockPresenceType::Have });
176		metrics.record_response(&ResponseType::Presence {
177			cid,
178			presence: BlockPresenceType::DontHave,
179		});
180		metrics.record_response(&ResponseType::Presence {
181			cid,
182			presence: BlockPresenceType::DontHave,
183		});
184		metrics.record_response(&ResponseType::Presence {
185			cid,
186			presence: BlockPresenceType::DontHave,
187		});
188
189		let inner = metrics.inner.as_ref().expect("inner should be present");
190		assert_eq!(inner.entries_total.with_label_values(&[outcomes::BLOCK_SERVED]).get(), 2);
191		assert_eq!(inner.entries_total.with_label_values(&[outcomes::HAVE]).get(), 1);
192		assert_eq!(inner.entries_total.with_label_values(&[outcomes::DONT_HAVE]).get(), 3);
193		assert_eq!(inner.entries_total.with_label_values(&[outcomes::UNSUPPORTED_CID]).get(), 0);
194	}
195
196	#[test]
197	fn enabled_metrics_register_and_increment() {
198		let registry = Registry::new();
199		let metrics = BitswapMetrics::new(Some(&registry)).unwrap();
200
201		metrics.record_entry(outcomes::BLOCK_SERVED);
202		metrics.record_entry(outcomes::BLOCK_SERVED);
203		metrics.record_entry(outcomes::HAVE);
204		metrics.record_error(errors::TOO_MANY_ENTRIES);
205		metrics.record_duration(Duration::from_millis(5));
206		metrics.add_response_bytes(1024);
207
208		let inner = metrics.inner.as_ref().expect("inner should be present when registry given");
209		assert_eq!(inner.entries_total.with_label_values(&[outcomes::BLOCK_SERVED]).get(), 2);
210		assert_eq!(inner.entries_total.with_label_values(&[outcomes::HAVE]).get(), 1);
211		assert_eq!(inner.entries_total.with_label_values(&[outcomes::DONT_HAVE]).get(), 0);
212		assert_eq!(
213			inner.request_errors_total.with_label_values(&[errors::TOO_MANY_ENTRIES]).get(),
214			1
215		);
216		assert_eq!(inner.response_bytes_total.get(), 1024);
217		assert_eq!(inner.inbound_request_duration_seconds.get_sample_count(), 1);
218	}
