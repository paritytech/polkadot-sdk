63/// Outbound bitswap command sent from [`super::service::Litep2pNetworkService`].
64pub(crate) struct BitswapOutboundCmd {
65	pub(crate) peer: litep2p::PeerId,
66	pub(crate) wants: Vec<(Cid, WantType)>,
67	pub(crate) response_tx: ResponseSender,
68}

70/// Pending outbound WANT batch.
71struct PendingBatch {
72	cids: Vec<Cid>,
73	responses: HashMap<Cid, ResponseType>,
74	response_bytes: usize,
75	response_tx: Option<ResponseSender>,
76	inserted: Instant,
77}
78
79impl PendingBatch {
80	fn new(cids: Vec<Cid>, response_tx: ResponseSender, inserted: Instant) -> Self {
81		Self {
82			cids,
83			responses: HashMap::new(),
84			response_bytes: 0,
85			response_tx: Some(response_tx),
86			inserted,
87		}
88	}
89
90	fn record_responses(&mut self, responses: &HashMap<Cid, ResponseType>) {
91		for cid in &self.cids {
92			if self.responses.contains_key(cid) {
93				continue;
94			}
95			let Some(resp) = responses.get(cid) else { continue };
96			self.response_bytes = self.response_bytes.saturating_add(response_retained_bytes(resp));
97			self.responses.insert(*cid, resp.clone());
98		}
99	}
100
101	fn is_complete(&self) -> bool {
102		self.cids.len() == self.responses.len()
103	}
104
105	fn is_over_limit(&self, max_response_bytes: usize) -> bool {
106		self.response_bytes > max_response_bytes
107	}
108
109	fn is_expired(&self, timeout: Duration, now: Instant) -> bool {
110		now.saturating_duration_since(self.inserted) >= timeout
111	}
112
113	fn send_success(&mut self) {
114		let responses: Vec<ResponseType> =
115			self.cids.iter().filter_map(|cid| self.responses.get(cid).cloned()).collect();
116		let encoded = encode_responses_as_bitswap_message(&responses);
117		if let Some(response_tx) = self.response_tx.take() {
118			let _ = response_tx.send(Ok((encoded, ProtocolName::from(PROTOCOL_NAME))));
119		}
120	}
121
122	fn send_failure(&mut self, failure: RequestFailure) {
123		if let Some(response_tx) = self.response_tx.take() {
124			let _ = response_tx.send(Err(failure));
125		}
126	}
127}

139/// Pending outbound WANT batches, indexed by peer.
140#[derive(Default)]
141struct PendingBatches {
142	by_peer: HashMap<litep2p::PeerId, Vec<PendingBatch>>,
143}
144
145impl PendingBatches {
146	fn insert(&mut self, peer: litep2p::PeerId, batch: PendingBatch) {
147		self.by_peer.entry(peer).or_default().push(batch);
148	}
149
150	fn handle_response(&mut self, peer: litep2p::PeerId, responses: Vec<ResponseType>) {
151		self.handle_response_with_limit(peer, responses, MAX_RESPONSE_SIZE as usize);
152	}
153
154	fn handle_response_with_limit(
155		&mut self,
156		peer: litep2p::PeerId,
157		responses: Vec<ResponseType>,
158		max_response_bytes: usize,
159	) {
160		log::debug!(
161			target: LOG_TARGET,
162			"bitswap: received response from {peer:?} with {} entries",
163			responses.len()
164		);
165
166		let Some(peer_batches) = self.by_peer.get_mut(&peer) else { return };
167		let best = select_best_response_per_cid(responses);
168
169		peer_batches.retain_mut(|batch| {
170			batch.record_responses(&best);
171
172			if batch.is_over_limit(max_response_bytes) {
173				log::warn!(
174					target: LOG_TARGET,
175					"bitswap: response from {peer:?} exceeded pending batch byte limit: {} > {}",
176					batch.response_bytes,
177					max_response_bytes,
178				);
179				batch.send_failure(RequestFailure::Network(OutboundFailure::ConnectionClosed));
180				false
181			} else if batch.is_complete() {
182				batch.send_success();
183				false
184			} else {
185				true
186			}
187		});
188
189		if peer_batches.is_empty() {
190			self.by_peer.remove(&peer);
191		}
192	}
193
194	fn expire(&mut self, timeout: Duration, now: Instant) {
195		self.by_peer.retain(|peer, peer_batches| {
196			peer_batches.retain_mut(|batch| {
197				if batch.is_expired(timeout, now) {
198					log::debug!(
199						target: LOG_TARGET,
200						"bitswap: expired pending batch for {} CIDs from {:?}",
201						batch.cids.len(),
202						peer,
203					);
204					batch.send_failure(RequestFailure::Network(OutboundFailure::Timeout));
205					false
206				} else {
207					true
208				}
209			});
210
211			!peer_batches.is_empty()
212		});
213	}
214
215	#[cfg(test)]
216	fn is_empty(&self) -> bool {
217		self.by_peer.is_empty()
218	}
219
220	#[cfg(test)]
221	fn len(&self) -> usize {
222		self.by_peer.len()
223	}
224
225	#[cfg(test)]
226	fn contains_key(&self, peer: &litep2p::PeerId) -> bool {
227		self.by_peer.contains_key(peer)
228	}
229}

231/// Bidirectional bitswap service for litep2p.
232pub(crate) struct BitswapService<Block: BlockT> {
233	handle: BitswapHandle,
234	client: Arc<dyn BlockBackend<Block> + Send + Sync>,
235	cmd_rx: mpsc::Receiver<BitswapOutboundCmd>,
236	pending: PendingBatches,
237	metrics: BitswapMetrics,
238}
239
240impl<Block: BlockT> BitswapService<Block> {
241	/// Create a new bidirectional bitswap service.
242	///
243	/// Returns the boxed task future (to be spawned on the executor) and the
244	/// [`BitswapConfig`] to be passed into the litep2p config builder.
245	///
246	/// If `metrics_registry` is `Some`, Prometheus metrics are registered with
247	/// it. A registration failure is logged and falls back to disabled metrics
248	/// (the service still runs).
249	pub(crate) fn new(
250		client: Arc<dyn BlockBackend<Block> + Send + Sync>,
251		metrics_registry: Option<&Registry>,
252	) -> (Pin<Box<dyn Future<Output = ()> + Send>>, BitswapConfig) {
253		let metrics = BitswapMetrics::new(metrics_registry).unwrap_or_else(|err| {
254			log::debug!(target: LOG_TARGET, "failed to register bitswap metrics: {err}");
255			BitswapMetrics::new(None).expect("registering with None registry never fails; qed")
256		});
257		let (litep2p_config, handle) = Config::new();
258		let (cmd_tx, cmd_rx) = mpsc::channel(CMD_CHANNEL_CAPACITY);
259		let service = Self { handle, client, cmd_rx, pending: PendingBatches::default(), metrics };
260		let future = Box::pin(async move { service.run().await });
261		let config = BitswapConfig { litep2p_config, cmd_tx };
262		(future, config)
263	}
264
265	/// Run the bitswap event loop.
266	async fn run(mut self) {
267		log::debug!(target: LOG_TARGET, "starting bidirectional bitswap service");
268		let mut expiry_ticker = tokio::time::interval(EXPIRY_TICK_INTERVAL);
269		expiry_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);
270		expiry_ticker.tick().await;
271
272		loop {
273			tokio::select! {
274				event = self.handle.next() => match event {
275					Some(BitswapEvent::Request { peer, cids }) =>
276						self.handle_inbound_request(peer, cids).await,
277					Some(BitswapEvent::Response { peer, responses }) =>
278						self.pending.handle_response(peer, responses),
279					None => {
280						log::debug!(target: LOG_TARGET, "bitswap handle stream ended");
281						return;
282					},
283				},
284				cmd = self.cmd_rx.recv() => match cmd {
285					Some(BitswapOutboundCmd { peer, wants, response_tx }) =>
286						self.handle_outbound_cmd(peer, wants, response_tx).await,
287					None => {
288						log::debug!(target: LOG_TARGET, "bitswap cmd channel closed");
289						return;
290					},
291				},
292				_ = expiry_ticker.tick() => {
293					self.pending.expire(REQUEST_TIMEOUT, Instant::now());
294				},
295			}
296		}
297	}

299	/// Handle an inbound bitswap WANT request from `peer`.
300	async fn handle_inbound_request(&mut self, peer: litep2p::PeerId, cids: Vec<(Cid, WantType)>) {
301		let started = Instant::now();
302		let want_count = cids.len();
303		if inbound_wantlist_exceeds_limit(want_count) {
304			self.metrics.record_error(errors::TOO_MANY_ENTRIES);
305			log::trace!(target: LOG_TARGET, "bitswap: ignored inbound request with {want_count} entries");
306			return;
307		}
308
309		log::debug!(target: LOG_TARGET, "bitswap: handle inbound request from {peer:?} for {cids:?}");
310
311		let metrics = &self.metrics;
312		let response: Vec<ResponseType> = cids
313			.into_iter()
314			.filter(|(cid, _)| {
315				let supported = is_cid_supported(cid);
316				if !supported {
317					metrics.record_entry(outcomes::UNSUPPORTED_CID);
318				}
319				supported
320			})
321			.map(|(cid, want_type)| {
322				let hash = H256::from_slice(&cid.hash().digest()[0..32]);
323				let transaction = match self.client.indexed_transaction(hash) {
324					Ok(ex) => ex,
325					Err(error) => {
326						metrics.record_error(errors::CLIENT);
327						log::error!(target: LOG_TARGET, "error retrieving transaction {hash}: {error}");
328						None
329					},
330				};
331				let response = match transaction {
332					Some(transaction) => match want_type {
333						WantType::Block => ResponseType::Block { cid, block: transaction },
334						_ => ResponseType::Presence { cid, presence: BlockPresenceType::Have },
335					},
336					None => ResponseType::Presence { cid, presence: BlockPresenceType::DontHave },
337				};
338				metrics.record_response(&response);
339				response
340			})
341			.collect();
342
343		// note: we assume the duplicate encode (litep2p re-serialises internally inside
344		// `send_response`) is cheap.
345		let response_bytes = encode_responses_as_bitswap_message(&response).len();
346		self.metrics.add_response_bytes(response_bytes as u64);
347
348		self.handle.send_response(peer, response).await;
349		self.metrics.record_duration(started.elapsed());
350	}
351
352	/// Handle an outbound bitswap command from the network service.
353	async fn handle_outbound_cmd(
354		&mut self,
355		peer: litep2p::PeerId,
356		wants: Vec<(Cid, WantType)>,
357		response_tx: ResponseSender,
358	) {
359		log::debug!(
360			target: LOG_TARGET,
361			"bitswap: outbound WANT for {} CIDs to {peer:?}",
362			wants.len(),
363		);
364		let cids: Vec<_> = wants.iter().map(|(cid, _)| *cid).collect();
365		self.pending.insert(peer, PendingBatch::new(cids, response_tx, Instant::now()));
366		self.handle.send_request(peer, wants).await;
367	}
368}
369
370fn inbound_wantlist_exceeds_limit(len: usize) -> bool {
371	len > MAX_WANTED_BLOCKS
372}
373
374/// Collapse a response list into at most one entry per CID, preferring `Block`
375/// over `Presence` when both arrive for the same CID.
376fn select_best_response_per_cid(responses: Vec<ResponseType>) -> HashMap<Cid, ResponseType> {
377	let mut best: HashMap<Cid, ResponseType> = HashMap::new();
378	for resp in responses {
379		let cid = match &resp {
380			ResponseType::Block { cid, .. } => *cid,
381			ResponseType::Presence { cid, .. } => *cid,
382		};
383		match best.entry(cid) {
384			std::collections::hash_map::Entry::Vacant(e) => {
385				e.insert(resp);
386			},
387			std::collections::hash_map::Entry::Occupied(mut e) => {
388				if matches!(resp, ResponseType::Block { .. }) &&
389					matches!(*e.get(), ResponseType::Presence { .. })
390				{
391					e.insert(resp);
392				}
393			},
394		}
395	}
396	best
397}

399/// Return the byte size of a response that counts toward the pending batch cap.
400fn response_retained_bytes(response: &ResponseType) -> usize {
401	match response {
402		ResponseType::Block { block, .. } => block.len(),
403		ResponseType::Presence { .. } => 0,
404	}
405}
406
407/// Encode litep2p [`ResponseType`] values into a [`BitswapProtoMessage`] byte vector.
408fn encode_responses_as_bitswap_message(responses: &[ResponseType]) -> Vec<u8> {
409	let mut msg = BitswapProtoMessage::default();
410
411	for resp in responses {
412		match resp {
413			ResponseType::Block { cid, block } => {
414				let prefix: Prefix = cid.into();
415				msg.payload
416					.push(MessageBlock { prefix: prefix.to_bytes(), data: block.clone() });
417			},
418			ResponseType::Presence { cid, presence } => {
419				msg.block_presences.push(BlockPresence {
420					cid: cid.to_bytes(),
421					r#type: match presence {
422						BlockPresenceType::Have => ProtoPresenceType::Have as i32,
423						BlockPresenceType::DontHave => ProtoPresenceType::DontHave as i32,
424					},
425				});
426			},
427		}
428	}
429
430	msg.encode_to_vec()
431}
