61/// Outbound bitswap command sent from [`super::service::Litep2pNetworkService`].
62pub(crate) struct BitswapOutboundCmd {
63	pub(crate) peer: litep2p::PeerId,
64	pub(crate) wants: Vec<(Cid, WantType)>,
65	pub(crate) response_tx: ResponseSender,
66}

68/// Pending outbound WANT batch.
69struct PendingBatch {
70	cids: Vec<Cid>,
71	responses: HashMap<Cid, ResponseType>,
72	response_bytes: usize,
73	response_tx: Option<ResponseSender>,
74	inserted: Instant,
75}
76
77impl PendingBatch {
78	fn new(cids: Vec<Cid>, response_tx: ResponseSender, inserted: Instant) -> Self {
79		Self {
80			cids,
81			responses: HashMap::new(),
82			response_bytes: 0,
83			response_tx: Some(response_tx),
84			inserted,
85		}
86	}
87
88	fn record_responses(&mut self, responses: &HashMap<Cid, ResponseType>) {
89		for cid in &self.cids {
90			if self.responses.contains_key(cid) {
91				continue;
92			}
93			let Some(resp) = responses.get(cid) else { continue };
94			self.response_bytes = self.response_bytes.saturating_add(response_retained_bytes(resp));
95			self.responses.insert(*cid, resp.clone());
96		}
97	}
98
99	fn is_complete(&self) -> bool {
100		self.cids.len() == self.responses.len()
101	}
102
103	fn is_over_limit(&self, max_response_bytes: usize) -> bool {
104		self.response_bytes > max_response_bytes
105	}
106
107	fn is_expired(&self, timeout: Duration, now: Instant) -> bool {
108		now.saturating_duration_since(self.inserted) >= timeout
109	}
110
111	fn send_success(&mut self) {
112		let responses: Vec<ResponseType> =
113			self.cids.iter().filter_map(|cid| self.responses.get(cid).cloned()).collect();
114		let encoded = encode_responses_as_bitswap_message(&responses);
115		if let Some(response_tx) = self.response_tx.take() {
116			let _ = response_tx.send(Ok((encoded, ProtocolName::from(PROTOCOL_NAME))));
117		}
118	}
119
120	fn send_failure(&mut self, failure: RequestFailure) {
121		if let Some(response_tx) = self.response_tx.take() {
122			let _ = response_tx.send(Err(failure));
123		}
124	}
125}

137/// Pending outbound WANT batches, indexed by peer.
138#[derive(Default)]
139struct PendingBatches {
140	by_peer: HashMap<litep2p::PeerId, Vec<PendingBatch>>,
141}
142
143impl PendingBatches {
144	fn insert(&mut self, peer: litep2p::PeerId, batch: PendingBatch) {
145		self.by_peer.entry(peer).or_default().push(batch);
146	}
147
148	fn handle_response(&mut self, peer: litep2p::PeerId, responses: Vec<ResponseType>) {
149		self.handle_response_with_limit(peer, responses, MAX_RESPONSE_SIZE as usize);
150	}
151
152	fn handle_response_with_limit(
153		&mut self,
154		peer: litep2p::PeerId,
155		responses: Vec<ResponseType>,
156		max_response_bytes: usize,
157	) {
158		log::debug!(
159			target: LOG_TARGET,
160			"bitswap: received response from {peer:?} with {} entries",
161			responses.len()
162		);
163
164		let Some(peer_batches) = self.by_peer.get_mut(&peer) else { return };
165		let best = select_best_response_per_cid(responses);
166
167		peer_batches.retain_mut(|batch| {
168			batch.record_responses(&best);
169
170			if batch.is_over_limit(max_response_bytes) {
171				log::warn!(
172					target: LOG_TARGET,
173					"bitswap: response from {peer:?} exceeded pending batch byte limit: {} > {}",
174					batch.response_bytes,
175					max_response_bytes,
176				);
177				batch.send_failure(RequestFailure::Network(OutboundFailure::ConnectionClosed));
178				false
179			} else if batch.is_complete() {
180				batch.send_success();
181				false
182			} else {
183				true
184			}
185		});
186
187		if peer_batches.is_empty() {
188			self.by_peer.remove(&peer);
189		}
190	}
191
192	fn expire(&mut self, timeout: Duration, now: Instant) {
193		self.by_peer.retain(|peer, peer_batches| {
194			peer_batches.retain_mut(|batch| {
195				if batch.is_expired(timeout, now) {
196					log::debug!(
197						target: LOG_TARGET,
198						"bitswap: expired pending batch for {} CIDs from {:?}",
199						batch.cids.len(),
200						peer,
201					);
202					batch.send_failure(RequestFailure::Network(OutboundFailure::Timeout));
203					false
204				} else {
205					true
206				}
207			});
208
209			!peer_batches.is_empty()
210		});
211	}
212
213	#[cfg(test)]
214	fn is_empty(&self) -> bool {
215		self.by_peer.is_empty()
216	}
217
218	#[cfg(test)]
219	fn len(&self) -> usize {
220		self.by_peer.len()
221	}
222
223	#[cfg(test)]
224	fn contains_key(&self, peer: &litep2p::PeerId) -> bool {
225		self.by_peer.contains_key(peer)
226	}
227}

229/// Bidirectional bitswap service for litep2p.
230pub(crate) struct BitswapService<Block: BlockT> {
231	handle: BitswapHandle,
232	client: Arc<dyn BlockBackend<Block> + Send + Sync>,
233	cmd_rx: mpsc::Receiver<BitswapOutboundCmd>,
234	pending: PendingBatches,
235}
236
237impl<Block: BlockT> BitswapService<Block> {
238	/// Create a new bidirectional bitswap service.
239	///
240	/// Returns the boxed task future (to be spawned on the executor) and the
241	/// [`BitswapConfig`] to be passed into the litep2p config builder.
242	pub(crate) fn new(
243		client: Arc<dyn BlockBackend<Block> + Send + Sync>,
244	) -> (Pin<Box<dyn Future<Output = ()> + Send>>, BitswapConfig) {
245		let (litep2p_config, handle) = Config::new();
246		let (cmd_tx, cmd_rx) = mpsc::channel(CMD_CHANNEL_CAPACITY);
247		let service = Self { handle, client, cmd_rx, pending: PendingBatches::default() };
248		let future = Box::pin(async move { service.run().await });
249		let config = BitswapConfig { litep2p_config, cmd_tx };
250		(future, config)
251	}
252
253	/// Run the bitswap event loop.
254	async fn run(mut self) {
255		log::debug!(target: LOG_TARGET, "starting bidirectional bitswap service");
256		let mut expiry_ticker = tokio::time::interval(EXPIRY_TICK_INTERVAL);
257		expiry_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);
258		expiry_ticker.tick().await;
259
260		loop {
261			tokio::select! {
262				event = self.handle.next() => match event {
263					Some(BitswapEvent::Request { peer, cids }) =>
264						self.handle_inbound_request(peer, cids).await,
265					Some(BitswapEvent::Response { peer, responses }) =>
266						self.pending.handle_response(peer, responses),
267					None => {
268						log::debug!(target: LOG_TARGET, "bitswap handle stream ended");
269						return;
270					},
271				},
272				cmd = self.cmd_rx.recv() => match cmd {
273					Some(BitswapOutboundCmd { peer, wants, response_tx }) =>
274						self.handle_outbound_cmd(peer, wants, response_tx).await,
275					None => {
276						log::debug!(target: LOG_TARGET, "bitswap cmd channel closed");
277						return;
278					},
279				},
280				_ = expiry_ticker.tick() => {
281					self.pending.expire(REQUEST_TIMEOUT, Instant::now());
282				},
283			}
284		}
285	}

287	/// Handle an inbound bitswap WANT request from `peer`.
288	async fn handle_inbound_request(&mut self, peer: litep2p::PeerId, cids: Vec<(Cid, WantType)>) {
289		let want_count = cids.len();
290		if inbound_wantlist_exceeds_limit(want_count) {
291			log::trace!(target: LOG_TARGET, "bitswap: ignored inbound request with {want_count} entries");
292			return;
293		}
294
295		log::debug!(target: LOG_TARGET, "bitswap: handle inbound request from {peer:?} for {cids:?}");
296
297		let response: Vec<ResponseType> = cids
298			.into_iter()
299			.filter(|(cid, _)| is_cid_supported(&cid))
300			.map(|(cid, want_type)| {
301				let hash = H256::from_slice(&cid.hash().digest()[0..32]);
302				let transaction = match self.client.indexed_transaction(hash) {
303					Ok(ex) => ex,
304					Err(error) => {
305						log::error!(target: LOG_TARGET, "error retrieving transaction {hash}: {error}");
306						None
307					},
308				};
309				match transaction {
310					Some(transaction) => match want_type {
311						WantType::Block => ResponseType::Block { cid, block: transaction },
312						_ => ResponseType::Presence { cid, presence: BlockPresenceType::Have },
313					},
314					None => ResponseType::Presence { cid, presence: BlockPresenceType::DontHave },
315				}
316			})
317			.collect();
318
319		self.handle.send_response(peer, response).await;
320	}
321
322	/// Handle an outbound bitswap command from the network service.
323	async fn handle_outbound_cmd(
324		&mut self,
325		peer: litep2p::PeerId,
326		wants: Vec<(Cid, WantType)>,
327		response_tx: ResponseSender,
328	) {
329		log::debug!(
330			target: LOG_TARGET,
331			"bitswap: outbound WANT for {} CIDs to {peer:?}",
332			wants.len(),
333		);
334		let cids: Vec<_> = wants.iter().map(|(cid, _)| *cid).collect();
335		self.pending.insert(peer, PendingBatch::new(cids, response_tx, Instant::now()));
336		self.handle.send_request(peer, wants).await;
337	}
338}

340fn inbound_wantlist_exceeds_limit(len: usize) -> bool {
341	len > MAX_WANTED_BLOCKS
342}

344/// Collapse a response list into at most one entry per CID, preferring `Block`
345/// over `Presence` when both arrive for the same CID.
346fn select_best_response_per_cid(responses: Vec<ResponseType>) -> HashMap<Cid, ResponseType> {
347	let mut best: HashMap<Cid, ResponseType> = HashMap::new();
348	for resp in responses {
349		let cid = match &resp {
350			ResponseType::Block { cid, .. } => *cid,
351			ResponseType::Presence { cid, .. } => *cid,
352		};
353		match best.entry(cid) {
354			std::collections::hash_map::Entry::Vacant(e) => {
355				e.insert(resp);
356			},
357			std::collections::hash_map::Entry::Occupied(mut e) => {
358				if matches!(resp, ResponseType::Block { .. }) &&
359					matches!(*e.get(), ResponseType::Presence { .. })
360				{
361					e.insert(resp);
362				}
363			},
364		}
365	}
366	best
367}

369/// Return the byte size of a response that counts toward the pending batch cap.
370fn response_retained_bytes(response: &ResponseType) -> usize {
371	match response {
372		ResponseType::Block { block, .. } => block.len(),
373		ResponseType::Presence { .. } => 0,
374	}
375}

377/// Encode litep2p [`ResponseType`] values into a [`BitswapProtoMessage`] byte vector.
378fn encode_responses_as_bitswap_message(responses: &[ResponseType]) -> Vec<u8> {
379	let mut msg = BitswapProtoMessage::default();
380
381	for resp in responses {
382		match resp {
383			ResponseType::Block { cid, block } => {
384				let prefix: Prefix = cid.into();
385				msg.payload
386					.push(MessageBlock { prefix: prefix.to_bytes(), data: block.clone() });
387			},
388			ResponseType::Presence { cid, presence } => {
389				msg.block_presences.push(BlockPresence {
390					cid: cid.to_bytes(),
391					r#type: match presence {
392						BlockPresenceType::Have => ProtoPresenceType::Have as i32,
393						BlockPresenceType::DontHave => ProtoPresenceType::DontHave as i32,
394					},
395				});
396			},
397		}
398	}
399
400	msg.encode_to_vec()
401}
