48type EventDetails<'a> = subxt::events::Event<'a, SrcChainConfig>;

50/// Outcome of decoding a single pallet-revive event.
51enum ReviveEvent {
52	Revert,
53	Log(Log),
54}
55
56/// Decode a single event detail into a [`ReviveEvent`], or `None` if it is not a pallet-revive
57/// event we care about.
58fn decode_revive_event(
59	event: &EventDetails<'_>,
60	block_number: U256,
61	transaction_hash: H256,
62	transaction_index: usize,
63	block_hash: H256,
64) -> Option<ReviveEvent> {
65	let pallet_name = event.pallet_name();
66	let event_name = event.event_name();
67
68	if EthExtrinsicRevert::is_event(pallet_name, event_name) {
69		return Some(ReviveEvent::Revert);
70	}
71	if ContractEmitted::is_event(pallet_name, event_name) {
72		match event.decode_fields_as::<ContractEmitted>() {
73			Some(Ok(evt)) => {
74				return Some(ReviveEvent::Log(Log {
75					address: evt.contract,
76					topics: evt.topics,
77					data: Some(evt.data.into()),
78					block_number,
79					transaction_hash,
80					transaction_index: transaction_index.into(),
81					block_hash,
82					log_index: event.index().into(),
83					..Default::default()
84				}));
85			},
86			Some(Err(err)) => log::warn!(
87				target: LOG_TARGET,
88				"Failed to decode ContractEmitted event {} in block {block_number} (tx {transaction_hash:?}): {err:?}, log dropped from receipt",
89				event.index()
90			),
91			// `is_event()` already confirmed the variant, so this is unreachable in practice.
92			None => {},
93		}
94	}
95	None
96}

98/// Iterate decoded block events and bucket revert flags and logs per extrinsic.
99/// Events for other extrinsics are skipped.
100///
101/// Events are stored sequentially without size markers, so a single
102/// undecodable event (e.g. from a runtime upgrade that shifted variant
103/// indices) corrupts the offset for all subsequent events.
104/// Decode errors are logged and skipped to avoid losing the entire receipt.
105///
106/// Returns `(reverted_extrinsics, logs_by_extrinsic)` keyed by extrinsic index.
107fn extract_revive_events(
108	block_events: &subxt::events::Events<SrcChainConfig>,
109	substrate_block_number: SubstrateBlockNumber,
110	eth_block_number: U256,
111	eth_block_hash: H256,
112	eth_tx_hash_for: impl Fn(usize) -> Option<H256>,
113) -> (HashSet<usize>, HashMap<usize, Vec<Log>>) {
114	let mut reverted_extrinsics: HashSet<usize> = HashSet::new();
115	let mut logs_by_extrinsic: HashMap<usize, Vec<Log>> = HashMap::new();
116
117	for (event_index, event_result) in block_events.iter().enumerate() {
118		let event = match event_result {
119			Ok(e) => e,
120			Err(err) => {
121				log::debug!(
122					target: LOG_TARGET,
123					"Failed to decode event {event_index} in block #{substrate_block_number}: {err:?}"
124				);
125				continue;
126			},
127		};
128
129		let extrinsic_index = match event.phase() {
130			Phase::ApplyExtrinsic(idx) => idx as usize,
131			_ => continue,
132		};
133
134		let Some(eth_tx_hash) = eth_tx_hash_for(extrinsic_index) else { continue };
135
136		match decode_revive_event(
137			&event,
138			eth_block_number,
139			eth_tx_hash,
140			extrinsic_index,
141			eth_block_hash,
142		) {
143			Some(ReviveEvent::Revert) => {
144				reverted_extrinsics.insert(extrinsic_index);
145			},
146			Some(ReviveEvent::Log(log)) => {
147				logs_by_extrinsic.entry(extrinsic_index).or_default().push(log);
148			},
149			None => {},
150		}
151	}
152
153	(reverted_extrinsics, logs_by_extrinsic)
154}

170/// Utility to extract receipts from extrinsics.
171#[derive(Clone)]
172pub struct ReceiptExtractor {
173	/// Fetch the receipt data info.
174	fetch_receipt_data: FetchReceiptDataFn,

177	fetch_eth_block_hash: FetchEthBlockHashFn,

179	/// Auto-discovered first EVM block on the chain.
180	/// Set once during backward sync when the first non-EVM block is encountered.
181	/// Uses `u64::MAX` as sentinel for "not yet discovered".

200	/// Create a new `ReceiptExtractor` with custom Ethereum address recovery logic.
201	///
202	/// Use `ReceiptExtractor::new` if the default Ethereum address recovery
203	/// logic ([`TransactionSigned::recover_eth_address`] based) is enough.
204	pub async fn new_with_custom_address_recovery(
205		runtime_api_provider: VersionAwareRuntimeApiProvider,
206		recover_eth_address_fn: RecoverEthAddressFn,
207	) -> Result<Self, ClientError> {
208		let provider = runtime_api_provider.clone();
209		let fetch_eth_block_hash = Arc::new(move |substrate_block_hash, substrate_block_number| {
210			let provider = provider.clone();
211
212			let fut = async move {
213				let runtime_api = provider
214					.at_block_hash_and_number(substrate_block_hash, substrate_block_number)
215					.await
216					.inspect_err(|err| {
217						log::debug!(
218							target: LOG_TARGET,
219							"Failed to access the runtime API at block #{substrate_block_number} \
220							({substrate_block_hash:?}) for an eth_block_hash query: {err:?}"
221						);
222					})
223					.ok()?;
224				runtime_api
225					.eth_block_hash(U256::from(substrate_block_number))?
226					.await
227					.inspect_err(|err| {
228						log::debug!(
229							target: LOG_TARGET,
230							"Failed to query eth_block_hash at block #{substrate_block_number} \
231							({substrate_block_hash:?}): {err:?}"
232						);
233					})
234					.ok()
235					.flatten()
236			};
237
238			Box::pin(fut) as Pin<Box<_>>
239		});
240
241		let provider = runtime_api_provider;
242		let fetch_receipt_data = Arc::new(move |block_hash| {
243			let provider = provider.clone();
244
245			let fut = async move {
246				let runtime_api = provider
247					.at(block_hash)
248					.await
249					.inspect_err(|err| {
250						log::debug!(
251							target: LOG_TARGET,
252							"Failed to access the runtime API at block {block_hash:?} for an \
253							eth_receipt_data query: {err:?}"
254						);
255					})
256					.ok()?;
257				runtime_api
258					.eth_receipt_data()?
259					.await
260					.inspect_err(|err| {
261						log::debug!(
262							target: LOG_TARGET,
263							"Failed to query eth_receipt_data at block {block_hash:?}: {err:?}"
264						);
265					})
266					.ok()
267			};
268
269			Box::pin(fut) as Pin<Box<_>>
270		});
271
272		Ok(Self {
273			fetch_receipt_data,
274			fetch_eth_block_hash,
275			first_evm_block: Arc::new(AtomicU64::new(u64::MAX)),
276			recover_eth_address: recover_eth_address_fn,
277		})
278	}

302	/// Check if the block is before the `first_evm_block` floor.
303	/// When sentinel (`u64::MAX`), no blocks are rejected (permissive default).
304	pub fn is_before_first_evm_block(&self, block_number: SubstrateBlockNumber) -> bool {
305		let val = self.first_evm_block.load(Ordering::Acquire);
306		val != u64::MAX && block_number < val
307	}
308
309	/// Set the first EVM block. Only stores if lower than the current value.
310	pub fn set_first_evm_block(&self, block_number: SubstrateBlockNumber) {
311		let prev = self.first_evm_block.fetch_min(block_number, Ordering::AcqRel);
312		if block_number > prev {
313			log::debug!(target: LOG_TARGET,
314				"Ignored attempt to raise first_evm_block to #{block_number}, current is #{prev}");
315		}
316	}
317
318	/// The auto-discovered first EVM block, or `None` if not yet discovered.
319	pub fn first_evm_block(&self) -> Option<SubstrateBlockNumber> {
320		let val = self.first_evm_block.load(Ordering::Acquire);
321		(val != u64::MAX).then_some(val)
322	}

324	/// Resolve the Ethereum block hash for a substrate block, falling back to the substrate hash.
325	async fn resolve_eth_block_hash(
326		&self,
327		substrate_block_hash: H256,
328		substrate_block_number: SubstrateBlockNumber,
329	) -> H256 {
330		match (self.fetch_eth_block_hash)(substrate_block_hash, substrate_block_number).await {
331			Some(hash) => hash,
332			None => {
333				log::trace!(target: LOG_TARGET,
334					"eth_block_hash returned None for substrate block \
335					 #{substrate_block_number} ({substrate_block_hash:?}), \
336					 falling back to substrate hash as ETH hash");
337				substrate_block_hash
338			},
339		}
340	}

342	/// Decode the raw call payload into a [`TransactionSigned`] and construct its [`ReceiptInfo`].
343	fn decode_transaction_and_build_receipt(
344		&self,
345		eth_block_hash: H256,
346		block_number: U256,
347		call: EthTransact,
348		transaction_hash: H256,
349		transaction_index: usize,
350		receipt_gas_info: ReceiptGasInfoV1,
351		reverted: bool,
352		logs: Vec<Log>,
353	) -> Result<(TransactionSigned, ReceiptInfo), ClientError> {
354		let signed_tx =
355			TransactionSigned::decode(&call.payload).map_err(|_| ClientError::TxDecodingFailed)?;
356		let from = (self.recover_eth_address)(&signed_tx).map_err(|_| {
357			log::error!(target: LOG_TARGET, "Failed to recover eth address from signed tx");
358			ClientError::RecoverEthAddressFailed
359		})?;
360
361		let tx_info = GenericTransaction::from_signed(
362			signed_tx.clone(),
363			receipt_gas_info.effective_gas_price,
364			Some(from),
365		);
366
367		let contract_address = if tx_info.to.is_none() {
368			Some(create1(
369				&from,
370				tx_info
371					.nonce
372					.unwrap_or_default()
373					.try_into()
374					.map_err(|_| ClientError::ConversionFailed)?,
375			))
376		} else {
377			None
378		};
379
380		let receipt = ReceiptInfo::new(
381			eth_block_hash,
382			block_number,
383			contract_address,
384			from,
385			logs,
386			tx_info.to,
387			receipt_gas_info.effective_gas_price,
388			U256::from(receipt_gas_info.gas_used),
389			!reverted,
390			transaction_hash,
391			transaction_index.into(),
392			tx_info.r#type.unwrap_or_default(),
393		);
394		Ok((signed_tx, receipt))
395	}

397	/// Extract receipts from block.
398	pub async fn extract_from_block(
399		&self,
400		block: &SubstrateBlock,
401	) -> Result<Vec<(TransactionSigned, ReceiptInfo)>, ClientError> {
402		let eth_block_hash =
403			self.resolve_eth_block_hash(block.block_hash(), block.block_number()).await;
404
405		self.extract_from_block_with_eth_hash(block, eth_block_hash).await
406	}

408	/// Extract receipts from block, using a pre-fetched ethereum block hash.
409	///
410	/// Fetches block events once in a single pass before building receipts.
411	pub async fn extract_from_block_with_eth_hash(
412		&self,
413		block: &SubstrateBlock,
414		eth_block_hash: H256,
415	) -> Result<Vec<(TransactionSigned, ReceiptInfo)>, ClientError> {
416		if self.is_before_first_evm_block(block.block_number()) {
417			return Ok(vec![]);
418		}
419
420		let eth_tx_by_index: BTreeMap<usize, (EthTransact, H256, ReceiptGasInfoV1)> = self
421			.get_block_extrinsics(block)
422			.await?
423			.map(|(call, receipt_gas_info, extrinsic_index)| {
424				let hash = H256(keccak_256(&call.payload));
425				(extrinsic_index, (call, hash, receipt_gas_info))
426			})
427			.collect();
428
429		if eth_tx_by_index.is_empty() {
430			return Ok(vec![]);
431		}
432
433		let substrate_block_number = block.block_number();
434		let eth_block_number: U256 = substrate_block_number.into();
435		let block_events = block.events().fetch().await.inspect_err(|err| {
436			log::debug!(target: LOG_TARGET, "Error fetching events for block #{substrate_block_number}: {err:?}");
437		})?;
438		let (reverted_extrinsics, mut logs_by_extrinsic) = extract_revive_events(
439			&block_events,
440			substrate_block_number,
441			eth_block_number,
442			eth_block_hash,
443			|idx| eth_tx_by_index.get(&idx).map(|(_, hash, _)| *hash),
444		);
445
446		eth_tx_by_index
447			.into_iter()
448			.map(|(transaction_index, (call, transaction_hash, receipt_gas_info))| {
449				let reverted = reverted_extrinsics.contains(&transaction_index);
450				let logs = logs_by_extrinsic.remove(&transaction_index).unwrap_or_default();
451				self.decode_transaction_and_build_receipt(
452					eth_block_hash,
453					eth_block_number,
454					call,
455					transaction_hash,
456					transaction_index,
457					receipt_gas_info,
458					reverted,
459					logs,
460				)
461				.inspect_err(|err| {
462					log::warn!(target: LOG_TARGET, "Error extracting extrinsic: {err:?}");
463				})
464			})
465			.collect()
466	}

468	/// Return the ETH extrinsics of the block grouped with reconstruction receipt info and
469	/// extrinsic index
470	async fn get_block_extrinsics(
471		&self,
472		block: &SubstrateBlock,
473	) -> Result<impl Iterator<Item = (EthTransact, ReceiptGasInfoV1, usize)>, ClientError> {
474		// Filter extrinsics from pallet_revive
475		let extrinsics = block.extrinsics().fetch().await.inspect_err(|err| {
476			log::debug!(target: LOG_TARGET, "Error fetching for #{:?} extrinsics: {err:?}", block.block_number());
477		})?;
478
479		let receipt_data =
480			(self.fetch_receipt_data)(block.block_hash()).await.ok_or_else(|| {
481				log::trace!(target: LOG_TARGET,
482				"Receipt data not found for block #{} ({:?})",
483				block.block_number(), block.block_hash());
484				ClientError::ReceiptDataNotFound
485			})?;
486		let block_number = block.block_number();
487		let extrinsics: Vec<_> = extrinsics
488			.iter()
489			.enumerate()
490			.flat_map(|(ext_idx, ext)| {
491				let ext = ext.ok()?;
492				match ext.decode_call_data_fields_as::<EthTransact>()? {
493					Ok(call) => Some((call, ext_idx)),
494					Err(err) => {
495						log::warn!(
496							target: LOG_TARGET,
497							"Failed to decode EthTransact call in extrinsic {ext_idx} of block #{block_number}: {err:?}, transaction dropped from receipts"
498						);
499						None
500					},
501				}
502			})
503			.collect();
504
505		// Sanity check we received enough data from the pallet revive.
506		if receipt_data.len() != extrinsics.len() {
507			log::error!(
508				target: LOG_TARGET,
509				"Receipt data length ({}) does not match extrinsics length ({})",
510				receipt_data.len(),
511				extrinsics.len()
512			);
513			Err(ClientError::ReceiptDataLengthMismatch)
514		} else {
515			Ok(extrinsics
516				.into_iter()
517				.zip(receipt_data)
518				.map(|((call, ext_idx), rec)| (call, rec, ext_idx)))
519		}
520	}

522	/// Extract a [`TransactionSigned`] and a [`ReceiptInfo`] for a specific transaction in a
523	/// [`SubstrateBlock`]
524	pub async fn extract_from_transaction(
525		&self,
526		block: &SubstrateBlock,
527		transaction_index: usize,
528	) -> Result<(TransactionSigned, ReceiptInfo), ClientError> {
529		let (eth_call, receipt_gas_info, transaction_hash) = self
530			.get_block_extrinsics(block)
531			.await?
532			.find_map(|(call, receipt_gas_info, extrinsic_index)| {
533				(extrinsic_index == transaction_index).then(|| {
534					let hash = H256(keccak_256(&call.payload));
535					(call, receipt_gas_info, hash)
536				})
537			})
538			.ok_or_else(|| {
539				log::trace!(target: LOG_TARGET,
540					"extract_from_transaction: no EVM extrinsic at tx_index {transaction_index} \
541					 in block #{} ({:?})", block.block_number(), block.block_hash());
542				ClientError::EthExtrinsicNotFound
543			})?;
544
545		let substrate_block_number = block.block_number();
546		let eth_block_number: U256 = substrate_block_number.into();
547		let eth_block_hash =
548			self.resolve_eth_block_hash(block.block_hash(), substrate_block_number).await;
549		let block_events = block.events().fetch().await.inspect_err(|err| {
550			log::debug!(target: LOG_TARGET, "Error fetching events for block #{substrate_block_number}: {err:?}");
551		})?;
552		let (reverted_extrinsics, mut logs_by_extrinsic) = extract_revive_events(
553			&block_events,
554			substrate_block_number,
555			eth_block_number,
556			eth_block_hash,
557			|idx| (idx == transaction_index).then_some(transaction_hash),
558		);
559
560		let reverted = reverted_extrinsics.contains(&transaction_index);
561		let logs = logs_by_extrinsic.remove(&transaction_index).unwrap_or_default();
562		self.decode_transaction_and_build_receipt(
563			eth_block_hash,
564			eth_block_number,
565			eth_call,
566			transaction_hash,
567			transaction_index,
568			receipt_gas_info,
569			reverted,
570			logs,
571		)
572	}

756	use codec::{Compact, Decode, Encode};
757	use frame_system::EventRecord;
758	use revive_dev_runtime::{Runtime, RuntimeEvent};
759	use subxt::{
760		PolkadotConfig,
761		client::OfflineClient,
762		config::{polkadot::PolkadotConfigBuilder, substrate::SpecVersionForRange},
763		events::Events,
764		metadata::Metadata,
765	};
766
767	/// Build `Events` by SCALE-encoding revive events against the generated runtime metadata.
768	struct EventsBuilder {
769		metadata: Metadata,
770		bytes: Vec<u8>,
771		count: u32,
772	}
773
774	impl EventsBuilder {
775		fn new() -> Self {
776			let metadata_bytes: &[u8] =
777				include_bytes!(concat!(env!("OUT_DIR"), "/revive_chain.scale"));
778			let metadata = Metadata::decode(&mut &metadata_bytes[..]).unwrap();
779			Self { metadata, bytes: Vec::new(), count: 0 }
780		}
781
782		fn push_event(
783			mut self,
784			phase: frame_system::Phase,
785			event: pallet_revive::Event<Runtime>,
786		) -> Self {
787			EventRecord::<RuntimeEvent, H256> {
788				phase,
789				event: RuntimeEvent::Revive(event),
790				topics: vec![],
791			}
792			.encode_to(&mut self.bytes);
793			self.count += 1;
794			self
795		}
796
797		fn build(self) -> Events<SrcChainConfig> {
798			let mut encoded_events = Vec::new();
799			Compact(self.count).encode_to(&mut encoded_events);
800			encoded_events.extend(self.bytes);
801
802			let config = PolkadotConfigBuilder::new()
803				.set_metadata_for_spec_versions(std::iter::once((0u32, self.metadata.into())))
804				.set_spec_version_for_block_ranges(std::iter::once(SpecVersionForRange {
805					block_range: 0..u64::MAX,
806					spec_version: 0,
807					transaction_version: 0,
808				}))
809				.build();
810			let client = OfflineClient::<PolkadotConfig>::new_with_config(config);
811			let at_block =
812				client.at_block(0u64).expect("spec version range covers all block numbers; qed");
813			at_block.events().from_bytes(encoded_events)
814		}
815	}

890		let tx1 = H256::from([0x02; 32]);
891		let tx2 = H256::from([0x03; 32]);
892		let emitted_by = |contract: H160| pallet_revive::Event::ContractEmitted {
893			contract,
894			data: vec![],
895			topics: vec![],
896		};

898			.push_event(frame_system::Phase::ApplyExtrinsic(0), emitted_by(H160::from([0xaa; 20])))
899			.push_event(frame_system::Phase::ApplyExtrinsic(0), emitted_by(H160::from([0xbb; 20])))
900			.push_event(
901				frame_system::Phase::ApplyExtrinsic(1),
902				pallet_revive::Event::EthExtrinsicRevert {
903					dispatch_error: sp_runtime::DispatchError::Other("tx-1 revert"),
904				},
905			)
906			.push_event(frame_system::Phase::ApplyExtrinsic(2), emitted_by(H160::from([0xcc; 20])))
907			.build();
908
909		let (reverts, logs) =
910			extract_revive_events(&events, 0, U256::zero(), H256::zero(), |idx| match idx {

920		// log_index is block-wide

922		assert_eq!(logs[&0][1].log_index, U256::from(1));
