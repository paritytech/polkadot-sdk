258	/// Wait for a given relay chain block in an async way.
259	///
260	/// The caller needs to pass the hash of a block it waits for and the function will return when
261	/// the block is available or an error occurred.
262	///
263	/// The waiting for the block is implemented as follows:
264	///
265	/// 1. Get a read lock on the import lock from the backend.
266	///
267	/// 2. Check if the block is already imported. If yes, return from the function.
268	///
269	/// 3. If the block isn't imported yet, add an import notification listener.
270	///
271	/// 4. Poll the import notification listener until the block is imported or the timeout is
272	/// fired.
273	///
274	/// The timeout is set to 6 seconds. This should be enough time to import the block in the
275	/// current round and if not, the new round of the relay chain already started anyway.
276	async fn wait_for_block(&self, hash: PHash) -> RelayChainResult<()> {
277		let mut listener =
278			match check_block_in_chain(self.backend.clone(), self.full_client.clone(), hash)? {
279				BlockCheckStatus::InChain => return Ok(()),
280				BlockCheckStatus::Unknown(listener) => listener,
281			};
282
283		let mut timeout = futures_timer::Delay::new(Duration::from_secs(TIMEOUT_IN_SECONDS)).fuse();
284
285		loop {
286			futures::select! {
287				_ = timeout => return Err(RelayChainError::WaitTimeout(hash)),
288				evt = listener.next() => match evt {
289					Some(evt) if evt.hash == hash => return Ok(()),
290					// Not the event we waited on.
291					Some(_) => continue,
292					None => return Err(RelayChainError::ImportListenerClosed(hash)),
293				}
294			}
295		}
296	}

361pub enum BlockCheckStatus {
362	/// Block is in chain
363	InChain,
364	/// Block status is unknown, listener can be used to wait for notification
365	Unknown(ImportNotifications<PBlock>),
366}
367
368// Helper function to check if a block is in chain.
369pub fn check_block_in_chain(
370	backend: Arc<FullBackend>,
371	client: Arc<FullClient>,
372	hash: PHash,
373) -> RelayChainResult<BlockCheckStatus> {
374	let _lock = backend.get_import_lock().read();
375
376	if backend.blockchain().status(hash)? == BlockStatus::InChain {
377		return Ok(BlockCheckStatus::InChain);
378	}
379
380	let listener = client.import_notification_stream();
381
382	Ok(BlockCheckStatus::Unknown(listener))
383}

385/// Build Polkadot full node with parachain bootnode request-response protocol.
386fn build_polkadot_with_paranode_protocol<Network>(
387	config: Configuration,
388	params: NewFullParams<CollatorOverseerGen>,
389) -> Result<(NewFull, async_channel::Receiver<IncomingRequest>), polkadot_service::Error>
390where
391	Network: NetworkBackend<PBlock, PHash>,
392{
393	let fork_id = config.chain_spec.fork_id().map(ToString::to_string);
394	let mut polkadot_builder = PolkadotServiceBuilder::<_, Network>::new(config, params)?;
395	let (config, request_receiver) = bootnode_request_response_config::<_, _, Network>(
396		polkadot_builder.genesis_hash(),
397		fork_id.as_deref(),
398	);
399	polkadot_builder.add_extra_request_response_protocol(config);
400
401	Ok((polkadot_builder.build()?, request_receiver))
402}

404/// Build the Polkadot full node using the given `config`.
405#[sc_tracing::logging::prefix_logs_with("Relaychain")]
406fn build_polkadot_full_node(
407	config: Configuration,
408	parachain_config: &Configuration,
409	telemetry_worker_handle: Option<TelemetryWorkerHandle>,
410	hwbench: Option<sc_sysinfo::HwBench>,
411) -> Result<
412	(NewFull, Option<CollatorPair>, async_channel::Receiver<IncomingRequest>),
413	polkadot_service::Error,
414> {
415	let (is_parachain_node, maybe_collator_key) = if parachain_config.role.is_authority() {
416		let collator_key = CollatorPair::generate().0;
417		(polkadot_service::IsParachainNode::Collator(collator_key.clone()), Some(collator_key))
418	} else {
419		(polkadot_service::IsParachainNode::FullNode, None)
420	};
421
422	let new_full_params = polkadot_service::NewFullParams {
423		is_parachain_node,
424		// Disable BEEFY. It should not be required by the internal relay chain node.
425		enable_beefy: false,
426		force_authoring_backoff: false,
427		telemetry_worker_handle,
428
429		// Cumulus doesn't spawn PVF workers, so we can disable version checks.
430		node_version: None,
431		secure_validator_mode: false,
432		workers_path: None,
433		workers_names: None,
434
435		overseer_gen: CollatorOverseerGen,
436		overseer_message_channel_capacity_override: None,
437		malus_finality_delay: None,
438		hwbench,
439		execute_workers_max_num: None,
440		prepare_workers_hard_max_num: None,
441		prepare_workers_soft_max_num: None,
442		keep_finalized_for: None,
443		invulnerable_ah_collators: HashSet::new(),
444		collator_protocol_hold_off: None,
445		experimental_collator_protocol: false,
446		collator_reputation_persist_interval: None,
447	};
448
449	let (relay_chain_full_node, paranode_req_receiver) = match config.network.network_backend {
450		NetworkBackendType::Libp2p => build_polkadot_with_paranode_protocol::<
451			sc_network::NetworkWorker<_, _>,
452		>(config, new_full_params)?,
453		NetworkBackendType::Litep2p => build_polkadot_with_paranode_protocol::<
454			sc_network::Litep2pNetworkBackend,
455		>(config, new_full_params)?,
456	};
457
458	Ok((relay_chain_full_node, maybe_collator_key, paranode_req_receiver))
459}

461/// Builds a relay chain interface by constructing a full relay chain node
462pub fn build_inprocess_relay_chain(
463	mut polkadot_config: Configuration,
464	parachain_config: &Configuration,
465	telemetry_worker_handle: Option<TelemetryWorkerHandle>,
466	task_manager: &mut TaskManager,
467	hwbench: Option<sc_sysinfo::HwBench>,
468) -> RelayChainResult<(
469	Arc<dyn RelayChainInterface + 'static>,
470	Option<CollatorPair>,
471	Arc<dyn NetworkService>,
472	async_channel::Receiver<IncomingRequest>,
473)> {
474	// This is essentially a hack, but we want to ensure that we send the correct node version
475	// to the telemetry.
476	polkadot_config.impl_version = polkadot_cli::Cli::impl_version();
477	polkadot_config.impl_name = polkadot_cli::Cli::impl_name();
478
479	let (full_node, collator_key, paranode_req_receiver) = build_polkadot_full_node(
480		polkadot_config,
481		parachain_config,
482		telemetry_worker_handle,
483		hwbench,
484	)
485	.map_err(|e| RelayChainError::Application(Box::new(e) as Box<_>))?;
486
487	let relay_chain_interface = Arc::new(RelayChainInProcessInterface::new(
488		full_node.client,
489		full_node.backend,
490		full_node.sync_service,
491		full_node.overseer_handle.clone().ok_or(RelayChainError::GenericError(
492			"Overseer not running in full node.".to_string(),
493		))?,
494	));
495
496	task_manager.add_child(full_node.task_manager);
497
498	Ok((relay_chain_interface, collator_key, full_node.network, paranode_req_receiver))
499}
