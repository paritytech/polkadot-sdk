40	/// Benchmarks the time it takes to read a single Storage item.
41	/// Uses the latest state that is available for the given client.
42	pub(crate) fn bench_read<B, BA, C>(
43		&self,
44		client: Arc<C>,
45		_shared_trie_cache: Option<sp_trie::cache::SharedTrieCache<HashingFor<B>>>,
46	) -> Result<BenchRecord>
47	where
48		C: UsageProvider<B> + StorageProvider<B, BA> + CallApiAt<B>,
49		B: BlockT + Debug,
50		BA: ClientBackend<B>,
51		<<B as BlockT>::Header as HeaderT>::Number: From<u32>,
52	{
53		if self.params.is_validate_block_mode() && self.params.disable_pov_recorder {
54			return Err("PoV recorder must be activated to provide a storage proof for block validation at runtime. Remove `--disable-pov-recorder` from the command line.".into());
55		}
56		if self.params.is_validate_block_mode() &&
57			self.params.batch_size > MAX_BATCH_SIZE_FOR_BLOCK_VALIDATION
58		{
59			return Err(format!("Batch size is too large. This may cause problems with runtime memory allocation. Better set `--batch-size {}` or less.", MAX_BATCH_SIZE_FOR_BLOCK_VALIDATION).into());
60		}
61
62		let mut record = BenchRecord::default();
63		let best_hash = client.usage_info().chain.best_hash;
64
65		info!("Preparing keys from block {}", best_hash);
66		let (keys, mut rng) = select_entries(
67			self.params.keys_limit,
68			self.params.random_seed,
69			|first_key_ref| {
70				let fk = first_key_ref.map(|b| sp_storage::StorageKey(b.to_vec()));
71				Ok(client.storage_keys(best_hash, None, fk.as_ref())?)
72			},
73			|| Ok(client.storage_keys(best_hash, None, None)?),
74			|k: &sp_storage::StorageKey| k.0.as_slice(),
75		)?;
76
77		let mut child_nodes = Vec::new();
78		// Interesting part here:
79		// Read all the keys in the database and measure the time it takes to access each.
80		info!("Reading {} keys", keys.len());
81
82		// Read using the same TrieBackend and recorder for up to `batch_size` keys.
83		// This would allow us to measure the amortized cost of reading a key.
84		let state = client
85			.state_at(best_hash)
86			.map_err(|_err| Error::Input("State not found".into()))?;
87		// We reassign the backend and recorder for every batch size.
88		// Using a new recorder for every read vs using the same for the entire batch
89		// produces significant different results. Since in the real use case we use a
90		// single recorder per block, simulate the same behavior by creating a new
91		// recorder every batch size, so that the amortized cost of reading a key is
92		// measured in conditions closer to the real world.
93		let (mut backend, mut recorder) = self.create_backend::<B, C>(&state);
94
95		let mut read_in_batch = 0;
96		let mut on_validation_batch = vec![];
97		let mut on_validation_size = 0;
98
99		let last_key = keys.last().expect("Checked above to be non-empty");
100		for key in keys.as_slice() {
101			match (self.params.include_child_trees, self.is_child_key(key.clone().0)) {
102				(true, Some(info)) => {
103					// child tree key: sample with select_entries when child_keys_limit is set
104					match select_entries(
105						self.params.child_keys_limit,
106						self.params.random_seed,
107						|first_key_ref| {
108							let fk = first_key_ref.map(|b| sp_storage::StorageKey(b.to_vec()));
109							Ok(client
110								.child_storage_keys(best_hash, info.clone(), None, fk.as_ref())?
111								.map(|ck| (ck, info.clone())))
112						},
113						|| {
114							Ok(client
115								.child_storage_keys(best_hash, info.clone(), None, None)?
116								.map(|ck| (ck, info.clone())))
117						},
118						|(k, _): &(sp_storage::StorageKey, ChildInfo)| k.0.as_slice(),
119					) {
120						Ok((entries, _)) => child_nodes.extend(entries),
121						Err(SelectEntriesEmptyStorage::Input(_)) => {},
122						Err(e) => return Err(e),
123					}
124				},
125				_ => {
126					// regular key
127					on_validation_batch.push((key.0.clone(), None));
128					let start = Instant::now();
129					let v = backend
130						.storage(key.0.as_ref())
131						.expect("Checked above to exist")
132						.ok_or("Value unexpectedly empty")?;
133					on_validation_size += v.len();
134					if self.params.is_import_block_mode() {
135						record.append(v.len(), start.elapsed())?;
136					}
137				},
138			}
139			read_in_batch += 1;
140			let is_batch_full = read_in_batch >= self.params.batch_size || key == last_key;
141
142			// Read keys on block validation
143			if is_batch_full && self.params.is_validate_block_mode() {
144				let root = backend.root();
145				let storage_proof = recorder
146					.clone()
147					.map(|r| r.drain_storage_proof())
148					.expect("Storage proof must exist for block validation");
149				let elapsed = measure_block_validation::<B>(
150					*root,
151					storage_proof,
152					on_validation_batch.clone(),
153					self.params.validate_block_rounds,
154				);
155				record.append(on_validation_size / on_validation_batch.len(), elapsed)?;
156
157				on_validation_batch = vec![];
158				on_validation_size = 0;
159			}
160
161			// Reload recorder
162			if is_batch_full {
163				(backend, recorder) = self.create_backend::<B, C>(&state);
164				read_in_batch = 0;
165			}
166		}
167
168		if self.params.include_child_trees && !child_nodes.is_empty() {
169			child_nodes.shuffle(&mut rng);
170
171			info!("Reading {} child keys", child_nodes.len());
172			let (last_child_key, last_child_info) =
173				child_nodes.last().expect("Checked above to be non-empty");
174			for (key, info) in child_nodes.as_slice() {
175				on_validation_batch.push((key.0.clone(), Some(info.clone())));
176				let start = Instant::now();
177				let v = backend
178					.child_storage(info, key.0.as_ref())
179					.expect("Checked above to exist")
180					.ok_or("Value unexpectedly empty")?;
181				on_validation_size += v.len();
182				if self.params.is_import_block_mode() {
183					record.append(v.len(), start.elapsed())?;
184				}
185				read_in_batch += 1;
186				let is_batch_full = read_in_batch >= self.params.batch_size ||
187					(last_child_key == key && last_child_info == info);
188
189				// Read child keys on block validation
190				if is_batch_full && self.params.is_validate_block_mode() {
191					let root = backend.root();
192					let storage_proof = recorder
193						.clone()
194						.map(|r| r.drain_storage_proof())
195						.expect("Storage proof must exist for block validation");
196					let elapsed = measure_block_validation::<B>(
197						*root,
198						storage_proof,
199						on_validation_batch.clone(),
200						self.params.validate_block_rounds,
201					);
202					record.append(on_validation_size / on_validation_batch.len(), elapsed)?;
203
204					on_validation_batch = vec![];
205					on_validation_size = 0;
206				}
207
208				// Reload recorder
209				if is_batch_full {
210					(backend, recorder) = self.create_backend::<B, C>(&state);
211					read_in_batch = 0;
212				}
213			}
214		}
215
216		Ok(record)
217	}
218
219	fn create_backend<'a, B, C>(
220		&self,
221		state: &'a C::StateBackend,
222	) -> (
223		sp_state_machine::TrieBackend<
224			&'a <C::StateBackend as AsTrieBackend<HashingFor<B>>>::TrieBackendStorage,
225			HashingFor<B>,
226			&'a sp_trie::cache::LocalTrieCache<HashingFor<B>>,
227		>,
228		Option<sp_trie::recorder::Recorder<HashingFor<B>>>,
229	)
230	where
231		C: CallApiAt<B>,
232		B: BlockT + Debug,
233	{
234		let recorder = (!self.params.disable_pov_recorder).then(|| Default::default());
235		let backend = sp_state_machine::TrieBackendBuilder::wrap(state.as_trie_backend())
236			.with_optional_recorder(recorder.clone())
237			.build();
238
239		(backend, recorder)
240	}
241}
242
243fn measure_block_validation<B: BlockT + Debug>(
244	root: B::Hash,
245	storage_proof: StorageProof,
246	on_validation_batch: Vec<(Vec<u8>, Option<ChildInfo>)>,
247	rounds: u32,
248) -> std::time::Duration {
249	debug!(
250		"POV: len {:?} {:?}",
251		storage_proof.len(),
252		storage_proof.clone().encoded_compact_size::<HashingFor<B>>(root)
253	);
254	let batch_size = on_validation_batch.len();
255	let wasm_module = get_wasm_module();
256	let mut instance = wasm_module
257		.new_instance(sc_executor_common::wasm_runtime::DEFAULT_HEAP_ALLOC_STRATEGY)
258		.expect("Failed to create wasm instance");
259	let params = StorageAccessParams::<B>::new_read(root, storage_proof, on_validation_batch);
260	let dry_run_encoded = params.as_dry_run().encode();
261	let encoded = params.encode();
262
263	let mut durations_in_nanos = Vec::new();
264
265	for i in 1..=rounds {
266		info!("validate_block with {} keys, round {}/{}", batch_size, i, rounds);
267
268		// Dry run to get the time it takes without storage access
269		let dry_run_start = Instant::now();
270		instance
271			.call_export("validate_block", &dry_run_encoded)
272			.expect("Failed to call validate_block");
273		let dry_run_elapsed = dry_run_start.elapsed();
274		debug!("validate_block dry-run time {:?}", dry_run_elapsed);
275
276		let start = Instant::now();
277		instance
278			.call_export("validate_block", &encoded)
279			.expect("Failed to call validate_block");
280		let elapsed = start.elapsed();
281		debug!("validate_block time {:?}", elapsed);
282
283		durations_in_nanos
284			.push(elapsed.saturating_sub(dry_run_elapsed).as_nanos() as u64 / batch_size as u64);
285	}
286
287	std::time::Duration::from_nanos(
288		durations_in_nanos.iter().sum::<u64>() / durations_in_nanos.len() as u64,
289	)
