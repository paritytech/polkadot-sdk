46	/// Benchmarks the time it takes to write a single Storage item.
47	///
48	/// Uses the latest state that is available for the given client.
49	///
50	/// Unlike reading benchmark, where we read every single key, here we write a batch of keys in
51	/// one time. So writing a remaining keys with the size much smaller than batch size can
52	/// dramatically distort the results. To avoid this, we skip the remaining keys.
53	pub(crate) fn bench_write<Block, BA, H, C>(
54		&self,
55		client: Arc<C>,
56		(db, state_col): (Arc<dyn sp_database::Database<DbHash>>, ColumnId),
57		storage: Arc<dyn sp_state_machine::Storage<HashingFor<Block>>>,
58		shared_trie_cache: Option<sp_trie::cache::SharedTrieCache<HashingFor<Block>>>,
59	) -> Result<BenchRecord>
60	where
61		Block: BlockT<Header = H, Hash = DbHash> + Debug,
62		H: HeaderT<Hash = DbHash>,
63		BA: ClientBackend<Block>,
64		C: UsageProvider<Block> + HeaderBackend<Block> + StorageProvider<Block, BA>,
65	{
66		if self.params.is_validate_block_mode() && self.params.disable_pov_recorder {
67			return Err("PoV recorder must be activated to provide a storage proof for block validation at runtime. Remove `--disable-pov-recorder`.".into());
68		}
69		if self.params.is_validate_block_mode() &&
70			self.params.batch_size > MAX_BATCH_SIZE_FOR_BLOCK_VALIDATION
71		{
72			return Err(format!("Batch size is too large. This may cause problems with runtime memory allocation. Better set `--batch-size {}` or less.", MAX_BATCH_SIZE_FOR_BLOCK_VALIDATION).into());
73		}
74
75		// Store the time that it took to write each value.
76		let mut record = BenchRecord::default();
77
78		let best_hash = client.usage_info().chain.best_hash;
79		let header = client.header(best_hash)?.ok_or("Header not found")?;
80		let original_root = *header.state_root();
81
82		let (trie, _) = self.create_trie_backend::<Block, H>(
83			original_root,
84			&storage,
85			shared_trie_cache.as_ref(),
86		);
87
88		info!("Preparing keys from block {}", best_hash);
89		let (kvs, mut rng) = select_entries(
90			self.params.keys_limit,
91			self.params.random_seed,
92			|first_key_ref| {
93				let mut iter_args = sp_state_machine::IterArgs::default();
94				iter_args.start_at = first_key_ref;
95				Ok(trie.pairs(iter_args)?)
96			},
97			|| Ok(trie.pairs(Default::default())?),
98			|r| r.as_ref().map(|(k, _)| k.as_slice()).unwrap_or(&[]),
99		)?;
100
101		info!("Writing {} keys in batches of {}", kvs.len(), self.params.batch_size);
102		let remainder = kvs.len() % self.params.batch_size;
103		if self.params.is_validate_block_mode() && remainder != 0 {
104			info!("Remaining `{remainder}` keys will be skipped");
105		}
106
107		let mut child_nodes = Vec::new();
108		let mut batched_keys = Vec::new();
109		// Generate all random values first; Make sure there are no collisions with existing
110		// db entries, so we can rollback all additions without corrupting existing entries.
111
112		for key_value in kvs {
113			let (k, original_v) = key_value?;
114			match (self.params.include_child_trees, self.is_child_key(k.to_vec())) {
115				(true, Some(info)) => {
116					match select_entries(
117						self.params.child_keys_limit,
118						self.params.random_seed,
119						|first_key_ref| {
120							let fk = first_key_ref.map(|b| sp_storage::StorageKey(b.to_vec()));
121							Ok(client
122								.child_storage_keys(best_hash, info.clone(), None, fk.as_ref())?
123								.map(|ck| (ck, info.clone())))
124						},
125						|| {
126							Ok(client
127								.child_storage_keys(best_hash, info.clone(), None, None)?
128								.map(|ck| (ck, info.clone())))
129						},
130						|(key, _): &(sp_storage::StorageKey, ChildInfo)| key.0.as_slice(),
131					) {
132						Ok((entries, _)) => child_nodes.push((
133							entries.into_iter().map(|(ck, _)| ck).collect::<Vec<_>>(),
134							info.clone(),
135						)),
136						Err(SelectEntriesEmptyStorage::Input(_)) => {},
137						Err(e) => return Err(e),
138					}
139				},
140				_ => {
141					// regular key
142					let mut new_v = vec![0; original_v.len()];
143
144					loop {
145						// Create a random value to overwrite with.
146						// NOTE: We use a possibly higher entropy than the original value,
147						// could be improved but acts as an over-estimation which is fine for now.
148						rng.fill_bytes(&mut new_v[..]);
149						if check_new_value::<Block>(
150							db.clone(),
151							&trie,
152							&k.to_vec(),
153							&new_v,
154							self.state_version(),
155							state_col,
156							None,
157						) {
158							break;
159						}
160					}
161
162					batched_keys.push((k.to_vec(), new_v.to_vec()));
163					if batched_keys.len() < self.params.batch_size {
164						continue;
165					}
166
167					// Write each value in one commit.
168					let (size, duration) = if self.params.is_validate_block_mode() {
169						self.measure_per_key_amortised_validate_block_write_cost::<Block, H>(
170							original_root,
171							&storage,
172							shared_trie_cache.as_ref(),
173							batched_keys.clone(),
174							None,
175						)?
176					} else {
177						self.measure_per_key_amortised_import_block_write_cost::<Block, H>(
178							original_root,
179							&storage,
180							shared_trie_cache.as_ref(),
181							db.clone(),
182							batched_keys.clone(),
183							self.state_version(),
184							state_col,
185							None,
186						)?
187					};
188					record.append(size, duration)?;
189					batched_keys.clear();
190				},
191			}
192		}
193
194		if self.params.include_child_trees && !child_nodes.is_empty() {
195			info!("Writing {} child keys", child_nodes.iter().map(|(c, _)| c.len()).sum::<usize>());
196			for (mut child_keys, info) in child_nodes {
197				if child_keys.len() < self.params.batch_size {
198					warn!(
199						"{} child keys will be skipped because it's less than batch size",
200						child_keys.len()
201					);
202					continue;
203				}
204
205				child_keys.shuffle(&mut rng);
206
207				for key in child_keys {
208					if let Some(original_v) = client
209						.child_storage(best_hash, &info, &key)
210						.expect("Checked above to exist")
211					{
212						let mut new_v = vec![0; original_v.0.len()];
213
214						loop {
215							rng.fill_bytes(&mut new_v[..]);
216							if check_new_value::<Block>(
217								db.clone(),
218								&trie,
219								&key.0,
220								&new_v,
221								self.state_version(),
222								state_col,
223								Some(&info),
224							) {
225								break;
226							}
227						}
228						batched_keys.push((key.0, new_v.to_vec()));
229						if batched_keys.len() < self.params.batch_size {
230							continue;
231						}
232
233						let (size, duration) = if self.params.is_validate_block_mode() {
234							self.measure_per_key_amortised_validate_block_write_cost::<Block, H>(
235								original_root,
236								&storage,
237								shared_trie_cache.as_ref(),
238								batched_keys.clone(),
239								None,
240							)?
241						} else {
242							self.measure_per_key_amortised_import_block_write_cost::<Block, H>(
243								original_root,
244								&storage,
245								shared_trie_cache.as_ref(),
246								db.clone(),
247								batched_keys.clone(),
248								self.state_version(),
249								state_col,
250								Some(&info),
251							)?
252						};
253						record.append(size, duration)?;
254						batched_keys.clear();
255					}
256				}
257			}
258		}
259
260		Ok(record)
261	}

263	fn create_trie_backend<Block, H>(
264		&self,
265		original_root: Block::Hash,
266		storage: &Arc<dyn sp_state_machine::Storage<HashingFor<Block>>>,
267		shared_trie_cache: Option<&sp_trie::cache::SharedTrieCache<HashingFor<Block>>>,
268	) -> (DbState<HashingFor<Block>>, Option<Recorder<HashingFor<Block>>>)
269	where
270		Block: BlockT<Header = H, Hash = DbHash> + Debug,
271		H: HeaderT<Hash = DbHash>,
272	{
273		let recorder = (!self.params.disable_pov_recorder).then(|| Default::default());
274		let trie = DbStateBuilder::<HashingFor<Block>>::new(storage.clone(), original_root)
275			.with_optional_cache(shared_trie_cache.map(|c| c.local_cache_trusted()))
276			.with_optional_recorder(recorder.clone())
277			.build();
278
279		(trie, recorder)
280	}

282	/// Measures write benchmark
283	/// if `child_info` exist then it means this is a child tree key
284	fn measure_per_key_amortised_import_block_write_cost<Block, H>(
285		&self,
286		original_root: Block::Hash,
287		storage: &Arc<dyn sp_state_machine::Storage<HashingFor<Block>>>,
288		shared_trie_cache: Option<&sp_trie::cache::SharedTrieCache<HashingFor<Block>>>,
289		db: Arc<dyn sp_database::Database<DbHash>>,
290		changes: Vec<(Vec<u8>, Vec<u8>)>,
291		version: StateVersion,
292		col: ColumnId,
293		child_info: Option<&ChildInfo>,
294	) -> Result<(usize, Duration)>
295	where
296		Block: BlockT<Header = H, Hash = DbHash> + Debug,
297		H: HeaderT<Hash = DbHash>,
298	{
299		let batch_size = changes.len();
300		let average_len = changes.iter().map(|(_, v)| v.len()).sum::<usize>() / batch_size;
301		// For every batched write use a different trie instance and recorder, so we
302		// don't benefit from past runs.
303		let (trie, _recorder) =
304			self.create_trie_backend::<Block, H>(original_root, storage, shared_trie_cache);
305
306		let start = Instant::now();
307		// Create a TX that will modify the Trie in the DB and
308		// calculate the root hash of the Trie after the modification.
309		let replace = changes
310			.iter()
311			.map(|(key, new_v)| (key.as_ref(), Some(new_v.as_ref())))
312			.collect::<Vec<_>>();
313		let stx = match child_info {
314			Some(info) => trie.child_storage_root(info, replace.iter().cloned(), version).2,
315			None => trie.storage_root(replace.iter().cloned(), version).1,
316		};
317		// Only the keep the insertions, since we do not want to benchmark pruning.
318		let tx = convert_tx::<Block>(db.clone(), stx.clone(), false, col);
319		db.commit(tx).map_err(|e| format!("Writing to the Database: {}", e))?;
320		let result = (average_len, start.elapsed() / batch_size as u32);
321
322		// Now undo the changes by removing what was added.
323		let tx = convert_tx::<Block>(db.clone(), stx.clone(), true, col);
324		db.commit(tx).map_err(|e| format!("Writing to the Database: {}", e))?;
325
326		Ok(result)
327	}

329	/// Measures write benchmark on block validation
330	/// if `child_info` exist then it means this is a child tree key
331	fn measure_per_key_amortised_validate_block_write_cost<Block, H>(
332		&self,
333		original_root: Block::Hash,
334		storage: &Arc<dyn sp_state_machine::Storage<HashingFor<Block>>>,
335		shared_trie_cache: Option<&sp_trie::cache::SharedTrieCache<HashingFor<Block>>>,
336		changes: Vec<(Vec<u8>, Vec<u8>)>,
337		maybe_child_info: Option<&ChildInfo>,
338	) -> Result<(usize, Duration)>
339	where
340		Block: BlockT<Header = H, Hash = DbHash> + Debug,
341		H: HeaderT<Hash = DbHash>,
342	{
343		let batch_size = changes.len();
344		let average_len = changes.iter().map(|(_, v)| v.len()).sum::<usize>() / batch_size;
345		let (trie, recorder) =
346			self.create_trie_backend::<Block, H>(original_root, storage, shared_trie_cache);
347		for (key, _) in changes.iter() {
348			let _v = trie
349				.storage(key)
350				.expect("Checked above to exist")
351				.ok_or("Value unexpectedly empty")?;
352		}
353		let storage_proof = recorder
354			.map(|r| r.drain_storage_proof())
355			.expect("Storage proof must exist for block validation");
356		let root = trie.root();
357		debug!(
358			"POV: len {:?} {:?}",
359			storage_proof.len(),
360			storage_proof.clone().encoded_compact_size::<HashingFor<Block>>(*root)
361		);
362		let params = StorageAccessParams::<Block>::new_write(
363			*root,
364			storage_proof,
365			(changes, maybe_child_info.cloned()),
366		);
367
368		let mut durations_in_nanos = Vec::new();
369		let wasm_module = get_wasm_module();
370		let mut instance = wasm_module
371			.new_instance(sc_executor_common::wasm_runtime::DEFAULT_HEAP_ALLOC_STRATEGY)
372			.expect("Failed to create wasm instance");
373		let dry_run_encoded = params.as_dry_run().encode();
374		let encoded = params.encode();
375
376		for i in 1..=self.params.validate_block_rounds {
377			info!(
378				"validate_block with {} keys, round {}/{}",
379				batch_size, i, self.params.validate_block_rounds
380			);
381
382			// Dry run to get the time it takes without storage access
383			let dry_run_start = Instant::now();
384			instance
385				.call_export("validate_block", &dry_run_encoded)
386				.expect("Failed to call validate_block");
387			let dry_run_elapsed = dry_run_start.elapsed();
388			debug!("validate_block dry-run time {:?}", dry_run_elapsed);
389
390			let start = Instant::now();
391			instance
392				.call_export("validate_block", &encoded)
393				.expect("Failed to call validate_block");
394			let elapsed = start.elapsed();
395			debug!("validate_block time {:?}", elapsed);
396
397			durations_in_nanos.push(
398				elapsed.saturating_sub(dry_run_elapsed).as_nanos() as u64 / batch_size as u64,
399			);
400		}
401
402		let result = (
403			average_len,
404			std::time::Duration::from_nanos(
405				durations_in_nanos.iter().sum::<u64>() / durations_in_nanos.len() as u64,
406			),
407		);
408
409		Ok(result)
410	}

413/// Converts a Trie transaction into a DB transaction.
414/// Removals are ignored and will not be included in the final tx.
415/// `invert_inserts` replaces all inserts with removals.
416fn convert_tx<B: BlockT>(
417	db: Arc<dyn sp_database::Database<DbHash>>,
418	mut tx: PrefixedMemoryDB<HashingFor<B>>,
419	invert_inserts: bool,
420	col: ColumnId,
421) -> Transaction<DbHash> {
422	let mut ret = Transaction::<DbHash>::default();
423
424	for (mut k, (v, rc)) in tx.drain().into_iter() {
425		if rc > 0 {
426			db.sanitize_key(&mut k);
427			if invert_inserts {
428				ret.remove(col, &k);
429			} else {
430				ret.set(col, &k, &v);
431			}
432		}
433		// < 0 means removal - ignored.
434		// 0 means no modification.
435	}
436	ret
437}

439/// Checks if a new value causes any collision in tree updates
440/// returns true if there is no collision
441/// if `child_info` exist then it means this is a child tree key
442fn check_new_value<Block: BlockT>(
443	db: Arc<dyn sp_database::Database<DbHash>>,
444	trie: &DbState<HashingFor<Block>>,
445	key: &Vec<u8>,
446	new_v: &Vec<u8>,
447	version: StateVersion,
448	col: ColumnId,
449	child_info: Option<&ChildInfo>,
450) -> bool {
451	let new_kv = vec![(key.as_ref(), Some(new_v.as_ref()))];
452	let mut stx = match child_info {
453		Some(info) => trie.child_storage_root(info, new_kv.iter().cloned(), version).2,
454		None => trie.storage_root(new_kv.iter().cloned(), version).1,
455	};
456	for (mut k, (_, rc)) in stx.drain().into_iter() {
457		if rc > 0 {
458			db.sanitize_key(&mut k);
459			if db.get(col, &k).is_some() {
460				trace!("Benchmark-store key creation: Key collision detected, retry");
461				return false;
462			}
463		}
464	}
465	true
