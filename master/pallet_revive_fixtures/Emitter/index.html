Module Emitter 

Source
Expand description

Emits a log and then either returns or reverts, from its own frame or from a sub-call, so a test can watch where the log ends up.

contract Emitter {
    event Emitted(uint64 value);
    event Doomed(uint64 value);
    function emitValue(uint64 value) public { <stmts> }
    function emitThenRevert(uint64 value) public { <stmts> }
    function emitAndCallReverting(uint64 kept, uint64 dropped) public { <stmts> }
    uint64 public touched;
    function touchAndEmitDoomed(uint64 value) public { <stmts> }
    function callThenRevert(uint64 dropped) public { <stmts> }
    function emitAndCallRevertingChain(uint64 kept, uint64 dropped) public { <stmts> }
}
