Module LogVolume 

Source
Expand description

Emits logs of a chosen size, so a test can fill a transaction’s receipt to an exact number of bytes.

contract LogVolume {
    event Outer(uint256 value);
    uint256 public stored;
    function emitLogs(uint32 count, uint32 size, uint32 lastSize) public { <stmts> }
    function emitLogsThenRevert(uint32 count, uint32 size, uint32 lastSize) public { <stmts> }
    function emitLogsGuarded(uint32 count, uint32 size, uint32 lastSize) public returns (bool ok) { <stmts> }
    function storeAndEmitLogs(uint32 count, uint32 size, uint32 lastSize) public { <stmts> }
    function emitLogsThenCall(uint32 count, uint32 size, uint32 lastSize, address target, bytes calldata data) public returns (bool success) { <stmts> }
}
