SubCaller with more code that its functions never run.

The extra functions do arithmetic on their arguments. They take and return single words, with no arrays, strings or hashing.

contract SubCallerLarge is SubCaller {
    function mix(uint256 x) external pure returns (uint256) { <stmts> }
    function mix2(uint256 x, uint256 y) external pure returns (uint256) { <stmts> }
    function mix3(int256 x, int256 y) external pure returns (int256) { <stmts> }
    function mix4(uint256 x, uint256 y) external pure returns (uint256) { <stmts> }
    function mix5(int256 x, int256 y) external pure returns (int256) { <stmts> }
    function mix6(uint256 x, uint256 y) external pure returns (uint256) { <stmts> }
    function mix7(uint256 x, uint256 y) external pure returns (uint256) { <stmts> }
}
